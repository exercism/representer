Function Get-ColorCodeValue () {
[ CmdletBinding ()]
Param (
[ string []] $Colors
)
$allColors = [ string []]@( 'black' ; 'brown' ; 'red' ; 'orange' ; 'yellow' ; 'green' ; 'blue' ; 'violet' ; 'grey' ; 'white' )
$Colors | Select -First 2 | % { $val = 0 } { $val *= 10 ; $val += $allColors . IndexOf ( $_ ) } { $val }
}
