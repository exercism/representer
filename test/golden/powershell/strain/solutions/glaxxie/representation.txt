(file "Strain.ps1")
Function Invoke-Keep () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
$Data | Where-Object { & $Predicate $_ }
}
Function Invoke-Discard () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
$Data | Where-Object { -not ( & $Predicate $_ ) }
}
