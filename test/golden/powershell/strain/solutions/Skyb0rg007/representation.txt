(file "Strain.ps1")
Function Invoke-Keep () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
Process
{
$result = @()
foreach ( $obj in $Data ) {
if ( & $Predicate $obj ) {
$result += , $obj
}
}
return $result
}
}
Function Invoke-Discard () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
Process
{
$result = @()
foreach ( $obj in $Data ) {
if ( -not ( & $Predicate $obj )) {
$result += , $obj
}
}
return $result
}
}
