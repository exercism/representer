Function Invoke-Keep () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
$results = New-Object System.Collections.Generic.List [ Object ]
foreach ( $item in $Data ){
if ( & $Predicate $item ){
$results . add ( $item )
}
}
return $results . ToArray ()
}
Function Invoke-Discard () {
[ CmdletBinding ()]
Param (
[ Object []] $Data ,
[ ScriptBlock ] $Predicate
)
$results = New-Object System.Collections.Generic.List [ Object ]
foreach ( $item in $Data ){
if ( -not ( & $Predicate $item )){
$results . add ( $item )
}
}
return $results . ToArray ()
}
