(file "SpaceAge.ps1")
Function Get-SpaceAge ( $Seconds , $Planet = "\"Earth\"" ) {
$list = @( "\"Earth\"" , "\"Mercury\"" , "\"Venus\"" , "\"Mars\"" , "\"Jupiter\"" , "\"Saturn\"" , "\"Uranus\"" , "\"Neptune\"" )
if ( ! $list . Contains ( $Planet )){
Throw "\"Cannot validate argument on parameter 'Planet'. Invalid planet\""
}
$OrbitalPeriods = @{
'Mercury' = 0.2408467
'Venus' = 0.61519726
'Earth' = 1
'Mars' = 1.8808158
'Jupiter' = 11.862615
'Saturn' = 29.447498
'Uranus' = 84.016846
'Neptune' = 164.79132
}
[ math ]:: Round ( $Seconds / 31557600 / $OrbitalPeriods [ $Planet ], 2 )
}
