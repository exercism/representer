(file "SecretHandshake.ps1")
Function Invoke-SecretHandshake () {
[ CmdletBinding ()]
Param (
[ int ] $Number
)
$str = [ System.Convert ]:: ToString ( $Number , 2 )
$resultArr = @()
if ( $str [ -1 ] -eq "\"1\"" ) { $resultArr += "\"wink\"" }
if ( $str [ -2 ] -eq "\"1\"" ) { $resultArr += "\"double blink\"" }
if ( $str [ -3 ] -eq "\"1\"" ) { $resultArr += "\"close your eyes\"" }
if ( $str [ -4 ] -eq "\"1\"" ) { $resultArr += "\"jump\"" }
if ( $str [ -5 ] -eq "\"1\"" ) {[ array ]:: Reverse ( $resultArr )}
return $resultArr
}
