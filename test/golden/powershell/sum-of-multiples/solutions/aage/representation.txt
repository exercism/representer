Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ Int []] $Multiples ,
[ Int ] $Limit
)
Function Is-Multiple {
Param ([ Int ] $Num )
ForEach ( $m in $Multiples ) {
If ( $m -eq 0 ) { Continue }
ElseIf ( $Num % $m -eq 0 ) { Return $true }
}
Return $false
}
$sum = 0
1 .. ( $Limit - 1 ) | ? { Is-Multiple -Num $_ } | ForEach { $sum += $_ }
Return $sum
}
