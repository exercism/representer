Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ int []] $Multiples ,
[ int ] $Limit
)
$LessThanLimit = @( 0 )
$Multiples | ForEach-Object {
$multiple = $_
$current = $multiple
while ( $current -lt $Limit )
{
$LessThanLimit += $current
$current += $multiple
}
}
$Sum = $LessThanLimit | Select-Object -Unique | Measure-Object -Sum | Select-Object -ExpandProperty Sum
return $Sum
}
