Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ int []] $Multiples ,
[ int ] $Limit
)
if ( $Multiples . Length -eq 0 ) {
return 0
}
$Multiples = $Multiples | Sort-Object
$Sum = 0
for ( $i = $Multiples [ 0 ]; $i -lt $Limit ; $i ++ ) {
foreach ( $Number in $Multiples ) {
if ( $i % $Number -eq 0 ) {
$Sum += $i
break
}
}
}
return $Sum
}
