Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ int []] $Multiples ,
[ int ] $Limit
)
$r = foreach ( $m in $Multiples ) {
for ( $i = 0 ; $i -le [ Math ]:: Floor (( $Limit - 1 ) / $m ); $i ++ ) {

}
}
$r | Select-Object -Unique | Measure-Object -Sum | Select-Object -ExpandProperty Sum
}
