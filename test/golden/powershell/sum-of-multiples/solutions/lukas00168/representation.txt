Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ int []] $Multiples ,
[ int ] $Limit
)
$res = 0 ;
for ( $i = 1 ; $i -lt $Limit ; $i ++ ) {
for ( $m = 0 ; $m -lt $Multiples . Length ; $m ++ ){
if ( $i % $Multiples [ $m ] -eq 0 ){
$res = $res + $i ;
break ;
}
}
}
return $res ;
Throw "\"Function not implemented\""
}
