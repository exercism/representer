Function Get-SumOfMultiples {
[ CmdletBinding ()]
Param (
[ int []] $Multiples ,
[ int ] $Limit
)
$allMultiples = @()
foreach ( $element in $Multiples ) {
$recurrent = $element
while ( $recurrent -lt $Limit ) { $allMultiples += $recurrent ; $recurrent += $element }
}
return ( $allMultiples | Select-Object -Unique | Measure-Object -Sum ) . Sum
}
