Function Get-GrainSquare () {
[ CmdletBinding ()]
Param (
[ BigInt ] $Number
)
if ( $Number -le 0 -or $Number -ge 65 ) {
throw "\"*square must be between 1 and 64*\"" ;
}
return [ BigInt ][ Math ]:: pow ( 2 , $Number -1 );
}
Function Get-GrainTotal () {
return [ BigInt ]([ Math ]:: pow ( 2 , 64 ) - [ BigInt ] 1 );
}
