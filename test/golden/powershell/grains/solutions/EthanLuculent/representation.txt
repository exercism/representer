Function Get-GrainSquare () {
[ CmdletBinding ()]
Param (
[ BigInt ] $Number
)
if ( $Number -gt 64 -or $Number -lt 1 ) {
throw "\"square must be between 1 and 64\""
}
return [ BigInt ]:: Pow ( 2 , $Number -1 )
}
Function Get-GrainTotal () {
$total = 0
for ( $i = 1 ; $i -lt 65 ; $i ++ ) {
$total += Get-GrainSquare -Number $i
}
$total
}
