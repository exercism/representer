(file "Raindrops.ps1")
Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
if ( $Rain % 3 -eq 0 ) { $result += "\"Pling\"" }
if ( $Rain % 5 -eq 0 ) { $result += "\"Plang\"" }
if ( $Rain % 7 -eq 0 ) { $result += "\"Plong\"" }
if ( $Rain % 3 -ne 0 -and $Rain % 5 -ne 0 -and $Rain % 7 -ne 0 ) { $result = $Rain }
return $result
}
Get-Raindrops -Rain 35
