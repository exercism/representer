(file "Raindrops.ps1")
Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
[ string ] $Result = "\"\""
$factorRain = @{
3 = "\"Pling\"" ;
5 = "\"Plang\"" ;
7 = "\"Plong\"" ;
}
Foreach ( $factor in ( $factorRain . GetEnumerator () | Sort-Object -Property Name )){
If ( $Rain % $factor . Name -eq 0 ){
$Result += $factor . Value
}
}
If ([ string ]:: IsNullOrEmpty ( $Result )){
$Result = $Rain . ToString ()
}
Return $Result
}
