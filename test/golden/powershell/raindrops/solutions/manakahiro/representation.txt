(file "Raindrops.ps1")
Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
$ans = "\"\""
if ( $Rain % 3 -eq 0 ) {
$ans = $ans + "\"Pling\""
}
if ( $Rain % 5 -eq 0 ) {
$ans = $ans + "\"Plang\""
}
if ( $Rain % 7 -eq 0 ) {
$ans = $ans + "\"Plong\""
}
if ( $ans . Length -eq 0 ) {
$ans = $ans + [ string ] $Rain
}
return $ans
}
