(file "Raindrops.ps1")
Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
$result = "\"\""
if ( $Rain % 3 -eq 0 ) {
$result += "\"Pling\""
}
if ( $Rain % 5 -eq 0 ) {
$result += "\"Plang\""
}
if ( $Rain % 7 -eq 0 ) {
$result += "\"Plong\""
}
if ( $result -eq "\"\"" ) {
$result = $Rain . ToString ()
}
return $result
}
