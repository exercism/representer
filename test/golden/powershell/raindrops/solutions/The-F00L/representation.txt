Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
$res = "\"\""
if ( $Rain % 3 -eq 0 ){
$res = $res + "\"Pling\"" ;
}
if ( $Rain % 5 -eq 0 ){
$res = $res + "\"Plang\"" ;
}
if ( $Rain % 7 -eq 0 ){
$res = $res + "\"Plong\"" ;
}
if ( $res -eq "\"\"" ){
return $Rain ;
} else {
return $res ;
}
Throw "\"Not implemented exception\""
}
