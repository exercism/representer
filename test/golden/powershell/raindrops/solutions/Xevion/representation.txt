Function Get-Raindrops () {
[ CmdletBinding ()]
Param (
[ int ] $Rain
)
[ String ] $result = "\"\""
if ( $Rain % 3 -eq 0 ) { $result += "\"Pling\"" }
if ( $Rain % 5 -eq 0 ) { $result += "\"Plang\"" }
if ( $Rain % 7 -eq 0 ) { $result += "\"Plong\"" }
if ( $result . Length -gt 0 ) { return $result }
return $Rain . ToString ()
}
