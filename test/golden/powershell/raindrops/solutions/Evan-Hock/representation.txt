function Get-Raindrops {
[ CmdletBinding ()]
param (
[ int ] $Rain
)
$conversion = "\"\""
if ( $Rain % 3 -eq 0 )
{
$conversion += "\"Pling\""
}
if ( $Rain % 5 -eq 0 )
{
$conversion += "\"Plang\""
}
if ( $Rain % 7 -eq 0 )
{
$conversion += "\"Plong\""
}
if ( $conversion -eq "\"\"" )
{
return [ string ] $Rain
}
else
{
return $conversion
}
}
