$ global : allergens =
$ global : THRESHOLD = 256 ;
Class Allergies {
$score = 0 ;
$allergicTo = @(
@{
"\"eggs\"" = 1 ;
"\"peanuts\"" = 2 ;
"\"shellfish\"" = 4 ;
"\"strawberries\"" = 8 ;
"\"tomatoes\"" = 16 ;
"\"chocolate\"" = 32 ;
"\"pollen\"" = 64 ;
"\"cats\"" = 128 ;
} .GetEnumerator( ) | Sort-Object -Property Value
)
Allergies ( $score ) {
if ( $score -gt $ global : THRESHOLD ){
$this . score = $score - $ global : THRESHOLD ;
} else {
$this . score = $score
}
$this . allergicTo . foreach ({
$_ . Value = $ ( $this . score -band $_ . Value ) -ne 0 ;
})
}
[ bool ] IsAllergicTo ([ string ] $allergen ){
return $this . allergicTo . where ({ $_ . Key -imatch $allergen }) . Value
}
[ array ] GetAllergies (){
return $this . allergicTo . where ({ $_ . Value -eq $true }) . Name
}
}
