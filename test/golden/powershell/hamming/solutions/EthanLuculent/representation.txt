function Get-HammingDifference ([ string ] $strand1 , [ string ] $strand2 ) {
if ( $strand1 . ToCharArray () . count -ne $strand2 . ToCharArray () . count ) {
throw "\"strands must be of equal length.\""
}
if ( $strand1 -eq "\"\"" -and $strand2 -eq "\"\"" ) {
return 0
}
$strand1Arr = $strand1 . ToCharArray ()
$dist = 0
for ( $i = 0 ; $i -lt $strand1Arr . count ; $i ++ ) {
$dist += $strand1 . Substring ( $i , 1 ) -ne $strand2 . Substring ( $i , 1 )
}
$dist
}
