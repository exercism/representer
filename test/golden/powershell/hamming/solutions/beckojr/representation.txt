function Get-HammingDifference ([ string ] $strand1 , [ string ] $strand2 ) {
$lg = $strand1 . Length
if ( $lg -ne $strand2 . Length ) { Throw "\"strands must be of equal length.\"" }
$dist = 0
for ( $i = 0 ; $i -lt $lg ; $i ++ ) { if ( $strand1 [ $i ] -ne $strand2 [ $i ]) { $dist += 1 } }
return $dist
}
