function Get-HammingDifference ([ string ] $strand1 , [ string ] $strand2 ) {
if ( $strand1 . Length -ne $strand2 . Length ) {
throw "\"strands must be of equal length.\""
}
$distance = 0
for ( $i = 0 ; $i -lt $strand1 . Length ; $i ++ ) {
if ( $strand1 [ $i ] -ne $strand2 [ $i ]) {
$distance += 1
}
}
$distance
}
