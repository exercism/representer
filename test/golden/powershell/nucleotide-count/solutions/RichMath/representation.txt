Function Get-NucleotideCount () {
[ CmdletBinding ()]
Param (
[ string ] $Strand
)
if ( $Strand . Length -gt 0 -AND
$Strand -cnotmatch "\"^[ACGT]+$\"" ){
Throw "\"Invalid nucleotide in strand\""
}
$nc = @{ A = 0 ; C = 0 ; G = 0 ; T = 0 }
for ( $i = 0 ; $i -lt $Strand . Length ; $i ++ ){
[ string ] $k = $Strand [ $i ]
$nc [ $k ] = ( $nc [ $k ]) += 1
}
$nc
}
