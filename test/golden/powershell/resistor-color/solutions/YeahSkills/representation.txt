Function Get-ColorCode {
[ CmdletBinding ()]
Param (
[ string ] $Color
)
$ColorValues = @{
"\"black\"" = 0
"\"brown\"" = 1
"\"red\"" = 2
"\"orange\"" = 3
"\"yellow\"" = 4
"\"green\"" = 5
"\"blue\"" = 6
"\"violet\"" = 7
"\"grey\"" = 8
"\"white\"" = 9
}
$Value = $ColorValues [ $Color ]
Write-Output ( $Value ?? "\"\u{1F937}\u200D\u2642\uFE0F\"" )
}
Function Get-Colors {
$Colors = @( "\"black\"" , "\"brown\"" , "\"red\"" , "\"orange\"" , "\"yellow\"" , "\"green\"" , "\"blue\"" , "\"violet\"" , "\"grey\"" , "\"white\"" )
Write-Output $Colors
}
