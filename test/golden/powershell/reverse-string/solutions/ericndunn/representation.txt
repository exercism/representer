Function Get-ReverseString {
[ CmdletBinding ()]
Param (
[ Parameter ( Position = 1 , ValueFromPipeline = $true )]
[ string ] $Forward
)
$reversed = "\"\""
for ( $i = $Forward . Length ; $i -gt 0 ; $i -- ){
$reversed += $Forward [ $i - 1 ]
}
return $reversed
Throw "\"Function not implemented\""
}
