Function Get-ReverseString {
[ CmdletBinding ()]
Param (
[ Parameter ( Position = 1 , ValueFromPipeline = $true )]
[ string ] $Forward
)
if ( $Forward -ne "\"\"" ) {
for ( $i = 1 ; $i -le $Forward . Length ; $i ++ ) {
$result += $Forward . chars ( $Forward . Length - $i )
}
}
else { $result = "\"\"" }
return $result
}
