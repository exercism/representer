Function Get-ReverseString {
[ CmdletBinding ()]
Param (
[ Parameter ( Position = 1 , ValueFromPipeline = $true )]
[ string ] $Forward
)
[ string ] $Reversed = ''
for ( $i = 0 ; $i -lt $Forward . Length ; $i ++ ) {
$index = $Forward . Length - $i - 1
$Reversed += $Forward [ $index ]
}
return $Reversed
}
