Class Character {
$Strength
$Dexterity
$Constitution
$Intelligence
$Wisdom
$Charisma
$HitPoints
Character () {
"\"Strength\"" , "\"Dexterity\"" , "\"Constitution\"" , "\"Intelligence\"" , "\"Wisdom\"" , "\"Charisma\"" | ForEach-Object { $this . $_ = 3 .. 18 | Get-Random }
$this . HitPoints = 10 + [ Character ]:: GetModifier ( $this . Constitution )
}
static [ int ] GetModifier ([ int ] $Score ) {
return [ math ]:: Floor ( ( $Score -10 ) /2 )
}
[ int ] Ability () {
$ability = ( $this | Get-Member -MemberType Properties ) . Name | Get-Random
return $this . $ability
}
}
