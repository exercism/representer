(file "RotationalCipher.ps1")
Function Invoke-RotationalCipher () {
[ CmdletBinding ()]
Param (
[ string ] $Text ,
[ int ] $Shift
)
$result = -join ( $Text . ToCharArray () | ForEach-Object {
if ( $_ -ge 'a' -and $_ -le 'z' ) {
$charCode = ([ int ][ char ] $_ - [ int ][ char ] 'a' + $Shift ) % 26 + [ int ][ char ] 'a'
} elseif ( $_ -ge 'A' -and $_ -le 'Z' ) {
$charCode = ([ int ][ char ] $_ - [ int ][ char ] 'A' + $Shift ) % 26 + [ int ][ char ] 'A'
} else {
$charCode = [ int ][ char ] $_
}
[ char ] $charCode
})
return $result
}
