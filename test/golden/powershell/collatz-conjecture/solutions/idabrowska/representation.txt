(file "CollatzConjecture.ps1")
Function Invoke-CollatzConjecture () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
if ( $Number -le 0 )
{
throw "'error: Only positive numbers are allowed'"
}
$stepCount = 0 ;
$num = $Number ;
while ( $num -gt 1 )
{
if ( $num % 2 -eq 0 ) { $num = $num / 2 }
else { $num = 3 * $num + 1 }
$stepCount ++
}
return $stepCount
}
