(file "CollatzConjecture.ps1")
Function Invoke-CollatzConjecture () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
if ( $Number -le 0 ) {
Throw "\"error: Only positive numbers are allowed\""
}
$Steps = 0
while ( $Number -gt 1 ) {
if ( $Number % 2 ) {
$Number *= 3
$Number += 1
} else {
$Number / = 2
}
$Steps ++
}
return $Steps
}
