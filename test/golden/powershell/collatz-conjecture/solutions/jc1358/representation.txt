Function Invoke-CollatzConjecture () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
if ( $number -lt 1 ) { throw "\"*error: Only positive numbers are allowed*\"" }
$n = $number
$iter = 0
while ( $n -gt 1 ) {
if ( $n % 2 -eq 0 ) { $n = $n / 2 } else { $n = ( $n * 3 ) + 1 }
$iter ++
}
return $iter
}
