Function Invoke-CollatzConjecture () {
[ CmdletBinding ()]
Param (
[ Parameter ( Mandatory = $true , ParameterSetName = "\"Number\"" )]
[ Int64 ] $Number
)
try {
if ( $Number -le 0 ) {
throw "\"Error: Only positive numbers are allowed\""
}
$steps = 0
while ( $Number -ne 1 ) {
$steps += 1
if ( $Number % 2 -eq 0 ) {
$Number = $Number / 2
} else {
$Number = 3 * $Number + 1
}
}
return $steps
} catch {
throw $_ . Exception . Message
}
}
