(file "RnaTranscription.ps1")
Function Invoke-RnaTranscription () {
[ CmdletBinding ()]
Param (
[ string ] $Strand
)
$complements = @{
"\"G\"" = "\"C\""
"\"C\"" = "\"G\""
"\"T\"" = "\"A\""
"\"A\"" = "\"U\""
}
$rna = "\"\""
foreach ( $nucleotide in $Strand . ToCharArray ()) {
$rna += $complements [ $nucleotide . ToString ()]
}
return $rna
}
