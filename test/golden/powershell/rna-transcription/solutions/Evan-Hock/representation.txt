using namespace System.Text
function Invoke-RnaTranscription () {
[ CmdletBinding ()]
param (
[ string ] $Strand
)
$rna = [ StringBuilder ]:: new ( $Strand . Length )
foreach ( $dnaCodon in $Strand . GetEnumerator ())
{
$rnaCodon = switch ( $dnaCodon )
{
'A' { 'U' }
'C' { 'G' }
'G' { 'C' }
'T' { 'A' }
}
[ void ] $rna . Append ( $rnaCodon )
}
return $rna . ToString ()
}
