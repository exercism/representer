Function Get-BobResponse () {
[ CmdletBinding ()]
Param (
[ string ] $HeyBob
)
if ( $HeyBob -cmatch "\"^[^a-z]*[A-Z]+[^a-z\\?]*$\"" ) {
return "'Whoa, chill out!'"
}
elseif ( $HeyBob -cmatch "\"^[^a-z]*[A-Z]+[a-z]*\\?\\s*$\"" ) {
return "\"Calm down, I know what I'm doing!\""
}
elseif ( $HeyBob -cmatch "\"^.+\\?\\s*$\"" ) {
return 'Sure.'
}
elseif ( $HeyBob -cmatch "\"^\\s*$\"" ) {
return "'Fine. Be that way!'"
}
else {
return 'Whatever.'
}
}
