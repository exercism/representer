(file "BobResponse.ps1")
Function Get-BobResponse () {
[ CmdletBinding ()]
Param (
[ string ] $HeyBob
)
if ( -not ( IsTalking ( $HeyBob ))) {
return "\"Fine. Be that way!\""
}
if (( IsAQuestion ( $HeyBob )) -and ( IsYelling ( $HeyBob ))) {
return "\"Calm down, I know what I'm doing!\""
}
if ( IsAQuestion ( $HeyBob )) {
return "\"Sure.\""
}
if ( IsYelling ( $HeyBob )) {
return "\"Whoa, chill out!\""
}
return "\"Whatever.\""
}
Function CleanText () {
Param (
[ string ] $Text
)
return $Text . Trim ()
}
Function IsTalking () {
Param (
[ string ] $Text
)
return CleanText ( $Text ) -ne "\"\""
}
Function IsAQuestion () {
Param (
[ string ] $Text
)
$NewText = CleanText ( $Text )
return $NewText . EndsWith ( "\"?\"" )
}
Function IsYelling () {
Param (
[ string ] $Text
)
return ( $Text -ceq $Text . ToUpper ()) -and ( $Text -cmatch "\"[A-Za-z]\"" )
}
