(file "BobResponse.ps1")
Function Get-BobResponse () {
[ CmdletBinding ()]
Param (
[ string ] $HeyBob
)
$HeyBob = $HeyBob . trim ()
if ( $HeyBob -eq '' )
{
return "'Fine. Be that way!'" ;
}
elseif ( $HeyBob . ToUpper () -ceq $HeyBob "`\n" -and $HeyBob . EndsWith ( '?' ) "`\n" -and $HeyBob -match '[a-z]+' )
{
return "\"Calm down, I know what I'm doing!\"" ;
}
elseif ( $HeyBob . EndsWith ( '?' ))
{
return 'Sure.'
}
elseif ( $HeyBob . ToUpper () -ceq $HeyBob "`\n" -and $HeyBob -match '[a-z]+' )
{
return "'Whoa, chill out!'"
}
else
{
return 'Whatever.'
}
}
