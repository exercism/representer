Function Invoke-BinarySearch () {
[ CmdletBinding ()]
Param (
[ Int64 []] $Array ,
[ Int64 ] $Value
)
$low = 0
$high = $Array . Length - 1
while ( $low -le $high ) {
[ int ] $mid = ( $low + $high ) / 2
if ( $Array [ $mid ] -gt $Value ) {
$high = $mid - 1
}
elseif ( $Array [ $mid ] -lt $Value ) {
$low = $mid + 1
}
else {
return $mid
}
}
throw "\"*error: value not in array*\""
}
