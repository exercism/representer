Function Invoke-PerfectNumbers () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
if ( $Number -lt 1 ){
Throw "\"error: Classification is only possible for positive integers.\""
}
$i = 1
$sum = 0
while ( $i -lt $Number ){
if ( $Number % $i -eq 0 ){
$sum += $i
}
$i ++
}
if ( $Number -eq $sum ){
Return "\"perfect\""
}
if ( $Number -lt $sum ){
Return "\"abundant\""
}
if ( $Number -gt $sum ){
Return "\"deficient\""
}
Throw "\"Please implement this function\""
}
