Function Invoke-Append () {
[ CmdletBinding ()]
Param (
[ object []] $List1 ,
[ object []] $List2
)
$output = @()
$output += $list1
$output += $list2
return $output
}
Function Invoke-Concatenate () {
[ CmdletBinding ()]
Param (
[ object []] $Lists
)
$output = @()
$output += $lists | foreach-object { $_ }
return $output
}
Function Invoke-Filter () {
[ CmdletBinding ()]
Param (
[ object []] $List ,
[ scriptblock ] $Predicate
)
return $list | Where-Object { & $predicate $_ }
}
Function Get-Length () {
[ CmdletBinding ()]
Param (
[ object []] $List
)
return $list . count
}
Function Invoke-Map () {
[ CmdletBinding ()]
Param (
[ object []] $List ,
[ scriptblock ] $Function
)
return $list | ForEach-Object { & $Function $_ }
}
Function Invoke-Foldl () {
[ CmdletBinding ()]
Param (
[ scriptblock ] $Function ,
[ object []] $List ,
[ object ] $Accumulator
)
$list | ForEach-Object { $Accumulator = & $Function $Accumulator $_ }
return $Accumulator
}
Function Invoke-Foldr () {
[ CmdletBinding ()]
Param (
[ scriptblock ] $Function ,
[ object []] $List ,
[ object ] $Accumulator
)
Invoke-Reverse $list | ForEach-Object { $Accumulator = & $Function $Accumulator $_ }
return $Accumulator
}
Function Invoke-Reverse () {
[ CmdletBinding ()]
Param (
[ object []] $List
)
[ array ]:: reverse ( $list )
return $list
}
