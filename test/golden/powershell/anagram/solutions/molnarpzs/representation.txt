Function Invoke-Anagram () {
[ CmdletBinding ()]
Param (
[ string ] $Subject ,
[ string []] $Candidates
)
$sortedSubject = ( $Subject . ToLower () . ToCharArray () | Sort-Object ) -join ''
$filtered = $Candidates | Where-Object {
$_ . ToLower () -ne $Subject . ToLower () -and
( $_ . ToLower () . ToCharArray () | Sort-Object ) -join '' -eq $sortedSubject
}
return $filtered
}
