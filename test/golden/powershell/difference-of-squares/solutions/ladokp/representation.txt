(file "DifferenceOfSquares.ps1")
Function Get-SquareOfSum () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
[ Math ]:: Pow (( $Number * ( $Number + 1 ) / 2 ), 2 )
}
Function Get-SumOfSquares () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
$Number * ( $Number + 1 ) * ( 2 * $Number + 1 ) / 6
}
Function Get-DifferenceOfSquares () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
( Get-SquareOfSum ( $Number )) - ( Get-SumOfSquares ( $Number ))
}
