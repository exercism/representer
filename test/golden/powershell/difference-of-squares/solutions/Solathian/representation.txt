Function Get-SquareOfSum () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
return [ Math ]:: Pow (( $Number + $Number * $Number ) /2 , 2 )
}
Function Get-SumOfSquares () {
[ CmdletBinding ()]
Param (
[ Int64 ] $Number
)
$sum = 0
for ( $i = 1 ; $i -le $Number ; $i ++ )

return $sum ;
