Function Get-TwoFer (){
[ CmdletBinding ()]
Param (
[ Parameter ( Mandatory = $false , Position = 0 )]
[ string ] $Name
)
if ( -not [ string ]:: IsNullOrEmpty ( $Name )) {
$output = "\"One for " $Name ", one for me\""
}
else
{
$output = "\"One for you, one for me\""
}
return $output
Throw "\"Exercise not implemented\""
}
