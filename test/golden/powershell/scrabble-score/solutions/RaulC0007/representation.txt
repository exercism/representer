Function Get-ScrabbleScore () {
[ CmdletBinding ()]
Param (
[ string ] $Word ,
[ switch ] $Bonus
)
if ([ string ]:: IsNullOrEmpty ( $Word )) {
return 0
}
$score = 0
foreach ( $char in $Word . ToUpper () . ToCharArray ()) {
switch ( $char ) {
{ $_ -in 'A' , 'E' , 'I' , 'O' , 'U' , 'L' , 'N' , 'R' , 'S' , 'T' } { $score += 1 }
{ $_ -in 'D' , 'G' } { $score += 2 }
{ $_ -in 'B' , 'C' , 'M' , 'P' } { $score += 3 }
{ $_ -in 'F' , 'H' , 'V' , 'W' , 'Y' } { $score += 4 }
'K' { $score += 5 }
{ $_ -in 'J' , 'X' } { $score += 8 }
{ $_ -in 'Q' , 'Z' } { $score += 10 }
}
}
if ( $Bonus ) {
$score *= 2
}
return $score
}
