module Anagram
  ( anagramsFor
  ) where
anagramsFor :: String -> Array String -> Array String
anagramsFor word candidates =
  filter ( \ w -> toUpper w /= upWord )
  $ map fst
  $ filter ( snd >>> ( _ == word' ))
  $ map ( \ w -> Tuple w ( normalizeWord w )) candidates
  where
    upWord = toUpper word
    normalizeWord = toUpper >>> toCharArray >>> sort
    word' = normalizeWord word
