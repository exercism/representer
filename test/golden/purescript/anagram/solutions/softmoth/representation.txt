module Anagram
  ( anagramsFor
  ) where
import Prelude
import Data.Array ( filter , sort )
import Data.String ( toUpper )
import Data.String.CodeUnits ( toCharArray )
import Data.Tuple ( Tuple ( .. ), fst , snd )
anagramsFor :: String -> Array String -> Array String
anagramsFor word candidates =
  filter ( \ w -> toUpper w /= upWord )
  $ map fst
  $ filter ( snd >>> ( _ == word' ))
  $ map ( \ w -> Tuple w ( normalizeWord w )) candidates
  where
    upWord = toUpper word
    normalizeWord = toUpper >>> toCharArray >>> sort
    word' = normalizeWord word
