module Etl
  ( transform
  ) where
transform :: Map Int ( Array Char ) -> Map Char Int
transform ary =
  fromFoldable $
    concatMap convertOne $
      toUnfoldable ary
  where
    convertOne :: Tuple Int ( Array Char ) -> Array ( Tuple Char Int )
    convertOne ( Tuple i cs ) = map ( \ c -> Tuple c i ) ( charArrayToLower cs )
charArrayToLower :: Array Char -> Array Char
charArrayToLower = toCharArray <<< toLower <<< fromCharArray
