module Allergies
  ( allergicTo
  , list
  ) where
allergicTo :: Int -> String -> Boolean
allergicTo scores allergy = case lookup allergy allergiesWithScore of
  Just score -> scores .&. score /= 0
  _ -> false
list :: Int -> Array String
list scores = map fst $ filter includedInScores allergiesWithScore
  where includedInScores ( Tuple _ score ) = scores .&. score /= 0
allergiesWithScore :: Array ( Tuple String Int )
allergiesWithScore =
  [ Tuple "\"eggs\"" 1
  , Tuple "\"peanuts\"" 2
  , Tuple "\"shellfish\"" 4
  , Tuple "\"strawberries\"" 8
  , Tuple "\"tomatoes\"" 16
  , Tuple "\"chocolate\"" 32
  , Tuple "\"pollen\"" 64
  , Tuple "\"cats\"" 128
  ]
