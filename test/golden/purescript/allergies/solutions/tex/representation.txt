module Allergies
  ( allergicTo
  , list
  ) where
import Prelude
import Data.Array ( elemIndex , filter )
import Data.Int ( pow )
import Data.Int.Bits (( .&. ))
import Data.Maybe ( maybe )
import Data.String ( Pattern ( .. ), split )
allergies ∷ Array String
allergies = split ( Pattern "\",\"" )
  "\"eggs,peanuts,shellfish,strawberries,tomatoes,chocolate,pollen,cats\""
allergicTo ∷ Int → String → Boolean
allergicTo n s = n > 0 && bit == n .&. bit
  where
    bit = maybe 0 identity $ pow 2 <$> elemIndex s allergies
list ∷ Int → Array String
list n = filter ( allergicTo n ) allergies
