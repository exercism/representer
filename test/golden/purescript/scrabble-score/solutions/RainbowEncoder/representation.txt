module ScrabbleScore
  ( scoreWord
  ) where
scoreLetter :: Pattern -> Int
scoreLetter p
  | contains p "\"AEIOULNRST\"" = 1
  | contains p "\"DG\"" = 2
  | contains p "\"BCMP\"" = 3
  | contains p "\"FHVWY\"" = 4
  | contains p "\"K\"" = 5
  | contains p "\"JX\"" = 8
  | contains p "\"QZ\"" = 10
  | otherwise = 0
scoreWord :: String -> Int
scoreWord str = toUpper str
  # toCharArray
  # map ( scoreLetter <<< Pattern <<< singleton )
  # foldl ( + ) 0
