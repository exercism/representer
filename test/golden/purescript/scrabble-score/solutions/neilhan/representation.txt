module ScrabbleScore
  ( scoreWord
  ) where
import Prelude
import Data.Array ( catMaybes , elem , filter , find )
import Data.Maybe ( Maybe ( .. ))
import Data.String ( toUpper )
import Data.String.CodeUnits ( toCharArray )
import Data.Traversable ( sequence , sum , traverse )
import Data.Tuple ( Tuple ( .. ), fst , snd )
scoreWord :: String -> Int
scoreWord s = sum $ catMaybes $ map charToScore $ toCharArray $ toUpper s
charToScore :: Char -> Maybe Int
charToScore c = snd <$> find ( \ tp -> elem c ( fst tp )) scores
scores :: Array ( Tuple ( Array Char ) Int )
scores =
  [ Tuple [ 'A' , 'E' , 'I' , 'O' , 'U' , 'L' , 'N' , 'R' , 'S' , 'T' ] 1
  , Tuple [ 'D' , 'G' ] 2
  , Tuple [ 'B' , 'C' , 'M' , 'P' ] 3
  , Tuple [ 'F' , 'H' , 'V' , 'W' , 'Y' ] 4
  , Tuple [ 'K' ] 5
  , Tuple [ 'J' , 'X' ] 8
  , Tuple [ 'Q' , 'Z' ] 10
  ]
