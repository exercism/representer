module Meetup
  ( meetup
  , Week ( .. )
  ) where
import Data.Date ( Date , Day , Month , Weekday , Year , canonicalDate , lastDayOfMonth , weekday )
import Data.Enum ( fromEnum , toEnum )
import Data.Int ( ceil , toNumber )
import Data.Maybe ( Maybe )
import Prelude ( bottom , mod , ( # ), ( * ), ( + ), ( - ), ( / ), ( < ), ( <$> ), ( > ), ( >>> ))
data Week = First
  | Second
  | Third
  | Fourth
  | Last
  | Teenth
meetup :: Year -> Month -> Week -> Weekday -> Maybe Date
meetup year month week weekday' =
  canonicalDate year month <$> toEnum ( daysFromMonthToDate + 1 )
  where
    nthWeekSinceMonth = case week of
      First -> 1
      Second -> 2
      Third -> 3
      Fourth -> 4
      _ -> 1
    firstWeekdayOfMonth = canonicalDate year month bottom # weekday # normalize
    daysFromWeekToDate = case week of
      Last ->
        let lastDayOfMonth' = lastDayOfMonth year month in
        let lastDateOfMonth = canonicalDate year month lastDayOfMonth' in
        let lastWeekdayOfMonth = weekday lastDateOfMonth # normalize in
        let daysInMonth = fromEnum lastDayOfMonth' - 1 in
        let weekdayDelta = normalize weekday' - lastWeekdayOfMonth in
          daysInMonth + weekdayDelta - if weekdayDelta > 0 then 7 else 0
      Teenth ->
        let weekdayOfThirteenth = firstWeekdayOfMonth + 12 # ( _ ` mod ` 7 ) in
        let weekdayDelta = normalize weekday' - weekdayOfThirteenth in
          12 + weekdayDelta + if weekdayDelta < 0 then 7 else 0
      _ ->
        let weekdayDelta = normalize weekday' - firstWeekdayOfMonth in
          weekdayDelta + if weekdayDelta < 0 then 7 else 0
    daysFromMonthToDate = ( nthWeekSinceMonth - 1 ) * 7 + daysFromWeekToDate
    toWeek :: Day -> Int
    toWeek = fromEnum >>> toNumber >>> ( _ / 7.0 ) >>> ceil >>> ( _ - 1 )
    normalize :: Weekday -> Int
    normalize = fromEnum >>> ( _ ` mod ` 7 )
