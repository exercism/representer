module Meetup
  ( meetup ,
    Week ( .. )
  ) where
import Data.Date ( Date , canonicalDate , day , lastDayOfMonth , weekday )
import Data.Date.Component ( Day , Month , Weekday , Year )
import Data.Enum ( fromEnum , toEnum )
import Data.Maybe ( Maybe )
import Data.Array ( elem , filter , find , last , mapMaybe , ( !! ), ( .. ))
import Prelude
data Week = First
  | Second
  | Third
  | Fourth
  | Last
  | Teenth
meetup ∷ Year → Month → Week → Weekday → Maybe Date
meetup y m we wd =
  case we of
    First → weekDays !! 0
    Second → weekDays !! 1
    Third → weekDays !! 2
    Fourth → weekDays !! 3
    Last → last weekDays
    Teenth → find ( \ d → ( fromEnum ( day d )) ` elem ` ( 13 .. 19 )) weekDays
  where
    weekDays ∷ Array Date
    weekDays = filter ( \ d → weekday d == wd ) $ daysInMonth y m
daysInMonth ∷ Year → Month → Array Date
daysInMonth y m =
  map ( canonicalDate y m ) $ mapMaybe toEnum $ 1 .. ( fromEnum $ lastDayOfMonth y m )
