module Leap
  ( isLeapYear
  ) where
import Prelude
isDivisibleBy :: Int -> Int -> Boolean
isDivisibleBy year divisor =
  ( div year divisor ) * divisor == year
isLeapYear :: Int -> Boolean
isLeapYear year =
  (( isDivisibleBy year 4 ) && ( not ( isDivisibleBy year 100 ))) || (( isDivisibleBy year 100 ) && ( isDivisibleBy year 400 ) && ( isDivisibleBy year 4 ))
