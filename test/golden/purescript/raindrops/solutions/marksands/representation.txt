module Raindrops ( raindrops ) where
import Data.Maybe ( Maybe ( .. ))
import Prelude ( mod , show , ( <> ), ( == ))
import Data.List ( foldl )
raindrops :: Int -> String
raindrops value = do
  case result of
    Just v -> v
    _ -> show value
  where
    pling = if ( mod value 3 ) == 0 then Just "\"Pling\"" else Nothing
    plang = if ( mod value 5 ) == 0 then Just "\"Plang\"" else Nothing
    plong = if ( mod value 7 ) == 0 then Just "\"Plong\"" else Nothing
    result = foldl ( <> ) Nothing [ pling , plang , plong ]
