module Raindrops
  ( raindrops
  ) where
import Prelude
raindrops :: Int -> String
raindrops n = if plingPlangPlong == "\"\"" then show n else plingPlangPlong
  where
    plingPlangPlong = pling <> plang <> plong
    pling = if 3 ` divides ` n then "\"Pling\"" else "\"\""
    plang = if 5 ` divides ` n then "\"Plang\"" else "\"\""
    plong = if 7 ` divides ` n then "\"Plong\"" else "\"\""
    divides a b = b ` mod ` a == 0
