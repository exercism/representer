module Raindrops
  ( raindrops
  ) where
raindrops :: Int -> String
raindrops n = if null $ res then show n else res
  where
  res =
    foldr processRaindrops "\"\""
      [ Tuple 3 "\"Pling\"" , Tuple 5 "\"Plang\"" , Tuple 7 "\"Plong\"" ]
  processRaindrops ( Tuple int str ) acc
    | n ` mod ` int == 0 = str <> acc
    | otherwise = acc
