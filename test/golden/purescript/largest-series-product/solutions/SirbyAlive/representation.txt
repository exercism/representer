module LargestSeriesProduct
  ( largestProduct
  )
  where
import Prelude
import Data.Array ( length , range , slice )
import Data.Foldable ( maximum , product )
import Data.Int ( fromString )
import Data.Maybe ( Maybe ( .. ))
import Data.String ( Pattern ( .. ), split )
import Data.String ( length ) as S
import Data.Traversable ( traverse )
largestProduct :: String -> Int -> Maybe Int
largestProduct str n
  | S . length str < n || n < 0 = Nothing
  | otherwise = do
    ints <- traverse fromString $ split ( Pattern "\"\"" ) str
    maximum $ product <$> genSubArray n ints
genSubArray :: forall a . Int -> Array a -> ( Array ( Array a ))
genSubArray n arr =
  let read i = slice i ( i + n ) arr
  in read <$> ( range 0 $ ( _ - n ) $ length arr )
