module Hamming
  ( distance
  ) where
import Prelude
import Data.Maybe ( Maybe ( .. ))
import Data.String ( uncons )
count :: Int -> String -> String -> Maybe Int
count c a b = case uncons a , uncons b of
  Just { head : x , tail : xs }, Just { head : y , tail : ys } -> count ( c + if x == y then 0 else 1 ) xs ys
  Just _ , Nothing -> Nothing
  Nothing , Just _ -> Nothing
  Nothing , Nothing -> Just c
distance :: String -> String -> Maybe Int
distance = count 0
