module Hamming
  ( distance
  ) where
import Prelude
import Data.Array ( filter , zipWith , length )
import Data.Maybe ( Maybe ( .. ))
import Data.String ( length ) as String
import Data.String.CodeUnits ( toCharArray )
distance :: String -> String -> Maybe Int
distance xs ys
  | String . length xs /= String . length ys = Nothing
  | otherwise = Just $ distance' xs ys
distance' :: String -> String -> Int
distance' xs ys = length <<< filter not $ zipWith ( == ) x y
  where x = toCharArray xs
    y = toCharArray ys
