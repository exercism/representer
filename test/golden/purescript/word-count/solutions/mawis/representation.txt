module WordCount
  ( wordCount
  ) where
wordCount :: String -> Map String Int
wordCount = fromFoldableWith ( + )
  <<< map ( \ w -> Tuple w 1 )
  <<< filter ( not <<< null )
  <<< map unquote
  <<< split ( unsafeRegex "\"[^a-z1-9']\"" noFlags )
  <<< toLower
  where unquote = replace ( unsafeRegex "\"^'(.*)'$\"" noFlags ) "\"$1\""
