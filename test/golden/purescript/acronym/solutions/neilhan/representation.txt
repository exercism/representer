module Acronym
  ( abbreviate
  ) where
abbreviate :: String -> String
abbreviate =
  separateMixCase
  >>> replace ( unsafeRegex "\"[^\\\\w]\"" global ) "\" \""
  >>> split ( Pattern "\" \"" )
  >>> map ( take 1 )
  >>> joinWith "\"\""
  >>> toUpper
separateMixCase :: String -> String
separateMixCase = replace ( unsafeRegex "\"([a-z])([A-Z])\"" global ) "\"$1 $2\""
