component {
this . drops = [
[ 3 , "\"Pling\"" ],
[ 5 , "\"Plang\"" ],
[ 7 , "\"Plong\"" ]
];
function forEach ( fn , arr ) {
for ( i = 1 ; i <= ArrayLen ( arr ); i ++ ) {
fn ( arr [ i ]);
}
}
function reduce ( fn , acc , arr ) {
forEach ( function ( item ) {
acc = fn ( acc , item );
}, arr );
return acc ;
};
function convert ( number ) {
res = reduce ( function ( acc , drop ) {
return ( number % drop [ 1 ] == 0 ? acc & drop [ 2 ] : acc );
}, "\"\"" , this . drops );
return ( res == "\"\"" ? res & number : res );
}
}
