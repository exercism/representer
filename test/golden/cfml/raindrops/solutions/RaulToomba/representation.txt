component {
function convert ( number ) {
raindrop = "\"\"" ;
if ( number MOD 3 == 0 ){
raindrop = "\"Pling\"" ;
}
if ( number MOD 5 == 0 ){
raindrop = raindrop & "\"Plang\"" ;
}
if ( number MOD 7 == 0 ){
raindrop = raindrop & "\"Plong\"" ;
}
if ( raindrop == "\"\"" ){
raindrop = number ;
}
return raindrop ;
}
}
