component {
function keep ( list , predicate ) {
var results = [];
for ( elem in list ) {
if ( predicate ( elem )) {
arrayAppend ( results , elem );
}
}
return results ;
}
function discard ( list , predicate ) {
var newPredicate = function ( x ) { return not predicate ( x ); }
return keep ( list , newPredicate );
}
}
