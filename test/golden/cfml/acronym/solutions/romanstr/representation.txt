component {
function abbreviate ( phrase ) {
phrase = rereplace ( phrase , "\"[^A-Za-z0-9]+\"" , "\" \"" , "\"all\"" );
var firstChars = refindnocase ( "\"( |^)[A-Z0-9]\"" , phrase , 1 , true , "\"all\"" );
var acronym = "\"\"" ;
firstChars.each ( function ( element ){
acronym &= trim ( element.match [ 1 ]);
});
return uCase ( acronym );
}
}
