component {
function create ( row , col ) {
if ( row < 0 || row > 7 || col < 0 || col > 7 )
return false ;
return { "\"row\"" : row , "\"col\"" : col };
}
function canAttack ( queen1 , queen2 ) {
if ( queen1 [ "\"row\"" ] == queen2 [ "\"row\"" ] )
return true ;
if ( queen1 [ "\"col\"" ] == queen2 [ "\"col\"" ] )
return true ;
if ( queen1 [ "\"row\"" ] + queen1 [ "\"col\"" ] == queen2 [ "\"row\"" ] + queen2 [ "\"col\"" ] )
return true ;
if ( queen1 [ "\"row\"" ] - queen1 [ "\"col\"" ] == queen2 [ "\"row\"" ] - queen2 [ "\"col\"" ] )
return true ;
return false ;
}
}
