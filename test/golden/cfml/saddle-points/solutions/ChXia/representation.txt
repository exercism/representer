component {
function saddlePoints ( matrix ) {
var arrayMaxR = [];
var arrayMinC = [];
var result = [];
var totalRow = matrix.len ();
var totalColumn = matrix [ 1 ]. len ();
for ( var r = 1 ; r <= totalRow ; r ++ ) {
var maxRow = - 1 ;
var candidateRow = [];
for ( var c = 1 ; c <= totalColumn ; c ++ ) {
if ( matrix [ r ][ c ] > maxRow ) {
maxRow = matrix [ r ][ c ];
candidateRow = [ { "\"row\"" : r - 1 , "\"column\"" : c - 1 } ];
} else if ( matrix [ r ][ c ] == maxRow ) {
candidateRow.append ({ "\"row\"" : r - 1 , "\"column\"" : c - 1 });
}
}
arrayMaxR.addAll ( candidateRow );
}
for ( var c = 1 ; c <= totalColumn ; c ++ ){
var minColumn = 10000000 ;
var canditateColumn = [];
for ( var r = 1 ; r <= totalRow ; r ++ ){
if ( matrix [ r ][ c ] < minColumn ){
minColumn = matrix [ r ][ c ];
canditateColumn = [ { "\"row\"" : r - 1 , "\"column\"" : c - 1 } ];
}
else if ( matrix [ r ][ c ] == minColumn ){
canditateColumn.append ({ "\"row\"" : r - 1 , "\"column\"" : c - 1 });
}
}
arrayMinC.addAll ( canditateColumn );
}
for ( var max in arrayMaxR ){
for ( var min in arrayMinC ){
if ( max.row == min.row && max.column == min.column )
result.append ( max );
}
}
return result ;
}
