component {
function squareOfSum ( number ) {
var sum = 0 ;
for ( var i = 1 ; i <= number ; i ++ ) {
sum += i ;
}
return sum * sum ;
}
function sumOfSquares ( number ) {
var numArr = [];
var sqNumArr = [];
var sum = 0
for ( var i = 1 ; i <= number ; i ++ ){
arrayAppend ( numArr , i );
}
for ( var j = 1 ; j <= arrayLen ( numArr ); j ++ ) {
arrayAppend ( sqNumArr , ( j * j ));
}
for ( num in sqNumArr ) {
sum += num
}
return sum ;
}
function differenceOfSquares ( number ) {
return squareOfSum ( number ) - sumOfSquares ( number );
}
}
