(file "DifferenceOfSquares.cfc")
component {
function squareOfSum ( number ) {
result = 0 ;
for ( i = 1 ; i <= number ; i ++ ) {
result += i ;
}
return result * result ;
}
function sumOfSquares ( number ) {
result = 0 ;
for ( i = 1 ; i <= number ; i ++ ) {
result += i * i ;
}
return result ;
}
function differenceOfSquares ( number ) {
return squareOfSum ( number ) - sumOfSquares ( number );
}
}
