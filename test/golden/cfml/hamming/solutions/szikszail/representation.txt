(file "Hamming.cfc")
component {
function distance ( strand1 , strand2 ) {
if ( len ( strand1 ) != len ( strand2 )) {
throw ( message = "\"left and right strands must be of equal length\"" )
}
var n = 0 ;
for ( var i = 1 ; i lte len ( strand1 ); i ++ ) {
if ( strand1 [ i ] != strand2 [ i ]) {
n ++ ;
}
}
return n
}
}
