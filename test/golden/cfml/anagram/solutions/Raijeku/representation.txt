component {
function anagrams ( string subject , array candidates ) {
result = [];
for ( candidate in candidates ) {
counter = 0 ;
candidate = lCase ( candidate );
subjectI = lCase ( subject );
for ( i = 1 ; i <= len ( candidate ); i ++ ){
char = right ( left ( candidate , i ), 1 )
if ( listContains ( subjectI , char )){
counter += 1 ;
subjectI = replace ( subjectI , char , "\"\"" , "\"one\"" );
}
}
if (( counter EQ len ( candidate )) AND ( len ( candidate ) EQ len ( subject )) AND ( subject NEQ candidate )) {
arrayAppend ( result , candidate );
}
}
return result ;
}
}
