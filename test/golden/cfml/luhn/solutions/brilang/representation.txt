(file "Luhn.cfc")
component {
function valid ( value ) {
var input = replace ( arguments . value , "' '" , '' , 'ALL' );
if ( not isNumeric ( input ) ) {
return false ;
}
var nDigits = Len ( input );
if ( nDigits eq 1 ) {
return false ;
}
var parity = nDigits MOD 2 ;
var digit = "\"\"" ;
var sum = 0 ;
var i = "\"\"" ;
for ( i = 0 ; i lte nDigits - 1 ; i ++ ) {
digit = Mid ( input , i + 1 , 1 );
if ( ( i mod 2 ) eq parity ) {
digit = digit * 2 ;
if ( digit gt 9 ) {
digit -= 9 ;
}
}
sum = sum + digit ;
}
if ( sum mod 10 ) {
return false ;
}
else {
return true ;
}
}
}
