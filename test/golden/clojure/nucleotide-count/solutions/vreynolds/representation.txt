(source
  (list_lit ns nucleotide-count)
  (list_lit def nucleotide-counts
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_1))
        (list_lit merge
          (map_lit "\\A" 0 "\\T" 0 "\\C" 0 "\\G" 0)
          (list_lit frequencies PLACEHOLDER_1)))))
  (list_lit def count-of-nucleotide-in-strand
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_2) (local PLACEHOLDER_1))
        (map_lit :pre
          (vec_lit
            (list_lit contains?
              (set_lit "\\A" "\\T" "\\C" "\\G") PLACEHOLDER_2)))
        (list_lit
          (list_lit nucleotide-counts PLACEHOLDER_1) PLACEHOLDER_2)))))
