(file "src/high_scores.clj"
  (source
    (list_lit ns high-scores)
    (list_lit def scores
      (list_lit fn*
        (list_lit
          (vec_lit scores) scores)))
    (list_lit def latest
      (list_lit fn*
        (list_lit
          (vec_lit scores)
          (list_lit last scores))))
    (list_lit def personal-best
      (list_lit fn*
        (list_lit
          (vec_lit scores)
          (list_lit apply max scores))))
    (list_lit def personal-top-three
      (list_lit fn*
        (list_lit
          (vec_lit scores)
          (list_lit take 3
            (list_lit sort > scores)))))))
