(file "src/log_levels.clj"
  (source
    (list_lit ns log-levels
      (list_lit :require
        (vec_lit clojure.string :as str)))
    (list_lit def (definition PLACEHOLDER_1) "#\"(]: )|(\\[)\"")
    (list_lit def (definition PLACEHOLDER_2)
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit let
            (vec_lit (local PLACEHOLDER_4)
              (list_lit zipmap
                (vec_lit :log-level :message)
                (list_lit map
                  (qualified str trim)
                  (list_lit remove
                    (qualified str blank?)
                    (list_lit
                      (qualified str split) PLACEHOLDER_3 PLACEHOLDER_1)))))
            (list_lit update PLACEHOLDER_4 :log-level
              (qualified str lower-case))))))
    (list_lit def message
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit :message
            (list_lit PLACEHOLDER_2 PLACEHOLDER_3)))))
    (list_lit def log-level
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit :log-level
            (list_lit PLACEHOLDER_2 PLACEHOLDER_3)))))
    (list_lit def reformat
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit let
            (vec_lit G__1
              (list_lit PLACEHOLDER_2 PLACEHOLDER_3) (local PLACEHOLDER_5)
              (list_lit get G__1 :message) (local PLACEHOLDER_6)
              (list_lit get G__1 :log-level))
            (list_lit format "\"%s (%s)\"" PLACEHOLDER_5 PLACEHOLDER_6)))))))
