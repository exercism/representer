(file "src/log_levels.clj"
  (source
    (list_lit ns log-levels
      (list_lit :require
        (vec_lit clojure.string :as str)))
    (list_lit def (definition PLACEHOLDER_1) "#\"\\[([A-Z]+)\\]\\:\\s*(.+)\\s*\"")
    (list_lit def (definition PLACEHOLDER_2) "\"%s (%s)\"")
    (list_lit def message
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit
            (qualified str trim)
            (list_lit first
              (list_lit next
                (list_lit next
                  (list_lit re-matches PLACEHOLDER_1 PLACEHOLDER_3))))))))
    (list_lit def log-level
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit
            (qualified str lower-case)
            (list_lit second
              (list_lit re-matches PLACEHOLDER_1 PLACEHOLDER_3))))))
    (list_lit def reformat
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_3))
          (list_lit format PLACEHOLDER_2
            (list_lit message PLACEHOLDER_3)
            (list_lit log-level PLACEHOLDER_3)))))))
