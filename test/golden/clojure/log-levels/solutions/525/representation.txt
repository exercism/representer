  (source
    (list_lit ns log-levels
      (list_lit :require
        (vec_lit clojure.string :as str)))
    (list_lit def message
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1))
          (list_lit
            (qualified str trim)
            (list_lit get
              (list_lit
                (qualified str split) PLACEHOLDER_1 "#\":\"") 1)))))
    (list_lit def log-level
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1))
          (list_lit
            (qualified clojure.string lower-case)
            (list_lit
              (qualified clojure.string replace)
              (list_lit first
                (list_lit re-seq "#\"\\W\\w+\\W\"" PLACEHOLDER_1)) "#\"\\W\"" "\"\"")))))
    (list_lit def reformat
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1))
          (list_lit def (definition PLACEHOLDER_2)
            (list_lit log-level PLACEHOLDER_1))
          (list_lit if
            (list_lit = PLACEHOLDER_2 "\"error\"")
            (list_lit str
              (list_lit
                (qualified clojure.string replace)
                (list_lit
                  (qualified str trim) PLACEHOLDER_1) "#\"\\WERROR\\W+\\s\"" "\"\"") "\" (error)\"")
            (list_lit if
              (list_lit = PLACEHOLDER_2 "\"info\"") "\"Disk defragmented (info)\""
              (list_lit if
                (list_lit = PLACEHOLDER_2 "\"warning\"") "\"Decreased performance (warning)\""
                (list_lit if :else nil nil)))))))))
