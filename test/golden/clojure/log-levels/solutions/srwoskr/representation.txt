(source
  (list_lit ns log-levels
    (list_lit :require
      (vec_lit clojure.string :as str)))
  (list_lit def message
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_1))
        (list_lit
          (qualified str trim)
          (list_lit last
            (list_lit
              (qualified str split) PLACEHOLDER_1 "#\":\""))))))
  (list_lit def log-level
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_1))
        (list_lit
          (qualified str lower-case)
          (list_lit subs
            (list_lit first
              (list_lit
                (qualified str split) PLACEHOLDER_1 "#\"]\"")) 1)))))
  (list_lit def reformat
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_1))
        (list_lit str
          (list_lit message PLACEHOLDER_1) "\" (\""
          (list_lit log-level PLACEHOLDER_1) "\")\"")))))
