    (list_lit ns beer-song
      (list_lit :require
        (vec_lit clojure.string)))
    (list_lit def verse
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1))
          (list_lit let
            (vec_lit (local PLACEHOLDER_2)
              (list_lit fn*
                (list_lit
                  (vec_lit (local PLACEHOLDER_3))
                  (list_lit str
                    (list_lit if
                      (list_lit zero? PLACEHOLDER_3) "\"no more\""
                      (list_lit str PLACEHOLDER_3)) "\" bottle\""
                    (list_lit if
                      (list_lit = PLACEHOLDER_3 1) nil
                      (list_lit do "\"s\""))))) (local PLACEHOLDER_4)
              (list_lit PLACEHOLDER_2 PLACEHOLDER_1) (local PLACEHOLDER_5)
              (list_lit PLACEHOLDER_2
                (list_lit mod
                  (list_lit dec PLACEHOLDER_1) 100)))
            (list_lit str
              (list_lit
                (qualified clojure.string capitalize) PLACEHOLDER_4) "\" of beer on the wall, \"" PLACEHOLDER_4 "\" of beer.\"" "\\newline"
              (list_lit if
                (list_lit zero? PLACEHOLDER_1) "\"Go to the store and buy some more, \""
                (list_lit str "\"Take \""
                  (list_lit if
                    (list_lit = PLACEHOLDER_1 1) "\"it\"" "\"one\"") "\" down and pass it around, \"")) PLACEHOLDER_5 "\" of beer on the wall.\\n\"")))))
    (list_lit def sing
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_6))
          (list_lit sing PLACEHOLDER_6 0))
        (list_lit
          (vec_lit (local PLACEHOLDER_6) (local PLACEHOLDER_7))
          (list_lit
            (qualified clojure.string join) "\\newline"
            (list_lit map verse
              (list_lit range PLACEHOLDER_6
                (list_lit dec PLACEHOLDER_7) -1))))))))
