(source
  (list_lit ns beer-song
    (list_lit :require
      (vec_lit clojure.string :as string)))
  (list_lit def (definition PLACEHOLDER_1)
    (list_lit fn*
      (list_lit
        (vec_lit num)
        (list_lit if
          (list_lit = num 1) "\"bottle\"" "\"bottles\""))))
  (list_lit def (definition PLACEHOLDER_2)
    (list_lit fn*
      (list_lit
        (vec_lit num)
        (list_lit if
          (list_lit > num 0)
          (list_lit print-str num)
          (list_lit if
            (list_lit = num 0) "\"no more\"" nil)))))
  (list_lit def (definition PLACEHOLDER_3)
    (list_lit fn*
      (list_lit
        (vec_lit num)
        (list_lit if
          (list_lit > num 1) "\"Take one down and pass it around\""
          (list_lit if
            (list_lit = num 1) "\"Take it down and pass it around\""
            (list_lit if
              (list_lit = num 0) "\"Go to the store and buy some more\"" nil))))))
  (list_lit def verse
    (list_lit fn*
      (list_lit
        (vec_lit num)
        (list_lit let
          (vec_lit next
            (list_lit if
              (list_lit > num 0)
              (list_lit - num 1) 99))
          (list_lit format "\"%s %s of beer on the wall, %s %s of beer.\\n%s, %s %s of beer on the wall.\\n\""
            (list_lit
              (qualified string capitalize)
              (list_lit PLACEHOLDER_2 num))
            (list_lit PLACEHOLDER_1 num)
            (list_lit PLACEHOLDER_2 num)
            (list_lit PLACEHOLDER_1 num)
            (list_lit PLACEHOLDER_3 num)
            (list_lit PLACEHOLDER_2 next)
            (list_lit PLACEHOLDER_1 next))))))
  (list_lit def sing
    (list_lit fn*
      (list_lit
        (vec_lit (local PLACEHOLDER_4))
        (list_lit sing PLACEHOLDER_4 0))
      (list_lit
        (vec_lit (local PLACEHOLDER_4) (local PLACEHOLDER_5))
        (list_lit
          (qualified string join) "\"\\n\""
          (list_lit map
            (list_lit fn*
              (list_lit
                (vec_lit num)
                (list_lit verse num)))
            (list_lit reverse
              (list_lit range PLACEHOLDER_5
                (list_lit + PLACEHOLDER_4 1)))))))))
