(file "src/beer_song.clj"
  (source
    (list_lit ns beer-song
      (list_lit :require
        (vec_lit clojure.string :as st)))
    (list_lit def verse
      (list_lit fn*
        (list_lit
          (vec_lit num)
          (list_lit case num 2 "\"2 bottles of beer on the wall, 2 bottles of beer.\nTake one down and pass it around, 1 bottle of beer on the wall.\\n\"" 1 "\"1 bottle of beer on the wall, 1 bottle of beer.\nTake it down and pass it around, no more bottles of beer on the wall.\\n\"" 0 "\"No more bottles of beer on the wall, no more bottles of beer.\nGo to the store and buy some more, 99 bottles of beer on the wall.\\n\""
            (list_lit str num "\" bottles of beer on the wall, \"" num "\" bottles of beer.\nTake one down and pass it around, \""
              (list_lit dec num) "\" bottles of beer on the wall.\\n\"")))))
    (list_lit def sing
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1))
          (list_lit sing PLACEHOLDER_1 0))
        (list_lit
          (vec_lit (local PLACEHOLDER_1) (local PLACEHOLDER_2))
          (list_lit
            (qualified st join) "\"\\n\""
            (list_lit map verse
              (list_lit range PLACEHOLDER_1
                (list_lit dec PLACEHOLDER_2) -1))))))))
