(file "src/clock.clj"
  (source
    (list_lit ns clock)
    (list_lit def clock
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_1) (local PLACEHOLDER_2))
          (list_lit let
            (vec_lit (local PLACEHOLDER_3)
              (list_lit mod
                (list_lit +
                  (list_lit * PLACEHOLDER_1 60) PLACEHOLDER_2)
                (list_lit * 60 24)) (local PLACEHOLDER_4)
              (list_lit mod
                (list_lit quot PLACEHOLDER_3 60) 24) (local PLACEHOLDER_5)
              (list_lit mod PLACEHOLDER_3 60))
            (map_lit :hour PLACEHOLDER_4 :minute PLACEHOLDER_5)))))
    (list_lit def clock->string
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_6))
          (list_lit format "\"%02d:%02d\""
            (list_lit :hour PLACEHOLDER_6)
            (list_lit :minute PLACEHOLDER_6)))))
    (list_lit def add-time
      (list_lit fn*
        (list_lit
          (vec_lit (local PLACEHOLDER_6) (local PLACEHOLDER_7))
          (list_lit clock
            (list_lit :hour PLACEHOLDER_6)
            (list_lit +
              (list_lit :minute PLACEHOLDER_6) PLACEHOLDER_7)))))))
