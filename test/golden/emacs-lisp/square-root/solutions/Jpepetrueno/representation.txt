( defun square-root ( radicand )
"\"Compute the square root of RADICAND.\nThis function uses the Newton-Raphson method to approximate the square root.\""
( if ( < radicand 0 )
( error "\"Square root of a negative number is undefined\"" )
( let (( x ( / radicand 2.0 ))
( epsilon 0.00001 ))
( while ( > ( abs ( - ( * x x ) radicand )) epsilon )
( setq x ( / ( + x ( / radicand x )) 2.0 )))
( truncate x ))))
( provide 'square-root )
