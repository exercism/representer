(file "bob.el")
( require 'rx )
( defun response-for ( phrase )
( let* (( case-fold-search nil )
( trimmed
( replace-regexp-in-string "\"\\n\"" "\"\"" ( string-trim phrase )))
( is-question ( bob--questionp trimmed )))
( cond (( bob--yellp trimmed )
( if is-question
"\"Calm down, I know what I'm doing!\""
"\"Whoa, chill out!\"" ))
( is-question "\"Sure.\"" )
(( bob--silentp trimmed ) "\"Fine. Be that way!\"" )
( t "\"Whatever.\"" ))))
( defun bob--yellp ( phrase )
( and
( string= phrase ( upcase phrase ))
( string-match ( rx alpha ) phrase )))
( defun bob--questionp ( phrase )
( string-match ( rx "\"?\"" eol ) phrase ))
( defun bob--silentp ( phrase )
( string= phrase "\"\"" ))
( provide 'bob )
