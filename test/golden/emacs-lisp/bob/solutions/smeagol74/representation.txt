(file "bob.el")
( defun response-for--silence-p ( phrase )
"\"Check if PHRASE is silence.\""
( string-match-p "\"^[[:space:]]*$\"" ( string-replace "\"\\n\"" "\" \"" phrase )))
( defun response-for--question-p ( phrase )
"\"Check if PHRASE is question.\""
( string-suffix-p "\"?\"" ( string-trim phrase )))
( defun response-for--yell-p ( phrase )
"\"Check if PHRASE is yelling.\""
( and ( string-equal phrase ( upcase phrase ))
( not ( string-equal ( upcase phrase ) ( downcase phrase )))))
( defun response-for ( phrase )
"\"Responds to PHRASE as Bob.\""
( cond (( response-for--silence-p phrase ) "\"Fine. Be that way!\"" )
(( and ( response-for--question-p phrase ) ( response-for--yell-p phrase )) "\"Calm down, I know what I'm doing!\"" )
(( response-for--question-p phrase ) "\"Sure.\"" )
(( response-for--yell-p phrase ) "\"Whoa, chill out!\"" )
( t "\"Whatever.\"" )))
( provide 'bob )
