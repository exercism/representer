( defun response-for ( phrase )
( let (( stripped ( string-trim phrase )))
( if ( string= stripped "\"\"" ) "\"Fine. Be that way!\"" ( response-for-something stripped ))
))
( defun response-for-something ( phrase )
( if ( is-question phrase )
( if ( all-caps-p phrase ) "\"Calm down, I know what I'm doing!\"" "\"Sure.\"" )
( if ( all-caps-p phrase ) "\"Whoa, chill out!\"" "\"Whatever.\"" ))
)
( defun is-question ( string ) ( = ( last-char string ) ?? ))
( defun has-letters-p ( string ) ( string-match "\"[[:alpha:]]\"" string ))
( defun all-caps-p ( string ) ( and ( string= ( upcase string ) string ) ( has-letters-p string )))
( defun last-char ( string ) ( aref string ( - ( length string ) 1 )))
