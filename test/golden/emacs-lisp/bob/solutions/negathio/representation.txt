(file "bob.el")
( defun bob-question-p ( expression )
"\"t if EXPRESSION ends in '?' and the last word is lowercased\""
( string-match-p "\"\\\\?[[:space:]]*\\\\'\"" expression ))
( defun bob-no-addressing-p ( expression )
"\"t if EXPRESSION contains only whitespace characters\""
( string-match-p "\"\\\\`[[:space:]]*\\\\'\"" expression ))
( defun bob-yell-p ( expression )
"\"t if EXPRESSION is uppercased\""
( let (( case-fold-search nil ))
( and ( not ( string-match-p "\"[[:lower:]]\"" expression ))
( string-match-p "\"[[:alpha:]]\"" expression ))))
( defun bob-yell-question-p ( expression )
"\"t if EXPRESSION is uppercased and ends with '?'\""
( and ( bob-question-p expression ) ( bob-yell-p expression )))
( defun response-for ( expression )
( cond
(( bob-yell-question-p expression )
"\"Calm down, I know what I'm doing!\"" )
(( bob-question-p expression )
"\"Sure.\"" )
(( bob-no-addressing-p expression )
"\"Fine. Be that way!\"" )
(( bob-yell-p expression )
"\"Whoa, chill out!\"" )
( t "\"Whatever.\"" )))
( provide 'bob )
