( defun bob--char-code-p ( char category )
"\"Check whetever the CHAR is from UTF-8 CATEGORY.\""
( equal category ( get-char-code-property char 'general-category )))
( defun bob--question-p ( phrase )
"\"Check whetever the PHRASE is a question.\n\nThis predicate returns t if the string ends with \\\"?\\\".\""
( equal ?? ( car ( last ( string-to-list phrase )))))
( defun bob--phrase-upcase-p ( phrase )
"\"Check whetever the PHRASE is upcase only.\n\nThis predicate returns t if some upcase letter is present but\nnone lowercase is in the string.\""
( let (( letters ( string-to-list phrase )))
( and ( seq-some ( lambda ( letter ) ( bob--char-code-p letter 'Lu )) letters )
( seq-every-p ( lambda ( letter ) ( not ( bob--char-code-p letter 'Ll ))) letters ))))
( defun bob--upcase-question-p ( phrase )
"\"Check whetever the PHRASE is upcase and a question.\n\nThis predicate returns t if `question-p' AND `phrase-upcase-p' as t.\""
( and ( bob--question-p phrase ) ( bob--phrase-upcase-p phrase )))
( defun bob--blank-p ( phrase )
"\"Check whetever the PHRASE is an empty string.\""
( or ( string= "\"\"" phrase ) ( null phrase )))
( defun response-for ( phrase )
"\"Return Bob's reply from PHRASE text.\""
( pcase ( string-trim phrase )
(( pred bob--upcase-question-p ) "\"Calm down, I know what I'm doing!\"" )
(( pred bob--question-p ) "\"Sure.\"" )
(( pred bob--phrase-upcase-p ) "\"Whoa, chill out!\"" )
(( pred bob--blank-p ) "\"Fine. Be that way!\"" )
( _ "\"Whatever.\"" )))
( provide 'bob )
