( defun str-to-upcase-sorted ( str )
( if ( null str ) nil
( sort ( string-to-vector ( upcase str )) #' < )))
( defun anagrams-for ( subject candidates )
( let (( sub ( str-to-upcase-sorted subject ))
( can ( str-to-upcase-sorted ( car candidates ))))
( cond (( null candidates ) nil )
(( string= ( upcase subject ) ( upcase ( car candidates )))
( anagrams-for subject ( cdr candidates )))
(( equal sub can )
( cons ( car candidates ) ( anagrams-for subject ( cdr candidates ))))
( t ( anagrams-for subject ( cdr candidates ))))))
( provide 'anagram )
