( defun to-roman ( num )
"\"Convert integer NUM to a Roman numeral string.\""
( format "\"%s%s%s%s\""
( thousands-to-roman num )
( hundreds-to-roman num )
( tens-to-roman num )
( ones-to-roman num )))
( defun thousands-to-roman ( num )
"\"Convert the thousands place of NUM to Roman numerals.\""
( make-string ( / num 1000 ) ?M ))
( defun hundreds-to-roman ( num )
"\"Convert the hundreds place of NUM to Roman numerals.\""
( let (( hundreds ( / ( mod num 1000 ) 100 )))
( cond (( = hundreds 9 ) "\"CM\"" )
(( > hundreds 4 ) ( format "\"D%s\"" ( make-string ( - hundreds 5 ) ?C )))
(( = hundreds 4 ) "\"CD\"" )
( t ( make-string hundreds ?C )))))
( defun tens-to-roman ( num )
"\"Convert the tens place of NUM to Roman numerals.\""
( let (( tens ( / ( mod num 100 ) 10 )))
( cond (( = tens 9 ) "\"XC\"" )
(( > tens 4 ) ( format "\"L%s\"" ( make-string ( - tens 5 ) ?X )))
(( = tens 4 ) "\"XL\"" )
( t ( make-string tens ?X )))))
( defun ones-to-roman ( num )
"\"Convert the ones place of NUM to Roman numerals.\""
( let (( ones ( mod num 10 )))
( cond (( = ones 9 ) "\"IX\"" )
(( > ones 4 ) ( format "\"V%s\"" ( make-string ( - ones 5 ) ?I )))
(( = ones 4 ) "\"IV\"" )
( t ( make-string ones ?I )))))
( provide 'roman-numerals )
