(file "roman-numerals.el")
( defun to-roman ( number )
"\"Convert the given NUMBER to a roman numeral representation.\n\nNote, this function does not compact thousands beyond repeating the character\n`M' as may be found in some roman texts using a horizontal line above the\ncharacter to indicate 1000 times its value.\""
( interactive )
( let* (( ones ( % number 10 ))
( after-ones ( / ( - number ones ) 10 ))
( tens ( % after-ones 10 ))
( after-tens ( / ( - after-ones tens ) 10 ))
( hundreds ( % after-tens 10 ))
( after-hundreds ( / ( - after-tens hundreds ) 10 ))
( thousands ( % after-hundreds 10 )))
( concat
( make-string thousands ?M )
( cond (( = hundreds 9 ) "\"CM\"" )
(( = hundreds 8 ) "\"DCCC\"" )
(( = hundreds 7 ) "\"DCC\"" )
(( = hundreds 6 ) "\"DC\"" )
(( = hundreds 5 ) "\"D\"" )
(( = hundreds 4 ) "\"CD\"" )
( t ( make-string hundreds ?C )))
( cond (( = tens 9 ) "\"XC\"" )
(( = tens 8 ) "\"LXXX\"" )
(( = tens 7 ) "\"LXX\"" )
(( = tens 6 ) "\"LX\"" )
(( = tens 5 ) "\"L\"" )
(( = tens 4 ) "\"XL\"" )
( t ( make-string tens ?X )))
( cond (( = ones 9 ) "\"IX\"" )
(( = ones 8 ) "\"VIII\"" )
(( = ones 7 ) "\"VII\"" )
(( = ones 6 ) "\"VI\"" )
(( = ones 5 ) "\"V\"" )
(( = ones 4 ) "\"IV\"" )
( t ( make-string ones ?I ))))))
( provide 'roman-numerals )
