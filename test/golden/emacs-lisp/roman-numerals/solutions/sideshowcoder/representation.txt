(file "roman-numerals.el")
( require 'cl-seq )
( defun to-roman ( value )
"\"Convert VALUE into the roman numeral representation.\""
( let (( roman "\"\"" )
( roman-to-value ' (( "\"M\"" . 1000 ) ( "\"CM\"" . 900 )
( "\"D\"" . 500 ) ( "\"CD\"" . 400 )
( "\"C\"" . 100 ) ( "\"XC\"" . 90 )
( "\"L\"" . 50 ) ( "\"XL\"" . 40 )
( "\"X\"" . 10 ) ( "\"IX\"" . 9 )
( "\"V\"" . 5 ) ( "\"IV\"" . 4 )
( "\"I\"" . 1 ))))
( cl-flet (( >-roman-value ( to-match roman-value )
( >= to-match ( cdr roman-value ))))
( while ( > value 0 )
( let (( roman-numeral ( cl-find value roman-to-value :test #' >-roman-value )))
( setf roman ( concat roman ( car roman-numeral )))
( setf value ( - value ( cdr roman-numeral ))))))
roman ))
( provide 'roman-numerals )
