( if ( >= number divisor )
( progn ( setq n ( / number divisor ))
( make-string n character ))

( concat ( progn
( setq result ( reduce number 1000 ?M ))

( progn
( setq number ( - number 900 ))
"\"CM\"" )

( setq result ( reduce number 500 ?D ))

( progn

"\"XL\"" )
