(file "roman-numerals.el")
( defun get-letters ( power )
( if ( equal power 0 )
' ( "\"I\"" "\"V\"" "\"X\"" )
( if ( equal power 1 )
' ( "\"X\"" "\"L\"" "\"C\"" )
( if ( equal power 2 )
' ( "\"C\"" "\"D\"" "\"M\"" )
' ( "\"M\"" ))))
)
( defun get-ones ( digit letters )
( setq number-of-ones
( if ( equal ( % digit 5 ) 4 )
1
( % digit 5 )))
( make-string number-of-ones ( string-to-char ( car letters )))
)
( defun get-major-digit ( digit letters )
( if ( and ( >= digit 4 ) ( <= digit 8 ))
( nth 1 letters )
( if ( equal digit 9 ) ( nth 2 letters ) "\"\"" )
)
)
( defun write-digit ( digit letters )
( setq ones ( get-ones digit letters ))
( setq major-digit ( get-major-digit digit letters ))
( if ( or ( <= digit 5 ) ( equal digit 9 )) ( concat ones major-digit )
( concat major-digit ones )
))
( defun digit-to-roman ( digit power )
( if ( >= power 4 ) nil
( if ( and ( equal power 3 ) ( >= digit 4 )) nil
( setq letters ( get-letters power ))
( write-digit digit letters )
)
)
)
( defun to-roman ( decimal-number )
( setq i 0 )
( setq roman "\"\"" )
( while ( < i 4 )
( setq ten-power ( expt 10 i ))
( setq next-ten-power ( expt 10 ( 1+ i )))
( setq digit ( / ( - ( % decimal-number next-ten-power )
( % decimal-number ten-power ))
ten-power ))
( setq i ( 1+ i ))
( setq roman ( concat ( digit-to-roman digit ( 1- i )) roman ))
)
roman
)
( provide 'roman-numerals )
