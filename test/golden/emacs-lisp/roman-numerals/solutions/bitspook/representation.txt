(file "roman-numerals.el")
( require 'cl-lib )
( defun to-roman ( num )
"\"Convert decimal number to roman\""
( let* (( ceils ' (( "\"I\"" . 1 ) ( "\"IV\"" . 4 ) ( "\"V\"" . 5 ) ( "\"IX\"" . 9 ) ( "\"X\"" . 10 ) ( "\"XL\"" . 40 ) ( "\"L\"" . 50 ) ( "\"XC\"" . 90 ) ( "\"C\"" . 100 ) ( "\"CD\"" . 400 ) ( "\"D\"" . 500 ) ( "\"CM\"" . 900 ) ( "\"M\"" . 1000 )))
( romans ( reverse ( mapcar #' car ceils )))
( result "\"\"" )
( cur num )
( roman ( pop romans ))
( val ( alist-get roman ceils nil nil #' string= )))
( while ( > cur 0 )
( if ( <= val cur )
( setq result ( concat result roman )
cur ( - cur val ))
( setq roman ( pop romans )
val ( alist-get roman ceils nil nil #' string= ))))
result ))
( provide 'roman-numerals )
