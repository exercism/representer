( defun convert-to-symbol ( num )
"\"Convert NUM to a list of roman characters.\""
( cond
(( >= num 1000 ) ( cons ?M ( convert-to-symbol ( - num 1000 ))))
(( >= num 900 ) ( cons ?C ( cons ?M ( convert-to-symbol ( - num 900 )))))
(( >= num 500 ) ( cons ?D ( convert-to-symbol ( - num 500 ))))
(( >= num 400 ) ( cons ?C ( cons ?D ( convert-to-symbol ( - num 400 )))))
(( >= num 100 ) ( cons ?C ( convert-to-symbol ( - num 100 ))))
(( >= num 90 ) ( cons ?X ( cons ?C ( convert-to-symbol ( - num 90 )))))
(( >= num 50 ) ( cons ?L ( convert-to-symbol ( - num 50 ))))
(( >= num 40 ) ( cons ?X ( cons ?L ( convert-to-symbol ( - num 40 )))))
(( >= num 10 ) ( cons ?X ( convert-to-symbol ( - num 10 ))))
(( >= num 9 ) ( cons ?I ' ( ?X )))
(( >= num 5 ) ( cons ?V ( convert-to-symbol ( - num 5 ))))
(( >= num 4 ) ( cons ?I ( cons ?V ( convert-to-symbol ( - num 4 )))))
(( >= num 1 ) ( cons ?I ( convert-to-symbol ( - num 1 ))))))
( defun to-roman ( num )
"\"Convert NUM to a roman number.\""
( mapconcat
'identity
( seq-map 'char-to-string ( convert-to-symbol num ))
"\"\"" ))
