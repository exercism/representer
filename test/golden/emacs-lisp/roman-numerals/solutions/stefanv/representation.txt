(file "roman-numerals.el")
( setq numerals
' (( 0 . "\"\"" )
( 1 . "\"I\"" )
( 2 . "\"II\"" )
( 3 . "\"III\"" )
( 4 . "\"IV\"" )
( 5 . "\"V\"" )
( 6 . "\"VI\"" )
( 7 . "\"VII\"" )
( 8 . "\"VIII\"" )
( 9 . "\"IX\"" )
( 10 . "\"X\"" )
( 20 . "\"XX\"" )
( 30 . "\"XXL\"" )
( 40 . "\"XL\"" )
( 50 . "\"L\"" )
( 60 . "\"LX\"" )
( 70 . "\"LXX\"" )
( 80 . "\"LXXX\"" )
( 90 . "\"XC\"" )
( 100 . "\"C\"" )
( 200 . "\"CC\"" )
( 300 . "\"CCC\"" )
( 400 . "\"CD\"" )
( 500 . "\"D\"" )
( 600 . "\"DC\"" )
( 700 . "\"DCC\"" )
( 800 . "\"DCCC\"" )
( 900 . "\"CM\"" )
( 1000 . "\"M\"" )
( 2000 . "\"MM\"" )
( 3000 . "\"MMM\"" )
( 4000 . "\"MMMM\"" )))
( defun str-to-roman ( s )
( let* (( N ( length s ))
( p ( expt 10 ( - N 1 ))))
( cond (( not ( eq 0 N ))
( concat
( alist-get
( * ( string-to-number ( substring s 0 1 )) p )
numerals )
( str-to-roman ( substring s 1 )))))))
( defun to-roman ( x )
"\"Convert X to roman numeral\""
( str-to-roman ( number-to-string x )))
( provide 'roman-numerals )
