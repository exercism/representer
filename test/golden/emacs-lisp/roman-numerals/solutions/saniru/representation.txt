(file "roman-numerals.el")
( defconst roman-numerals-thousands [ "\"\"" "\"M\"" "\"MM\"" "\"MMM\"" ] )
( defconst roman-numerals-hundreds [ "\"\"" "\"C\"" "\"CC\"" "\"CCC\"" "\"CD\"" "\"D\"" "\"DC\"" "\"DCC\"" "\"DCCC\"" "\"CM\"" ] )
( defconst roman-numerals-tens [ "\"\"" "\"X\"" "\"XX\"" "\"XXX\"" "\"XL\"" "\"L\"" "\"LX\"" "\"LXX\"" "\"LXXX\"" "\"XC\"" ] )
( defconst roman-numerals-units [ "\"\"" "\"I\"" "\"II\"" "\"III\"" "\"IV\"" "\"V\"" "\"VI\"" "\"VII\"" "\"VIII\"" "\"IX\"" ] )
( defconst roman-numerals-romans [roman-numerals-units roman-numerals-tens roman-numerals-hundreds roman-numerals-thousands] )
( defun char-to-roman ( place char )
"\"Return Roman representation of CHAR in decimal PLACE.\""
( condition-case nil
( aref ( eval ( aref roman-numerals-romans place )) char )
( args-out-of-range "\"\"" )))
( defun get-digit ( number place )
"\"Get PLACEth digit from NUMBER.\""
( ignore-errors ( - ( aref ( number-to-string number ) place ) 48 )))
( defun to-roman ( number )
"\"Convert decimal representation of NUMBER into roman.\""
( let (( result ' ())
( len ( length ( number-to-string number ))))
( dotimes ( i len )
( push ( ignore-errors ( char-to-roman ( - len ( 1+ i )) ( get-digit number i ))) result ))
( apply 'concat ( nreverse result ))))
( provide 'roman-numerals )
