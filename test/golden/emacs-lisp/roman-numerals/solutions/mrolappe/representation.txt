( defun cvt-thousands ( number )
"\"Convert thousands digit of number to roman numeral representation.\n   Conversion will only be done for a number <= 3999.\""
( let (( thousands ( / number 1000 )))
( if ( and ( >= thousands 1 ) ( <= thousands 3 ))
( make-string ( / number 1000 ) ?M )
"\"\"" )))
( defun cvt-hundreds ( number )
"\"Convert hundreds digit of number to roman numeral representation.\""
( let (( hundreds ( / ( mod number 1000 ) 100 )))
( cond
(( and ( >= hundreds 1 ) ( <= hundreds 3 ))
( make-string hundreds ?C ))
(( = hundreds 4 ) "\"CD\"" )
(( = hundreds 5 ) "\"D\"" )
(( and ( >= hundreds 6 ) ( <= hundreds 8 ))
( concat "\"D\"" ( make-string ( - hundreds 5 ) ?C )))
(( = hundreds 9 ) "\"CM\"" )
( t "\"\"" ))))
( defun cvt-tens ( number )
"\"Convert tens digit to roman numeral representation.\""
( let (( tens ( / ( mod number 100 ) 10 )))
( cond
(( and ( >= tens 1 ) ( <= tens 3 ))
( make-string tens ?X ))
(( = tens 4 ) "\"XL\"" )
(( = tens 5 ) "\"L\"" )
(( and ( >= tens 6 ) ( <= tens 8 ))
( concat "\"L\"" ( make-string ( - tens 5 ) ?X )))
(( = tens 9 ) "\"XC\"" )
( t "\"\"" ))))
( defun cvt-ones ( number )
"\"Convert ones digit to roman numeral representation\""
( let (( ones ( mod number 10 )))
( cond
(( and ( >= ones 1 ) ( <= ones 3 ))
( make-string ones ?I ))
(( = ones 4 ) "\"IV\"" )
(( = ones 5 ) "\"V\"" )
(( and ( >= ones 6 ) ( <= ones 8 ))
( concat "\"V\"" ( make-string ( - ones 5 ) ?I )))
(( = ones 9 ) "\"IX\"" )
( t "\"\"" ))))
( defun to-roman ( number )
"\"Convert number to roman numeral representation\""
( concat ( cvt-thousands number ) ( cvt-hundreds number ) ( cvt-tens number ) ( cvt-ones number )))
( provide 'roman-numerals )
