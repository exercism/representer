(file "roman-numerals.el")
( require 'cl-lib )
( defun to-roman ( number )
"\"Convert NUMBER to roman number.\""
( cl-labels (( split-numbers ( number )
( if ( = number 0 )
nil
( cons ( % number 10 ) ( split-numbers ( / number 10 )))))
( convert-to-symbolic ( number )
( let (( roman-list ' (( I ) ( I I ) ( I I I ) ( I V ) ( V ) ( V I ) ( V I I ) ( V I I I ) ( I X ))))
( if ( > number 0 ) ( nth ( 1- number ) roman-list ))))
( stepped-sequence ( func sequence )
( if ( null sequence )
nil
( cons sequence ( stepped-sequence func ( funcall func sequence )))))
( substitute ( symbolic subst )
( cl-sublis ( cl-mapcar 'cons ' ( X V I ) subst ) symbolic )))
( let* (( romans-list ' ( M D C L X V I ))
( numbers ( reverse ( split-numbers number )))
( intermediate ( mapcar #' convert-to-symbolic
numbers ))
( converted ( cl-mapcar #' substitute intermediate
( nthcdr ( - 4 ( ceiling ( log number 10 )))
( cons ' ( nil nil M ) ( stepped-sequence 'cddr romans-list )))))
( cleaned ( cl-remove-if 'null converted ))
( combined ( apply 'append cleaned ))
( as-strings ( mapconcat 'symbol-name combined "\"\"" )))
as-strings )))
( provide 'roman-numerals )
