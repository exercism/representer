( defun to-roman ( value )
( let (( result "\"\"" ))
( setq n ( / value 1000 ))
( setq value ( - value ( * n 1000 )))
( setq result ( make-string n ?M ))
( when ( >= value 900 )
( setq result ( concat result "\"CM\"" ))
( setq value ( - value 900 ))
)
( when ( >= value 500 )
( setq result ( concat result "\"D\"" ))
( setq value ( - value 500 ))
)
( when ( >= value 400 )
( setq result ( concat result "\"CD\"" ))
( setq value ( - value 400 ))
)
( setq n ( / value 100 ))
( setq result ( concat result ( make-string n ?C )))
( setq value ( - value ( * n 100 )))
( when ( >= value 90 )
( setq result ( concat result "\"XC\"" ))
( setq value ( - value 90 ))
)
( when ( >= value 50 )
( setq result ( concat result "\"L\"" ))
( setq value ( - value 50 ))
)
( when ( >= value 40 )
( setq result ( concat result "\"XL\"" ))
( setq value ( - value 40 ))
)
( setq n ( / value 10 ))
( setq result ( concat result ( make-string n ?X )))
( setq value ( - value ( * n 10 )))
( when ( = value 9 )
( setq result ( concat result "\"IX\"" ))
( setq value ( - value 9 ))
)
( when ( >= value 5 )
( setq result ( concat result "\"V\"" ))
( setq value ( - value 5 ))
)
( when ( >= value 4 )
( setq result ( concat result "\"IV\"" ))
( setq value ( - value 4 ))
)
( setq result ( concat result ( make-string value ?I )))
result
))
