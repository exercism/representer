( defun to-roman ( n )
( let (( symbols
' (( "\"M\"" 1000 )
( "\"CM\"" 900 )
( "\"D\"" 500 )
( "\"CD\"" 400 )
( "\"C\"" 100 )
( "\"XC\"" 90 )
( "\"L\"" 50 )
( "\"XL\"" 40 )
( "\"X\"" 10 )
( "\"IX\"" 9 )
( "\"V\"" 5 )
( "\"IV\"" 4 )
( "\"I\"" 1 )))
result )
( while ( > n 0 )
( let (( symbol ( caar symbols ))
( i ( cadar symbols )))
( if ( > i n )
( setq symbols ( cdr symbols ))
( setq result ( concat result symbol ))
( setq n ( - n i )))))
result ))
( provide 'roman-numerals )
