( defvar roman-values
' (( 1000 . "\"M\"" )
( 900 . "\"CM\"" )
( 500 . "\"D\"" )
( 400 . "\"CD\"" )
( 100 . "\"C\"" )
( 90 . "\"XC\"" )
( 50 . "\"L\"" )
( 40 . "\"XL\"" )
( 10 . "\"X\"" )
( 9 . "\"IX\"" )
( 5 . "\"V\"" )
( 4 . "\"IV\"" )
( 1 . "\"I\"" )))
( defun to-roman ( n )
( if ( zerop n )
"\"\""
( let (( highest-match ( first-match n )))
( concat ( cdr highest-match )
( to-roman ( - n ( car highest-match )))))))
( defun first-match ( n )
( car ( seq-drop-while ( lambda ( x ) ( > ( car x ) n )) roman-values )))
( provide 'roman-numerals )
