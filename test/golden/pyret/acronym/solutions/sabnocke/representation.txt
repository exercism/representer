use context starter2024
provide: abbreviate end
fun helper(item :: String):
if string-equal(string-char-at(item, 0), "\"_\"):"
string-char-at(item, 1)
else:
string-char-at(item, 0)
end
end
fun abbreviate(phrase):
a = string-split-all(string-replace(phrase, "\"-\"," "\"" "\")," "\"" "\").filter(lam(item):" not(string-equal(item, "\"\"))" end)
b = a.map(lam(n): string-to-upper(helper(n)) end)
join-str(b, "\"\")"
end
