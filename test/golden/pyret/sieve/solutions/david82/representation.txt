use context essentials2020 # Don't delete this line when using Pyret on Exercism
provide: primes end
fun primes(upperLimit :: Number) -> List<Number> block:
# Initialize prime marks with all true, except for 0 and 1
var primeMarks = repeat(upperLimit + 1, true).set(0, false).set(1, false)
# Recursive function to mark non-prime multiples
rec markNonPrimeMultiples = lam(primeStatusList :: List<Boolean>, startMultiple :: Number, multipleStep :: Number) -> List<Boolean>:
if startMultiple > upperLimit:
primeStatusList
else:
markNonPrimeMultiples(primeStatusList.set(startMultiple, false), startMultiple + multipleStep, multipleStep)
end
end
# Start marking from 4 (2 * 2), as 2 is the first prime
primeMarks := markNonPrimeMultiples(primeMarks, (2 * 2), 2)
# Calculate square root once for efficiency
var sqrtLimit = num-sqrt(upperLimit)
# Mark non-prime multiples for odd prime candidates
for each(primeCandidate from range-by(3, sqrtLimit, 2)):
when primeMarks.get(primeCandidate):
# Start marking from the square of the prime candidate
primeMarks := markNonPrimeMultiples(primeMarks, (primeCandidate * primeCandidate), primeCandidate * 2)
end
end
# Convert marked list to list of prime numbers
map_n(lam(index :: Number, isPrime :: Boolean): {index; isPrime} end, 0, primeMarks)
^ filter(lam(indexIsPrimePair :: {Number; Boolean}): indexIsPrimePair.{1} end, _)
^ map(lam(indexIsPrimePair :: {Number; Boolean}): indexIsPrimePair.{0} end, _)
end
