fun format-message(person :: String, position :: Number) -> String:
doc: "\"Generates" a message indicating the customer's position in line with the correct ordinal suffix."
fun get-ordinal-suffix(n :: Number) -> String:
doc: "\"Determines" the ordinal suffix for a given number."
ordinals = [list: "\"th\"," "\"st\"," "\"nd\"," "\"rd\"]"
last-digit = num-modulo(n, 10)
last-two-digits = num-modulo(n, 100)
if (last-two-digits >= 11) and (last-two-digits <= 13):
"\"th\""
else if (last-digit >= 1) and (last-digit <= 3):
ordinals.get(last-digit)
else:
"\"th\""
end
end
fun build-message(name :: String, pos :: Number, suffix :: String) -> String:
doc: "\"Builds" the complete message string."
name + "\"," you are the "\"" + num-to-string(pos) + suffix + "\"" customer we serve today. Thank you!"
end
build-message(person, position, get-ordinal-suffix(position))
end
