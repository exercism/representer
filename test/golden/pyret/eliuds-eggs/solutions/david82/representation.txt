use context essentials2020 # Don't delete this line when using Pyret on Exercism
provide: egg-count end
# Helper function to convert a decimal number to binary
fun to-binary(n :: Number) -> List<Number>:
if n == 0:
[list: ]
else:
link(num-modulo(n, 2), to-binary(num-floor(n / 2)))
end
end
# Main function to count the number of 1's in the binary representation of a number
fun egg-count(n :: Number) -> Number:
to-binary(n).filter(lam(x): x == 1 end).length()
end
