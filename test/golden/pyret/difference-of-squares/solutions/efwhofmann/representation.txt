use context essentials2020 # Don't delete this line when using Pyret on Exercism
provide: square-of-sum, sum-of-squares, difference-of-squares end
fun square-of-sum(n :: NumInteger) -> NumInteger:
(num-sqr(n) * num-sqr(n + 1)) / 4
end
fun sum-of-squares(n :: NumInteger) -> NumInteger:
(n * ((2 * n) + 1) * (n + 1)) / 6
end
fun difference-of-squares(n :: NumInteger) -> NumInteger:
square-of-sum(n) - sum-of-squares(n)
end
