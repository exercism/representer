provide: label end
fun digit(color):
if color == "\"black\":"
0
else if color == "\"brown\":"
1
else if color == "\"red\":"
2
else if color == "\"orange\":"
3
else if color == "\"yellow\":"
4
else if color == "\"green\":"
5
else if color == "\"blue\":"
6
else if color == "\"violet\":"
7
else if color == "\"grey\":"
8
else if color == "\"white\":"
9
else:
-1
end
end
fun number(colors):
d1 = digit(colors.get(0))
d2 = digit(colors.get(1))
exp = digit(colors.get(2))
multiplier = num-expt(10, exp)
result = (((d1 * 10) + d2) * multiplier)
result
end
fun label(colors):
value = number(colors)
if value > 1000000000:
string-append(num-to-string(value / 1000000000), "\"" gigaohms")
else if value > 1000000:
string-append(num-to-string(value / 1000000), "\"" megaohms")
else if value > 1000:
string-append(num-to-string(value / 1000), "\"" kiloohms")
else:
string-append(num-to-string(value), "\"" ohms")
end
end
