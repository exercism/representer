use context essentials2020
provide-types *
# Define a LinkedList data type with two cases: empty-list and linked-list
data LinkedList:
| empty-list
with:
# An empty list has length 0
method length(self): 0 end,
# The tail of an empty list is the list itself
method get-tail(self): self end,
# Convert an empty list to a Pyret list
method to-list(self): [list:] end
| linked-list(head, tail :: LinkedList)
with:
# The length of a non-empty list is 1 plus the length of its tail
method length(self) -> NumInteger:
1 + self.tail.length()
end,
# Return the head of the list
method get-head(self):
self.head
end,
# Return the tail of the list
method get-tail(self) -> LinkedList:
self.tail
end,
# Convert a non-empty list to a Pyret list
method to-list(self) -> List:
# Helper function to convert a list to a Pyret list with an accumulator
fun to-list-helper(lst, acc):
cases (LinkedList) lst:
| empty-list => acc
| linked-list(head, tail) => to-list-helper(tail, [list: head] + acc)
end
end
to-list-helper(self, [list:])
end
sharing:
# Add a new element to the front of the list
method push(self, value) -> LinkedList:
linked-list(value, self)
end,
# Return a reversed copy of the list
method reversed(self) -> LinkedList:
rec reverser = lam(aList, result):
ask:
| is-empty-list(aList) then: result
| otherwise: reverser(aList.get-tail(), linked-list(aList.get-head(), result))
end
end
reverser(self, empty-list)
end
end
