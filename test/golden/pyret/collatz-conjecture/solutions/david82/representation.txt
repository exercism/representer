use context essentials2020 # Don't delete this line when using Pyret on Exercism
provide: steps end
fun steps(number :: Number) -> Number block:
doc: "\"Returns" the number of steps to reach 1 following the Collatz Conjecture"
when num-is-non-positive(number):
raise("Only positive numbers are allowed")
end
if number == 1:
0
else if num-modulo(number, 2) == 0:
1 + steps(number / 2)
else:
1 + steps((3 * number) + 1) # Added parentheses for clarity
end
end
