(file "collatz-conjecture.arr")
provide: steps end
fun steps(n :: NumInteger) -> NumInteger:
fun aux(curr, cnt):
if curr == 1:
cnt
else if num-modulo(curr, 2) == 0:
aux(curr / 2, cnt + 1)
else:
aux((3 * curr) + 1, cnt + 1)
end
end
if n > 0:
aux(n, 0)
else:
raise("Only positive numbers are allowed")
end
end
