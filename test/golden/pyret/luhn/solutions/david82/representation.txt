use context essentials2020 # Don't delete this line when using Pyret on Exercism
provide: is-valid end
import lists as L
fun normalize-digit(digit :: Number, threshold :: Number) -> Number:
if digit > threshold: digit - threshold else: digit end
end
fun convert-char-to-digit(char :: String) -> Option<Number>:
string-to-number(char)
end
fun calculate-digit(index :: Number, char :: String, doubling-step :: Number) -> Number:
cases (Option) convert-char-to-digit(char):
| none => 0
| some(digit) =>
ask:
| num-modulo(index, 2) == doubling-step then: normalize-digit(digit * 2, 9)
| otherwise: digit
end
end
end
fun calculate-checksum(cleaned-card-number :: String, doubling-step :: Number) -> Number:
L.fold_n(
lam(index :: Number, accumulated-sum :: Number, char :: String):
accumulated-sum + calculate-digit(index, char, doubling-step)
end,
0,
0,
string-explode(cleaned-card-number))
end
fun clean-card-number(card-number :: String) -> String:
string-replace(card-number, "\"" "\"," "\"\")"
end
fun all-digits(code-points :: List<Number>) -> Boolean:
L.all(lam(cp): (cp >= 48) and (cp <= 57) end, code-points)
end
fun is-valid(card-number :: String) -> Boolean:
cleaned-card-number = clean-card-number(card-number)
card-length = string-length(cleaned-card-number)
code-points = string-to-code-points(cleaned-card-number)
ask:
| (card-length > 1) and all-digits(code-points) then:
doubling-step = num-modulo(card-length, 2)
checksum = calculate-checksum(cleaned-card-number, doubling-step)
num-modulo(checksum, 10) == 0
| otherwise:
false
end
end
