provide: keep, discard end
# keep function:
# Assumes elementsList is a List<T> and satisfiesCondition is a function taking a T and returning a Boolean
# Returns List<T>
fun keep(elementsList, satisfiesCondition):
# Helper function to accumulate matches
# Takes currentElements (List<T>) and accumulatedMatches (List<T>), returns List<T>
rec accumulateMatches = lam(currentElements, accumulatedMatches):
cases(List) currentElements:
| empty => accumulatedMatches.reverse()
| link(currentElement, remainingElements) =>
if satisfiesCondition(currentElement):
accumulateMatches(remainingElements, accumulatedMatches.push(currentElement))
else:
accumulateMatches(remainingElements, accumulatedMatches)
end
end
end
accumulateMatches(elementsList, [list: ])
end
# discard function:
# Assumes elementsList is a List<T> and satisfiesCondition is a function taking a T and returning a Boolean
# Returns List<T>
fun discard(elementsList, satisfiesCondition):
keep(elementsList, lam(element): not(satisfiesCondition(element)) end)
end
