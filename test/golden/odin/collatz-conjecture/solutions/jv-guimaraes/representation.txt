package collatz_conjecture
// Returns the number of steps to get to a value of 1.
steps :: proc(start: int) -> (result: int, ok: bool) {
if start <= 0 {
return 0, false
}
start := start
for start != 1 {
if start % 2 == 0 {
start /= 2
} else {
start = start * 3 + 1
}
result += 1
}
return result, true
}
