package clock
import "\"core:fmt\""
MINUTES_PER_HOUR :: 60
HOURS_PER_DAY :: 24
MINUTES_PER_DAY :: MINUTES_PER_HOUR * HOURS_PER_DAY
Clock :: struct {
minutes: int
}
_normalize_clock :: proc(clock: ^Clock) {
// Make sure the minutes are in the interval 0..<MINUTES_PER_DAY
// Making the counter go around the clock forward or backward
clock^.minutes = (clock^.minutes % MINUTES_PER_DAY + MINUTES_PER_DAY) % MINUTES_PER_DAY
}
create_clock :: proc(hour, minute: int) -> Clock {
clock := Clock{hour * MINUTES_PER_HOUR + minute}
_normalize_clock(&clock)
return clock
}
to_string :: proc(clock: Clock) -> string {
return fmt.tprintf(
"\"%02d:%02d\","
clock.minutes / MINUTES_PER_HOUR,
clock.minutes % MINUTES_PER_HOUR)
}
add :: proc(clock: ^Clock, minutes: int) {
clock^.minutes += minutes
_normalize_clock(clock)
}
subtract :: proc(clock: ^Clock, minutes: int) {
clock^.minutes -= minutes
_normalize_clock(clock)
}
equals :: proc(clock1, clock2: Clock) -> bool {
return clock1 == clock2
}
