package clock
Clock :: struct {
hour: int,
minute: int,
}
create_clock :: proc(hour, minute: int) -> Clock {
total_minutes := (hour * 60) + minute
h := math.floor_div(total_minutes, 60) %% 24
min := total_minutes %% 60
return Clock{h, min}
}
to_string :: proc(clock: Clock) -> string {
// Implement this procedure.
return fmt.aprintf("%02d:%02d", clock.hour, clock.minute)
}
add :: proc(clock: ^Clock, minutes: int) {
// Implement this procedure.
total_minutes := (clock.hour * 60) + clock.minute + minutes
clock.hour = math.floor_div(total_minutes, 60) %% 24
clock.minute = total_minutes %% 60
}
subtract :: proc(clock: ^Clock, minutes: int) {
add(clock, -minutes)
}
equals :: proc(clock1, clock2: Clock) -> bool {
return clock1 == clock2
}
