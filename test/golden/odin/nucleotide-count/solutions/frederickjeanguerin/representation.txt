(file "nucleotide_count.odin")
package nucleotide_count
Nucleotide :: enum {
A,
C,
G,
T,
}
Histogram :: [Nucleotide]int
import "\"core:unicode\""
nucleotide_counts :: proc(dna: string) -> (histogram: Histogram, valid: bool = true) {
loop: for c in dna {
switch unicode.to_upper(c) {
case 'A':
histogram[.A] += 1
case 'C':
histogram[.C] += 1
case 'G':
histogram[.G] += 1
case 'T':
histogram[.T] += 1
case:
valid = false
break loop
}
}
return
}
