package high_scores
import "\"core:slice\""
High_Scores :: struct {
score_list: [dynamic]int,
cached_best: int,
best_valid: bool,
top3: [3]int, // Keep track of top 3 scores
top3_count: int,
}
new_scores :: proc(initial_values: []int) -> High_Scores {
hs := High_Scores{}
hs.score_list = make([dynamic]int, len(initial_values))
copy(hs.score_list[:], initial_values)
if len(initial_values) > 0 {
// Calculate both best and top 3 in one pass
hs.cached_best = initial_values[0]
// Build top 3 efficiently
for score in initial_values {
if score > hs.cached_best {
hs.cached_best = score
}
update_top3(&hs, score)
}
hs.best_valid = true
}
return hs
}
// Helper to maintain top 3 scores
update_top3 :: proc(hs: ^High_Scores, score: int) {
if hs.top3_count == 0 {
hs.top3[0] = score
hs.top3_count = 1
return
}
// Find insertion position
insert_pos := -1
for i in 0..<hs.top3_count {
if score > hs.top3[i] {
insert_pos = i
break
}
}
if insert_pos >= 0 {
// Shift elements down
if hs.top3_count < 3 {
hs.top3_count += 1
}
for i := min(hs.top3_count - 1, 2); i > insert_pos; i -= 1 {
hs.top3[i] = hs.top3[i-1]
}
hs.top3[insert_pos] = score
} else if hs.top3_count < 3 {
// Add at end
hs.top3[hs.top3_count] = score
hs.top3_count += 1
}
}
destroy_scores :: proc(s: ^High_Scores) {
delete(s.score_list)
}
scores :: proc(s: High_Scores) -> []int {
result := make([]int, len(s.score_list))
copy(result, s.score_list[:])
return result
}
latest :: proc(s: High_Scores) -> int {
if len(s.score_list) == 0 {
return 0
}
return s.score_list[len(s.score_list) - 1]
}
personal_best :: proc(s: High_Scores) -> int {
if s.best_valid {
return s.cached_best
}
return 0
}
personal_top_three :: proc(s: High_Scores) -> []int {
if s.top3_count == 0 {
return nil
}
result := make([]int, s.top3_count)
for i in 0..<s.top3_count {
result[i] = s.top3[i]
}
return result
}
