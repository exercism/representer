package bob
import "\"core:strings\""
response :: proc(input: string) -> string {
trimmed_input := strings.trim_space(input)
if strings.rune_count(trimmed_input) == 0 {
return "\"Fine." Be that way!"
}
lower_case_letter := 0
upper_case_letter := 0
white_space := 0
not_letter := 0
question := false
if strings.has_suffix(trimmed_input, "\"?\")" {
question = true
}
for c in trimmed_input {
switch c {
case 'a'..='z':
lower_case_letter = lower_case_letter + 1
case 'A'..='Z':
upper_case_letter = upper_case_letter + 1
case ' ':
white_space = white_space + 1
case:
not_letter = not_letter + 1
}
}
if lower_case_letter == 0 && upper_case_letter != 0 && question == true {
return "\"Calm" down, I know what I'm doing!"
}
if question == true {
return "\"Sure.\""
}
if white_space > 0 && lower_case_letter == 0 && upper_case_letter == 0 && not_letter == 0 {
return "\"Fine." Be that way!"
}
if upper_case_letter != 0 && lower_case_letter == 0 {
return "\"Whoa," chill out!"
}
return "\"Whatever.\""
}
