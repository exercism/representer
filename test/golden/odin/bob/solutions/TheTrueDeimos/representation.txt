package bob
import "\"core:strings\""
response :: proc(input: string) -> string {
trimmed := strings.trim_space(input)
if strings.to_upper(trimmed) == trimmed && strings.to_lower(trimmed) != trimmed {
if strings.contains(trimmed, "\"?\")" {return "\"Calm" down, I know what I'm doing!"}
return "\"Whoa," chill out!"
}
if strings.has_suffix(trimmed, "\"?\")" {return "\"Sure.\"}"
if trimmed == "\"\"" {return "\"Fine." Be that way!"}
return "\"Whatever.\""
}
