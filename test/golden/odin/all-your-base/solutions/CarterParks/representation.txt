package all_your_base
Error :: enum { None, Invalid_Input_Digit, Input_Base_Too_Small, Output_Base_Too_Small, Unimplemented, }
rebase :: proc(input_base: int, digits: []int, output_base: int) -> ([]int, Error) {
non_zero, out_range: bool
for d in digits {
if d < 0 || d >= input_base do out_range = true
if d > 0 do non_zero = true
}
zero :: proc() -> (z: []int) {
z = make([]int, 1)
z[0] = 0
return
}
switch {
case input_base < 2 : return nil, .Input_Base_Too_Small
case output_base < 2 : return nil, .Output_Base_Too_Small
case out_range : return nil, .Invalid_Input_Digit
case len(digits) == 0 : return zero(), .None
case !non_zero : return zero(), .None
} // end error checking
ipow :: proc(a, b: int) -> (c := 1) { // integer exponentiation
for _ in 0..<b do c *= a
return
}
acc, l: int // decimal accumulator, decimal count
for d, i in digits do acc += d * ipow(input_base, len(digits) - 1 - i) // convert digits to decimal
for a := acc; a > 0; a /= output_base do l += 1 // compute output decimal count
out := make([]int, l) // output decimal array
for _, i in out { // compute output decimals
out[l - 1 - i] = acc %% output_base // assign remainder to decimal place
acc /= output_base // on to next decimal place
}
return out, .None
}
