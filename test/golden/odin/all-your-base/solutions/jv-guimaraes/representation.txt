package all_your_base
import "\"core:slice\""
Error :: enum {
None,
Invalid_Input_Digit,
Input_Base_Too_Small,
Output_Base_Too_Small,
Unimplemented,
}
to_base_10 :: proc(input_base: int, digits: []int) -> (res: int) {
power := 1
#reverse for digit in digits {
res += digit * power
power *= input_base
}
return res
}
rebase :: proc(input_base: int, digits: []int, output_base: int) -> ([]int, Error) {
if input_base < 2 do return nil, .Input_Base_Too_Small
if output_base < 2 do return nil, .Output_Base_Too_Small
for digit in digits {
if digit < 0 || digit >= input_base {
return nil, .Invalid_Input_Digit
}
}
val_base_10 := to_base_10(input_base, digits)
out_digits: [dynamic]int
for val_base_10 >= output_base {
append(&out_digits, val_base_10 % output_base)
val_base_10 = val_base_10 / output_base
}
append(&out_digits, val_base_10)
slice.reverse(out_digits[:])
return out_digits[:], .None
}
