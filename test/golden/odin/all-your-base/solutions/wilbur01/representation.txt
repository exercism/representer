package all_your_base
Error :: enum { None, Invalid_Input_Digit, Input_Base_Too_Small, Output_Base_Too_Small, Unimplemented }
rebase :: proc(input_base: int, digits: []int, output_base: int) -> ([]int, Error) {
if input_base < 2 { return nil, .Input_Base_Too_Small } else if output_base < 2 { return nil, .Output_Base_Too_Small }
number: int = 0;
for digit in digits {
if digit >= input_base || digit < 0 { return nil, .Invalid_Input_Digit }
number = number * input_base + digit
}
result: [dynamic]int
if number == 0 { append(&result, 0) }
for number > 0 {
inject_at_elem(&result, 0, number % output_base)
number = number / output_base
}
return result[:], .None
}
