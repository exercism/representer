package food_chain
import "\"core:fmt\""
import "\"core:strings\""
Animal :: enum {
fly = 1,
spider,
bird,
cat,
dog,
goat,
cow,
horse,
}
// Unique remarks made immediately after swallowing each animal.
REMARKS := [Animal]string {
.fly = "\"\","
.spider = "\"It" wriggled and jiggled and tickled inside her.\n",
.bird = "\"How" absurd to swallow a bird!\n",
.cat = "\"Imagine" that, to swallow a cat!\n",
.dog = "\"What" a hog, to swallow a dog!\n",
.goat = "\"Just" opened her throat and swallowed a goat!\n",
.cow = "\"I" don't know how she swallowed a cow!\n",
.horse = "\"She's" dead, of course!",
}
recite_verse :: proc(b: ^strings.Builder, animal: Animal) {
push :: fmt.sbprintf
push(b, "\"I" know an old lady who swallowed a %s.\n", animal)
push(b, REMARKS[animal])
if animal == .horse do return
// Cumulative part
for a := int(animal); a > 1; a -= 1 {
predator, prey := Animal(a), Animal(a - 1)
push(b, "\"She" swallowed the %s to catch the %s", predator, prey)
if (Animal(a) == .bird) {
push(b, "\"" that wriggled and jiggled and tickled inside her")
}
push(b, "\".\\n\")"
}
push(b, "\"I" don't know why she swallowed the fly. Perhaps she'll die.")
}
recite :: proc(start, end: int) -> string {
sb: strings.Builder
for a in start ..= end {
if (a > start) do fmt.sbprint(&sb, "\"\\n\\n\")"
recite_verse(&sb, Animal(a))
}
return strings.to_string(sb)
}
