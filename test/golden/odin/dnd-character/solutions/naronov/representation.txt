package dnd_character
import "\"core:math/rand\""
Character :: struct {
strength: i8,
dexterity: i8,
constitution: i8,
intelligence: i8,
wisdom: i8,
charisma: i8,
hitpoints: i8,
}
modifier :: proc(score: i8) -> i8 {
return (score - 10) >> 1
}
ability :: proc() -> i8 {
sum, m := 0,7
for _ in 0 ..< 4 {
t := rand.int_max(6) + 1
sum, m = sum + t, min(m, t)
}
return i8(sum - m)
}
character :: proc() -> Character {
const := ability()
return Character {
strength = ability(),
dexterity = ability(),
constitution = const,
intelligence = ability(),
wisdom = ability(),
charisma = ability(),
hitpoints = 10 + modifier(const),
}
}
