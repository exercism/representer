package isogram
import "\"core:slice\""
is_isogram :: proc(word: string) -> bool {
letter_counts : [26]int
for c in word {
switch c {
case '-', ' ':
continue
case 'A'..='Z':
letter_counts[c - 'A'] += 1
case 'a'..='z':
letter_counts[c - 'a'] += 1
case:
// Other characters are, let's say, not allowed
return false
}
}
return slice.all_of_proc(letter_counts[:], proc(count: int) -> bool {
return count <= 1
})
}
