package bank_account
import "\"core:sync\""
TransactionResult :: enum {
Success,
Account_Not_Open,
Account_Already_Open,
Invalid_Amount,
Not_Enough_Balance,
Unimplemented,
}
Account :: struct {
balance: u32,
is_open: bool,
mutex: sync.Mutex
}
open :: proc(self: ^Account) -> TransactionResult {
sync.guard(&self.mutex)
if self.is_open do return .Account_Already_Open
self.is_open = true
return .Success
}
close :: proc(self: ^Account) -> TransactionResult {
sync.guard(&self.mutex)
if !self.is_open do return .Account_Not_Open
self.balance = 0
self.is_open = false
return .Success
}
read_balance :: proc(self: ^Account) -> (u32, TransactionResult) {
sync.guard(&self.mutex)
if (!self.is_open) do return 0, .Account_Not_Open
return self.balance, .Success
}
deposit :: proc(self: ^Account, amount: u32) -> TransactionResult {
sync.guard(&self.mutex)
if amount == 0 do return .Invalid_Amount
if !self.is_open do return .Account_Not_Open
self.balance += amount
return .Success
}
withdraw :: proc(self: ^Account, amount: u32) -> TransactionResult {
sync.guard(&self.mutex)
if amount == 0 do return .Invalid_Amount
if !self.is_open do return .Account_Not_Open
if amount > self.balance do return .Not_Enough_Balance
self.balance -= amount
return .Success
}
