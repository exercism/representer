package matching_brackets
is_balanced :: proc(input: string) -> bool {
stack := [dynamic]rune{}
for val in input {
switch val {
case '{', '[', '(':
append(&stack, val)
case '}', ']', ')':
if len(stack) == 0 {
delete(stack)
return false
}
match: rune = '('
if val == '}' {
match = '{'
} else if val == ']' {
match = '['
}
if stack[len(stack) - 1] == match {
pop(&stack)
} else {
delete(stack)
return false
}
}
}
result := len(stack) == 0
delete(stack)
return result
}
