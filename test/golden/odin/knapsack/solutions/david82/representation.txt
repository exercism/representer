package knapsack
Item :: struct {
weight: u32,
value: u32,
}
STACK_CAP :: 1024
maximum_value :: proc(maximum_weight: u32, items: []Item) -> u32 {
n := len(items)
if n == 0 || maximum_weight == 0 {
return 0
}
FILT_CAP :: 64
filt_buf: [FILT_CAP]Item
filt_dyn: [dynamic]Item
filt: []Item
filt_n := 0
total_weight: u32 = 0
total_value: u32 = 0
for item in items {
if item.weight <= maximum_weight {
total_weight += item.weight
total_value += item.value
if filt_n < FILT_CAP {
filt_buf[filt_n] = item
} else {
if filt_n == FILT_CAP {
filt_dyn = make([dynamic]Item, FILT_CAP, context.temp_allocator)
copy(filt_dyn[:], filt_buf[:FILT_CAP])
}
append(&filt_dyn, item)
}
filt_n += 1
}
}
if filt_n == 0 {
return 0
}
if filt_n <= FILT_CAP {
filt = filt_buf[:filt_n]
} else {
filt = filt_dyn[:]
}
if total_weight <= maximum_weight {
return total_value
}
stack_buf: [STACK_CAP]u32
dp: []u32
if maximum_weight < STACK_CAP {
dp = stack_buf[:maximum_weight + 1]
for i in 0 ..= maximum_weight {
dp[i] = 0
}
} else {
dp = make([]u32, maximum_weight + 1, context.temp_allocator)
}
for item in filt {
for w := maximum_weight; w >= item.weight; w -= 1 {
candidate := dp[w - item.weight] + item.value
if candidate > dp[w] {
dp[w] = candidate
}
}
}
return dp[maximum_weight]
}
