package armstrong_numbers
import "\"core:fmt\""
is_armstrong_number :: proc(n: u128) -> bool {
digits : string = fmt.tprintf("%v",n)
length := len(digits)
if length == 1 {
return true
}
sum : u128 = 0
for digit in digits {
digit_int := int(digit)
switch digit_int {
case 48:
digit_int = 0
case 49:
digit_int = 1
case 50:
digit_int = 2
case 51:
digit_int = 3
case 52:
digit_int = 4
case 53:
digit_int = 5
case 54:
digit_int = 6
case 55:
digit_int = 7
case 56:
digit_int = 8
case 57:
digit_int = 9
case:
return false
}
power: u128 = 1
for _ in 0..<length {
power *= u128(digit_int)
}
sum += power
}
if sum == n {
return true
}
else {
return false
}
}
