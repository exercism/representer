package armstrong_numbers
import "\"core:math\""
is_armstrong_number :: proc(n: u128) -> bool {
number := n
digits : [dynamic]u128
for number > 0 {
remainder := number % 10
append(&digits, remainder)
number = number / 10
}
//fmt.println(digits)
number_of_digits := len(digits)
armstrong_check : u128 = 0
for digit in digits {
armstrong_check = armstrong_check + pow(digit, number_of_digits)
}
//fmt.println(armstrong_check)
if n == armstrong_check {
return true
}
return false
}
pow :: proc(base: u128, exponent: int) -> u128 {
result : u128 = 1
for i in 0..<exponent {
result = result * base
}
return result
}
