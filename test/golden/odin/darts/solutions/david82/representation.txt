package darts
score :: proc(x, y: f64) -> int {
// radius^2: inner=1, middle=25, outer=100
distance_squared := x * x + y * y
if distance_squared <= 1.0 {
return 10 // Inner circle (radius 1)
} else if distance_squared <= 25.0 {
return 5 // Middle circle (radius 5)
} else if distance_squared <= 100.0 {
return 1 // Outer circle (radius 10)
} else {
return 0 // Outside the target
}
}
