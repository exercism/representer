package space_age
Planet :: enum {
Mercury,
Venus,
Earth,
Mars,
Jupiter,
Saturn,
Uranus,
Neptune,
}
Relative_Periods := [Planet]f64 {
.Mercury = 0.2408467,
.Venus = 0.61519726,
.Earth = 1,
.Mars = 1.8808158,
.Jupiter = 11.862615,
.Saturn = 29.447498,
.Uranus = 84.016846,
.Neptune = 164.79132,
}
Earth_Year_In_Seconds :: 31_557_600
age :: proc(planet: Planet, seconds: int) -> f64 {
return f64(seconds) / (Earth_Year_In_Seconds * Relative_Periods[planet])
}
