(file "hamming.odin")
package hamming
Error :: enum {
None,
UnequalLengths,
Unimplemented,
}
distance :: proc(strand1, strand2: string) -> (count: int, err: Error) {
if len(strand1) != len(strand2) {
err = .UnequalLengths
return
}
for i in 0 ..< len(strand1) {
if strand1[i] != strand2[i] {
count += 1
}
}
return
}
