{ #category : #private }
SumOfMultiples >> makeFilter: anArray [
^ [ : n |
( anArray
collect:
[ : divisor | n isDivisibleBy: divisor ])
inject: false into:
[ : acc : each | acc or: each ]]
]
{ #category : #actions }
SumOfMultiples >> sumFactors: aCollection limit: anInteger [
| filter |
filter := self makeFilter: aCollection .
^ (( 1 to: anInteger - 1 )
select: filter )
inject: 0 into:
[ : acc : each | acc + each ]
]
