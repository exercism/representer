Class {
#name : #IsbnVerifier ,
#superclass : #Object ,
#category : # 'Exercise@IsbnVerifier'
}
{ #category : #private }
IsbnVerifier >> containsOnlyValidChars: aString [
^ ( aString asArray
collectWithIndex: [ : c : i | c isDigit or: [ c = $X and: [ i = aString size ] ] ])
allSatisfy: [ : x | x ]
]
{ #category : #private }
IsbnVerifier >> convertToInteger: aDigitOrX [
^ aDigitOrX = $X
ifTrue: [ 10 ]
ifFalse: [ aDigitOrX digitValue ]
]
{ #category : #testing }
IsbnVerifier >> isValidIsbn: aString [
| cleanString checksum |
cleanString := aString select: [ : c | c ~= $- ].
( cleanString size = 10
and: [ self containsOnlyValidChars: cleanString ])
ifFalse: [ ^ false ].
checksum := cleanString reverse asArray
withIndexCollect: [ : c : i | ( self convertToInteger: c ) * i ].
^ checksum sum % 11 = 0
]
