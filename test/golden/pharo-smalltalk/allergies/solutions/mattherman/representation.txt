Class {
#name : #Allergies ,
#superclass : #Object ,
#classVars : [
'allergens'
] ,
#category : # 'Exercise@Allergies'
}
{ #category : #solution }
Allergies >> allergicToItem: aString forScore: anInteger [
^ ( anInteger bitAt: ( allergens indexOf: aString )) = 1
]
{ #category : #solution }
Allergies >> allergyListForScore: anInteger [
^ allergens select: [ : allergen | self allergicToItem: allergen forScore: anInteger ]
]
{ #category : #initialization }
Allergies >> initialize [
super initialize .
allergens := #('eggs' 'peanuts' 'shellfish' 'strawberries' 'tomatoes' 'chocolate' 'pollen' 'cats')
]
