{ #category : #public }
Allergies >> allergicToItem: item forScore: score [
^ ( self allergyListForScore: score )
includes: item
]
{ #category : #public }
Allergies >> allergyListForScore: score [
^ items withIndexSelect: [ : e : i |
0 ~= ( 2 ** ( i - 1 ) & score ) ]
]
