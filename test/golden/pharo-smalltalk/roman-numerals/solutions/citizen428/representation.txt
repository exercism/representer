RomanNumerals >> romanNumber: anInteger [
| remainder |
remainder := anInteger .
^ String streamContents: [ : aStream |
self class numberMappings keysAndValuesDo: [ : n : roman |
[ remainder >= n ] whileTrue: [
aStream nextPutAll: roman .
remainder := remainder - n ] ] ]
]
