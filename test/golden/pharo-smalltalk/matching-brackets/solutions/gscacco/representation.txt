MatchingBrackets >> isPaired: aString [
| dict stk myBlock |
stk := Stack new .
myBlock := [ : c |
stk isEmpty
ifTrue: [ ^ false ].
stk top ~= c
ifFalse: [ stk pop ]
ifTrue: [ ^ false ] ].
dict := {( $( -> [ stk push: $( ]).
( $[ -> [ stk push: $[ ]).
( ${ -> [ stk push: ${ ]).
( $) -> [ myBlock value: $( ]).
( $] -> [ myBlock value: $[ ]).
( $} -> [ myBlock value: ${ ])} asDictionary .
aString asArray do: [ : c | ( dict at: c ifAbsent: [ ]) value ].
^ stk isEmpty
]
