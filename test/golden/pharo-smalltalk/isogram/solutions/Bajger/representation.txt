Isogram >> isIsogramPhrase: aString [
| aBag |
aBag := ( aString
reject: [ : char | char = $- or: [ char = Character space ] ]
thenCollect: [ : char | char asLowercase ]) asBag .
^ aBag
noneSatisfy: [ : character | ( aBag occurrencesOf: character ) > 1 ]
]
