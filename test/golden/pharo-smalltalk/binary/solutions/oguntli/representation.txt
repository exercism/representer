Binary >> decimalFromBinary: aString [
| number s |
s := aString asOrderedCollection reject: [ : e | e = $0 or: [ e = $1 ] ].
s size > 0 ifTrue: [ ^ nil ] ifFalse: [ ^ ( '2r' , aString ) asNumber ].
]
