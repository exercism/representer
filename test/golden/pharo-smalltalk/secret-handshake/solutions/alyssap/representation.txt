{ #category : # "'as yet unclassified'" }
SecretHandshake >> commandsNumber: anInteger [
| result |
result := WriteStream on: ( Array new: 4 ).
( anInteger allMask: 1 ) and: [ result nextPut: 'wink' ].
( anInteger allMask: 2 ) and: [ result nextPut: "'double blink'" ].
( anInteger allMask: 4 ) and: [ result nextPut: "'close your eyes'" ].
( anInteger allMask: 8 ) and: [ result nextPut: 'jump' ].
( anInteger allMask: 16 ) and: [ ^ result contents reverse ].
^ result contents
]
