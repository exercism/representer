SecretHandshake >> commandsNumber: anInteger [
| outcollection |
outcollection := OrderedCollection new .
1 to: 4 do: [ : index |
(( anInteger bitAt: index ) = 1 ) ifTrue: [
outcollection add: ( signalDictionary at: index ) ] ].
(( anInteger bitAt: 5 ) = 1 )
ifTrue: [ outcollection := outcollection reverse ].
^ outcollection asArray
]
