Acronym >> abbreviatePhrase: aString [
^ (( aString substrings: separators )
collect: [ : word | word first asUppercase ]
thenSelect: [ : word | word isAlphaNumeric ]) joinUsing: ''
]

Acronym >> initialize [
separators := "' -_'" .
