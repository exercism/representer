Class {
#name : #Acronym ,
#superclass : #Object ,
#instVars : [
'stream' ,
'state'
] ,
#category : # 'Exercise@Acronym'
}
{ #category : #computing }
Acronym >> abbreviatePhrase: aString [
self reset .
aString do: [ : c | self perform: state with: c ].
^ self result asUppercase
]
{ #category : #computing }
Acronym >> feed: aCharacter [
self perform: state with: aCharacter
]
{ #category : #initialization }
Acronym >> initialize [
self reset
]
{ #category : #initialization }
Acronym >> reset [
state := #space: .
stream := WriteStream on: String new
]
{ #category : #accessing }
Acronym >> result [
^ stream contents
]
{ #category : # "'changing state'" }
Acronym >> space: aCharacter [
aCharacter isLetter
and: [ stream nextPut: aCharacter .
state := #word: ]
]
{ #category : # "'changing state'" }
Acronym >> word: aCharacter [
aCharacter = $' | aCharacter isLetter or: [ state := #space: ]
]
