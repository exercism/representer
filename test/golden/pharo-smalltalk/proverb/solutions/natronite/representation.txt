Proverb >> reciteStrings: anArray [
^ anArray withIndexCollect: [ : w : i |
i < anArray size
ifTrue: [
"'For want of a {1} the {2} was lost.'" format:
( anArray copyFrom: i to: i + 1 ) ]
ifFalse: [
"'And all for the want of a {1}.'" format:
( anArray copyFrom: 1 to: 1 ) ] ]
]
