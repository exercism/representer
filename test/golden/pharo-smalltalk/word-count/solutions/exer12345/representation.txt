Class {
#name : 'WordCount' ,
#superclass : 'Object' ,
#category : 'Exercise@WordCount' ,
#package : 'Exercise@WordCount'
}
{ #category : 'exercism' }
WordCount >> countWordsSentence: aString [
| theDict workingString |
theDict := Dictionary new .
workingString := ( aString asLowercase
regex: '[[:alnum:]'']+'
matchesCollect: [ : m | m ]) collect: [ : str |
str trimBoth: [ : e | e = $' ] ].
^ workingString histogram
]
Extension { #name : #Collection }
{ #category : '*Exercise@WordCount' }
Collection >> histogram [
| dict setOfInputSequence |
dict := Dictionary new .
setOfInputSequence := self asSet .
setOfInputSequence do: [ : eachOfSeq |
dict
at: eachOfSeq
put:
( self count: [ : each | ( Set with: eachOfSeq ) includes: each ]) ].
^ dict
]
