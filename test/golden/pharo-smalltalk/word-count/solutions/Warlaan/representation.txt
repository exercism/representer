Class {
#name : #WordCount ,
#superclass : #Object ,
#category : # 'Exercise@WordCount'
}
{ #category : #accessing }
WordCount >> countWordsSentence: aString [
^ ((( aString substrings: "' ,'" )
groupedBy: [ : word |
( word asLowercase select: [ : c | c isAlphaNumeric or: c = $' ])
trimBoth: [ : c | c = $' ] ]) collect: [ : list | list size ])
asDictionary associationsSelect: [ : pair | pair key ~= '' ]
]
