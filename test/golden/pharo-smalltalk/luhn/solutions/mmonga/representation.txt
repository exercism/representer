Class {
#name : #Luhn ,
#superclass : #Object ,
#instVars : [
'evens' ,
'odds'
] ,
#category : # 'Exercise@Luhn'
}
{ #category : #private }
Luhn >> makeEvensAndOdds: aDigitString [
evens := LinkedList new .
odds := LinkedList new .
aDigitString reverse
withIndexDo: [ : d : i |
( i isDivisibleBy: 2 )
ifTrue: [ | n2 |
n2 := d digitValue * 2 .
evens
add:
( n2 <= 9
ifTrue: [ n2 ]
ifFalse: [ n2 - 9 ]) ]
ifFalse: [ odds add: d digitValue ] ].
]
{ #category : #operations }
Luhn >> validValue: aString [
| digits |
digits := aString select: [ : d | d ~= Character space ].
self makeEvensAndOdds: digits .
^ ( digits size < 2
or: [ digits size ~= ( digits select: [ : d | d isDigit ]) size ])
ifTrue: [ false ]
ifFalse: [ evens sum + odds sum isDivisibleBy: 10 ]
]
