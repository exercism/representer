Class {
#name : #SpaceAge ,
#superclass : #Object ,
#instVars : [
'earthYearInSeconds' ,
'factorsByPlanet'
] ,
#category : # 'Exercise@SpaceAge'
}
{ #category : # "'as yet unclassified'" }
SpaceAge >> ageOnPlanet: aPlanet at: seconds [
^ seconds / ( factorsByPlanet at: aPlanet ) / earthYearInSeconds
]
{ #category : #initialization }
SpaceAge >> initialize [
earthYearInSeconds := 31557600 .
factorsByPlanet := Dictionary newFrom: {
( 'Earth' -> 1 ).
( 'Mercury' -> 0.2408467 ).
( 'Venus' -> 0.61519726 ).
( 'Mars' -> 1.8808158 ).
( 'Jupiter' -> 11.862615 ).
( 'Saturn' -> 29.447498 ).
( 'Uranus' -> 84.016846 ).
( 'Neptune' -> 164.79132 ) }
]
