Class {
#name : #Clock ,
#superclass : #Object ,
#instVars : [
'hour' ,
'minute'
] ,
#category : # 'Exercise@Clock'
}
{ #category : # "'as yet unclassified'" }
Clock class >> hour: aHour minute: aMinute [
^ self new
addHour: aHour ;
addMinute: aMinute ;
yourself
]
{ #category : #arithmetic }
Clock >> + aMinute [
^ self addMinute: aMinute
]
{ #category : #arithmetic }
Clock >> - aMinute [
^ self subtractMinute: aMinute
]
{ #category : #comparing }
Clock >> = aClock [
^ self equals: aClock
]
{ #category : #adding }
Clock >> addHour: aHour [
hour := ( hour + aHour ) \\ 24
]
{ #category : #adding }
Clock >> addMinute: aMinute [
self addHour: ( minute + aMinute ) // 60 .
minute := ( minute + aMinute ) \\ 60 .
]
{ #category : #converting }
Clock >> asString [
^ '{hour}:{minute}'
format:
{( #hour -> ( hour asString padLeftTo: 2 with: $0 )).
( #minute -> ( minute asString padLeftTo: 2 with: $0 ))} asDictionary
]
{ #category : #comparing }
Clock >> equals: aClock [
^ ( minute = aClock minute ) and: [ hour = aClock hour ]
]
{ #category : #accessing }
Clock >> hour [
^ hour
]
{ #category : #initialization }
Clock >> initialize [
super initialize .
hour := 0 .
minute := 0 .
]
{ #category : #accessing }
Clock >> minute [
^ minute
]
{ #category : # "'as yet unclassified'" }
Clock >> subtractMinute: aMinute [
^ self addMinute: ( aMinute abs negated )
]
