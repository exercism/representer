Class {
#name : #Diamond ,
#superclass : #Object ,
#category : # 'Exercise@Diamond'
}
{ #category : #operations }
Diamond >> halfRowsLetter: aString [
| letter n prevString |
letter := aString at: 1 .
n := letter asInteger - $A asInteger .
prevString := ( n - 1 + $A asInteger ) asCharacter asString .
^ n = 0
ifTrue: [ self makeRow: n ]
ifFalse: [ | base |
base := ( self halfRowsLetter: prevString )
collect: [ : s | "' '" , s , "' '" ].
base , ( self makeRow: n ) ]
]
{ #category : #operations }
Diamond >> makeRow: n [
| letter midSpaces |
letter := ( n + $A asInteger ) asCharacter .
midSpaces := '' join: (( 0 to: 2 * ( n - 1 )) collect: [ : x | "' '" ]).
^ {( n = 0
ifTrue: [ letter asString ]
ifFalse: [ ''
join:
{ letter .
midSpaces .
letter } ])}
]
{ #category : #operations }
Diamond >> rowsLetter: aString [
| letter n prevString |
letter := aString at: 1 .
n := letter asInteger - $A asInteger .
prevString := ( n - 1 + $A asInteger ) asCharacter asString .
^ n = 0
ifTrue: [ self makeRow: n ]
ifFalse: [ | base |
base := ( self halfRowsLetter: prevString )
collect: [ : s | "' '" , s , "' '" ].
base , ( self makeRow: n ) , base reverse ]
]
