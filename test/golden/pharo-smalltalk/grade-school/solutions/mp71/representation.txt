Class {
#name : #GradeSchool ,
#superclass : #Object ,
#instVars : [
'students'
] ,
#category : # 'Exercise@GradeSchool'
}
{ #category : #initialization }
GradeSchool >> addStudents: listOfStudentGrades [
listOfStudentGrades
do: [ : sg |
| student grade |
student := sg at: 1 .
grade := sg at: 2 .
( students at: grade ifAbsentPut: [ SortedCollection new ])
add: student ].
^ self .
]
{ #category : #initialization }
GradeSchool >> desiredGrade: grade [
^ Array
streamContents:
[ : s | ( students at: grade ifAbsent: [ #() ]) do: [ : student | s nextPut: student ] ]
]
{ #category : #initialization }
GradeSchool >> initialize [
students := Dictionary new .
]
{ #category : #initialization }
GradeSchool >> roster [
^ Array streamContents: [ : s |
students keys sort do: [ : grade |
( students at: grade ) do: [ : student |
s nextPut: student ]
]
].
]
