Class {
#name : #Grains ,
#superclass : #Object ,
#category : # 'Exercise@Grains'
}
{ #category : #computing }
Grains >> atSquare: squareNumber [
self checkCorrectRanges: squareNumber .
^ self basicAtSquare: squareNumber .
]
{ #category : #private }
Grains >> basicAtSquare: squareNumber [
squareNumber = 1
ifTrue: [ ^ squareNumber ].
^ (( self basicAtSquare: squareNumber - 1 ) * 2 )
]
{ #category : # 'private-exceptions' }
Grains >> checkCorrectRanges: anInteger [
( anInteger between: 1 and: 64 ) ifFalse: [ self error: "'square must be between 1 and 64'" ]
]
{ #category : #computing }
Grains >> total [
^ ( 1 to: 64 ) sum: [ : aNumber | self basicAtSquare: aNumber ]
]
