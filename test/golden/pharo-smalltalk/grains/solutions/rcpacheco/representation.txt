Class {
#name : #Grains ,
#superclass : #Object ,
#category : # 'Exercise@Grains'
}
{ #category : #action }
Grains >> atSquare: anInteger [
anInteger = 1
ifTrue: [ ^ 1 ].
anInteger > 1 & ( anInteger < 65 )
ifTrue: [ ^ 2 * ( self atSquare: anInteger - 1 ) ].
self error: "'square must be between 1 and 64'"
]
{ #category : #actions }
Grains >> total [
| result |
result := 0 .
1 to: 64 do: [ : each | result := result + ( self atSquare: each ) ].
^ result
]
