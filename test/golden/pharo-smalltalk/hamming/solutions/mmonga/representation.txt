Class {
#name : #Hamming ,
#superclass : #Object ,
#category : # 'Exercise@Hamming'
}
{ #category : #operations }
Hamming >> distanceStrand1: leftString strand2: rightString [
| comparisons |
self errorMsgFor: leftString and: rightString .
comparisons := leftString asArray
with: rightString asArray
collect: [ : a : b | a = b ].
^ ( comparisons reject: [ : x | x ]) size
]
{ #category : #private }
Hamming >> errorMsgFor: leftString and: rightString [
( leftString isEmpty and: [ rightString isNotEmpty ])
ifTrue: [ self error: "'left strand must not be empty'" ].
( rightString isEmpty and: [ leftString isNotEmpty ])
ifTrue: [ self error: "'right strand must not be empty'" ].
leftString size = rightString size
ifFalse: [ self error: "'left and right strands must be of equal length'" ].
]
