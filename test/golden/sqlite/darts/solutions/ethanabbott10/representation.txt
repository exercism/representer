SET score = 0

UPDATE darts
SET score = 1
WHERE ( x * x + y * y ) <= 100 AND ( x * x + y * y ) > 25 ;
UPDATE darts
SET score = 5
WHERE ( x * x + y * y ) <= 25 AND ( x * x + y * y ) > 1 ;
UPDATE darts
SET score = 10
WHERE ( x * x + y * y ) <= 1 ;
