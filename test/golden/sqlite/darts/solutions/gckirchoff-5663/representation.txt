DROP TABLE IF EXISTS darts ;
CREATE TABLE IF NOT EXISTS darts (
x REAL NOT NULL ,
y REAL NOT NULL ,
score INT
);
CREATE TRIGGER update_score AFTER INSERT ON darts
FOR EACH ROW
BEGIN
UPDATE darts
SET score = CASE
WHEN SQRT ( NEW . X * NEW . x + NEW . y * NEW . y ) <= 1 then 10
WHEN SQRT ( NEW . X * NEW . x + NEW . y * NEW . y ) <= 5 then 5
WHEN SQRT ( NEW . X * NEW . x + NEW . y * NEW . y ) <= 10 then 1
ELSE 0
END
WHERE rowid = NEW . rowid ;
END ;
. mode csv
. import . / data . csv darts
