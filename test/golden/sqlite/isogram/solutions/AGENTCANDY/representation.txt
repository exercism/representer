CREATE TABLE alphabet ( letter );
INSERT INTO alphabet ( letter ) VALUES
( 'a' ), ( 'b' ), ( 'c' ), ( 'd' ), ( 'e' ), ( 'f' ), ( 'g' ), ( 'h' ), ( 'i' ), ( 'j' ), ( 'k' ), ( 'l' ), ( 'm' ),
( 'n' ), ( 'o' ), ( 'p' ), ( 'q' ), ( 'r' ), ( 's' ), ( 't' ), ( 'u' ), ( 'v' ), ( 'w' ), ( 'x' ), ( 'y' ), ( 'z' );
UPDATE isogram
SET is_isogram = (
SELECT min ( norep ) FROM (
SELECT phrase , length ( phrase ) - length ( replace ( lower ( phrase ), letter , '' )) < 2 AS norep
FROM isogram CROSS JOIN alphabet
)
GROUP BY phrase
HAVING phrase = isogram . phrase
);
