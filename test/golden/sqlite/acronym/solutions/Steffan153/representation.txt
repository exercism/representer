UPDATE acronym
SET result = (
WITH words ( s , b , w ) AS (
VALUES ( UPPER ( phrase ) || "' '" , '' , '' )
UNION ALL
SELECT
substr ( s , 2 ),
IIF ( s GLOB '[A-Z'']*' , b || substr ( s , 1 , 1 ), '' ),
IIF ( S GLOB '[A-Z'']*' , '' , b )
FROM words
WHERE s != ''
)
SELECT group_concat ( substr ( w , 1 , 1 ), '' )
FROM words
);
