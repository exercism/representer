WITH color_number AS (
SELECT
CASE
WHEN color1 = 'black' THEN 0
WHEN color1 = 'brown' THEN 1
WHEN color1 = 'red' THEN 2
WHEN color1 = 'orange' THEN 3
WHEN color1 = 'yellow' THEN 4
WHEN color1 = 'green' THEN 5
WHEN color1 = 'blue' THEN 6
WHEN color1 = 'violet' THEN 7
WHEN color1 = 'grey' THEN 8
WHEN color1 = 'white' THEN 9
END AS color1_num ,
CASE
WHEN color2 = 'black' THEN 0
WHEN color2 = 'brown' THEN 1
WHEN color2 = 'red' THEN 2
WHEN color2 = 'orange' THEN 3
WHEN color2 = 'yellow' THEN 4
WHEN color2 = 'green' THEN 5
WHEN color2 = 'blue' THEN 6
WHEN color2 = 'violet' THEN 7
WHEN color2 = 'grey' THEN 8
WHEN color2 = 'white' THEN 9
END AS color2_num ,
rowid
FROM color_code
)
UPDATE color_code
SET result = (
SELECT color1_num || color2_num
FROM color_number
WHERE color_code . rowid = color_number . rowid
);
