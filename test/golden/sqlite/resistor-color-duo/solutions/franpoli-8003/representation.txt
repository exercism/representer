CREATE TABLE color_code (
color1 TEXT ,
color2 TEXT ,
result INT
);
INSERT INTO color_code
SELECT c1 . color , c2 . color , ( c1 . value * 10 + c2 . value )
FROM color_values c1 , color_values c2 ;
