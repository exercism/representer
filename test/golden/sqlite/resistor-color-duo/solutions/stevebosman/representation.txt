CREATE TABLE "\"color_value\"" ( "\"color\"" TEXT , "\"value\"" INT );
INSERT
INTO "\"color_value\"" ( color , value )
VALUES
( 'black' , 0 ),
( 'brown' , 1 ),
( 'red' , 2 ),
( 'orange' , 3 ),
( 'yellow' , 4 ),
( 'green' , 5 ),
( 'blue' , 6 ),
( 'violet' , 7 ),
( 'grey' , 8 ),
( 'white' , 9 );
UPDATE color_code AS cc
SET result = cv1 . value * 10 + cv2 . value
FROM color_value AS cv1 , color_value AS cv2
WHERE cv1 . color = cc . color1
AND cv2 . color = cc . color2 ;
