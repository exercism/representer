UPDATE results
SET result = CASE
WHEN property = 'scores' THEN
(
SELECT GROUP_CONCAT ( score , ',' )
FROM scores
WHERE scores . game_id = results . game_id
)
WHEN property = 'latest' THEN
(
SELECT score
FROM scores
WHERE scores . game_id = results . game_id
ORDER BY rowid DESC
LIMIT 1
)
WHEN property = 'personalBest' THEN
(
SELECT score
FROM scores
WHERE scores . game_id = results . game_id
ORDER BY score DESC
LIMIT 1
)
ELSE
(
SELECT GROUP_CONCAT ( score , ',' )
FROM ( SELECT score
FROM scores
WHERE scores . game_id = results . game_id
ORDER BY score DESC
LIMIT 3
)
)
END
