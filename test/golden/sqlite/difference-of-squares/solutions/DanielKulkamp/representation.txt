WHEN property == "\"squareOfSum\"" THEN pow (( number * ( 1 + number ) / 2 ), 2 )
WHEN property == "\"sumOfSquares\"" THEN ( number * number + number ) * ( 2 * number + 1 ) / 6
WHEN property == "\"differenceOfSquares\"" THEN pow (( number * ( 1 + number ) / 2 ), 2 ) - ( number * number + number ) * ( 2 * number + 1 ) / 6
ELSE NULL END ;
