UPDATE "\"difference-of-squares\""
SET "\"result\"" = POWER ((( 1 + "\"number\"" ) * "\"number\"" / 2 ), 2 )
WHERE "\"property\"" = 'squareOfSum' ;
UPDATE "\"difference-of-squares\""
SET result = "\"number\"" * ( "\"number\"" + 1 ) * ( 2 * "\"number\"" + 1 ) / 6
WHERE property = 'sumOfSquares' ;
WITH a AS
(
select "\"number\"" , "\"result\"" from "\"difference-of-squares\"" WHERE "\"property\"" = 'squareOfSum'
),
b AS
(
select "\"number\"" , "\"result\"" from "\"difference-of-squares\"" WHERE "\"property\"" = 'sumOfSquares'
)
UPDATE "\"difference-of-squares\""
SET "\"result\"" = ( a . result - b . result )
from ( a JOIN b ON a . "\"number\"" = b . "\"number\"" )
WHERE "\"difference-of-squares\"" . "\"property\"" = 'differenceOfSquares' AND "\"difference-of-squares\"" . "\"number\"" = a . "\"number\"" ;
