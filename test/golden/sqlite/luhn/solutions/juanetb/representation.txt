(file "luhn.sql")
WITH RECURSIVE
clean_numbers AS (
SELECT value ,
CASE
WHEN value IS NULL THEN ''
WHEN value GLOB "'*[^0-9 ]*'" THEN ''
ELSE REPLACE ( value , "' '" , '' )
END AS clean_value
FROM luhn
),
digits AS (
SELECT value ,
clean_value ,
LENGTH ( clean_value ) as str_length ,
1 AS position ,
CAST ( CASE WHEN LENGTH ( clean_value ) > 0
THEN SUBSTR ( clean_value , LENGTH ( clean_value ), 1 )
ELSE '0' END AS INTEGER ) AS running_sum
FROM clean_numbers
UNION ALL
SELECT value ,
clean_value ,
str_length ,
position + 1 ,
running_sum + (
CASE
WHEN position % 2 = 1 THEN
CASE
WHEN CAST ( SUBSTR ( clean_value , str_length - position , 1 ) AS INTEGER ) * 2 > 9
THEN CAST ( SUBSTR ( clean_value , str_length - position , 1 ) AS INTEGER ) * 2 - 9
ELSE CAST ( SUBSTR ( clean_value , str_length - position , 1 ) AS INTEGER ) * 2
END
ELSE
CAST ( SUBSTR ( clean_value , str_length - position , 1 ) AS INTEGER )
END
)
FROM digits
WHERE position < str_length
)
UPDATE luhn
SET result = COALESCE (
(
SELECT CASE
WHEN clean_value = '' THEN 0
WHEN str_length <= 1 THEN 0
WHEN running_sum % 10 = 0 THEN 1
ELSE 0
END
FROM digits
WHERE position = str_length
AND value = luhn . value
),
0
);
