(file "armstrong-numbers.sql")
WITH RECURSIVE `temp` ( `number` , `sum` , `remainder` , `power` ) AS (
SELECT `number` , 0 , `number` , ( FLOOR ( LOG ( 10 , `number` )) + 1 ) FROM `armstrong-numbers`
UNION ALL
SELECT `number` , ( `sum` + POWER ( MOD ( `remainder` , 10 ), `power` )), FLOOR ( `remainder` / 10 ), `power`
FROM `temp`
WHERE `remainder` > 0
)
UPDATE `armstrong-numbers`
SET `result` = ( SELECT `armstrong-numbers` . `number` = MAX ( `sum` ) FROM `temp` WHERE `temp` . `number` = `armstrong-numbers` . `number` );
