with recursive
impl ( id , number ) as (
select rowid , number from collatz
union all
select id , iif ( number & 1 , 3 * number + 1 , cast ( number / 2 as integer ))
from impl where number > 1
),
impl2 ( id , steps ) as (
select id , count ( * ) - 1 from impl group by id
)
update collatz
set steps = impl2 . steps
from impl2
where rowid = impl2 . id ;
