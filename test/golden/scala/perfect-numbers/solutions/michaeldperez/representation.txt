import NumberType.NumberType
object NumberType extends Enumeration {
  type NumberType = Value
  val Abundant = Value ( "\"Abundant\"" )
  val Deficient = Value ( "\"Deficient\"" )
  val Perfect = Value ( "\"Perfect\"" )
}
object PerfectNumbers {
  def classify ( num : Int ) : Either [ String , NumberType ] = {
    if ( num < 1 ) {
      return Left ( "\"Classification is only possible for natural numbers.\"" )
    }
    val sumOfFactors : Int = ( 1 until num ). foldLeft ( 0 ) {
      ( sum : Int , int : Int ) => if ( num % int == 0 ) sum + int else sum
    }
    if ( sumOfFactors == num ) {
      Right ( NumberType . Perfect )
    } else if ( sumOfFactors > num ) {
      Right ( NumberType . Abundant )
    } else {
      Right ( NumberType . Deficient )
    }
  }
}
