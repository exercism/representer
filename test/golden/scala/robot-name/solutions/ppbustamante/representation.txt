import scala.util.Random
class Robot () {
  private val seed = new Random ( System . nanoTime ())
  private var _name = generateName ()
  private def generateName () : String = {
    val str = Iterator . continually ( Random . nextPrintableChar ()). filter ( _ . isUpper ). take ( 2 ). mkString
    val num = Iterator . continually ( Random . nextPrintableChar ()). filter ( _ . isDigit ). take ( 3 ). mkString
    s "\"$str$num\""
  }
  def name : String = _name
  def reset () : Unit = _name = generateName ()
}
