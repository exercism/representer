object Bob {
  def response ( statement : String ) : String = ( isEmpty ( statement ), isAllUpperCase ( statement ), isQuestion ( statement )) match {
    case ( true , _ , _ ) => "\"Fine. Be that way!\""
    case ( false , true , true ) => "\"Calm down, I know what I'm doing!\""
    case ( false , true , false ) => "\"Whoa, chill out!\""
    case ( false , false , true ) => "\"Sure.\""
    case ( false , false , false ) => "\"Whatever.\""
  }
  def isEmpty ( statement : String ) : Boolean = statement . trim (). isEmpty
  def isAllUpperCase ( statement : String ) : Boolean =
    statement . toUpperCase (). equals ( statement ) && ! statement . toLowerCase (). equals ( statement )
  def isQuestion ( statement : String ) : Boolean = {
    val trimmed = statement . trim ()
    trimmed . length > 0 && trimmed . last == '?'
  }
}
