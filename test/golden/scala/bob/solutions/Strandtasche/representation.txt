object Bob {
  def shouting ( statement : String ) : Boolean = {
    statement . exists ( _ . isLetter ) && statement . forall ( x => ! x . isLetter || x . isUpper )
  }
  def response ( statement : String ) : String = {
    val statementF = statement . trim ()
    if ( statementF . isEmpty ) {
      "\"Fine. Be that way!\""
    }
    else if ( statementF . takeRight ( 1 ) == "\"?\"" ) {
      if ( shouting ( statementF )) {
        "\"Calm down, I know what I'm doing!\""
      }
      else {
        "\"Sure.\""
      }
    }
    else if ( shouting ( statementF )) {
      "\"Whoa, chill out!\""
    }
    else {
      "\"Whatever.\""
    }
  }
}
