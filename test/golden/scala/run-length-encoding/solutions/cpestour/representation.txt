object RunLengthEncoding {
  def encode ( str : String ) : String = str match {
    case "\"\"" => "\"\""
    case _ => {
      val c = str . head
      val ( init , tail ) = str . span ( _ == c )
      if ( init . length == 1 ) c + encode ( tail )
      else init . length . toString + c + encode ( tail )
    }
  }
  def decode ( str : String ) : String = str match {
    case "\"\"" => "\"\""
    case _ => {
      val ( init , tail ) = str . span ( _ . isDigit )
      if ( init . length > 0 ) tail . head . toString * init . toInt + decode ( tail . tail )
      else tail . head + decode ( tail . tail )
    }
  }
}
