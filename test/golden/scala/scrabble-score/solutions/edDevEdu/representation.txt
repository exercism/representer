object ScrabbleScore {
  private val letter1Points : Regex = "\"[aeioulnrst]\"" . r
  private val letter2Points : Regex = "\"[dg]\"" . r
  private val letter3Points : Regex = "\"[bcmp]\"" . r
  private val letter4Points : Regex = "\"[fhvwy]\"" . r
  private val letter5Points : Regex = "\"[k]\"" . r
  private val letter8Points : Regex = "\"[jx]\"" . r
  private val letter10Points : Regex = "\"[qz]\"" . r
  def score ( s : String ) : Int = {
    s . map (( c : Char ) => {
      c . toLower match {
        case letter1Points () => 1
        case letter2Points () => 2
        case letter3Points () => 3
        case letter4Points () => 4
        case letter5Points () => 5
        case letter8Points () => 8
        case letter10Points () => 10
        case _ => 0
      }
    }). sum
  }
}
