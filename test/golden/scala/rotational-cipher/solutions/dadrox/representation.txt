object RotationalCipher {
  def rotate ( s : String , n : Int ) : String = RotationalCipher ( n ). encode ( s )
}
case class RotationalCipher ( n : Int ) {
  private val abc = ( 'a' to 'z' ). mkString
  private val cba = abc . substring ( n ) + abc . take ( n )
  private val lookup = abc . zip ( cba ). toMap
  private def encode ( it : Char ) : Char = lookup . get ( it . toLower ). map { c =>
    if ( it . isUpper ) c . toUpper else c
  }. getOrElse ( it )
  def encode ( s : String ) : String = s . map ( encode ). mkString
}
