object Hamming {
  def distance ( str : String , str1 : String ) : Option [ Int ] = {
    if ( str . length != str1 . length ) None
    else if ( str . equals ( str1 )) Some ( 0 )
    else {
      Some ( str . zipWithIndex . foldLeft ( 0 ) { case ( acc , ( char , index )) =>
        if ( str1 . charAt ( index ) == char ) acc else acc + 1
      })
    }
  }
}
