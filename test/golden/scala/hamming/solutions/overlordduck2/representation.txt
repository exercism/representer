object Hamming {
  def distance ( dna1 : String , dna2 : String ) : Option [ Int ] = {
    if ( dna1 . length != dna2 . length ){
      return None
    }
    if ( dna1 . equals ( dna2 )){
      return Some ( 0 )
    }
    var counter = 0
    var i = 0
    for ( i <- 0 until dna1 . length ){
      if ( dna1 . charAt ( i ) != dna2 . charAt ( i ))
        counter += 1
    }
    Some ( counter )
  }
}
