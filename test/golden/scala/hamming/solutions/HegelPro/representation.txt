object Hamming {
  def distance ( strOne : String , strTwo : String ) : Option [ Int ] =
    if ( strOne . isEmpty && strTwo . isEmpty ) Some ( 0 )
    else if ( strOne . isEmpty || strTwo . isEmpty ) None
    else if ( strOne . head != strTwo . head ) distance ( strOne . tail , strTwo . tail ). map ( _ + 1 )
    else distance ( strOne . tail , strTwo . tail )
}
