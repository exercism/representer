object Hamming {
  def distance ( string1 : String , string2 : String ) : Option [ Int ] = {
    if ( string1 . length != string2 . length ) return None
    var count = 0 ;
    string1 . zipWithIndex . foreach {
      case ( char , index ) => if ( char != string2 . charAt ( index )) count += 1
    }
    return Some ( count )
  }
}
