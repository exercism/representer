object Hamming {
  def distance ( s1 : String , s2 : String ) : Option [ Int ] = {
    var l1 = s1 . length
    var l2 = s2 . length
    if ( l1 != l2 )
      None
    else {
      var l = 0
      for ( a <- 0 to l1 - 1 ) {
        if ( s1 . slice ( a , a + 1 ) != s2 . slice ( a , a + 1 )) {
          l = l + 1
        }
      }
      Some ( l )
    }
  }
}
