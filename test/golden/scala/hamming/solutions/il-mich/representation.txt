object Hamming {
  def distance ( dnaStrandOne : String , dnaStrandTwo : String ) : Option [ Int ] = {
    if ( dnaStrandOne . length != dnaStrandTwo . length ) {
      return None
    } else {
      return Some ( matcher (( dnaStrandOne . toList , dnaStrandTwo . toList ), 0 ))
    }
  }
  def matcher ( pair : ( List [ Char ], List [ Char ]), num : Int ) : Int = {
    pair match {
      case ( Nil , Nil ) => return num
      case ( hda :: tla , hdb :: tlb ) if hda == hdb => return matcher (( tla , tlb ), num )
      case ( hda :: tla , hdb :: tlb ) => return matcher (( tla , tlb ), num + 1 )
    }
  }
}
