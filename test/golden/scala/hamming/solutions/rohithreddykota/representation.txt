object Hamming {
  def distance ( dna1 : String , dna2 : String ) : Option [ Int ] = {
    val ( seq1 , seq2 ) = ( dna1 . trim , dna2 . trim )
    if ( seq1 . length != seq2 . length ) {
      None
    } else {
      var count = 0
      for (( letter1 , letter2 ) <- seq1 zip seq2 ) {
        count = if ( letter1 equals letter2 ) count else count + 1
      }
      Some ( count )
    }
  }
}
