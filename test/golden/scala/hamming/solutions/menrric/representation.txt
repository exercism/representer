object Hamming {
  def distance ( dnaStrandOne : String , dnaStrandTwo : String ) : Option [ Int ] = {
    var contador = 0
    if ( dnaStrandOne . length == dnaStrandTwo . length ()){
      for ( letra <- 0 until dnaStrandOne . length ){
        if ( dnaStrandOne ( letra ) != dnaStrandTwo ( letra )) contador += 1
      }
    Some ( contador )
    } else None
  }
}
