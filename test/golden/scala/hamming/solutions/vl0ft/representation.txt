import scala.annotation.tailrec
object Hamming {
  def distance ( str : String , str1 : String ) : Option [ Int ] = {
    if ( str . length != str1 . length ) return None
    return Some ( recDistance ( str . toList , str1 . toList , 0 ))
  }
  @tailrec
  def recDistance ( str : List [ Char ], str1 : List [ Char ], acc : Int ) : Int = {
    if ( str . isEmpty ) return acc
    val diff : Int = if ( str . head != str1 . head ) {
      1
    } else {
      0
    }
    recDistance ( str . tail , str1 . tail , acc + diff )
  }
}
