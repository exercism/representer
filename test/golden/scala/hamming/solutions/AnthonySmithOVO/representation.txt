object Hamming {
  def distance ( firstStrand : String , secondStrand : String ) : Option [ Int ] = ( firstStrand , secondStrand ) match {
    case ( first , second ) if firstStrand . length == secondStrand . length =>
      Some ( calculateDistance ( first , second ))
    case _ => None
    }
  private def calculateDistance ( firstStrand : String , secondStrand : String ) : Int =
    firstStrand . zip ( secondStrand ). foldLeft ( 0 )(( i , strands ) => strands match {
      case ( l , r ) if l == r => i
      case _ => i + 1
    })
}
