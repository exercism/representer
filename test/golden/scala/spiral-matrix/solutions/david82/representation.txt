object SpiralMatrix {
  def spiralMatrix ( size : Int ) : List [ List [ Int ]] = {
    if ( size <= 0 ) Nil
    else {
      val matrix = Array . ofDim [ Int ]( size , size )
      var top = 0
      var bottom = size - 1
      var left = 0
      var right = size - 1
      var value = 1
      while ( value <= size * size ) {
        var c = left
        while ( c <= right ) {
          matrix ( top )( c ) = value
          value += 1
          c += 1
        }
        top += 1
        var r = top
        while ( r <= bottom ) {
          matrix ( r )( right ) = value
          value += 1
          r += 1
        }
        right -= 1
        if ( top <= bottom ) {
          c = right
          while ( c >= left ) {
            matrix ( bottom )( c ) = value
            value += 1
            c -= 1
          }
          bottom -= 1
        }
        if ( left <= right ) {
          r = bottom
          while ( r >= top ) {
            matrix ( r )( left ) = value
            value += 1
            r -= 1
          }
          left += 1
        }
      }
      List . tabulate ( size )( r => matrix ( r ). toList )
    }
  }
}
