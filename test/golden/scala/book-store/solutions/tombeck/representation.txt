object BookStore {
  def total ( books : List [ Int ]) : Int = {
    if ( books . isEmpty ) 0 else {
      def price ( group : Set [ Int ]) : Int = group . size match {
        case 1 => 800
        case 2 => 2 * 800 * 95 / 100
        case 3 => 3 * 800 * 90 / 100
        case 4 => 4 * 800 * 80 / 100
        case 5 => 5 * 800 * 75 / 100
      }
      books . groupBy ( identity ). values . toList . sortWith ( _ . size > _ . size ). foldLeft ( Nil . asInstanceOf [ List [ List [ Set [ Int ]]]]) {
        case ( Nil , copies ) => List ( copies . map ( Set ( _ )))
        case ( acc , copies ) => for {
          group <- acc
          perm <- copies . map ( Set ( _ )). padTo ( group . size , Set ()). permutations
        } yield group . zip ( perm ). map { case ( a , b ) => a ++ b }
      }. map ( _ . map ( price ). sum ). min
    }
  }
}
