func dailyRateFrom ( hourlyRate : Int ) -> Double {
return Double ( hourlyRate ) * 8
}
func monthlyRateFrom ( hourlyRate : Int , withDiscount discount : Double ) -> Double {
let dailyRate = Double ( hourlyRate ) * 8.0
let billableDaysPerMonth = 22.0
let monthlyRate = dailyRate * billableDaysPerMonth
let discountedMonthlyRate = monthlyRate * ( 1.0 - discount / 100.0 )
return Double ( discountedMonthlyRate ) . rounded ()
}
func workdaysIn ( budget : Double , hourlyRate : Int , withDiscount discount : Double ) -> Double {
let dailyRate = Double ( hourlyRate ) * 8.0
let discountedDailyRate = dailyRate * ( 1.0 - discount / 100.0 )
let workdaysCovered = budget / discountedDailyRate
return Double ( Int ( workdaysCovered ))
}
