func timeToPrepare ( drinks : [ String ]) -> Double {
var totalTime = 0.0
let drinksPrepTimes : [ String : Double ] = [
"\"beer\"" : 0.5 ,
"\"water\"" : 0.5 ,
"\"soda\"" : 0.5 ,
"\"shot\"" : 1.0 ,
"\"mixed drink\"" : 1.5 ,
"\"fancy drink\"" : 2.5 ,
"\"frozen drink\"" : 3.0
]
for drink in drinks {
if let time = drinksPrepTimes [ drink ] {
totalTime += time
}
}
return totalTime
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var haveSoFar = 0
var limesCutUp = 0
var count = 0
let wedgesFromLimes : [ String : Int ] = [
"\"small\"" : 6 ,
"\"medium\"" : 8 ,
"\"large\"" : 10
]
while haveSoFar < needed && count < limes . count {
let currentSize = limes [ count ]
if let size = wedgesFromLimes [ currentSize ] {
haveSoFar += size
limesCutUp += 1
}
count += 1
}
return limesCutUp
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var count = 0
var remainingTime = Double ( minutesLeft )
while remainingTime > 0 && remainingOrders . count > count {
let orderTime = timeToPrepare ( drinks : remainingOrders [ count ])
remainingTime -= orderTime
count += 1
}
if count == remainingOrders . count {
return []
}
return Array ( remainingOrders [ count ... remainingOrders . count - 1 ])
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var beer : ( first : String , last : String , total : Int )? = nil
var soda : ( first : String , last : String , total : Int )? = nil
var beerTimes : [ String ] = []
var sodaTimes : [ String ] = []
for order in orders {
if order . drink == "\"beer\"" {
beerTimes . append ( order . time )
}
if order . drink == "\"soda\"" {
sodaTimes . append ( order . time )
}
}
beer = beerTimes . count > 0 ? ( first : beerTimes [ 0 ], last : beerTimes [ beerTimes . count - 1 ], total : beerTimes . count ) : nil
soda = sodaTimes . count > 0 ? ( first : sodaTimes [ 0 ], last : sodaTimes [ sodaTimes . count - 1 ], total : sodaTimes . count ) : nil
return ( beer : beer , soda : soda )
}
