func timeToPrepare ( drinks : [ String ]) -> Double {
drinks . reduce ( 0 ) { ( runningTotal , drink ) in
switch drink {
case "\"beer\"" , "\"soda\"" , "\"water\"" :
return runningTotal + 0.5
case "\"shot\"" :
return runningTotal + 1.0
case "\"mixed drink\"" :
return runningTotal + 1.5
case "\"fancy drink\"" :
return runningTotal + 2.5
case "\"frozen drink\"" :
return runningTotal + 3.0
default :
fatalError ( "\"Unknown drink \\(" drink )." )
}
}
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var wedgesMade = 0
for ( index , lime ) in limes . enumerated () {
if wedgesMade >= needed {
return index
}
switch lime {
case "\"small\"" :
wedgesMade += 6
case "\"medium\"" :
wedgesMade += 8
case "\"large\"" :
wedgesMade += 10
default :
fatalError ( "\"Unknown lime size \\(" lime )." )
}
}
return limes . count
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var minutesLeft = Double ( minutesLeft )
var remainingOrders = remainingOrders
while minutesLeft > 0 , ! remainingOrders . isEmpty {
minutesLeft -= timeToPrepare ( drinks : remainingOrders . removeFirst ())
}
return remainingOrders
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?,
soda : ( first : String , last : String , total : Int )?
) {
var beerTracker : ( first : String , last : String , total : Int )?
var sodaTracker : ( first : String , last : String , total : Int )?
for ( drink , time ) in orders {
switch drink {
case "\"beer\"" :
guard beerTracker != nil else {
beerTracker = ( first : time , last : time , total : 1 )
continue
}
beerTracker ? . last = time
beerTracker ? . total += 1
case "\"soda\"" :
guard sodaTracker != nil else {
sodaTracker = ( first : time , last : time , total : 1 )
continue
}
sodaTracker ? . last = time
sodaTracker ? . total += 1
default :
continue
}
}
return ( beer : beerTracker , soda : sodaTracker )
}
