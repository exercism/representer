let times = [
"\"beer\"" : 0.5 ,
"\"soda\"" : 0.5 ,
"\"water\"" : 0.5 ,
"\"shot\"" : 1.0 ,
"\"mixed drink\"" : 1.5 ,
"\"fancy drink\"" : 2.5 ,
"\"frozen drink\"" : 3.0
]
let wedges = [
"\"small\"" : 6 ,
"\"medium\"" : 8 ,
"\"large\"" : 10
]
func timeToPrepare ( drinks : [ String ]) -> Double {
drinks . reduce ( 0.0 , { ( result , drink ) in
result + times [ drink , default : 0.0 ]
})
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
guard needed > 0 else {
return 0
}
var total = 0
for ( index , lime ) in limes . enumerated () {
total += wedges [ lime , default : 0 ]
if total > needed {
return index + 1
}
}
return limes . count
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var result = remainingOrders
var totalMinutes = 0.0
for order in remainingOrders {
if totalMinutes < Double ( minutesLeft ) {
result . removeFirst ()
}
totalMinutes += timeToPrepare ( drinks : order )
}
return result
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var result : ( beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?) = ( beer : nil , soda : nil )
for order in orders {
switch order . drink {
case "\"beer\"" :
result . beer = result . beer ?? ( first : order . time , last : order . time , total : 0 )
result . beer ? . last = order . time
result . beer ? . total += 1
case "\"soda\"" :
result . soda = result . soda ?? ( first : order . time , last : order . time , total : 0 )
result . soda ? . last = order . time
result . soda ? . total += 1
default :
break
}
}
return result
}
