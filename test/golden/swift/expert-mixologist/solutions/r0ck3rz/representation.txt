func timeToPrepare ( drinks : [ String ]) -> Double {
enum Drinks : String {
case beer
case soda
case water
case shot
case mixedDrink = "\"mixed drink\""
case fancyDrink = "\"fancy drink\""
case frozenDrink = "\"frozen drink\""
var prepTime : Double {
switch self {
case . beer , . soda , . water :
return 0.5
case . shot :
return 1.0
case . mixedDrink :
return 1.5
case . fancyDrink :
return 2.5
default :
return 3.0
}
}
}
return drinks . compactMap ({ Drinks ( rawValue : $0 )}) . reduce ( . zero , {( time , drink ) -> Double in
return time + drink . prepTime
})
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
guard needed != . zero || ! limes . isEmpty else {
return . zero
}
enum LimeSize : String {
case small
case medium
case large
var num : Int {
switch self {
case . small :
return 6
case . medium :
return 8
default :
return 10
}
}
}
var numOfLimesCut = 0
var stillNeeded = needed
for lime in limes . compactMap ({ LimeSize ( rawValue : $0 )}){
if stillNeeded <= . zero {
return numOfLimesCut
} else {
stillNeeded -= lime . num
numOfLimesCut += 1
}
}
return numOfLimesCut
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var orders = remainingOrders
var minsSpent = timeToPrepare ( drinks : orders [ 0 ])
orders . removeFirst ()
var minsLeft = Double ( minutesLeft ) - minsSpent
while ! orders . isEmpty {
minsSpent = timeToPrepare ( drinks : orders [ 0 ])
orders . removeFirst ()
if minsLeft >= minsSpent {
minsLeft -= minsSpent
} else {
break
}
}
return orders
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
let beer = orders . filter { $0 . drink == "\"beer\"" }
let soda = orders . filter { $0 . drink == "\"soda\"" }
var result :( beer :( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?)
if ! beer . isEmpty {
result . beer = ( beer [ 0 ] . time , beer . last !. time , beer . count )
}
if ! soda . isEmpty {
result . soda = ( soda [ 0 ] . time , soda . last !. time , soda . count )
}
return result
}
