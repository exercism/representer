func timeToPrepare ( drinks : [ String ]) -> Double {
let totDur = drinks . reduce ( 0.0 ){ ( time , drink ) in
var dur = 0.0
switch ( drink ){
case "\"beer\"" , "\"soda\"" , "\"water\"" :
dur = 0.5
break
case "\"shot\"" :
dur = 1.0
break
case "\"mixed drink\"" :
dur = 1.5
break
case "\"fancy drink\"" :
dur = 2.5
break
case "\"frozen drink\"" :
dur = 3.0
break
default :
dur = 0.0
}
return ( time + dur )
}
return totDur
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
guard needed > 0 else { return 0 }
var cut = 0
var wedges = 0
for lime in limes {
cut = cut + 1
switch ( lime ){
case "\"small\"" :
wedges += 6
break
case "\"medium\"" :
wedges += 8
break
case "\"large\"" :
wedges += 10
break
default :
wedges += 0
}
if ( wedges >= needed ) {
break
} else { continue }
}
return cut
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var minRemaining = Double ( minutesLeft )
var ordersRemaining = remainingOrders
while minRemaining > 0 , ! ordersRemaining . isEmpty {
minRemaining -= timeToPrepare ( drinks : ordersRemaining . removeFirst ())
}
return ordersRemaining
}
