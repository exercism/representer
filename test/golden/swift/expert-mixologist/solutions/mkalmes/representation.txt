func timeToPrepare ( drinks : [ String ]) -> Double {
var preparationTime = 0.0
for drink in drinks {
switch drink {
case "\"beer\"" , "\"soda\"" , "\"water\"" :
preparationTime += 0.5
case "\"shot\"" :
preparationTime += 1.0
case "\"mixed drink\"" :
preparationTime += 1.5
case "\"fancy drink\"" :
preparationTime += 2.5
case "\"frozen drink\"" :
preparationTime += 3.0
default :
preparationTime += 0.0
}
}
return preparationTime
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
guard limes . isEmpty == false else { return 0 }
var prepared = 0
var myLimes = limes
while prepared < needed && myLimes . isEmpty == false {
let lime = myLimes . remove ( at : myLimes . startIndex )
switch lime {
case "\"small\"" :
prepared += 6
case "\"medium\"" :
prepared += 8
case "\"large\"" :
prepared += 10
default :
prepared += 0
}
}
return limes . count - myLimes . count
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var minutes = Double ( minutesLeft )
var myOrders = remainingOrders
while minutes > . zero && myOrders . isEmpty == false {
let order = myOrders . remove ( at : myOrders . startIndex )
minutes -= timeToPrepare ( drinks : order )
}
return myOrders
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var beerTimes = [ String ]()
var sodaTimes = [ String ]()
for order in orders {
if order . drink == "\"beer\"" {
beerTimes . append ( order . time )
}
if order . drink == "\"soda\"" {
sodaTimes . append ( order . time )
}
}
let beer : ( String , String , Int )? = beerTimes . isEmpty == false ? ( beerTimes . first ! , beerTimes . last ! , beerTimes . count ) : nil
let soda : ( String , String , Int )? = sodaTimes . isEmpty == false ? ( sodaTimes . first ! , sodaTimes . last ! , sodaTimes . count ) : nil
return ( beer , soda )
}
