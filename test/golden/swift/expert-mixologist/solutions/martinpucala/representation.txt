func timeToPrepare ( drinks : [ String ]) -> Double {
var time = 0.0
for drink in drinks {
if [ "\"beer\"" , "\"soda\"" , "\"water\"" ] . contains ( drink ) {
time += 0.5
}
if drink == "\"shot\"" {
time += 1.0
}
if drink == "\"mixed drink\"" {
time += 1.5
}
if drink == "\"fancy drink\"" {
time += 2.5
}
if drink == "\"frozen drink\"" {
time += 3.0
}
}
return time
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var needWedges = needed
var cutLimes = 0
if needWedges <= 0 {
return 0
}
for lime in limes {
if lime == "\"small\"" {
needWedges -= 6
}
if lime == "\"medium\"" {
needWedges -= 8
}
if lime == "\"large\"" {
needWedges -= 10
}
cutLimes += 1
if needWedges <= 0 {
break
}
}
return cutLimes
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var minutes = Double ( minutesLeft )
var orders = Array ( remainingOrders )
while ! ( orders . isEmpty || minutes <= 0 ) {
let order = orders . removeFirst ()
minutes -= timeToPrepare ( drinks : order )
}
return orders
}
func updateStats ( _ stats : inout ( first : String , last : String , total : Int ), _ order : ( drink : String , time : String )) {
let ( _ , time ) = order
stats . total += 1
stats . first = stats . first == "\"\"" ? time : min ( stats . first , time )
stats . last = stats . last == "\"\"" ? time : max ( stats . last , time )
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var beerStats : ( first : String , last : String , total : Int ) = ( "\"\"" , "\"\"" , 0 )
var sodaStats : ( first : String , last : String , total : Int ) = ( "\"\"" , "\"\"" , 0 )
for order in orders {
if order . drink == "\"beer\"" {
updateStats ( & beerStats , order )
} else if order . drink == "\"soda\"" {
updateStats ( & sodaStats , order )
}
}
if beerStats . total > 0 && sodaStats . total > 0 {
return ( beer : beerStats , soda : sodaStats )
} else if beerStats . total > 0 {
return ( beer : beerStats , soda : nil )
} else if sodaStats . total > 0 {
return ( beer : nil , soda : sodaStats )
} else {
return ( beer : nil , soda : nil )
}
}
