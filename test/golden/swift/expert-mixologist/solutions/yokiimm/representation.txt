func timeToPrepare ( drinks : [ String ]) -> Double {
var totalTime = 0.0
for drink in drinks {
switch drink {
case "\"beer\"" , "\"soda\"" , "\"water\"" : totalTime += 0.5
case "\"shot\"" : totalTime += 1.0
case "\"mixed drink\"" : totalTime += 1.5
case "\"fancy drink\"" : totalTime += 2.5
case "\"frozen drink\"" : totalTime += 3.0
default : continue
}
}
return totalTime
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var limesUsed : Int = 0
var neededVar : Int = needed
var limesVar : [ String ] = limes
while neededVar > 0 && ! limesVar . isEmpty {
let lime : String = limesVar . removeFirst ()
switch lime {
case "\"small\"" :
neededVar -= 4
case "\"medium\"" :
neededVar -= 8
case "\"large\"" :
neededVar -= 10
default :
break
}
limesUsed += 1
}
return limesUsed
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
guard ! remainingOrders . isEmpty , minutesLeft >= 0 else { return [] }
var minutes = Double ( minutesLeft )
var remains = remainingOrders
while ! remains . isEmpty && minutes > 0 {
minutes -= timeToPrepare ( drinks : remains . removeFirst ())
}
return remains
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var result : ( beer : ( first : String , last : String , total : Int )?,
soda : ( first : String , last : String , total : Int )?) =
( beer : ( first : "\"default\"" , last : "\"default\"" , total : 0 ),
soda : ( first : "\"default\"" , last : "\"default\"" , total : 0 ))
var beertime : [ String ] = []
var sodatime : [ String ] = []
for ( drink , time ) in orders {
switch drink {
case "\"beer\"" :
beertime . append ( time )
case "\"soda\"" :
sodatime . append ( time )
default :
continue
}
}
if beertime . isEmpty {
result . beer = nil
} else {
result . beer ? . first = beertime . first !
result . beer ? . last = beertime . last !
result . beer ? . total = beertime . count
}
if sodatime . isEmpty {
result . soda = nil
} else {
result . soda ? . first = sodatime . first !
result . soda ? . last = sodatime . last !
result . soda ? . total = sodatime . count
}
return result
}
