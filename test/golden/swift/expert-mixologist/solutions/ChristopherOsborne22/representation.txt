func timeToPrepare ( drinks : [ String ]) -> Double {
var totalTime : Double = 0
for drink in drinks {
switch drink {
case "\"beer\"" , "\"soda\"" , "\"water\"" :
totalTime += 0.5
case "\"shot\"" :
totalTime += 1.0
case "\"mixed drink\"" :
totalTime += 1.5
case "\"fancy drink\"" :
totalTime += 2.5
case "\"frozen drink\"" :
totalTime += 3.0
default :
totalTime += 0
}
}
return totalTime
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var newLimes = limes
var limesCount = 0
var wedgesCount = 0
while wedgesCount < needed {
if newLimes . count != 0 {
let firstWedge = newLimes . removeFirst ()
switch firstWedge {
case "\"small\"" :
wedgesCount += 6
limesCount += 1
case "\"medium\"" :
wedgesCount += 8
limesCount += 1
case "\"large\"" :
wedgesCount += 10
limesCount += 1
default :
wedgesCount += 0
}
} else {
break
}
}
return limesCount
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var newMinutesLeft = Double ( minutesLeft )
var newRemainingOrders = remainingOrders
repeat {
if newRemainingOrders . count != 0 {
let firstOder = newRemainingOrders . removeFirst ()
let timeForEachOrder = timeToPrepare ( drinks : firstOder )
newMinutesLeft -= timeForEachOrder
} else {
break
}
} while newMinutesLeft > 0
return newRemainingOrders
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var beer : ( first : String , last : String , total : Int )? = nil
var soda : ( first : String , last : String , total : Int )? = nil
for order in orders {
if order . drink == "\"beer\"" {
if beer == nil {
beer = ( first : order . time , last : order . time , total : 1 )
} else {
beer !. last = order . time
beer !. total = beer !. total + 1
}
}
if order . drink == "\"soda\"" {
if soda == nil {
soda = ( first : order . time , last : order . time , total : 1 )
} else {
soda !. last = order . time
soda !. total = soda !. total + 1
}
}
}
return ( beer : beer , soda : soda )
}
