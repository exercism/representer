func timeToPrepare ( drinks : [ String ]) -> Double {
var time = 0.0
drinks . forEach {
if $0 == "\"beer\"" || $0 == "\"soda\"" || $0 == "\"water\"" {
time += 0.5
} else if $0 == "\"shot\"" {
time += 1.0
} else if $0 == "\"mixed drink\"" {
time += 1.5
} else if $0 == "\"fancy drink\"" {
time += 2.5
} else if $0 == "\"frozen drink\"" {
time += 3.0
} else {
time += . zero
}
}
return time
}
func makeWedges ( needed : Int , limes : [ String ]) -> Int {
var wedges = 0
var limesUsed = 0
limes . forEach {
switch $0 {
case "\"small\"" :
if wedges < needed {
wedges += 6
limesUsed += 1
}
case "\"medium\"" :
if wedges < needed {
wedges += 8
limesUsed += 1
}
case "\"large\"" :
if wedges < needed {
wedges += 10
limesUsed += 1
}
default :
break
}
}
return limesUsed
}
func finishShift ( minutesLeft : Int , remainingOrders : [[ String ]]) -> [[ String ]] {
var timeElaspsed = 0.0
var orders = remainingOrders
remainingOrders . forEach {
if timeElaspsed < Double ( minutesLeft ) {
timeElaspsed += timeToPrepare ( drinks : $0 )
orders . remove ( at : . zero )
}
}
return orders
}
func orderTracker ( orders : [( drink : String , time : String )]) -> (
beer : ( first : String , last : String , total : Int )?, soda : ( first : String , last : String , total : Int )?
) {
var firstBeer = "\"\""
var lastBeer = "\"\""
var totalBeers = 0
var firstSoda = "\"\""
var lastSoda = "\"\""
var totalSodas = 0
orders . forEach {
if $0 . drink == "\"beer\"" {
totalBeers += 1
if firstBeer . isEmpty {
firstBeer = $0 . time
}
if lastBeer . isEmpty {
lastBeer = $0 . time
}
if $0 . time < firstBeer {
firstBeer = $0 . time
} else if $0 . time > lastBeer {
lastBeer = $0 . time
}
} else if $0 . drink == "\"soda\"" {
totalSodas += 1
if firstSoda . isEmpty {
firstSoda = $0 . time
}
if lastSoda . isEmpty {
lastSoda = $0 . time
}
if $0 . time < firstSoda {
firstSoda = $0 . time
} else if $0 . time > lastSoda {
lastSoda = $0 . time
}
}
}
let beerInfo = totalBeers == 0 ? nil : ( first : firstBeer , last : lastBeer , total : totalBeers )
let sodaInfo = totalSodas == 0 ? nil : ( first : firstSoda , last : lastSoda , total : totalSodas )
return ( beer : beerInfo , soda : sodaInfo )
}
