extension Array where Element : Equatable {
func isSublist ( of other : [ Element ]) -> Bool {
guard self . count <= other . count else { return false }
for i in 0 ... ( other . count - self . count ) {
if Array ( other [ i ..< ( i + self . count )]) == self {
return true
}
}
return false
}
func isSuperlist ( of other : [ Element ]) -> Bool {
return other . isSublist ( of : self )
}
}
func classifier ( listOne : [ Int ], listTwo : [ Int ]) -> Classification {
if listOne == listTwo { return . equal }
if listOne . isSublist ( of : listTwo ) { return . sublist }
if listOne . isSuperlist ( of : listTwo ) { return . superlist }
return . unequal
}
enum Classification {
case equal
case sublist
case superlist
case unequal
}
