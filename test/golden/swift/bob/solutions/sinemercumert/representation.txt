import Foundation
class Bob {
static func response ( _ message : String ) -> String {
let editMessage = message . trimmingCharacters ( in : . whitespacesAndNewlines )
if editMessage . isEmpty {
return "\"Fine. Be that way!\""
} else if editMessage == editMessage . uppercased () && editMessage != editMessage . lowercased () {
if editMessage . hasSuffix ( "\"?\"" ) {
return "\"Calm down, I know what I'm doing!\""
} else {
return "\"Whoa, chill out!\""
}
} else if editMessage . hasSuffix ( "\"?\"" ) {
return "\"Sure.\""
} else {
return "\"Whatever.\""
}
}
}
