import Foundation
class Bob {
private static func is_uppercase ( _ message : String ) -> Bool {
return message . contains { $0 . isLetter } && message == message . uppercased ()
}
static func response ( _ message : String ) -> String {
let message = message . trimmingCharacters ( in : . whitespacesAndNewlines )
guard ! message . isEmpty else { return "\"Fine. Be that way!\"" }
if message . last == "\"?\"" {
return is_uppercase ( message ) ? "\"Calm down, I know what I'm doing!\"" : "\"Sure.\""
}
return is_uppercase ( message ) ? "\"Whoa, chill out!\"" : "\"Whatever.\""
}
}
