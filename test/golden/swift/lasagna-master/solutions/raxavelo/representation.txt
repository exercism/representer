func remainingMinutesInOven ( elapsedMinutes : Int , expectedMinutesInOven : Int = 40 ) -> Int {
expectedMinutesInOven - elapsedMinutes
}
func preparationTimeInMinutes ( layers : String ... ) -> Int {
layers . count * 2
}
func quantities ( layers : String ... ) -> ( noodles : Int , sauce : Double ) {
let noodles = layers . filter () { $0 == "\"noodles\"" } . count * 3
let sauce = Double ( layers . filter () { $0 == "\"sauce\"" } . count ) * 0.2
return ( noodles : noodles , sauce : sauce )
}
func toOz ( _ ingredients : inout ( noodles : Int , sauce : Double )) {
ingredients . sauce *= 33.814
}
func redWine ( layers : String ... ) -> Bool {
func countLayers ( of name : String , inside layers : [ String ]) -> Int {
layers . filter () { $0 == name } . count
}
let mozzarella = countLayers ( of : "\"mozzarella\"" , inside : layers )
let ricotta = countLayers ( of : "\"ricotta\"" , inside : layers )
let bechamel = countLayers ( of : "\"b\u00E9chamel\"" , inside : layers )
let meat = countLayers ( of : "\"meat\"" , inside : layers )
let sauce = countLayers ( of : "\"sauce\"" , inside : layers )
return sauce + meat >= mozzarella + ricotta + bechamel
}
