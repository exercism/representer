struct Anagram {
let word : String
func match (
_ candidates : [ String ]) -> [ String ]
{
return candidates
. filter ({ $0 . lowercased () . isAnagram ( of : word . lowercased ()) })
}
}
extension String {
func isAnagram ( of other : String ) -> Bool {
guard self != other else { return false }
return self . sorted () == other . sorted ()
}
}
