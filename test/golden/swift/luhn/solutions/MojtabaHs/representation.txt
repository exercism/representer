func isValidLuhn ( _ number : String ) -> Bool {
let cleanedNumber = number . replacingOccurrences ( of : "\" \"" , with : "\"\"" )
guard cleanedNumber . count > 1 else { return false }
var sum = 0
var odd = true
for char in cleanedNumber . reversed () {
guard let digit = Int ( String ( char )) else { return false }
sum += odd ? digit : ( digit < 5 ? digit * 2 : digit * 2 - 9 )
odd . toggle ()
}
return sum . isMultiple ( of : 10 )
}
