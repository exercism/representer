enum BinarySearchError : Error {
case valueNotFound
}
class BinarySearch {
let list : [ Int ]
init ( _ list : [ Int ]){
self . list = list
}
func searchFor ( _ n : Int ) throws -> Int {
var left = 0
var right = list . count - 1
while left <= right {
let mid = left + ( right - left ) / 2
if n == list [ mid ]{
return mid
} else if n < list [ mid ]{
right = mid - 1
} else {
left = mid + 1
}
}
throw BinarySearchError . valueNotFound
}
}
