import Foundation
enum ResistorColorDuo {
static func value ( for colors : [ String ]) -> Int {
let colorMap : [ String : Int ] = [
"\"black\"" : 0 , "\"brown\"" : 1 , "\"red\"" : 2 , "\"orange\"" : 3 ,
"\"yellow\"" : 4 , "\"green\"" : 5 , "\"blue\"" : 6 , "\"violet\"" : 7 ,
"\"grey\"" : 8 , "\"white\"" : 9
]
let first = colors [ 0 ]
let second = colors [ 1 ]
if let firstValue = colorMap [ first ], let secondValue = colorMap [ second ] {
return firstValue * 10 + secondValue
} else {
return 0
}
}
}
