(file "triangle.vim")
function ! Equilateral ( triangle ) abort
let [ a , b , c ] = a:triangle
return a == b && a == c && a > 0
endfunction
function ! Isosceles ( triangle ) abort
let [ a , b , c ] = a:triangle
return s:IsValidTriangle ( a:triangle ) && ( a == b || b == c || a == c )
endfunction
function ! Scalene ( triangle ) abort
return s:IsValidTriangle ( a:triangle ) && ! Isosceles ( a:triangle )
endfunction
function s:IsValidTriangle ( triangle ) abort
let [ a , b , c ] = a:triangle
return a + b > c && a + c > b && b + c > a
endfunction
