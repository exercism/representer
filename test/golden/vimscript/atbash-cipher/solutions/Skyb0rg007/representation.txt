(file "atbash_cipher.vim")
function ! s:transpose ( str ) abort
return list2str ( map ( str2list ( a:str ), "\"48 <= v:val && v:val <= 57 ? v:val : 219 - v:val\"" ))
endfunction
function ! AtbashDecode ( cipher ) abort
return s:transpose ( substitute ( a:cipher , "\" \"" , "\"\"" , "\"g\"" ))
endfunction
function ! AtbashEncode ( plaintext ) abort
let clean = substitute ( a:plaintext , "\"[^a-zA-Z0-9]\"" , "\"\"" , "\"g\"" )
return substitute ( s:transpose ( tolower ( clean )), "\"......\\\\@=\"" , "\"& \"" , "\"g\"" )
endfunction
