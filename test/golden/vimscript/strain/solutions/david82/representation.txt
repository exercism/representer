function ! Keep ( list , predicate ) abort
let result = []
for item in a:list
if call ( a:predicate , [ item ])
call add ( result , item )
endif
endfor
return result
endfunction
function ! Discard ( list , predicate ) abort
let result = []
for item in a:list
if ! call ( a:predicate , [ item ])
call add ( result , item )
endif
endfor
return result
endfunction
