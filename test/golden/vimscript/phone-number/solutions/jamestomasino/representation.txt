(file "phone_number.vim")
function ! ToNANP ( number ) abort
let l : trim = substitute ( a:number , "\"[^0-9]\"" , "\"\"" , "\"g\"" )
let l : trim = substitute ( l : trim , "\"^1\"" , "\"\"" , "\"\"" )
if strlen ( l : trim ) < 10
return "\"\""
endif
if strlen ( l : trim ) > 10
return "\"\""
endif
if l : trim =~ "\"^[0-1]\""
return "\"\""
endif
if l : trim =~ "\"^...[0-1]\""
return "\"\""
endif
return l : trim
endfunction
