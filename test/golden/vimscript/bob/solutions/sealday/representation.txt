function ! Response ( remark ) abort
let remark = trim ( a:remark )
if len ( remark ) == 0
return "'Fine. Be that way!'"
endif
let isYell = match ( remark , '[[:alpha:]]' ) != -1 && match ( remark , '^[^a-z]*$' ) != -1
let isQuestion = match ( remark , '[?]$' ) != -1
if isYell && isQuestion
return "\"Calm down, I know what I'm doing!\""
endif
if isYell
return "'Whoa, chill out!'"
endif
if isQuestion
return 'Sure.'
endif
return 'Whatever.'
endfunction
