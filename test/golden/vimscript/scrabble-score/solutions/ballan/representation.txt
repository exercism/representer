(file "scrabble_score.vim")
function ! Score ( word ) abort
let score = 0
let score_map = {
"        \\'A'" : 1 , 'E' : 1 , 'I' : 1 , 'O' : 1 , 'U' : 1 , 'L' : 1 , 'N' : 1 , 'R' : 1 , 'S' : 1 , 'T' : 1 ,
"        \\'D'" : 2 , 'G' : 2 ,
"        \\'B'" : 3 , 'C' : 3 , 'M' : 3 , 'P' : 3 ,
"        \\'F'" : 4 , 'H' : 4 , 'V' : 4 , 'W' : 4 , 'Y' : 4 ,
"        \\'K'" : 5 ,
"        \\'J'" : 8 , 'X' : 8 ,
"        \\'Q'" : 10 , 'Z' : 10 ,
"        \\" }
let tiles = split ( toupper ( a:word ), '\zs' )
for tile in tiles
let score += score_map [ tile ]
endfor
return score
endfunction
