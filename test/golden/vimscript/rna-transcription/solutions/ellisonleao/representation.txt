(file "rna_transcription.vim")
function ! ToRna ( strand ) abort
let s:strand = toupper ( a:strand )
let s:mapping = { 'G' : 'C' , 'C' : 'G' , 'T' : 'A' , 'A' : 'U' }
let s:out = ''
for char in split ( s:strand , '\zs' )
let s:out .= get ( s:mapping , char , '' )
endfor
if len ( s:out ) != len ( s:strand )
return ''
endif
return s:out
endfunction
