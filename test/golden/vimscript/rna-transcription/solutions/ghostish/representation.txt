function ! ToRna ( strand ) abort
let l : complement = {
"\t\t\\'G'" : 'C' ,
"\t\t\\'C'" : 'G' ,
"\t\t\\'A'" : 'U' ,
"\t\t\\'T'" : 'A' ,
"        \\" }
let l : res = join ( map ( range ( strlen ( a:strand )), { i -> get ( l : complement , a:strand [ i ], "\"\"" )}), "\"\"" )
if strlen ( l : res ) == strlen ( a:strand )
return l : res
else
return "\"\""
endif
endfunction
