let anagrams = ( target : string , candidates : list ( string )) : list ( string ) => {
let sortString = ( xs : string ) : string =>
xs
-> Js . String . toLowerCase
-> Js . String . split ( "\"\"" , _ )
-> Js . Array . sortInPlace ( _ )
-> Js . Array . joinWith ( "\"\"" , _ );
let sortedTarget = sortString ( target );
let isAnagram = candidate =>
sortString ( candidate ) == sortedTarget
&& Js . String . toLowerCase ( candidate ) != Js . String . toLowerCase ( target )
? Some ( candidate ) : None ;
candidates -> Belt . List . keepMap ( isAnagram );
};
