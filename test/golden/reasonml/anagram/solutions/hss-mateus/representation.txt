(file "src/Anagram.re")
open Js ;
let anagrams = x => {
let normalize = y =>
y |> String . toLowerCase |> String . split ( "\"\"" ) |> Array . sortInPlace ;
let isAnagram =
( . y ) =>
normalize ( x ) == normalize ( y )
&& String . toLowerCase ( x ) != String . toLowerCase ( y );
List . filter ( isAnagram );
};
