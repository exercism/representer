let num_digits = ( s , n ) => String . length ( s ) == n ;
let valid_num_digits = ( s ) => ( num_digits ( s , 10 ) || num_digits ( s , 11 )) ? s : "\"\"" ;
let valid_country_code = ( s ) => {
if ( num_digits ( s , 11 )) {
( s . [ 0 ] == '1' ) ? String . sub ( s , 1 , 10 ) : "\"\"" ;
} else {
s ;
}
};
let valid_area_code = ( s ) => ( s != "\"\"" && s . [ 0 ] != '0' && s . [ 0 ] != '1' ) ? s : "\"\"" ;
let format_number = ( s ) => s != "\"\"" ? Some ( s ) : None ;
let phoneNumber = ( str ) => {
str |> Js . String . replaceByRe ([ % re "\"/\\\\D/g\"" ] , "\"\"" )
|> valid_num_digits
|> valid_country_code
|> valid_area_code
|> format_number ;
};
