(file "src/RnaTranscription.re")
type dna =
| A
| C
| G
| T ;
type rna =
| A
| C
| G
| U ;
let toRna = ( l : list ( dna )) : list ( rna ) => {
let rec aux = (( curr : list ( dna )) , ( res : list ( rna ))) : list ( rna ) => {
switch ( curr ) {
| [] => Js . List . rev ( res );
| [ h ,... t ] => {
switch ( h ) {
| A => aux ( t , [ U , ... res ]);
| C => aux ( t , [ G , ... res ]);
| G => aux ( t , [ C , ... res ]);
| T => aux ( t , [ A , ... res ]);
};
};
};
};
aux ( l , [] );
};
