let isLeapYear = year => {
let is4thYear = year mod 4 == 0 ;
let is100thYear = year mod 100 == 0 ;
let is400thYear = year mod 400 == 0 ;
switch ( is4thYear , is100thYear , is400thYear ) {
| ( true , false , false ) => true
| ( true , true , false ) => false
| ( true , true , true ) => true
| _ => false
};
};
