open Js ;
open Belt . Set . String ;
let is_isogram = x => {
let filtered =
String . toLowerCase ( x )
|> String . split ( "\"\"" )
|> Array . filter ( y => y != "\"-\"" && y != "\" \"" );
let uniques = fromArray ( filtered ) |> size ;
Array . length ( filtered ) == uniques ;
};
