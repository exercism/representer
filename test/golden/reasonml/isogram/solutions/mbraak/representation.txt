let create_letter_map = ( s : string ) => {
Js . String . split ( "\"\"" , s )
|> Array . to_list
|> List . fold_left (
( letter_map , letter : string ) => {
switch ( Js . Dict . get ( letter_map , letter )) {
| None => Js . Dict . set ( letter_map , letter , 1 )
| Some ( count ) => Js . Dict . set ( letter_map , letter , count + 1 )
};
letter_map ;
} ,
Js . Dict . empty ()
)
}
let filter_alphanumeric : string => string =
Js . String . replaceByRe ([ % re "\"/[^a-zA-Z]/g\"" ] , "\"\"" );
let is_isogram = s => {
let letter_map = String . lowercase ( s ) |> filter_alphanumeric |> create_letter_map ;
let counts = Js . Dict . values ( letter_map ) |> Array . to_list ;
List . for_all ( count => count == 1 , counts )
}
