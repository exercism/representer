(file "src/Isogram.re")
let moreThanOne = ( str : string , letter : string ) =>
( Js . String . split ( letter , str ) |> Js . Array . length ) - 1 > 1 ;
let parseToIsogram = str =>
str |> Js . String . replaceByRe ([ % re "\"/-| /\"" ] , "\"\"" ) |> Js . String . toLowerCase ;
let is_isogram = str => {
let parsedStr = str |> parseToIsogram ;
! Js . Array . some ( moreThanOne ( parsedStr ) , Js . String . split ( "\"\"" , parsedStr ));
};
