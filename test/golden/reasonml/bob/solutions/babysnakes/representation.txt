(file "src/Bob.re")
let hey = input => {
let isQuestion = s => Js . Re . test ( s , [ % re "\"/" \ ? \ s*$/" ]);
let isYelling = s =>
Js . String . toUpperCase ( s ) === s && Js . Re . test ( s , [ % re "\"/[A-Z]/\"" ]);
let isSayNothing = s => Js . Re . test ( s , [ % re "\"/^" \ s*$/" ]);
switch ( input ) {
| s when isYelling ( s ) && isQuestion ( s ) => "\"Calm down, I know what I'm doing!\""
| s when isQuestion ( s ) => "\"Sure.\""
| s when isYelling ( s ) => "\"Whoa, chill out!\""
| s when isSayNothing ( s ) => "\"Fine. Be that way!\""
| _ => "\"Whatever.\""
};
};
