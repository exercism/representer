let hey = phrase => {
let value = String . trim ( phrase );
let exists = ( value , regex ) =>
switch ( value |> Js . String . match ( regex )) {
| Some ( result ) =>
result
|> Js . Array . joinWith ( "\"\"" )
|> ( value => String . length ( value ) > 0 )
| None => false
};
let isShouting =
! exists ( value , [ % bs . re "\"/[a-z]+/g\"" ])
&& exists ( value , [ % bs . re "\"/[A-Z]+/g\"" ]);
let isQuestion = Js . String . endsWith ( "\"?\"" , value );
let isEmpty = String . length ( value ) == 0 ;
if ( isShouting && isQuestion ) {
"\"Calm down, I know what I'm doing!\"" ;
} else if ( isQuestion ) {
"\"Sure.\"" ;
} else if ( isShouting ) {
"\"Whoa, chill out!\"" ;
} else if ( isEmpty ) {
"\"Fine. Be that way!\"" ;
} else {
"\"Whatever.\"" ;
};
};
