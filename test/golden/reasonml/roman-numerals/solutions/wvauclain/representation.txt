let makeRomanDigit = ( ~ ones : string , ~ fives = "\"\"" , ~ tens = "\"\"" , digitNum : int ) : string =>
switch ( digitNum ) {
| 1 => ones
| 2 => Js . String . repeat ( 2 , ones )
| 3 => Js . String . repeat ( 3 , ones )
| 4 => ones ++ fives
| 5 => fives
| 6 => fives ++ ones
| 7 => fives ++ Js . String . repeat ( 2 , ones )
| 8 => fives ++ Js . String . repeat ( 3 , ones )
| 9 => ones ++ tens
| _ => "\"\""
};
let toRoman = ( num : int ) : string =>
makeRomanDigit ( ~ ones = "\"M\"" , num / 1000 mod 10 )
++ makeRomanDigit ( ~ ones = "\"C\"" , ~ fives = "\"D\"" , ~ tens = "\"M\"" , num / 100 mod 10 )
++ makeRomanDigit ( ~ ones = "\"X\"" , ~ fives = "\"L\"" , ~ tens = "\"C\"" , num / 10 mod 10 )
++ makeRomanDigit ( ~ ones = "\"I\"" , ~ fives = "\"V\"" , ~ tens = "\"X\"" , num mod 10 );
