type planet =
| Mercury
| Venus
| Earth
| Mars
| Jupiter
| Saturn
| Neptune
| Uranus ;
let earth = 31557600 . 0 ;
let mercury = earth *. 0 . 2408467 ;
let venus = earth *. 0 . 61519726 ;
let mars = earth *. 1 . 8808158 ;
let jupiter = earth *. 11 . 862615 ;
let saturn = earth *. 29 . 447498 ;
let uranus = earth *. 84 . 016846 ;
let neptune = earth *. 164 . 79132 ;
let ageOn = ( planet , seconds ) =>
switch ( planet ) {
| Mercury => seconds /. mercury
| Venus => seconds /. venus
| Earth => seconds /. earth
| Mars => seconds /. mars
| Jupiter => seconds /. jupiter
| Saturn => seconds /. saturn
| Neptune => seconds /. neptune
| Uranus => seconds /. uranus
};
