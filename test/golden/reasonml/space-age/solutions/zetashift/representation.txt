type planet =
| Mercury
| Venus
| Earth
| Mars
| Jupiter
| Saturn
| Neptune
| Uranus ;
module PlanetMap = Map . Make ({
type t = planet ;
let compare = Pervasives . compare
});
let m = PlanetMap .( empty |> add ( Mercury , 0 . 2408467 )
|> add ( Venus , 0 . 61519726 )
|> add ( Earth , 1 . )
|> add ( Mars , 1 . 8808158 )
|> add ( Jupiter , 11 . 862615 )
|> add ( Saturn , 29 . 447498 )
|> add ( Uranus , 84 . 016846 )
|> add ( Neptune , 164 . 79132 ));
let ageOn = ( planet , seconds ) => {
let earthYears = seconds /. 31557600 . 0 ;
earthYears /. PlanetMap . find ( planet , m );
}
