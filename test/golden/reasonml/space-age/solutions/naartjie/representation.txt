type planet =
| Mercury
| Venus
| Earth
| Mars
| Jupiter
| Saturn
| Neptune
| Uranus ;
let earthPeriod = 31557600 . 0 ;
let ageOn : ( planet , float ) => float =
( planet , seconds ) =>
seconds
/. (
earthPeriod
*. (
switch ( planet ) {
| Mercury => 0 . 2408467
| Venus => 0 . 61519726
| Earth => 1 . 0
| Mars => 1 . 8808158
| Jupiter => 11 . 862615
| Saturn => 29 . 447498
| Neptune => 164 . 79132
| Uranus => 84 . 016846
}
)
);
