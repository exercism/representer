type ordering =
| LT
| EQ
| GT ;
let ( <=> ) = ( l , r ) => l < r ? LT : l == r ? EQ : GT ;
let rec binarySearch = ( haystack , needle , low , high ) =>
if ( low > high ) {
None ;
} else {
let mid = ( low + high ) / 2 ;
switch ( needle <=> haystack [ mid ]) {
| LT => binarySearch ( haystack , needle , low , mid - 1 )
| EQ => Some ( mid )
| GT => binarySearch ( haystack , needle , mid + 1 , high )
};
};
let find = ( haystack , needle ) =>
binarySearch ( haystack , needle , 0 , Array . length ( haystack ) - 1 );
