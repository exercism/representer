(file "src/Change.re")
type coins = list ( int );
let rec makeChange' = ( amount , coins , memo ) => {
let amountString = amount |> string_of_int ;
let coinsString =
coins |> List . map ( string_of_int ) |> List . fold_left (( ++ ) , "\"\"" );
let key = amountString ++ "\"-\"" ++ coinsString ;
switch ( memo -> Js . Dict . get ( key )) {
| Some ( change ) => Some ( change )
| None =>
let change =
switch ( coins ) {
| [ hd , ... tail ] =>
if ( amount == hd ) {
Some ([ amount ]);
} else if ( amount > hd ) {
let r1 =
makeChange' ( amount - hd , coins , memo )
-> Belt . Option . map ( lst => lst |> List . cons ( hd ));
let r2 = makeChange' ( amount , tail , memo );
switch ( r1 , r2 ) {
| ( Some ( r1 ) , Some ( r2 )) =>
if ( r1 -> List . length < r2 -> List . length ) {
Some ( r1 );
} else {
Some ( r2 );
}
| ( None , Some ( r2 )) => Some ( r2 )
| ( Some ( r1 ) , None ) => Some ( r1 )
| ( None , None ) => None
};
} else {
makeChange' ( amount , tail , memo );
}
| [] => None
};
change -> Belt . Option . forEach ( memo -> Js . Dict . set ( key ));
change ;
};
};
let makeChange = ( amount , coins ) =>
if ( amount > 0 ) {
makeChange' ( amount , coins -> List . rev , Js . Dict . empty () )
-> Belt . Option . map ( List . rev );
} else if ( amount == 0 ) {
Some ([]);
} else {
None ;
};
