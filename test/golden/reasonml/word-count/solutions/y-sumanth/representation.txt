(file "src/WordCount.re")
open Js . String ;
let wordCount = str => {
let words =
str
|> toLowerCase
|> replaceByRe ([ % re "\"/[^'a-zA-Z0-9]+/g\"" ] , "\" \"" )
|> replaceByRe ([ % re "\"/'?\\\\s+'?/g\"" ] , "\" \"" )
|> trim
|> split ( "\" \"" );
let countedWords = Js . Dict . empty () ;
words
-> Belt . Array . map ( word =>
Js . Dict . set (
countedWords ,
word ,
Belt . Option . getWithDefault ( Js . Dict . get ( countedWords , word ) , 0 ) + 1 ,
)
);
countedWords ;
};
wordCount ( "\"car: carpet as java: javascript!!&@$%^&\"" );
