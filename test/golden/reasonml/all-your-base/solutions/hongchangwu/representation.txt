open Belt ;
let ipow = ( base , exp ) => {
let base = ref ( base );
let exp = ref ( exp );
let result = ref ( 1 );
try (
while ( true ) {
if ( exp ^ land 1 != 0 ) {
result := result ^ * base ^;
};
exp := exp ^ lsr 1 ;
if ( exp ^ == 0 ) {
raise ( End_of_file );
};
base := base ^ * base ^;
}
) {
| End_of_file => ()
};
result ^;
};
let rec ofDigits = ( digits , base ) =>
switch ( digits , base ) {
| ([ 0 , ... digits' ] , base ) => ofDigits ( digits' , base )
| ([] , _ ) => None
| ( _ , base ) when base <= 0 => None
| ( digits , base ) =>
let rec aux = ( acc , i ) => {
(
fun
| [] => acc
| [ digit , ..._ ] when digit < 0 || digit >= base => None
| [ digit , ... digits' ] => {
let acc' = Option . map ( acc , acc => acc + digit * ipow ( base , i ));
let i' = succ ( i );
aux ( acc' , i' , digits' );
}
);
};
aux ( Some ( 0 ) , 0 , List . reverse ( digits ));
};
let toDigits = ( num , base ) => {
switch ( num , base ) {
| ( num , _ ) when num < 0 => None
| ( _ , base ) when base <= 0 => None
| ( num , base ) =>
let rec aux = acc => {
(
fun
| 0 => acc
| num => {
let acc' = [ num mod base , ... acc ];
let num' = num / base ;
aux ( acc' , num' );
}
);
};
Some ( aux ([] , num ));
};
};
let rebase = ( fromBase , digits , toBase ) => {
digits -> ofDigits ( fromBase ) -> Option . flatMap ( toDigits ( _ , toBase ));
};
