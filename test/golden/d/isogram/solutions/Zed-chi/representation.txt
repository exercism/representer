module isogram ;
import std ;
bool isIsogram ( string text ){
text = toLower ( text );
string chs = "\"!@\#$%^& *()-=,.\"" ;
char [] set ;
for ( int i ; i < text . length ; i ++){
if ( count ( text , text [ i ]) > 1 && count ( chs , text [ i ]) == 0 ) {
return false ;
}
}
return true ;
}
