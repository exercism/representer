module collatz_conjecture ;
int steps ( int a )
{
int cnt = 0 ;
if ( a <= 0 )
{
throw new Exception ( "\"msg\"" );
}
else if ( a == 1 )
{
return 0 ;
}
else
{
while ( a != 1 )
{
if ( a % 2 == 0 ) {
a = a / 2 ;
} else {
a = 3 * a + 1 ;
}
cnt += 1 ;
}
}
return cnt ;
}
unittest
{
import std . exception : assertThrown ;
const int allTestsEnabled = 0 ;
assert ( steps ( 1 ) == 0 );
static if ( allTestsEnabled )
{
assert ( steps ( 16 ) == 4 );
assert ( steps ( 12 ) == 9 );
assert ( steps ( 1000000 ) == 152 );
assertThrown ( steps ( 0 ));
assertThrown ( steps (- 15 ));
}
}
