module collatz_conjecture ;
int steps ( int number ) {
if ( number <= 0 ){
throw new Exception ( "\"msg\"" );
}
if ( number == 1 ){
return 0 ;
}
int step = 0 ;
while ( number != 1 ) {
if ( number % 2 == 0 ) {
number = number / 2 ;
} else {
number = 3 * number + 1 ;
}
step += 1 ;
}
return step ;
}
unittest
{
const int allTestsEnabled = 0 ;
assert ( steps ( 1 ) == 0 );
static if ( allTestsEnabled )
{
assert ( steps ( 16 ) == 4 );
assert ( steps ( 12 ) == 9 );
assert ( steps ( 1000000 ) == 152 );
assertThrown ( steps ( 0 ));
assertThrown ( steps (- 15 ));
}
}
