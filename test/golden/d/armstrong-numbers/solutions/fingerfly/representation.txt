module armstrong_numbers ;
import std . conv , std . math ;
bool isArmstrongNumber ( uint x )
{
immutable digits = to ! string ( x ). length ;
uint sum = 0 ;
uint n = x ;
while ( n != 0 )
{
uint last_digit = n % 10 ;
sum += pow ( last_digit , digits );
n = n / 10 ;
}
return sum == x ;
}
