pure long answer ( immutable string question )
{
if (! question . startsWith ( "\"What is \"" ) || ! question . endsWith ( "\"?\"" ))
{
throw new Exception ( "\"Invalid question format\"" );
}
string content = question [ 8 .. $ - 1 ];
auto tokens = content . split ();
if ( tokens . empty )
{
throw new Exception ( "\"No tokens found\"" );
}
long result ;
size_t i = 0 ;
try
{
result = tokens [ i ]. to ! long ;
i ++;
}
catch ( ConvException )
{
throw new Exception ( "\"First token is not a valid number\"" );
}
while ( i < tokens . length )
{
string op = tokens [ i ];
i ++;
if ( op == "\"multiplied\"" || op == "\"divided\"" )
{
if ( i >= tokens . length || tokens [ i ] != "\"by\"" )
{
throw new Exception ( "\"Invalid syntax for multi-word operation\"" );
}
op ~= "\" by\"" ;
i ++;
}
if ( i >= tokens . length )
{
throw new Exception ( "\"Missing operand after operation\"" );
}
long operand ;
try
{
operand = tokens [ i ]. to ! long ;
i ++;
}
catch ( ConvException )
{
throw new Exception ( "\"Expected a number after operation\"" );
}
switch ( op )
{
case "\"plus\"" :
result += operand ;
break ;
case "\"minus\"" :
result -= operand ;
break ;
case "\"multiplied by\"" :
result *= operand ;
break ;
case "\"divided by\"" :
result /= operand ;
break ;
default :
throw new Exception ( "\"Unknown operation: \"" ~ op );
}
}
return result ;
}
