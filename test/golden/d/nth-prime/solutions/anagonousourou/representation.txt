module nth_prime ;
int prime ( int n )
in
{
if ( n <= 0 )
{
throw new Exception ( "\"parameter must be superior to zero.\"" );
}
}
do
{
int counter = 0 ;
for ( int i = 2 ;; i ++)
{
if ( isPrime ( i ))
{
counter ++;
}
if ( counter == n )
{
return i ;
}
}
}
bool isPrime ( long n )
{
if ( n <= 1 )
{
return false ;
}
int limit = cast ( int ) sqrt ( cast ( float ) n );
for ( int i = 2 ; i <= limit ; i ++)
{
if ( n % i == 0 )
{
return false ;
}
}
return true ;
}
unittest
{
immutable int allTestsEnabled = 1 ;
assert ( prime ( 1 ) == 2 );
static if ( allTestsEnabled )
{
assert ( prime ( 2 ) == 3 );
assert ( prime ( 6 ) == 13 );
assert ( prime ( 10_001 ) == 10_4743 );
assertThrown ( prime ( 0 ));
}
}
