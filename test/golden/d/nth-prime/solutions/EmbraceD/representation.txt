module nth_prime ;
uint prime ( in uint n )
{
if (! n )
{
throw new Exception ( "\"Start counting at one, comedian!\"" );
}
uint primes ;
uint number ;
while ( primes < n )
{
++ number ;
if ( isPrime ( number ))
{
++ primes ;
}
}
import std . stdio ;
writeln ( number );
return number ;
}
bool isPrime ( in uint n )
{
if ( n < 2 )
{
return false ;
}
for ( uint i = 2 ; i < n ; ++ i )
{
if (!( n % i ))
{
return false ;
}
}
return true ;
}
unittest
{
import std . exception : assertThrown ;
immutable int allTestsEnabled = 0 ;
assert ( prime ( 1 ) == 2 );
static if ( allTestsEnabled )
{
assert ( prime ( 2 ) == 3 );
assert ( prime ( 6 ) == 13 );
assert ( prime ( 10_001 ) == 10_4743 );
assertThrown ( prime ( 0 ));
}
}
