module pythagorean_triplet ;
import std . numeric : gcd ;
import std . algorithm . sorting : sort ;
struct Triplet {
uint a ;
uint b ;
uint c ;
}
pure Triplet [] tripletsWithSum ( uint total )
{
Triplet [] triples ;
for ( uint m = 2 ; 2 * m * ( m + 1 ) <= total ; ++ m ) {
for ( uint n = 1 + m % 2 ; n < m ; n += 2 ) {
if ( gcd ( m , n ) != 1 ) {
continue ;
}
uint a = m * m - n * n ;
uint b = 2 * m * n ;
uint c = m * m + n * n ;
uint p = a + b + c ;
if ( total % p == 0 ) {
uint q = total / p ;
if ( a > b ) {
uint t = a ;
a = b ;
b = t ;
}
triples ~= Triplet ( q * a , q * b , q * c );
}
}
}
triples . sort !( "\"a.a < b.a\"" );
return triples ;
}
