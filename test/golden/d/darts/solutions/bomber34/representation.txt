(file "source/darts.d")
module darts ;
import std . math ;
immutable double outerRegionRadius = 10 ;
immutable double middleRegionRadius = 5 ;
immutable double innerRegionRadius = 1 ;
int score ( double x , double y ) {
double euclideanDistanceFromOrigin = sqrt ( x * x + y * y );
int outerRegionScore = euclideanDistanceFromOrigin <= outerRegionRadius ? 1 : 0 ;
int middleRegionExtraScore = euclideanDistanceFromOrigin <= middleRegionRadius ? 4 : 0 ;
int innerRegionExtraScore = euclideanDistanceFromOrigin <= innerRegionRadius ? 5 : 0 ;
return outerRegionScore + middleRegionExtraScore + innerRegionExtraScore ;
}
unittest
{
immutable int allTestsEnabled = 0 ;
assert ( score (- 9 , 9 ) == 0 );
static if ( allTestsEnabled )
{
assert ( score ( 0 , 10 ) == 1 );
assert ( score (- 5 , 0 ) == 5 );
assert ( score ( 0 , - 1 ) == 10 );
assert ( score ( 0 , 0 ) == 10 );
assert ( score (- 0.1 , - 0.1 ) == 10 );
assert ( score ( 0.7 , 0.7 ) == 10 );
assert ( score ( 0.8 , - 0.8 ) == 5 );
assert ( score (- 3.5 , 3.5 ) == 5 );
assert ( score (- 3.6 , - 3.6 ) == 1 );
assert ( score (- 7.0 , 7.0 ) == 1 );
assert ( score ( 7.1 , - 7.1 ) == 0 );
assert ( score ( 0.5 , - 4 ) == 5 );
}
}
