module crypto ;
import std . ascii : isDigit , isAlpha , toLower ;
import std . string : strip ;
import std . array : appender ;
import std . conv : to ;
import std . algorithm : min ;
class Cipher
{
string text ;
int r , c ;
this ( string st )
{
text = st ;
doNormalize ();
doCalculateSize ();
}
void doNormalize ()
{
auto w = appender ! string ;
w . reserve ( 100 );
foreach ( ch ; text )
{
if ( isDigit ( ch ))
{
w ~= ch ;
}
else if ( isAlpha ( ch ))
{
w ~= toLower ( ch );
}
}
text = w . data ;
}
string normalizePlainText ()
{
return text ;
}
void doCalculateSize ()
{
foreach ( n ; 2 .. 100 )
{
if ( n * n >= text . length )
{
r = n ;
break ;
}
}
}
int size ()
{
return r ;
}
string [] plainTextSegments ()
{
string [] res ;
int len , i ;
len = text . length . to ! int ;
for ( i = 0 ; len > 0 ;)
{
int w = min ( i + r , text . length );
res ~= text [ i .. w ];
len -= r ;
i += r ;
}
return res ;
}
string doCipher ( string pad )
{
auto w = appender ! string ;
immutable auto len = text . length ;
ulong ii ;
foreach ( i ; 0 .. r )
{
foreach ( j ; 0 .. r )
{
ii = j * r + i ;
if ( ii >= len )
{
w ~= pad ;
break ;
}
w ~= text [ ii ];
}
}
return strip ( w . data );
}
string cipherText ()
{
return doCipher ( "\"\"" );
}
string normalizedCipherText ()
{
return doCipher ( "\" \"" );
}
}
