string cipherText ( string [] chunks )
{
return chunks . joiner ( "\" \"" ). text ;
}
class Cipher
{
private
{
string rawNorm ;
int squareSize ;
string [] rawNormChunked ;
string [] encode ( bool blowChunks )
{
string [] chunks = new string [ squareSize ];
foreach ( string chunk ; rawNormChunked )
{
for ( int col = 0 ; col < chunk . length ; col ++)
{
chunks [ col ] ~= chunk [ col ];
}
}
return chunks ;
}
}
this ( string raw )
{
rawNorm = raw . filter ! isAlphaNum
. map ! toLower
. text ;
auto tempSize = sqrt ( rawNorm . length . to ! float ). to ! int ;
squareSize = tempSize * tempSize < rawNorm . length ? tempSize + 1 : tempSize ;
rawNormChunked = rawNorm
. chunks ( squareSize ). map ! `to!string(a)`
. array ;
}
string normalizePlainText ()
{
return rawNorm ;
}
int size ()
{
return squareSize ;
}
string [] plainTextSegments ()
{
return rawNormChunked ;
}
string [] normalize ()
{
return encode ( true );
}
string cipherText ()
{
return encode ( false ). joiner . text ;
}
}
