ulong sqrt ( ulong n ) {
ulong r = 1 , tmp1 , tmp2 ;
while ( true ) {
r = r / 2 + n /( 2 * r );
tmp1 = r * r ;
tmp2 = ( r + 1 )*( r + 1 );
if ( tmp1 <= n && tmp2 >= n ) { break ; }
}
return r ;
}
struct Cipher {
string s ;
string n ;
string ct ;
ulong r ;
ulong c ;
string [] chnks ;
string [] trans ;
auto normalizePlainText () { return n ; }
auto size () { return c ; }
auto plainTextSegments () { return chnks ; }
auto cipherText () { return ct ; }
Normalized normalize ;
static struct Normalized {
string ct ;
auto cipherText () { return ct ; }
}
this ( string s ) {
foreach ( x ; s ) {
if ( ( x >= 'a' && x <= 'z' )
|| ( x >= '0' && x <= '9' )
) {
n ~= x ;
}
else if ( x >= 'A' && x <= 'Z' ) {
n ~= cast ( char )( x + 'a' - 'A' );
}
}
c = r = sqrt ( n . length );
if ( c * r < n . length ) { c ++; }
if ( c * r < n . length ) { r ++; }
chnks = n . chunks ( c ). map !( to ! string ). array ;
assert ( chnks . length == r );
chnks [$- 1 ] ~= iota ( c - chnks [$- 1 ]. length ). map !( x => "' '" ). array . to ! string ;
char [][] tmp ;
tmp . length = c ;
foreach ( ref x ; tmp ) { x . length = r ; }
foreach ( i , row ; chnks ) {
foreach ( j , x ; row ) {
tmp [ j ][ i ] = x ;
}
}
foreach ( row ; tmp ) { trans ~= row . to ! string . strip ;}
chnks [$- 1 ] = chnks [$- 1 ]. strip ;
foreach ( i , row ; trans ) {
ct ~= row ;
normalize . ct ~= row ;
if ( i + 1 < trans . length ) { normalize . ct ~= "\" \"" ; }
}
}
}
