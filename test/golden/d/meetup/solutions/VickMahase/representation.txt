pure string meetup ( ushort year , Month month , Week week , DayOfWeek day )
{
int startDay ;
int endDay ;
final switch ( week )
{
case Week . first :
startDay = 1 ;
endDay = 7 ;
break ;
case Week . second :
startDay = 8 ;
endDay = 14 ;
break ;
case Week . third :
startDay = 15 ;
endDay = 21 ;
break ;
case Week . fourth :
startDay = 22 ;
endDay = 28 ;
break ;
case Week . teenth :
startDay = 13 ;
endDay = 19 ;
break ;
case Week . last :
int monthEnd = Date ( year , month , 1 ). daysInMonth ;
startDay = monthEnd - 6 ;
endDay = monthEnd ;
break ;
}
for ( int d = startDay ; d <= endDay ; d ++)
{
Date dt = Date ( year , month , d );
if ( dt . dayOfWeek == day )
{
return format ( "\"%04d-%02d-%02d\"" , year , cast ( int ) month , d );
}
}
return "\"\"" ;
}
