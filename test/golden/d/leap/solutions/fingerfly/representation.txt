(file "source/leap.d")
module leap ;
bool isLeap ( int year )
{
return year % 400 == 0 || ( year % 4 == 0 && year % 100 != 0 );
}
unittest
{
immutable int allTestsEnabled = 0 ;
assert (! isLeap ( 2015 ));
static if ( allTestsEnabled )
{
assert (! isLeap ( 1970 ));
assert ( isLeap ( 1996 ));
assert ( isLeap ( 1960 ));
assert (! isLeap ( 2100 ));
assert (! isLeap ( 1900 ));
assert ( isLeap ( 2000 ));
assert ( isLeap ( 2400 ));
assert (! isLeap ( 1800 ));
}
}
