module pangram ;
import std . ascii ;
pure bool isPangram ( immutable string str )
{
bool [ 26 ] seenLetters ;
foreach ( char c ; str )
{
char lowerC = c . toLower ;
if ( lowerC . isAlpha )
{
size_t index = lowerC - 'a' ;
seenLetters [ index ] = true ;
}
}
foreach ( bool seen ; seenLetters )
{
if (! seen )
{
return false ;
}
}
return true ;
}
