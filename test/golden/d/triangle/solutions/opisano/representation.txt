module triangle ;
enum TriangleType
{
equilateral ,
isosceles ,
scalene
}
TriangleType kind ( T )( T a , T b , T c ) pure
if ( isNumeric ! T )
{
if (! checkSize ( a , b , c ))
{
throw new Exception ( "\"Invalid size\"" );
}
if (! inequality ( a , b , c ))
{
throw new Exception ( "\"Not a triangle\"" );
}
if ( a == b && b == c )
{
return TriangleType . equilateral ;
}
else if ( a == b || a == c || b == c )
{
return TriangleType . isosceles ;
}
return TriangleType . scalene ;
}
private bool checkSize ( T )( T a , T b , T c )
if ( isNumeric ! T )
{
return a > 0 && b > 0 && c > 0 ;
}
private bool inequality ( T )( T a , T b , T c )
if ( isNumeric ! T )
{
return ( a + b >= c ) && ( a + c >= b ) && ( b + c >= a );
}
