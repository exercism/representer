module triangle ;
import std . container . rbtree ;
import std . exception ;
enum TriangleType
{
equilateral ,
isosceles ,
scalene
}
@safe pure TriangleType kind ( float a , float b , float c ) {
enforce ( a > 0 && b > 0 && c > 0 , "\"One or more invalid values. Each must be greater than zero.\"" );
enforce ( a + b >= c && b + c >= a && c + a >= b , "\"One or more invalid values. Any two sides must be greater than or equal to the other.\"" );
switch ( redBlackTree ( a , b , c ). length ) {
case 1 : return TriangleType . equilateral ;
case 3 : return TriangleType . scalene ;
default : return TriangleType . isosceles ;
}
}
unittest
{
import std . exception : assertThrown ;
immutable int allTestsEnabled = 1 ;
{
assert ( TriangleType . equilateral == kind ( 2 , 2 , 2 ));
}
static if ( allTestsEnabled )
{
{
assert ( TriangleType . equilateral == kind ( 10 , 10 , 10 ));
}
{
assert ( TriangleType . isosceles == kind ( 3 , 4 , 4 ));
}
{
assert ( TriangleType . isosceles == kind ( 4 , 3 , 4 ));
}
{
assert ( TriangleType . isosceles == kind ( 4 , 4 , 3 ));
}
{
assert ( TriangleType . isosceles == kind ( 10 , 10 , 2 ));
}
{
assert ( TriangleType . scalene == kind ( 3 , 4 , 5 ));
}
{
assert ( TriangleType . scalene == kind ( 10 , 11 , 12 ));
}
{
assert ( TriangleType . scalene == kind ( 5 , 4 , 2 ));
}
{
assert ( TriangleType . scalene == kind ( 0.4 , 0.6 , 0.3 ));
}
{

}

}

}

}
}
