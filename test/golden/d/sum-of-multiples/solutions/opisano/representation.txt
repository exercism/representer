module sum_of_multiples ;
import std . algorithm . iteration : joiner , map , sum , uniq ;
import std . algorithm . sorting : sort ;
import std . range : array ;
struct Multiples
{
this ( int n , int limit )
{
this . n = n ;
this . current = n ;
this . limit = limit ;
}
bool empty ()
{
return n == 0 || current >= limit ;
}
int front ()
{
return current ;
}
void popFront ()
{
current += n ;
}
private :
int n ;
int current ;
int limit ;
}
int calculateSum ( int [] numbers , int limit )
{
auto result = numbers . map !( x => Multiples ( x , limit ))
. joiner
. array
. sort
. uniq
. sum ;
return result ;
}
