module anagram ;
pure string [] findAnagrams ( immutable string subject , immutable string [] candidates )
{
string lsubj = toLower ( subject );
auto key = sort ( lsubj . to !( dchar []));
string [] matches ;
foreach ( c ; candidates ) {
string lc = toLower ( c );
if ( lc != lsubj ) {
auto letters = sort ( lc . to !( dchar []));
if ( letters == key ) {
matches ~= c ;
}
}
}
return matches ;
}
