module grains ;
ulong square ( int n ) {
if ( n < 1 || n > 64 ) {
throw new Exception ( "\"Input must be between 1 and 64\"" );
}
return 2u L ^^ ( n - 1 );
}
ulong total () {
return ~ 0 ;
}

{
import std . exception : assertThrown ;
immutable int allTestsEnabled = 0 ;
assert ( square ( 1 ) == 1 );
static if ( allTestsEnabled )
{
assert ( square ( 2 ) == 2 );

assert ( square ( 4 ) == 8 );
