module prime_factors ;
pure long [] factors ( long num ) {
if ( num < 2 ) return [];
long [] primes ;
while ( num % 2 == 0 ) {
primes ~= 2 ;
num = num / 2 ;
}
for ( long i = 3 ; i <= cast ( long ) sqrt ( cast ( double ) num ); i = i + 2 ) {
while ( num % i == 0 ) {
primes ~= i ;
num = num / i ;
}
}
if ( num > 2 ) primes ~= num ;
return primes ;
}
unittest
{
immutable int allTestsEnabled = 1 ;
assert ( factors ( 1 ) == []);
static if ( allTestsEnabled )
{
assert ( factors ( 2 ) == [ 2 ]);
assert ( factors ( 9 ) == [ 3 , 3 ]);
assert ( factors ( 8 ) == [ 2 , 2 , 2 ]);
assert ( factors ( 12 ) == [ 2 , 2 , 3 ]);
assert ( factors ( 901_255 ) == [ 5 , 17 , 23 , 461 ]);
assert ( factors ( 93_819_012_551 ) == [ 11 , 9539 , 894_119 ]);
}
}
