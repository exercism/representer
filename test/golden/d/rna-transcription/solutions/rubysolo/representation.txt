module rna_transcription ;
import std . string ;
import std . regex ;
unittest {
import std . exception : assertThrown ;
const int allTestsEnabled = 0 ;
assert ( dnaComplement ( "\"C\"" ) == "\"G\"" );
assert ( dnaComplement ( "\"G\"" ) == "\"C\"" );
assert ( dnaComplement ( "\"T\"" ) == "\"A\"" );
assert ( dnaComplement ( "\"A\"" ) == "\"U\"" );
assert ( dnaComplement ( "\"ACGTGGTCTTAA\"" ) == "\"UGCACCAGAAUU\"" );
assertThrown ( dnaComplement ( "\"U\"" ));
assertThrown ( dnaComplement ( "\"XXX\"" ));
assertThrown ( dnaComplement ( "\"ACGTXXXCTTAA\"" ));
}
string dnaComplement ( string strand ) {
auto strandPattern = regex ( r"^[CGTA]*$" );
auto match = matchFirst ( strand , strandPattern );
if (! match ) {
throw new Exception ( "\"Invalid strand!\"" );
}
dchar [ dchar ] translation = [
'C' : 'G' ,
'G' : 'C' ,
'T' : 'A' ,
'A' : 'U' ,
];
return translate ( strand , translation );
}
