module circular ;
class Buffer ( T ) {
import std . exception : enforce ;
public :
this ( size_t size ) {
_buf = new T [ size ];
}
T pop () {
enforce (! _empty , "\"Buffer is empty\"" );
T val = _buf [ _begin ];
_begin = ( _begin + 1 ) % _buf . length ;
_empty = _begin == _end ;
return val ;
}
void push ( T val ) {
enforce (! isFull , "\"Buffer is full\"" );
_buf [ _end ] = val ;
_end = ( _end + 1 ) % _buf . length ;
_empty = false ;
}
void clear () {
_empty = true ;
_begin = _end = 0 ;
}
void forcePush ( T val ) {
if ( isFull ()) {
pop ();
}
push ( val );
}
private :
T [] _buf ;
size_t _begin , _end = 0 ;
bool _empty = true ;
bool isFull () { return ! _empty && _begin == _end ; }
}
unittest
{
import std . exception : assertThrown ;
{
auto myBuffer = new Buffer !( int )( 1U L );
assertThrown ( myBuffer . pop (), "\"Empty buffer should throw exception if popped!\"" );
}

auto myBuffer = new Buffer !( char )( 1 );
myBuffer . push ( '1' );
assert ( myBuffer . pop () == '1' );
}
{
