module circular ;
class Buffer ( T )
{
private immutable size_t capacity ;
private size_t size = 0 ;
private T [] ptr ;
private size_t inputIndex = 0 ;
private size_t outputIndex = 0 ;
this ( size_t capacity )
{
this . capacity = capacity ;
ptr = new T [ capacity ];
}
T pop ()
{
if ( size == 0 )
{
throw new Exception ( "\"Empty buffer\"" );
}
size --;
T value = ptr [ outputIndex ];
outputIndex = ( outputIndex + 1 ) % capacity ;
return value ;
}
void push ( T elt )
{
if ( size == capacity )
{
throw new Exception ( "\"Full buffer\"" );
}
ptr [ inputIndex ] = elt ;
inputIndex = ( inputIndex + 1 ) % capacity ;
size ++;
}
void clear ()
{
inputIndex = 0 ;
outputIndex = 0 ;
size = 0 ;
}
void forcePush ( T elt )
{
if ( size == capacity )
{
pop ();
}
push ( elt );
}
}
