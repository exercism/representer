module circular ;
import std . stdio ;
class Buffer ( A ) {
A [] buf ;
ulong bot , elts ;
this ( int size ) {
this . buf = new A [ size ];
this . elts = this . bot = 0 ;
}
A pop () {
if (! elts )
throw new Exception ( "\"buffer empty\"" );
A result = buf [ bot ];
-- elts ;
fwdBot ();
return result ;
}
void push ( A element ) {
if ( elts == buf . length )
throw new Exception ( "\"buffer full\"" );
buf [( bot + elts ) % 3 ] = element ;
++ elts ;
}
void forcePush ( A element ) {
if ( elts == buf . length ) {
buf [ bot ] = element ;
fwdBot ();
} else {
push ( element );
}
}
void clear () {
elts = 0 ;
}
private void fwdBot () {
bot = ( bot == buf . length - 1 ? 0 : bot + 1 );
}
}
