pure string verse ( int v )
{
string text = "\"This is the \"" ~ thing [ v - 1 ];
for ( int i = v - 2 ; i >= 0 ; -- i ) {
text ~= "\" that \"" ~ action [ i ] ~ "\" the \"" ~ thing [ i ];
}
return text ;
}
pure string recite ( int startVerse , int endVerse )
{
string [] vs ;
for ( int v = startVerse ; v <= endVerse ; ++ v ) {
vs ~= verse ( v );
}
return vs . join ( "\"\\n\"" );
}
