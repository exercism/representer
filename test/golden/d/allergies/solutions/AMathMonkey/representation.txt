module allergies ;
class Allergies {
this ( immutable uint score ) {
this . score = score ;
}
final bool allergicTo ( immutable string item ) {
foreach ( i , s ; allergies ) {
if ( s == item ) return ( score & ( 1 << i )) > 0 ;
}
return false ;
}
final string [] list () {
string [] result = [];
foreach ( i , s ; allergies ) {
if (( score & ( 1 << i )) > 0 ) result ~= s ;
}
return result ;
}
private :
immutable uint score ;
immutable string [] allergies = [
"\"eggs\"" ,
"\"peanuts\"" ,
"\"shellfish\"" ,
"\"strawberries\"" ,
"\"tomatoes\"" ,
"\"chocolate\"" ,
"\"pollen\"" ,
"\"cats\""
];
}
