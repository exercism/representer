module allergies ;
class Allergies
{
string [] allergies ;
this ( immutable uint score )
{
auto possibleAllergies = [
1 : "\"eggs\"" ,
2 : "\"peanuts\"" ,
4 : "\"shellfish\"" ,
8 : "\"strawberries\"" ,
16 : "\"tomatoes\"" ,
32 : "\"chocolate\"" ,
64 : "\"pollen\"" ,
128 : "\"cats\"" ,
];
for ( uint allergyKey = 1 ; allergyKey <= 128 ; allergyKey *= 2 ) {
if ( score & allergyKey ) {
allergies ~= possibleAllergies [ allergyKey ];
}
}
}
final bool allergicTo ( immutable string item )
{
import std . algorithm . searching : any ;
return allergies . any !( allergy => allergy == item );
}
final string [] list ()
{
return allergies ;
}
}
