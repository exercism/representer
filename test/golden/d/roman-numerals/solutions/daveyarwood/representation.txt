RomanNumeral ( 1000 , "\"M\"" ),
RomanNumeral ( 900 , "\"CM\"" ),
RomanNumeral ( 500 , "\"D\"" ),
RomanNumeral ( 400 , "\"CD\"" ),
RomanNumeral ( 100 , "\"C\"" ),

string convert ( int n ) {
string result ;
foreach ( RomanNumeral r ; romanNumerals ) {
while ( n >= r . arabic ) {
result ~= r . roman ;
n -= r . arabic ;
}
}
return result ;
}
