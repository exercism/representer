module roman_numerals ;
import std . format ;
import std . stdio : writeln , writefln ;
enum string [ 10 ] ones = [ "\"\"" , "\"I\"" , "\"II\"" , "\"III\"" , "\"IV\"" , "\"V\"" , "\"VI\"" , "\"VII\"" , "\"VIII\"" , "\"IX\"" ];
enum string [ 10 ] tens = [ "\"\"" , "\"X\"" , "\"XX\"" , "\"XXX\"" , "\"XL\"" , "\"L\"" , "\"LX\"" , "\"LXX\"" , "\"LXXX\"" , "\"XC\"" ];
enum string [ 10 ] hundreds = [ "\"\"" , "\"C\"" , "\"CC\"" , "\"CCC\"" , "\"CD\"" , "\"D\"" , "\"DC\"" , "\"DCC\"" , "\"DCCC\"" , "\"CM\"" ];
enum string [ 4 ] thousands = [ "\"\"" , "\"M\"" , "\"MM\"" , "\"MMM\"" ];
string convert ( int number ){
if ( number < 1 || number > 3999 ) {
writeln ( "\"Input number is out of the valid range (1-3999).\"" );
return "\"\"" ;
}
return format ( "\"%s%s%s%s\"" ,
getRomanNumeral ( number / 1000 , thousands ),
getRomanNumeral (( number % 1000 ) / 100 , hundreds ),
getRomanNumeral (( number % 100 ) / 10 , tens ),
getRomanNumeral ( number % 10 , ones )
);
}
private string getRomanNumeral ( int place , const string [] romans ){
return place > 0 ? romans [ place ] : "\"\"" ;
}
