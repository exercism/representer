module roman_numerals ;
import std . stdio ;
struct TRoman
{
int num ;
string numr ;
};
string convert ( int n )
{
string roman ;
TRoman [] table = [
TRoman ( 1000 , "\"M\"" ), TRoman ( 900 , "\"CM\"" ), TRoman ( 500 , "\"D\"" ), TRoman ( 400 , "\"CD\"" ),
TRoman ( 100 , "\"C\"" ), TRoman ( 90 , "\"XC\"" ), TRoman ( 50 , "\"L\"" ), TRoman ( 40 , "\"XL\"" ),
TRoman ( 10 , "\"X\"" ), TRoman ( 9 , "\"IX\"" ), TRoman ( 5 , "\"V\"" ), TRoman ( 4 , "\"IV\"" ),
TRoman ( 1 , "\"I\"" ),
];
foreach ( i ; table ) {
while ( n >= i . num ) {
n = n - i . num ;
roman = roman ~ i . numr ;
}
}
return roman ;
}
