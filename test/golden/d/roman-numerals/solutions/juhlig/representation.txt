writefln ( "\"Conversion of 9: %s\"" , convert ( 9 ));

writefln ( "\"Conversion of 27: %s\"" , convert ( 27 ));
assert ( "\"XXVII\"" == convert ( 27 ));
}
{
writefln ( "\"Conversion of 48: %s\"" , convert ( 48 ));

string convert ( uint n )
{
string s = "\"\"" ;
for (; n >= 1000 ; n -= 1000 ) s ~= 'M' ;
if ( n >= 900 )
{
s ~= "\"CM\"" ;
n -= 900 ;
}
else if ( n >= 500 )
{
s ~= 'D' ;
n -= 500 ;
}
else if ( n >= 400 )
{
s ~= "\"CD\"" ;
n -= 400 ;
}
for (; n >= 100 ; n -= 100 ) s ~= 'C' ;
if ( n >= 90 )
{
s ~= "\"XC\"" ;
n -= 90 ;
}
else if ( n >= 50 )
{
s ~= 'L' ;
n -= 50 ;
}
else if ( n >= 40 )
{
s ~= "\"XL\"" ;
n -= 40 ;
}
for (; n >= 10 ; n -= 10 ) s ~= 'X' ;
if ( n >= 9 )
{
s ~= "\"IX\"" ;
n -= 9 ;
}
else if ( n >= 5 )
{
s ~= 'V' ;
n -= 5 ;
}
else if ( n >= 4 )
{
s ~= "\"IV\"" ;
n -= 4 ;
}
for (; n >= 1 ; n --) s ~= 'I' ;
return s ;
}
