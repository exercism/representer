module roman_numerals ;
import std . stdio ;
import std . conv : to ;
string convert ( uint decimal ) {
string decStr = to ! string ( decimal ), result ;
foreach ( i , d ; decStr ) {
auto pos = decStr . length - i ;
uint digit = d . to ! int - '0' ;
string numeral ;
outer : while ( true ) {
final switch ( digit ) {
case 0 : break outer ;
case 9 :
final switch ( pos ) {
case 1 : numeral = "\"IX\"" ; break ;
case 2 : numeral = "\"XC\"" ; break ;
case 3 : numeral = "\"CM\"" ; break ;
}
break outer ;
case 1 , 2 , 3 :
final switch ( pos ) {
case 1 : numeral ~= "\"I\"" ; break ;
case 2 : numeral ~= "\"X\"" ; break ;
case 3 : numeral ~= "\"C\"" ; break ;
case 4 : numeral ~= "\"M\"" ; break ;
}
if ( digit == 1 ) {
break outer ;
} else {
-- digit ;
break ;
}
case 4 :
final switch ( pos ) {
case 1 : numeral = "\"I\"" ; break ;
case 2 : numeral = "\"X\"" ; break ;
case 3 : numeral = "\"C\"" ; break ;
}
++ digit ;
break ;
case 5 , 6 , 7 , 8 :
final switch ( pos ) {
case 1 : numeral ~= "\"V\"" ; break ;
case 2 : numeral ~= "\"L\"" ; break ;
case 3 : numeral ~= "\"D\"" ; break ;
}
if ( digit == 5 ) {
break outer ;
} else {
digit -= 5 ;
break ;
}
}
}
result ~= numeral ;
}
return result ;
}
