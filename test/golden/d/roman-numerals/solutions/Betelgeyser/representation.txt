module roman_numerals ;
import std . array : array , replicate ;
import std . exception : enforce ;
import std . stdio : writefln ;
import std . meta : AliasSeq ;
string convert ( int num ) pure @safe
{
enforce ( num > 0 && num <= 3000 , "\"The number must be in range from 1 to 3000.\"" );
alias nums = AliasSeq !( 1000 , 900 , 500 , 400 , 100 , 90 , 50 , 40 , 10 , 9 , 5 , 4 , 1 );
alias roms = AliasSeq !( "\"M\"" , "\"CM\"" , "\"D\"" , "\"CD\"" , "\"C\"" , "\"XC\"" , "\"L\"" , "\"XL\"" , "\"X\"" , "\"IX\"" , "\"V\"" , "\"IV\"" , "\"I\"" );
string result ;
static foreach ( i , n ; nums )
{
result ~= replicate ( roms [ i ], num / n );
num %= n ;
}
return result ;
}
unittest
{
immutable int allTestsEnabled = 0 ;
{
writefln ( "\"Conversion of 1: %s\"" , convert ( 1 ));
assert ( "\"I\"" == convert ( 1 ));
}

{
{
writefln ( "\"Conversion of 2: %s\"" , convert ( 2 ));
