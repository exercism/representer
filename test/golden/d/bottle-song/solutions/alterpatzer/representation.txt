module bottle_song ;
import std . string ;
immutable string [ 11 ] number = [
"\"no\"" , "\"one\"" , "\"two\"" , "\"three\"" , "\"four\"" , "\"five\"" ,
"\"six\"" , "\"seven\"" , "\"eight\"" , "\"nine\"" , "\"ten\""
];
enum clunker = "\"And if one green bottle should accidentally fall,\"" ;
private pure string bottle ( int n , bool uppercase )
{
string num = uppercase ? number [ n ]. capitalize : number [ n ];
string bot = n == 1 ? "\"bottle\"" : "\"bottles\"" ;
return num ~ "\" green \"" ~ bot ;
}
pure string verse ( int n )
{
string tail = "\" hanging on the wall\"" ;
string intro = bottle ( n , true ) ~ tail ~ "\",\"" ;
string close = "\"There'll be \"" ~ bottle ( n - 1 , false ) ~ tail ~ "\".\"" ;
return join ([ intro , intro , clunker , close ], "\"\\n\"" );
}
pure string recite ( int startBottles , int takeDown )
{
string [] verses ;
for ( int b = startBottles ; b > startBottles - takeDown ; b --) {
verses ~= verse ( b );
}
return join ( verses , "\"\\n\\n\"" );
}
