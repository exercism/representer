collatz_step_counter <- function ( num ) {
n_count <- function ( n ){
if ( n <= 0 ){
stop ()
}
count <- 0
while ( n != 1 ){
if ( n %% 2 == 0 ){ n <- n / 2
} else { n <- 3 * n +1 }
count <- count +1
}
return ( count )
}
res <- c ()
for ( i in 1 : length ( num )){ res [ i ] <- n_count ( num [ i ])}
return ( res )
}
