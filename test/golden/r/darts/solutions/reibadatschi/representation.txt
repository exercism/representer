score <- function ( x , y ) {
circle_out = 10 ^ 2
circle_mid = 5 ^ 2
circle_in = 1 ^ 2
val = x ^ 2 + y ^ 2
if ( val > circle_out ){
score = 0
}
if (( val <= circle_out ) & ( val >= circle_mid )){
score = 1
}
if (( val <= circle_mid ) & ( val >= circle_in )){
score = 5
}
if ( val <= circle_in ){
score = 10
}
return ( score )
}
