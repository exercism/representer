is_prime <- function ( n ) {
if ( n < 2 ) return ( FALSE )
if ( n == 2 || n == 3 ) return ( TRUE )
if ( n %% 2 == 0 ) return ( FALSE )
for ( i in 3 : sqrt ( n )) {
if ( n %% i == 0 ) return ( FALSE )
}
return ( TRUE )
}
prime <- function ( n ) {
if ( n == 0 ) { stop ( "\"There is no zeroth prime\"" )}
i <- 2
j <- 0
k <- 0
while ( k < n ) {
if ( is_prime ( i )) {
j <- i
k <- k + 1
i <- i + 1
} else {
i <- i + 1
}
}
return ( j )
}
