(file "word-count.R")
word_count <- function ( input ) {
sentence <- gsub ( "\"^'|'$| '|' \"" , "\" \"" , input )
sentence <- gsub ( "\"[\\\\.:;\\\"!&@%*$^,\\n]\"" , "\" \"" , sentence )
sentence <- trimws ( tolower ( gsub ( "\" +\"" , "\" \"" , sentence )))
split_words_list <- strsplit ( sentence , "\" \"" )
table ( split_words_list ) %>% as.list ()
}
