(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
counter <- 0
if ( nchar ( strand1 ) == nchar ( strand2 )){
s1 = unlist ( strsplit ( strand1 , "\"\"" ))
s2 = unlist ( strsplit ( strand2 , "\"\"" ))
counter = sum ( ! ( s1 == s2 ))
return ( counter )
}
else
stop ( "\"Error: length not equal\"" )
}
