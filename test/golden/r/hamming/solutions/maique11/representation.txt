(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
point <- nchar ( strand1 )
if ( point != nchar ( strand2 )){
stop ()
} else {
dist <- 0
while ( point > 0 ){
if ( substring ( strand1 , point , point ) != substring ( strand2 , point , point )){
dist <- dist + 1
point <- point - 1
} else {
point <- point - 1
}
}
}
return ( dist )
}
