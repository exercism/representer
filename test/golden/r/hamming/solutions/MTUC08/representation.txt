(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) != nchar ( strand2 )) {
stop ( "\"strands are not equal length\"" )
} else {
strand1 <- unlist ( strsplit ( strand1 , "\"\"" ))
strand2 <- unlist ( strsplit ( strand2 , "\"\"" ))
length ( strand2 [ strand1 != strand2 ])
}
}
