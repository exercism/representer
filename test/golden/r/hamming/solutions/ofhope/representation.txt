hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) != nchar ( strand2 )) {
throw ( "\"strings are not of equal length\"" )
}
strand1Vector <- strsplit ( strand1 , "\"\"" )[[ 1 ]]
strand2Vector <- strsplit ( strand2 , "\"\"" )[[ 1 ]]
i <- 0
hammingDistance <- nchar ( strand1 )
for ( char in strand1Vector ) {
i <- i + 1
if ( char == strand2Vector [ i ]) {
hammingDistance <- hammingDistance - 1
}
}
return ( hammingDistance )
}
