(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) != nchar ( strand2 )) {
stop ( "\"The two strands must have the same length.\"" )
}
count <- 0
for ( i in 1 : nchar ( strand1 )) {
if ( substr ( strand1 , i , i ) != substr ( strand2 , i , i )) {
count <- count + 1
}
}
return ( count )
}
