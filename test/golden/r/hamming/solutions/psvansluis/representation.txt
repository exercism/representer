(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) == nchar ( strand2 )){
strlength <- ( nchar ( strand1 ))
} else {
stop ( "\"strings have different lengths\"" )
}
truefalselist <- c ( rep ( 0 , times = strlength ))
for ( i in 1 : strlength ) {
truefalselist [ i ] <- substring ( strand1 , i , i ) != substring ( strand2 , i , i )
}
print ( sum ( truefalselist ))
}
