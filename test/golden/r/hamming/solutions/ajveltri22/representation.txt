(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) != nchar ( strand2 )) stop ( "\"Sequence lengths must be identical.\"" )
strand1 <- strsplit ( strand1 , split = "\"\"" )[[ 1 ]]
strand2 <- strsplit ( strand2 , split = "\"\"" )[[ 1 ]]
h_dist <- sum ( strand1 != strand2 )
return ( h_dist )
}
