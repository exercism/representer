(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
count <- 0
num <- nchar ( strand1 )
if ( nchar ( strand1 ) != nchar ( strand2 )) { break }
if ( num == 0 ) { count <- 0 }
else if ( num == 1 ){
if ( strand1 != strand2 ){
count <- 1
}
} else {
for ( i in 1 : num ){
if ( substring ( strand1 , i , i ) != substring ( strand2 , i , i ))
{ count = count + 1 }
}
}
return ( count )
}
