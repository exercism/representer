(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
chars_1 <- strsplit ( strand1 , "\"\"" )[[ 1 ]]
chars_2 <- strsplit ( strand2 , "\"\"" )[[ 1 ]]
total <- 0
l_1 <- length ( chars_1 )
l_2 <- length (( chars_2 ))
if ( l_1 == 0 & l_2 == 0 ){
return ( 0 )
}
else {
stopifnot ( l_1 == l_2 )
for ( i in 1 : l_1 ){
if ( chars_1 [[ i ]] != chars_2 [[ i ]]){
total <- total + 1
}
}
return ( total )
}
}
