(file "hamming.R")
hamming <- function ( strand1 , strand2 ) {
strand1 <- strsplit ( strand1 , split = "\"\"" )[[ 1 ]]
strand2 <- strsplit ( strand2 , split = "\"\"" )[[ 1 ]]
counter <- 0
if ( length ( strand1 ) == 0 && length ( strand2 ) == 0 ) {
return ( counter )
}
if ( length ( strand1 ) > length ( strand2 )) {
stop ( "\"First is longer!\"" )
}
if ( length ( strand1 ) < length ( strand2 )) {
stop ( "\"Second is longer!\"" )
}
if ( length ( strand1 ) == length ( strand2 )) {
for ( i in 1 : length ( strand1 )) {
if ( strand1 [ i ] != strand2 [ i ]) {
counter <- counter + 1
} else {
counter <- counter + 0
}
}
return ( counter )
}
}
