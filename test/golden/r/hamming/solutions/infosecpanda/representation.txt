hamming <- function ( strand1 , strand2 ) {
s1 = strsplit ( strand1 , "\"\"" )[[ 1 ]]
s2 = strsplit ( strand2 , "\"\"" )[[ 1 ]]
count = 0
if ( length ( s1 ) != length ( s2 ))
stop ()
if ( length ( s1 ) == 0 )
return ( 0 )
for ( i in 1 : length ( s1 ))
if ( s1 [ i ] != s2 [ i ])
count = count + 1
return ( count )
}
