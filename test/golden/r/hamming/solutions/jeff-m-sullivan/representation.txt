(file "hamming.R")
library ( stringr )
hamming <- function ( strand1 , strand2 ) {
if ( str_length ( strand1 ) == str_length ( strand2 )) {
s1 <- strsplit ( strand1 , "\"\"" )[[ 1 ]]
s2 <- strsplit ( strand2 , "\"\"" )[[ 1 ]]
return ( sum ( s1 != s2 ))
} else {
stop ( "\"Strings are not of the same length.\"" )
}
}
