(file "hamming.R")
hamming <- function ( strand_a , strand_b ) {
if ( nchar ( strand_a ) != nchar ( strand_b )) {
stop ( "\"DNA strands not of same lenght!\"" )
} else {
list_a <- unlist ( strsplit ( strand_a , "\"\"" ))
list_b <- unlist ( c ( strsplit ( strand_b , "\"\"" )))
return ( sum ( list_a != list_b ))
}
}
