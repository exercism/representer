hamming <- function ( strand1 , strand2 ) {
if ( nchar ( strand1 ) != nchar ( strand2 ))
stop ( "\"argument lengths not equal\"" )
strand_length <- nchar ( strand1 )
if ( strand_length == 0 ) return ( 0 )
distance <- 0
for ( i in 1 : strand_length ) {
if ( substr ( strand1 , i , i ) != substr ( strand2 , i , i ))
distance <- distance + 1
}
return ( distance )
}
