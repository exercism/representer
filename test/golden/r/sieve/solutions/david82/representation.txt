sieve <- function ( limit ) {
if ( limit < 2 ) {
return ( NULL )
}
is_prime <- rep ( TRUE , limit )
is_prime [ 1 ] <- FALSE
for ( p in 2 : limit ) {
if ( is_prime [ p ]) {
if ( p + p <= limit ) {
multiples <- seq ( from = p + p , to = limit , by = p )
is_prime [ multiples ] <- FALSE
}
}
}
which ( is_prime )
}
