scores_list <- function ( scores ) {
return ( scores )
}
latest <- function ( scores ) {
score = scores [ length ( scores )]
return ( score )
}
personal_best <- function ( scores ) {
score = max ( scores )
return ( score )
}
personal_top_three <- function ( scores ) {
score = sort ( scores , decreasing = TRUE )
if ( length ( score ) > 2 ){
score2 = score [ 1 : 3 ]
} else {
score2 = score
}
return ( score2 )
}
