parse_phone_number <- function ( number_string ) {
number_string <- strsplit ( number_string , split = NULL )[[ 1 ]]
numbers <- suppressWarnings ( na.omit ( as.integer ( number_string )))
if (( length ( numbers ) == 11 && numbers [ 1 ] == 1 && all ( numbers [ c ( 2 , 5 )] %in% 2 : 9 )) ||
( length ( numbers ) == 10 && all ( numbers [ c ( 1 , 4 )] %in% 2 : 9 ))) {
if ( length ( numbers ) == 11 ) numbers <- numbers [ -1 ]
return ( paste0 ( numbers , collapse = "\"\"" ))
}
}
