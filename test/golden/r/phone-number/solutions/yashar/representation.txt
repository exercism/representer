(file "phone-number.R")
library ( 'stringr' )
parse_phone_number <- function ( number_string ) {
m <- str_match ( trimws ( number_string ), "\"^(\\\\+1|1)? ?((\\\\(([2-9][0-9]{2})\\\\))|([2-9][0-9]{2}))[-. ]*([2-9][0-9]{2})[-. ]*([0-9]{4})$\"" )
if ( is.na ( m [ 1 ]))
return ( NULL )
else
return ( paste ( tail ( m [ ! is.na ( m )], 3 ), collapse = "\"\"" ))
}
