(file "phone-number.R")
library ( stringr )
parse_phone_number <- function ( number_string ) {
pattern <- paste0 ( "\"^\\\\s*(\\\\+?1)?\"" ,
"\"[- .(]*([2-9][0-9]{2})[- .)]*\"" ,
"\"([2-9][0-9]{2})[- .]*\"" ,
"\"([0-9]{4})\\\\s*$\"" )
elements <- str_match ( number_string , pattern )[ 1 , 3 : 5 ]
if ( any ( is.na ( elements ))) return ( NULL )
paste ( elements , collapse = "\"\"" )
}
