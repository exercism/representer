parse_phone_number <- function ( number_string ) {
rgx.INVALID <- "'[^ .()0-9+-]'"
rgx.DECORATIONS <- '[^0-9]'
rgx.PHONE_NUMBER <- "'(?x) ^1? ((?: [2-9][0-9]{2} ){2} [0-9]{4}) $'"
valid_format <- grepl ( rgx.INVALID , number_string ) == FALSE
digits <- gsub ( rgx.DECORATIONS , '' , number_string )
lookup <- regexec ( rgx.PHONE_NUMBER , digits , perl = TRUE )
phone_number <- if ( lookup [[ 1 ]][ 1 ] != -1 )
regmatches ( digits , lookup )[[ 1 ]][ 2 ]
if ( valid_format )
phone_number
}
