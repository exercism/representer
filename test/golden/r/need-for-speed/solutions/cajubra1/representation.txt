new_car <- function ( speed , battery_drain ) { list ( "\"speed\"" = speed , "\"battery_drain\"" = battery_drain , "\"battery\"" = 100 , "\"distance_traveled\"" = 0 )
}
new_track <- function ( track_length ) { list ( "\"track_length\"" = track_length )
}
battery_drained <- function ( car ) { car $ battery < car $ battery_drain
}
drive <- function ( car ) { if ( ! battery_drained ( car )) {
car $ distance_traveled <- car $ distance_traveled + car $ speed
car $ battery <- car $ battery - car $ battery_drain
return ( car )}
else { return ( car )}
}
can_finish <- function ( car , track ) { car $ battery >= car $ battery_drain * ceiling ( track $ track_length / car $ speed )
}
store_track <- function ( car , track , name ) { carfull <- car
carfull $ battery < -100
car [[ name ]] <- list ( "\"track_length\"" = track $ track_length , "\"complete\"" = can_finish ( carfull , track ))
return ( car )
}
