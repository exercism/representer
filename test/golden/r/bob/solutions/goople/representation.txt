bob <- function ( input ) {
is_empty <- grepl ( "\"^\\\\s*$\"" , input , perl = TRUE )
is_question <- grepl ( "\"\\\\?\\\\s*$\"" , input , perl = TRUE )
is_angry <- grepl (
"\"^[A-Z]+$\"" ,
gsub ( "\"[\\\\W\\\\_\\\\d]\"" , "\"\"" , input , perl = TRUE ),
perl = TRUE
)
if ( is_empty ) {
return ( "\"Fine. Be that way!\"" )
}
if ( is_angry ) {
if ( is_question ) {
return ( "\"Calm down, I know what I'm doing!\"" )
} else {
return ( "\"Whoa, chill out!\"" )
}
} else {
if ( is_question ) {
return ( "\"Sure.\"" )
} else {
return ( "\"Whatever.\"" )
}
}
}
