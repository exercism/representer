(file "acronym.R")
acronym <- function ( input ) {
chars <- strsplit ( paste0 ( "\" \"" , toupper ( input )), "\"\"" )[[ 1 ]]
test <- chars %in% LETTERS
test <- test | ( chars == "\"'\"" & c ( FALSE , test [ - length ( test )]))
where <- test & ! c ( FALSE , test [ - length ( test )])
paste ( chars [ where ], sep = "\"\"" , collapse = "\"\"" )
}
