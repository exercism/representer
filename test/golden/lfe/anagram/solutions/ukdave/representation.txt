( defmodule anagram
( export ( find 2 )))
( defun find ( word candidates )
( lists:filter
( lambda ( candidate ) ( is-anagram word candidate ))
candidates ))
( defun is-anagram ( word candidate )
( let (( word-lower ( string:to_lower word ))
( candidate-lower ( string:to_lower candidate )))
( and ( /= word-lower candidate-lower )
( == ( lists:sort word-lower ) ( lists:sort candidate-lower )))))
