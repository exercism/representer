(file "src/anagram.lfe")
( defmodule anagram
( export ( find 2 )))
( defun find ( original wordlist )
( let* (( lower-original ( string:to_lower original ))
( sorted-lower-original ( lists:sort lower-original )))
( lists:filter ( lambda ( word )
( let* (( lower-word ( string:to_lower word ))
( sorted-lower-word ( lists:sort lower-word )))
( and ( == sorted-lower-word sorted-lower-original )
( /= lower-word lower-original ))))
wordlist )))
