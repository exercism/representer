( defmodule anagram
( export ( find 2 )))
( defun find ( target candidates )
( let* (( target-lower ( string:casefold target ))
( target-lower-sorted ( lists:sort target-lower )))
( lists:filter
( lambda ( candidate ) ( is-anagram target-lower target-lower-sorted ( string:casefold candidate )))
candidates )))
( defun is-anagram ( target-lower target-lower-sorted word-lower )
( andalso ( /= target-lower word-lower ) ( == target-lower-sorted ( lists:sort word-lower ))))
