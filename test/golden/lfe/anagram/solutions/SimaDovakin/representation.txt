(file "src/anagram.lfe")
( defmodule anagram
( export ( find 2 )))
( defun find ( subject candidates )
( lists:filter ( lambda ( c ) ( anagram? subject c )) candidates ))
( defun anagram? ( subject candidate )
( let (( lowered-subject ( string:to_lower subject ))
( lowered-candidate ( string:to_lower candidate )))
( and ( not ( string:equal lowered-subject lowered-candidate ))
( string:equal ( lists:sort lowered-subject ) ( lists:sort lowered-candidate )))))
