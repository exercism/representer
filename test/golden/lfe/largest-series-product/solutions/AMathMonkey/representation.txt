( defmodule largest-series-product
( export ( from-string 2 )))
( defun product ( s )
( lists:foldl
( match-lambda
(( c acc ) ( when ( and ( >= c #\0 ) ( =< c #\9 )))
( * acc ( - c #\0 ))))
1
s ))
( defun from-string ( s n )
( lists:max
( lc (( <- start ( lists:seq 1 ( - ( length s ) n -1 ))))
( product ( lists:sublist s start n )))))
