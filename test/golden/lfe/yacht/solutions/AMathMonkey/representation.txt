( defmodule yacht
( export ( score 2 )))
( defun filter-sum ( dice n )
( lists:sum ( lc (( <- d dice ) ( =:= d n )) d )))
( defun score ( dice category )
( case category
( 'ones ( filter-sum dice 1 ))
( 'twos ( filter-sum dice 2 ))
( 'threes ( filter-sum dice 3 ))
( 'fours ( filter-sum dice 4 ))
( 'fives ( filter-sum dice 5 ))
( 'sixes ( filter-sum dice 6 ))
( 'choice ( lists:sum dice ))
( 'little-straight ( if ( =:= ( lists:sort dice ) ' ( 1 2 3 4 5 )) 30 0 ))
( 'big-straight ( if ( =:= ( lists:sort dice ) ' ( 2 3 4 5 6 )) 30 0 ))
( 'yacht ( case ( lists:usort dice ) (( list _ ) 50 ) ( _ 0 )))
( 'full-house ( case ( lists:sort dice )
(( list x x y y y ) ( when ( =/= x y )) ( lists:sum dice ))
(( list x x x y y ) ( when ( =/= x y )) ( lists:sum dice ))
( _ 0 )))
( 'four-of-a-kind ( case ( lists:sort dice )
(( list x x x x _ ) ( * x 4 ))
(( list _ x x x x ) ( * x 4 ))
( _ 0 )))))
