( defmodule triangle
( export ( kind 3 )))
( defun valid ( a b c )
( andalso
( > a 0 ) ( > b 0 ) ( > c 0 )
( > ( + a b ) c ) ( > ( + b c ) a ) ( > ( + a c ) b )))
( defun kind ( a b c )
( andalso ( valid a b c )
( cond
(( andalso ( == a b ) ( == b c )) 'equilateral )
(( orelse ( == a b ) ( == b c ) ( == a c )) 'isosceles )
( 'true 'scalene ))))
