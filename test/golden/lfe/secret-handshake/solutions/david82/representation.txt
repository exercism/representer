( defmodule secret-handshake
( export ( commands 1 )))
( defun bit-set? ( n mask )
( /= 0 ( band n mask )))
( defun commands ( n )
( let* (( v ( band n 31 ))
( pairs ' (( 1 "\"wink\"" )
( 2 "\"double blink\"" )
( 4 "\"close your eyes\"" )
( 8 "\"jump\"" )))
( actions
( lists:foldl
( lambda ( pair acc )
( let ((( list mask action ) pair ))
( if ( bit-set? v mask )
( lists:append acc ( list action ))
acc )))
' ()
pairs )))
( if ( bit-set? v 16 )
( lists:reverse actions )
actions )))
