(file "src/binary-string.lfe")
( defmodule binary-string
( export ( to-decimal 1 )))
( defun to-decimal
(( s ) ( when ( is_list s )) ( to-decimal-acc s 0 ))
(( _ ) 0 ))
( defun to-decimal-acc
(( ' () acc ) acc )
((( cons h t ) acc )
( case h
( 48 ( to-decimal-acc t ( * 2 acc )))
( 49 ( to-decimal-acc t ( + ( * 2 acc ) 1 )))
( _ 0 ))))
