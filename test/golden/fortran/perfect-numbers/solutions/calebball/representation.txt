(file "perfect_numbers.f90")
module perfect_numbers
implicit none
contains
function classify ( num )
integer , intent ( in ) :: num
character ( len = 9 ) :: classify
integer :: aliquot , n
if ( num < 1 ) then
classify = "\"ERROR\""
return
end if
aliquot = 0
do n = 1 , num / 2
if ( mod ( num , n ) == 0 ) aliquot = aliquot + n
end do
if ( aliquot < num ) then
classify = "\"deficient\""
else if ( aliquot > num ) then
classify = "\"abundant\""
else
classify = "\"perfect\""
end if
end function
end module
