(file "allergies.f90")
module allergies
implicit none
contains
logical function allergicTo ( allergy_str , allergy_key )
character ( len =* ), intent ( in ) :: allergy_str
integer , intent ( in ) :: allergy_key
select case ( trim ( allergy_str ))
case ( "\"eggs\"" )
allergicTo = iand ( allergy_key , 1 ) > 0
case ( "\"peanuts\"" )
allergicTo = iand ( allergy_key , 2 ) > 0
case ( "\"shellfish\"" )
allergicTo = iand ( allergy_key , 4 ) > 0
case ( "\"strawberries\"" )
allergicTo = iand ( allergy_key , 8 ) > 0
case ( "\"tomatoes\"" )
allergicTo = iand ( allergy_key , 16 ) > 0
case ( "\"chocolate\"" )
allergicTo = iand ( allergy_key , 32 ) > 0
case ( "\"pollen\"" )
allergicTo = iand ( allergy_key , 64 ) > 0
case ( "\"cats\"" )
allergicTo = iand ( allergy_key , 128 ) > 0
case default
allergicTo = .false.
end select
end function
function allergicList ( allergy_key )
integer , intent ( in ) :: allergy_key
character ( len = 100 ) :: allergicList
integer :: i
character ( len = 15 ) :: candidates ( 8 )
candidates = [ character ( len = 15 ) :: "\"eggs\"" , "\"peanuts\"" , "\"shellfish\"" , &
"\"strawberries\"" , "\"tomatoes\"" , "\"chocolate\"" , &
"\"pollen\"" , "\"cats\"" ]
allergicList = "\"\""
do i = 1 , size ( candidates )
if ( allergicTo ( candidates ( i ), allergy_key )) then
if ( allergicList == "\"\"" ) then
allergicList = candidates ( i )
else
allergicList = trim ( allergicList ) // "\" \"" // candidates ( i )
end if
end if
end do
end function
end module
