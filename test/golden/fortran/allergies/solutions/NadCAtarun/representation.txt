module allergies
implicit none
contains
logical function allergicTo ( allergy_str , allergy_key )
character ( len = * ), intent ( in ) :: allergy_str
integer , intent ( in ) :: allergy_key
select case ( allergy_str )
case ( "\"eggs\"" )
allergicTo = btest ( allergy_key , 0 )
case ( "\"peanuts\"" )
allergicTo = btest ( allergy_key , 1 )
case ( "\"shellfish\"" )
allergicTo = btest ( allergy_key , 2 )
case ( "\"strawberries\"" )
allergicTo = btest ( allergy_key , 3 )
case ( "\"tomatoes\"" )
allergicTo = btest ( allergy_key , 4 )
case ( "\"chocolate\"" )
allergicTo = btest ( allergy_key , 5 )
case ( "\"pollen\"" )
allergicTo = btest ( allergy_key , 6 )
case ( "\"cats\"" )
allergicTo = btest ( allergy_key , 7 )
case default
allergicTo = .false.
end select
end function
function allergicList ( allergy_key )
integer , intent ( in ) :: allergy_key
character ( len = 100 ) :: allergicList
allergicList = "\"\""
if ( allergicTo ( "\"eggs\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" eggs\""
if ( allergicTo ( "\"peanuts\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" peanuts\""
if ( allergicTo ( "\"shellfish\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" shellfish\""
if ( allergicTo ( "\"strawberries\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" strawberries\""
if ( allergicTo ( "\"tomatoes\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" tomatoes\""
if ( allergicTo ( "\"chocolate\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" chocolate\""
if ( allergicTo ( "\"pollen\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" pollen\""
if ( allergicTo ( "\"cats\"" , allergy_key )) allergicList = trim ( allergicList ) // "\" cats\""
if ( len ( allergicList ) > 1 ) then
allergicList = adjustl ( allergicList )
end if
end function
end module
