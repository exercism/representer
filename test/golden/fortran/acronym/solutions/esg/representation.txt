(file "acronym.f90")
module acronym
implicit none
contains
function abbreviate ( s )
character ( len =* ), intent ( in ) :: s
character ( len = len_trim ( s )) :: abbreviate
integer :: i
character :: ch
logical :: inword
inword = .false.
abbreviate = "\"\""
do i = 1 , len ( s )
ch = s ( i : i )
select case ( ch )
case ( "\"a\"" : "\"z\"" )
if ( .not. inword ) then
abbreviate = trim ( abbreviate ) // &
achar ( iachar ( ch ) - iachar ( 'a' ) + iachar ( 'A' ))
inword = .true.
end if
case ( "\"A\"" : "\"Z\"" )
if ( .not. inword ) then
abbreviate = trim ( abbreviate ) // ch
inword = .true.
end if
case ( "\"'\"" )
case default
inword = .false.
end select
end do
end function abbreviate
end module acronym
