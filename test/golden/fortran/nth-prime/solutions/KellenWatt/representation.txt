(file "nth_prime.f90")
module nth_prime
implicit none
contains
integer function prime ( n )
integer , intent ( in ) :: n
integer , dimension ( n ) :: primes
integer :: prime_count
integer :: candidate
integer :: i
if ( n < 1 ) then
prime = -1
return
end if
primes ( 1 ) = 2
prime_count = 1
candidate = 1
i = 0
check_loop : do while ( prime_count < n )
candidate = candidate + 2
do i = 1 , prime_count
if ( mod ( candidate , primes ( i )) == 0 ) cycle check_loop
end do
prime_count = prime_count + 1
primes ( prime_count ) = candidate
end do check_loop
prime = primes ( n )
end function
end module
