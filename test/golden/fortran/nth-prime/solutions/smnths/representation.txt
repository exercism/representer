module nth_prime
implicit none
contains
function prime ( n )
integer , intent ( in ) :: n
integer :: prime
integer :: num , cnt
prime = -1
if ( n <= 0 ) return
cnt = 0
num = 1
do
num = num + 1
if ( is_prime ( num )) then
cnt = cnt + 1
if ( cnt == n ) then
prime = num
return
endif
endif
enddo
end function
function is_prime ( num )
integer , intent ( in ) :: num
integer :: i
logical :: is_prime
is_prime = .true.
do i = 2 , num -1
if ( modulo ( num , i ) == 0 ) then
is_prime = .false.
return
endif
enddo
end function
end module
