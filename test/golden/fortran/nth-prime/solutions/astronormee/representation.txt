(file "nth_prime.f90")
module nth_prime
implicit none
contains
integer function prime ( n )
integer :: n , i , j , p
real , allocatable :: x (:)
logical :: isPrime
p = 0
allocate ( x ( n ))
if ( n .eq. 0 ) then
prime = -1
else
do i = 2 , ( 2 * n ) ** ( 2 )
isPrime = .true.
do j = 2 , int ( sqrt ( real ( i )))
if ( mod ( i , j ) == 0 ) then
isPrime = .false.
exit
endif
enddo
if ( isPrime ) then
p = p + 1
x ( p ) = i
endif
if ( p .ge. n ) then
exit
endif
enddo
prime = x ( n )
deallocate ( x )
endif
end function
end module
