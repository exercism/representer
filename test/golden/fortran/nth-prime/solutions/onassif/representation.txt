(file "nth_prime.f90")
module nth_prime
implicit none
contains
integer function prime ( n )
integer , intent ( in ) :: n
integer :: i , nth
logical :: isPrime
prime = -1
if ( n == 1 ) THEN
prime = 2
else if ( n > 1 ) then
prime = 2
nth = 1
do while ( nth < n )
prime = prime +1
isPrime = .TRUE.
do i = 2 , prime -1
if ( mod ( prime , i ) == 0 ) THEN
isPrime = .FALSE.
EXIT
end if
end do
if ( isPrime ) THEN
nth = nth + 1
end if
end do
end if
end function
end module
