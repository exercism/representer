module armstrong_numbers
implicit none
contains
logical function isArmstrongNumber ( i )
integer , intent ( in ) :: i
integer :: l , myI , di , d , sum
character ( 12 ) :: iStr
write ( iStr , '(I0)' ) i
l = len_trim ( iStr )
sum = 0
myI = i
do di = 1 , l
d = mod ( myI , 10 )
sum = sum + d ** l
myI = myI / 10
enddo
isArmstrongNumber = i == sum
end function
end module
