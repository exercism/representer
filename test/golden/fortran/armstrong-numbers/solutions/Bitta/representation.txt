module armstrong_numbers
implicit none
contains
logical function isArmstrongNumber ( i )
integer , intent ( in ) :: i
integer :: sum , n , last_digit , number_of_digits
n = i
number_of_digits = int ( log10 ( real ( n ))) + 1
sum = 0
do while ( n / = 0 )
last_digit = mod ( n , 10 )
sum = sum + last_digit ** number_of_digits
n = n / 10
end do
isArmstrongNumber = .false.
if ( sum == i ) isArmstrongNumber = .true.
end function
end module
