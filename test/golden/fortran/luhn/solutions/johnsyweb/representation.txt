(file "luhn.f90")
module luhn
implicit none
contains
logical function validate ( number )
character ( * ), intent ( in ) :: number
integer :: sum_of_digits , digit_count , i , d
validate = .false.
sum_of_digits = 0
digit_count = 0
do i = len ( number ), 1 , -1
if ( number ( i : i ) == "' '" ) cycle
d = iachar ( number ( i : i )) - iachar ( '0' )
if ( 0 > d .or. d > 9 ) return
sum_of_digits = sum_of_digits + d
if ( mod ( digit_count , 2 ) / = 0 ) then
sum_of_digits = sum_of_digits + d
if ( d >= 5 ) then
sum_of_digits = sum_of_digits - 9
end if
end if
digit_count = digit_count + 1
end do
validate = ( digit_count > 1 .and. mod ( sum_of_digits , 10 ) == 0 )
end function
end module
