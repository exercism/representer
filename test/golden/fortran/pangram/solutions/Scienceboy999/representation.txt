module pangram
implicit none
contains
logical function is_pangram ( sentence )
character ( * ) :: sentence
logical usedLetter ( 26 )
integer i
usedLetter (:) = .false.
do i = 1 , len ( sentence )
if (( sentence ( i : i ) .ge. 'a' ) .and. ( sentence ( i : i ) .le. 'z' )) then
usedLetter ( ichar ( sentence ( i : i )) - ichar ( 'a' ) + 1 ) = .true.
end if
if (( sentence ( i : i ) .ge. 'A' ) .and. ( sentence ( i : i ) .le. 'Z' )) then
usedLetter ( ichar ( sentence ( i : i )) - ichar ( 'A' ) + 1 ) = .true.
end if
end do
do i = 1 , 26
if ( .not. usedLetter ( i )) then
is_pangram = .false.
return
end if
end do
is_pangram = .true.
end function is_pangram
end module pangram
