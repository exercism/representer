module leap
implicit none
contains
logical function is_leap_year ( year )
integer :: year
IF ( MOD ( year , 4 ) > 0 ) THEN
is_leap_year = .false.
ELSE IF ( MOD ( year , 100 ) > 0 ) THEN
is_leap_year = .true.
ELSE IF ( MOD ( year , 400 ) > 0 ) THEN
is_leap_year = .false.
ELSE
is_leap_year = .true.
END IF
end function
end module
