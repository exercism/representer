module leap
implicit none
contains
logical function is_leap_year ( year )
integer :: year
is_leap_year = .false.
if ( mod ( year , 4 ) == 0 ) then
if ( mod ( year , 100 ) == 0 .and. mod ( year , 400 )/ = 0 ) then
is_leap_year = .false.
else
is_leap_year = .true.
end if
end if
end function
