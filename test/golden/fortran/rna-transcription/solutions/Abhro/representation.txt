(file "rna_transcription.f90")
module rna_transcription
implicit none
contains
function to_rna ( dna )
character ( len =* ), intent ( in ) :: dna
character ( len = len ( dna )) :: to_rna
integer :: i
do i = 1 , len ( dna )
to_rna ( i : i ) = to_rna_char ( dna ( i : i ))
end do
end function to_rna
function to_rna_char ( dna )
character , intent ( in ) :: dna
character :: to_rna_char
select case ( dna )
case ( 'A' )
to_rna_char = 'U'
case ( 'G' )
to_rna_char = 'C'
case ( 'C' )
to_rna_char = 'G'
case ( 'T' )
to_rna_char = 'A'
end select
end function to_rna_char
end module rna_transcription
