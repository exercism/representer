(file "rna_transcription.f90")
module rna_transcription
implicit none
contains
function to_rna ( dna )
character ( * ) :: dna
character ( len ( dna )) :: to_rna
integer :: i
to_rna = dna
do i = 1 , len_trim ( dna )
select case ( dna ( i : i ))
case ( 'G' )
to_rna ( i : i ) = 'C'
case ( 'C' )
to_rna ( i : i ) = 'G'
case ( 'T' )
to_rna ( i : i ) = 'A'
case ( 'A' )
to_rna ( i : i ) = 'U'
case default
print * , "\"Invalid DNA sequence!\""
stop
end select
end do
end function to_rna
end module rna_transcription
