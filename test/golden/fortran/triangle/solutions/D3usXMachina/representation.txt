module triangle
implicit none
interface equilateral
module procedure equilateral_real
module procedure equilateral_int
end interface
interface scalene
module procedure scalene_real
module procedure scalene_int
end interface
interface isosceles
module procedure isosceles_real
module procedure isosceles_int
end interface
interface degenerate
module procedure degenerate_real
module procedure degenerate_int
end interface
interface istriangle
module procedure triangle_real
module procedure triangle_int
end interface
contains
logical pure function equilateral_real ( edges )
real , dimension ( 3 ), intent ( in ) :: edges
equilateral_real = .FALSE.
if ( istriangle ( edges ) ) then
equilateral_real = ( edges ( 1 ) == edges ( 2 ) ) &
.AND. ( edges ( 1 ) == edges ( 3 ) )
end if
end function
logical pure function equilateral_int ( edges )
integer , dimension ( 3 ), intent ( in ) :: edges
equilateral_int = .FALSE.
if ( istriangle ( edges ) ) then
equilateral_int = ( edges ( 1 ) == edges ( 2 ) ) &
.AND. ( edges ( 1 ) == edges ( 3 ) )
end if
end function
logical pure function isosceles_real ( edges )
real , dimension ( 3 ), intent ( in ) :: edges
isosceles_real = .FALSE.
if ( istriangle ( edges ) ) then
isosceles_real = ( edges ( 1 ) == edges ( 2 ) ) &
.OR. ( edges ( 1 ) == edges ( 3 ) ) &
.OR. ( edges ( 2 ) == edges ( 3 ) )
end if
end function
logical pure function isosceles_int ( edges )
integer , dimension ( 3 ), intent ( in ) :: edges
isosceles_int = .FALSE.
if ( istriangle ( edges ) ) then
isosceles_int = ( edges ( 1 ) == edges ( 2 ) ) &
.OR. ( edges ( 1 ) == edges ( 3 ) ) &
.OR. ( edges ( 2 ) == edges ( 3 ) )
end if
end function
logical pure function scalene_real ( edges )
real , dimension ( 3 ), intent ( in ) :: edges
scalene_real = .FALSE.
if ( istriangle ( edges ) ) then
scalene_real = ( edges ( 1 ) / = edges ( 2 ) ) &
.AND. ( edges ( 1 ) / = edges ( 3 ) ) &
.AND. ( edges ( 2 ) / = edges ( 3 ) )
end if
end function
logical pure function scalene_int ( edges )
integer , dimension ( 3 ), intent ( in ) :: edges
scalene_int = .FALSE.
if ( istriangle ( edges ) ) then
scalene_int = ( edges ( 1 ) / = edges ( 2 ) ) &
.AND. ( edges ( 1 ) / = edges ( 3 ) ) &
.AND. ( edges ( 2 ) / = edges ( 3 ) )
end if
end function
logical pure function triangle_real ( edges )
real , dimension ( 3 ), intent ( in ) :: edges
triangle_real = .FALSE.
if ( ( edges ( 1 ) <= 0 ) &
.OR. ( edges ( 2 ) <= 0 ) &
.OR. ( edges ( 3 ) <= 0 ) ) then
return
end if
if ( ( edges ( 1 ) + edges ( 2 ) < edges ( 3 ) ) &
.OR. ( edges ( 1 ) + edges ( 3 ) < edges ( 2 ) ) &
.OR. ( edges ( 2 ) + edges ( 3 ) < edges ( 1 ) ) ) then
return
end if
triangle_real = .TRUE.
end function
logical pure function triangle_int ( edges )
integer , dimension ( 3 ), intent ( in ) :: edges
triangle_int = .FALSE.
if ( ( edges ( 1 ) <= 0 ) &
.OR. ( edges ( 2 ) <= 0 ) &
.OR. ( edges ( 3 ) <= 0 ) ) then
return
end if
if ( ( edges ( 1 ) + edges ( 2 ) < edges ( 3 ) ) &
.OR. ( edges ( 1 ) + edges ( 3 ) < edges ( 2 ) ) &
.OR. ( edges ( 2 ) + edges ( 3 ) < edges ( 1 ) ) ) then
return
end if
triangle_int = .TRUE.
end function
logical pure function degenerate_real ( edges )
real , dimension ( 3 ), intent ( in ) :: edges
degenerate_real = .FALSE.
if ( istriangle ( edges ) ) then
degenerate_real = ( edges ( 1 ) + edges ( 2 ) == edges ( 3 ) ) &
.OR. ( edges ( 1 ) + edges ( 3 ) == edges ( 2 ) ) &
.OR. ( edges ( 2 ) + edges ( 3 ) == edges ( 1 ) )
end if
end function
logical pure function degenerate_int ( edges )
integer , dimension ( 3 ), intent ( in ) :: edges
degenerate_int = .FALSE.
if ( istriangle ( edges ) ) then
degenerate_int = ( edges ( 1 ) + edges ( 2 ) == edges ( 3 ) ) &
.OR. ( edges ( 1 ) + edges ( 3 ) == edges ( 2 ) ) &
.OR. ( edges ( 2 ) + edges ( 3 ) == edges ( 1 ) )
end if
end function
end module
