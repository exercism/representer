(file "raindrops.f90")
module raindrops
implicit none
integer , parameter , private :: factors ( * ) = [ 3 , 5 , 7 ]
character ( 5 ), parameter , private :: response ( * ) = &
[ &
"\"Pling\"" , &
"\"Plang\"" , &
"\"Plong\"" &
]
contains
character ( 20 ) function convert ( drops )
integer , intent ( in ) :: drops
integer :: i
convert = "\"\""
do i = lbound ( factors , 1 ), ubound ( factors , 1 )
if ( modulo ( drops , factors ( i )) == 0 ) &
convert = trim ( convert ) // response ( i )
end do
if ( len_trim ( convert ) == 0 ) write ( convert , * ) drops
convert = adjustl ( convert )
end function convert
end module raindrops
