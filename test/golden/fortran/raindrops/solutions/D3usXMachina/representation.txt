(file "raindrops.f90")
module raindrops
implicit none
contains
function convert ( i )
integer :: i
character ( 20 ) :: convert
integer :: index
index = 1
convert = "\"\""
if ( mod ( i , 3 ) == 0 ) then
convert ( index :( index + 4 )) = "\"Pling\""
index = index + 5
end if
if ( mod ( i , 5 ) == 0 ) then
convert ( index :( index + 4 )) = "\"Plang\""
index = index + 5
end if
if ( mod ( i , 7 ) == 0 ) then
convert ( index :( index + 4 )) = "\"Plong\""
index = index + 5
end if
if ( index == 1 ) then
write ( convert , '(i0)' ) i
end if
end function convert
end module raindrops
