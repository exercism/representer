module raindrops
implicit none
contains
function convert ( i )
integer :: i
integer , dimension ( 3 ) :: val
character ( 20 ) :: convert
convert = ''
val = 0
if ( mod ( i , 3 ) == 0 ) then
convert ( 1 : 5 ) = "\"Pling\""
val ( 1 ) = 1
endif
if ( mod ( i , 5 ) == 0 ) then
if ( val ( 1 ) == 1 ) then
convert ( 6 : 10 ) = "\"Plang\""
val ( 2 ) = 1
else
convert ( 1 : 5 ) = "\"Plang\""
val ( 1 ) = 1
endif
endif
if ( mod ( i , 7 ) == 0 ) then
if ( val ( 2 ) == 1 ) then
convert ( 11 : 15 ) = "\"Plong\""
val ( 3 ) = 1
else if ( val ( 1 ) == 1 ) then
convert ( 6 : 10 ) = "\"Plong\""
val ( 2 ) = 1
else
convert ( 1 : 5 ) = "\"Plong\""
val ( 1 ) = 1
endif
endif
if (( val ( 1 ) + val ( 2 ) + val ( 3 )) == 0 ) then
write ( convert , '(I0)' ) i
endif
end function convert
end module raindrops
