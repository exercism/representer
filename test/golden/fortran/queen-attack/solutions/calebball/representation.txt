(file "queen_attack.f90")
module queen_attack
implicit none
contains
logical function isValid ( pos )
integer , dimension ( 2 ) :: pos
isValid = .true.
if ( any ( pos <= 0 )) isValid = .false.
if ( any ( pos > 8 )) isValid = .false.
end function
logical function canAttack ( white_pos , black_pos )
integer , dimension ( 2 ) :: white_pos , black_pos
integer :: diff ( 2 )
canAttack = .false.
if ( .not. ( isValid ( white_pos ) .and. isValid ( black_pos ))) return
diff = abs ( white_pos - black_pos )
if ( any ( white_pos == black_pos )) canAttack = .true.
if ( diff ( 1 ) == diff ( 2 )) canAttack = .true.
end function
end module
