(file "queen_attack.f90")
module queen_attack
implicit none
contains
logical function isValid ( pos )
integer , dimension ( 2 ) :: pos
isValid = ( pos ( 1 ) > 0 .and. pos ( 1 ) < 9 .and. pos ( 2 ) > 0 .and. pos ( 2 ) < 9 )
end function
logical function canAttack ( white_pos , black_pos )
integer , dimension ( 2 ) :: white_pos , black_pos
integer :: diff_white , diff_black
if ( .not. isValid ( white_pos ) .or. .not. isValid ( black_pos ) ) then
canAttack = .false.
return
end if
if ( white_pos ( 1 ) == black_pos ( 1 )) then
canAttack = .true.
return
end if
if ( white_pos ( 2 ) == black_pos ( 2 )) then
canAttack = .true.
return
end if
diff_white = abs ( white_pos ( 1 ) - black_pos ( 1 ))
diff_black = abs ( white_pos ( 2 ) - black_pos ( 2 ))
if ( diff_white == diff_black ) then
canAttack = .true.
return
end if
canAttack = .false.
end function
end module
