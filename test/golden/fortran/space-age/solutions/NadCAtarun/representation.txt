module space_age
implicit none
contains
double precision function earth_years ( seconds )
double precision , intent ( in ) :: seconds
earth_years = seconds / 31557600d0
end function
double precision function planet_factor ( planet )
character ( len = * ), intent ( in ) :: planet
select case ( planet )
case ( "\"Mercury\"" )
planet_factor = 0.2408467d0
case ( "\"Venus\"" )
planet_factor = 0.61519726d0
case ( "\"Earth\"" )
planet_factor = 1d0
case ( "\"Mars\"" )
planet_factor = 1.8808158d0
case ( "\"Jupiter\"" )
planet_factor = 11.862615d0
case ( "\"Saturn\"" )
planet_factor = 29.447498d0
case ( "\"Uranus\"" )
planet_factor = 84.016846d0
case ( "\"Neptune\"" )
planet_factor = 164.79132d0
end select
end function
double precision function age_in_years ( planet , seconds )
character ( len = * ), intent ( in ) :: planet
double precision , intent ( in ) :: seconds
age_in_years = earth_years ( seconds ) / planet_factor ( planet )
end function
end module
