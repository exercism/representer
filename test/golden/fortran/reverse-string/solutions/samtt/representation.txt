module reverse_string
implicit none
contains
function reverse ( input ) result ( reversed )
character ( len =* ), intent ( in ) :: input
character ( len = len ( input )) :: reversed
integer :: i , L
L = len ( input )
reversed = "\"\""
do i = 1 , L
reversed ( i : i ) = input ( L - i +1 : L - i +1 )
end do
end function
end module
