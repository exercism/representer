module rational_numbers
implicit none
private
public :: add , sub , mul , div , rational_abs , &
rational_to_pow , real_to_rational_pow , reduce
contains
function add ( r1 , r2 ) result ( s )
integer :: r1 ( 2 ), r2 ( 2 ), s ( 2 )
associate ( a1 => r1 ( 1 ), b1 => r1 ( 2 ), a2 => r2 ( 1 ), b2 => r2 ( 2 ))
s = reduce ([ a1 * b2 + a2 * b1 , b1 * b2 ])
end associate
end function add
function sub ( r1 , r2 ) result ( s )
integer :: r1 ( 2 ), r2 ( 2 ), s ( 2 )
s = add ( r1 , neg ( r2 ))
end function sub
function mul ( r1 , r2 ) result ( s )
integer :: r1 ( 2 ), r2 ( 2 ), s ( 2 )
s = reduce ( r1 * r2 )
end function mul
function div ( r1 , r2 ) result ( s )
integer :: r1 ( 2 ), r2 ( 2 ), s ( 2 )
s = mul ( r1 , recip ( r2 ))
end function div
function rational_abs ( r ) result ( s )
integer :: r ( 2 ), s ( 2 )
s = abs ( r )
end function rational_abs
function rational_to_pow ( r , n ) result ( s )
integer :: r ( 2 ), n , s ( 2 )
if ( n < 0 ) then
s = reduce ( recip ( r ) ** abs ( n ))
else
s = reduce ( r ** n )
end if
end function rational_to_pow
function real_to_rational_pow ( x , r ) result ( y )
integer :: r ( 2 )
real :: x , y
y = x ** ( 1.0 * r ( 1 ) / r ( 2 ))
end function real_to_rational_pow
function neg ( r ) result ( s )
integer :: r ( 2 ), s ( 2 )
s = r * [ -1 , 1 ]
end function neg
function reduce ( r ) result ( s )
integer :: r ( 2 ), s ( 2 )
s = sign ( 1 , r ( 2 )) * r / gcd ( r ( 1 ), r ( 2 ))
end function reduce
function recip ( r ) result ( s )
integer :: r ( 2 ), s ( 2 )
s = r ( 2 : 1 : -1 )
end function recip
recursive function gcd ( a , b ) result ( c )
integer :: a , b , c
if ( b == 0 ) then
c = abs ( a )
else
c = gcd ( b , mod ( a , b ))
end if
end function gcd
end module rational_numbers
