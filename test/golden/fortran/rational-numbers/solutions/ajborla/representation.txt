(file "rational_numbers.f90")
module rational_numbers
implicit none
private :: gcd , makeRatNum
contains
recursive function gcd ( x , y ) result ( gcdval )
integer :: x , y , gcdval
if ( y == 0 ) then ; gcdval = 0 ; return ; end if
if ( MOD ( x , y ) == 0 ) then ; gcdval = y ; return ; end if
gcdval = gcd ( y , MOD ( x , y ))
end function
function makeRatNum ( a , b ) result ( ratNum )
integer :: a , b
integer , dimension ( 2 ) :: ratNum
ratNum = (/ a , b /)
if ( a > 0 .and. b < 0 ) ratNum = (/ a * ( -1 ), b * ( -1 ) /)
end function
function add ( r1 , r2 )
integer , dimension ( 2 ) :: add , r1 , r2
integer :: a , b
a = r1 ( 1 ) * r2 ( 2 ) + r2 ( 1 ) * r1 ( 2 )
b = r1 ( 2 ) * r2 ( 2 )
add = reduce ( makeRatNum ( a , b ))
end function
function sub ( r1 , r2 )
integer , dimension ( 2 ) :: sub , r1 , r2
integer :: a , b
a = r1 ( 1 ) * r2 ( 2 ) - r2 ( 1 ) * r1 ( 2 )
b = r1 ( 2 ) * r2 ( 2 )
sub = reduce ( makeRatNum ( a , b ))
end function
function mul ( r1 , r2 )
integer , dimension ( 2 ) :: mul , r1 , r2
integer :: a , b
a = r1 ( 1 ) * r2 ( 1 )
b = r1 ( 2 ) * r2 ( 2 )
mul = reduce ( makeRatNum ( a , b ))
end function
function div ( r1 , r2 )
integer , dimension ( 2 ) :: div , r1 , r2
integer :: a , b
a = r1 ( 1 ) * r2 ( 2 )
b = r2 ( 1 ) * r1 ( 2 )
div = reduce ( makeRatNum ( a , b ))
end function
function rational_abs ( r1 )
integer , dimension ( 2 ) :: rational_abs , r1
integer :: a , b
a = ABS ( r1 ( 1 )) ; b = ABS ( r1 ( 2 ))
rational_abs = reduce ( makeRatNum ( a , b ))
end function
function rational_to_pow ( r1 , ex )
integer , dimension ( 2 ) :: rational_to_pow , r1
integer :: ex
integer :: a , b , m
a = r1 ( 1 ) ; b = r1 ( 2 ) ; m = ABS ( ex )
select case ( ex )
case (: -1 ) ; rational_to_pow = reduce ( makeRatNum ( b ** m , a ** m ))
case ( 0 ) ; rational_to_pow = makeRatNum ( 1 , 1 )
case ( 1 :) ; rational_to_pow = reduce ( makeRatNum ( a ** ex , b ** ex ))
end select
end function
function real_to_rational_pow ( ex , r1 )
integer , dimension ( 2 ) :: r1
real :: real_to_rational_pow , ex
real :: a_real , b_real
a_real = REAL ( r1 ( 1 )) ; b_real = REAL ( r1 ( 2 ))
real_to_rational_pow = ex ** ( a_real / b_real )
end function
function reduce ( r1 )
integer , dimension ( 2 ) :: reduce , r1
integer :: gcdval
gcdval = gcd ( r1 ( 1 ), r1 ( 2 ))
if ( gcdval == 0 ) then ; reduce = makeRatNum ( 0 , 1 )
else ; reduce = makeRatNum ( r1 ( 1 ) / gcdval , r1 ( 2 ) / gcdval ) ; end if
end function
end module
