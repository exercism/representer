module darts
implicit none
contains
function score ( x , y ) result ( points )
real , intent ( in ):: x , y
real :: distance
integer :: points
distance = sqrt ( x ** 2 + y ** 2 )
if ( distance <= 1 ) then
points = 10
else if ( distance <= 5 ) then
points = 5
else if ( distance <= 10 ) then
points = 1
else
points = 0
end if
return
end function score
end module darts
