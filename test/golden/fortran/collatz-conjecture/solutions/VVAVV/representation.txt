module collatz_conjecture
implicit none
contains
integer function steps ( i )
integer , intent ( in ) :: i
integer :: n
if ( i .le. 0 ) then
steps = -1
return
endif
n = i
steps = 0
do while ( n .ne. 1 )
n = f ( n )
steps = steps + 1
end do
end function steps
integer function f ( n )
integer , intent ( in ) :: n
if ( mod ( n , 2 ) .eq. 0 ) then
f = n / 2
else
f = 3 * n + 1
endif
end function f
end module
