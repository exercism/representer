(file "collatz_conjecture.f90")
module collatz_conjecture
implicit none
contains
integer function steps ( i ) result ( k )
integer :: i , n
k = 0
n = i
if ( i <= 0 ) then
k = -1
return
end if
do while ( n .ne. 1 )
if ( MOD ( n , 2 ) == 0 ) then
n = n / 2
else
n = ( 3 * n ) + 1
end if
k = k + 1
end do
end function
end module
