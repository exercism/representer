module collatz_conjecture
implicit none
contains
integer function steps ( i )
integer , intent ( in ) :: i
integer :: n
steps = -1
if ( i < 1 ) then
return
end if
n = i
steps = 0
do while ( n > 1 )
if ( mod ( n , 2 ) == 0 ) then
n = n / 2
else
n = 3 * n + 1
end if
steps = steps + 1
end do
end function
end module
