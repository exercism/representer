(file "anagram.rb"
  (program
    (class Anagram
      (body_statement
        (method initialize
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment (instance_variable PLACEHOLDER_2) =
              (call PLACEHOLDER_1 downcase))))
        (method match
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (call PLACEHOLDER_3 select
              (block
                (block_parameters PLACEHOLDER_4)
                (body_statement
                  (binary
                    (binary
                      (call PLACEHOLDER_4 downcase) != (instance_variable PLACEHOLDER_2)) &&
                    (binary
                      (call
                        (call
                          (call PLACEHOLDER_4 downcase) chars) sort) ==
                      (call
                        (call (instance_variable PLACEHOLDER_2) chars) sort))))))))))))
