(program
  (module Blackjack
    (body_statement
      (singleton_method self parse_card
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (case PLACEHOLDER_1
            (when
              (pattern "ace")
              (then 11))
            (when
              (pattern "two")
              (then 2))
            (when
              (pattern "three")
              (then 3))
            (when
              (pattern "four")
              (then 4))
            (when
              (pattern "five")
              (then 5))
            (when
              (pattern "six")
              (then 6))
            (when
              (pattern "seven")
              (then 7))
            (when
              (pattern "eight")
              (then 8))
            (when
              (pattern "nine")
              (then 9))
            (when
              (pattern "ten")
              (pattern "jack")
              (pattern "queen")
              (pattern "king")
              (then 10))
            (else 0))))
      (singleton_method self card_range
        (method_parameters PLACEHOLDER_2 PLACEHOLDER_3)
        (body_statement
          (assignment PLACEHOLDER_4 =
            (binary
              (call parse_card
                (argument_list PLACEHOLDER_2)) +
              (call parse_card
                (argument_list PLACEHOLDER_3))))
          (call debug
            (argument_list
              (string "The score is "
                (interpolation PLACEHOLDER_4) ".")))
          (case PLACEHOLDER_4
            (when
              (pattern
                (range 4 .. 11))
              (then "low"))
            (when
              (pattern
                (range 12 .. 16))
              (then "mid"))
            (when
              (pattern
                (range 17 .. 20))
              (then "high"))
            (when
              (pattern 21)
              (then "blackjack")))))
      (singleton_method self first_turn
        (method_parameters PLACEHOLDER_2 PLACEHOLDER_3 PLACEHOLDER_5)
        (body_statement
          (assignment PLACEHOLDER_6 =
            (call card_range
              (argument_list PLACEHOLDER_2 PLACEHOLDER_3)))
          (assignment PLACEHOLDER_7 =
            (call parse_card
              (argument_list PLACEHOLDER_5)))
          (if
            (binary
              (binary PLACEHOLDER_2 == "ace") &&
              (binary PLACEHOLDER_3 == "ace"))
            (then "P")
            (else
              (case PLACEHOLDER_6
                (when
                  (pattern "blackjack")
                  (then
                    (if
                      (binary PLACEHOLDER_7 >= 10)
                      (then "S")
                      (else "W"))))
                (when
                  (pattern "high")
                  (then "S"))
                (when
                  (pattern "mid")
                  (then
                    (if
                      (binary PLACEHOLDER_7 >= 7)
                      (then "H")
                      (else "S"))))
                (else "H")))))))))
