(program
  (module Blackjack
    (body_statement
      (singleton_method self parse_card
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (case PLACEHOLDER_1
            (when
              (pattern "ace")
              (then 11))
            (when
              (pattern "two")
              (then 2))
            (when
              (pattern "three")
              (then 3))
            (when
              (pattern "four")
              (then 4))
            (when
              (pattern "five")
              (then 5))
            (when
              (pattern "six")
              (then 6))
            (when
              (pattern "seven")
              (then 7))
            (when
              (pattern "eight")
              (then 8))
            (when
              (pattern "nine")
              (then 9))
            (when
              (pattern "ten")
              (pattern "jack")
              (pattern "queen")
              (pattern "king")
              (then 10))
            (else 0))))
      (singleton_method self card_range
        (method_parameters PLACEHOLDER_2 PLACEHOLDER_3)
        (body_statement
          (case
            (binary
              (call parse_card
                (argument_list PLACEHOLDER_2)) +
              (call parse_card
                (argument_list PLACEHOLDER_3)))
            (when
              (pattern
                (range 4 .. 11))
              (then "low"))
            (when
              (pattern
                (range 12 .. 16))
              (then "mid"))
            (when
              (pattern
                (range 17 .. 20))
              (then "high"))
            (when
              (pattern 21)
              (then "blackjack")))))
      (singleton_method self first_turn
        (method_parameters PLACEHOLDER_2 PLACEHOLDER_3 PLACEHOLDER_4)
        (body_statement
          (case
            (when
              (pattern
                (binary
                  (binary PLACEHOLDER_2 == "ace") &&
                  (binary PLACEHOLDER_3 == "ace")))
              (then "P"))
            (when
              (pattern
                (binary
                  (binary
                    (call card_range
                      (argument_list PLACEHOLDER_2 PLACEHOLDER_3)) == "blackjack") &&
                  (binary
                    (call parse_card
                      (argument_list PLACEHOLDER_4)) < 10)))
              (then "W"))
            (when
              (pattern
                (binary
                  (call card_range
                    (argument_list PLACEHOLDER_2 PLACEHOLDER_3)) == "high"))
              (then "S"))
            (when
              (pattern
                (binary
                  (binary
                    (call card_range
                      (argument_list PLACEHOLDER_2 PLACEHOLDER_3)) == "mid") &&
                  (binary
                    (call parse_card
                      (argument_list PLACEHOLDER_4)) >= 7)))
              (then "H"))
            (when
              (pattern
                (binary
                  (call card_range
                    (argument_list PLACEHOLDER_2 PLACEHOLDER_3)) == "low"))
              (then "H"))
            (else "S")))))))
