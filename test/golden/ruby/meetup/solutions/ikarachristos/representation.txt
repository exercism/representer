          (assignment (instance_variable PLACEHOLDER_3) PLACEHOLDER_1)
          (assignment (instance_variable PLACEHOLDER_4) PLACEHOLDER_2)))
      (method day
        (method_parameters PLACEHOLDER_5 PLACEHOLDER_6)
        (body_statement
          (assignment PLACEHOLDER_7
            (call Date new
              (argument_list (instance_variable PLACEHOLDER_4) (instance_variable PLACEHOLDER_3) 1)))
          (assignment PLACEHOLDER_8
            (element_reference WEEK_DAYS PLACEHOLDER_5))
          (assignment PLACEHOLDER_9
            (array))
          (while
            (binary
              (call PLACEHOLDER_7 month) == (instance_variable PLACEHOLDER_3))
            (do
              (if
                (binary
                  (call PLACEHOLDER_7 cwday) == PLACEHOLDER_8)
                (then
                  (binary PLACEHOLDER_9 << PLACEHOLDER_7)))
              (operator_assignment PLACEHOLDER_7 += 1)))
          (case PLACEHOLDER_6
            (when
              (pattern :first)
              (then
                (element_reference PLACEHOLDER_9 0)))
            (when
              (pattern :second)
              (then
                (element_reference PLACEHOLDER_9 1)))
            (when
              (pattern :third)
              (then
                (element_reference PLACEHOLDER_9 2)))
            (when
              (pattern :fourth)
              (then
                (element_reference PLACEHOLDER_9 3)))
            (when
              (pattern :last)
              (then
                (element_reference PLACEHOLDER_9
                  (unary - 1))))
            (when
              (pattern :teenth)
              (then
                (call PLACEHOLDER_9 find
                  (block
                    (block_parameters PLACEHOLDER_10)
                    (body_statement
                      (binary
                        (binary
                          (call PLACEHOLDER_10 day) > 12) &&
                        (binary
                          (call PLACEHOLDER_10 day) < 20)))))))
            (else
              (call PLACEHOLDER_9 first))))))))
