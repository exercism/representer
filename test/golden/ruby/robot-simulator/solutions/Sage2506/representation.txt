(program
  (class Robot
    (body_statement
      (assignment INCREMENTS
        (hash
          (pair :north
            (array 0 1))
          (pair :east
            (array 1 0))
          (pair :south
            (array 0
              (unary - 1)))
          (pair :west
            (array
              (unary - 1) 0))))
      (assignment VALID_BEARINGS
        (call INCREMENTS keys))
      (call attr_reader
        (argument_list :bearing :coordinates))
      (method orient
        (method_parameters direction)
        (body_statement
          (if
            (call VALID_BEARINGS include?
              (argument_list direction))
            (else
              (call raise
                (argument_list ArgumentError))))
          (assignment (instance_variable PLACEHOLDER_1) direction)))
      (method turn_right
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1)
            (binary
              (element_reference VALID_BEARINGS
                (binary
                  (call VALID_BEARINGS index
                    (argument_list (instance_variable PLACEHOLDER_1))) + 1)) || :north))))
      (method turn_left
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1)
            (element_reference VALID_BEARINGS
              (binary
                (call VALID_BEARINGS index
                  (argument_list (instance_variable PLACEHOLDER_1))) - 1)))))
      (method at
        (method_parameters
          (splat_parameter coordinates))
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) coordinates)))
      (method advance
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2)
            (call
              (call (instance_variable PLACEHOLDER_2) zip
                (argument_list
                  (element_reference INCREMENTS (instance_variable PLACEHOLDER_1)))) map
              (block
                (block_parameters PLACEHOLDER_3 PLACEHOLDER_4)
                (body_statement
                  (binary PLACEHOLDER_3 + PLACEHOLDER_4)))))))))
  (class Simulator
    (body_statement
      (assignment COMMANDS
        (hash
          (pair "L" :turn_left)
          (pair "R" :turn_right)
          (pair "A" :advance)))
      (method instructions
        (method_parameters PLACEHOLDER_5)
        (body_statement
          (call
            (call PLACEHOLDER_5 chars) map
            (block
              (block_parameters PLACEHOLDER_6)
              (body_statement
                (element_reference COMMANDS PLACEHOLDER_6))))))
      (method place
        (method_parameters robot
          (keyword_parameter x 0)
          (keyword_parameter y 0)
          (keyword_parameter direction :north))
        (body_statement
          (call robot at
            (argument_list x y))
          (call robot orient
            (argument_list direction))))
      (method evaluate
        (method_parameters robot PLACEHOLDER_5)
        (body_statement
          (call
            (call instructions
              (argument_list PLACEHOLDER_5)) each
            (block
              (block_parameters PLACEHOLDER_6)
              (body_statement
                (call robot send
                  (argument_list PLACEHOLDER_6))))))))))
