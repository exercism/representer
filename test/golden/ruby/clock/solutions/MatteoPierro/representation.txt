(file "clock.rb"
  (program
    (class Clock
      (body_statement
        (call attr_reader
          (argument_list :hour :minute))
        (method initialize
          (method_parameters
            (keyword_parameter hour 0)
            (keyword_parameter minute 0))
          (body_statement
            (assignment (instance_variable PLACEHOLDER_1) =
              (binary
                (binary hour +
                  (binary minute / 60)) % 24))
            (assignment (instance_variable PLACEHOLDER_2) =
              (binary minute % 60))))
        (method
          (operator +)
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (assignment hour =
              (binary
                (call self hour) +
                (call PLACEHOLDER_3 hour)))
            (assignment minute =
              (binary
                (call self minute) +
                (call PLACEHOLDER_3 minute)))
            (call Clock new
              (argument_list
                (pair :hour hour)
                (pair :minute minute)))))
        (method
          (operator -)
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (assignment hour =
              (binary
                (call self hour) -
                (call PLACEHOLDER_3 hour)))
            (assignment minute =
              (binary
                (call self minute) -
                (call PLACEHOLDER_3 minute)))
            (call Clock new
              (argument_list
                (pair :hour hour)
                (pair :minute minute)))))
        (method
          (operator ==)
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (binary
              (binary hour ==
                (call PLACEHOLDER_3 hour)) &&
              (binary minute ==
                (call PLACEHOLDER_3 minute)))))
        (method to_s
          (body_statement
            (binary "%02d:%02d" %
              (array hour minute))))))))
