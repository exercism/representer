(program
  (module SavingsAccount
    (body_statement
      (singleton_method self interest_rate
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (if
            (binary
              (binary PLACEHOLDER_1 >= 0) &&
              (binary PLACEHOLDER_1 < 1000))
            (then 0.5)
            (elsif
              (binary
                (binary PLACEHOLDER_1 >= 100) &&
                (binary PLACEHOLDER_1 < 5000))
              (then 1.621)
              (elsif
                (binary PLACEHOLDER_1 >= 5000)
                (then 2.475)
                (else 3.213))))))
      (singleton_method self annual_balance_update
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (binary PLACEHOLDER_1 *
            (binary 1.0 +
              (binary
                (call interest_rate
                  (argument_list PLACEHOLDER_1)) / 100.0)))))
      (singleton_method self years_before_desired_balance
        (method_parameters PLACEHOLDER_2 PLACEHOLDER_3)
        (body_statement
          (assignment PLACEHOLDER_4 = 0)
          (assignment PLACEHOLDER_5 = PLACEHOLDER_2)
          (until
            (binary PLACEHOLDER_5 >= PLACEHOLDER_3)
            (do
              (operator_assignment PLACEHOLDER_4 += 1)
              (assignment PLACEHOLDER_5 =
                (call annual_balance_update
                  (argument_list PLACEHOLDER_5))))) PLACEHOLDER_4)))))
