(file "savings_account.rb"
  (program
    (module SavingsAccount
      (body_statement
        (singleton_method self interest_rate
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (if
              (binary PLACEHOLDER_1 < 0)
              (then
                (return
                  (argument_list 3.213))))
            (if
              (binary PLACEHOLDER_1 < 1_000)
              (then
                (return
                  (argument_list 0.500))))
            (if
              (binary PLACEHOLDER_1 < 5_000)
              (then
                (return
                  (argument_list 1.621))))
            (return
              (argument_list 2.475))))
        (singleton_method self annual_balance_update
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (binary PLACEHOLDER_1 +
              (binary PLACEHOLDER_1 *
                (binary
                  (call interest_rate
                    (argument_list PLACEHOLDER_1)) / 100)))))
        (singleton_method self years_before_desired_balance
          (method_parameters PLACEHOLDER_2 PLACEHOLDER_3)
          (body_statement
            (assignment PLACEHOLDER_4 = 0)
            (while
              (binary PLACEHOLDER_2 < PLACEHOLDER_3)
              (do
                (assignment PLACEHOLDER_2 =
                  (call annual_balance_update
                    (argument_list PLACEHOLDER_2)))
                (operator_assignment PLACEHOLDER_4 += 1))) PLACEHOLDER_4))))))
