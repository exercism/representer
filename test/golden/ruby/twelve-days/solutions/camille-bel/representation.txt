(file "twelve_days.rb"
  (program
    (class TwelveDays
      (body_statement
        (assignment (constant PLACEHOLDER_1) =
          (call
            (array "a Partridge in a Pear Tree." "two Turtle Doves, and " "three French Hens, " "four Calling Birds, " "five Gold Rings, " "six Geese-a-Laying, " "seven Swans-a-Swimming, " "eight Maids-a-Milking, " "nine Ladies Dancing, " "ten Lords-a-Leaping, " "eleven Pipers Piping, " "twelve Drummers Drumming, ") freeze))
        (assignment (constant PLACEHOLDER_2) =
          (call
            (string_array "%w("
              (bare_string first)
              (bare_string second)
              (bare_string third)
              (bare_string fourth)
              (bare_string fifth)
              (bare_string sixth)
              (bare_string seventh)
              (bare_string eighth)
              (bare_string ninth)
              (bare_string tenth)
              (bare_string eleventh)
              (bare_string twelfth)) freeze))
        (singleton_method self song
          (body_statement
            (assignment PLACEHOLDER_3 = 0)
            (assignment song = "")
            (call
              (range 0 ... 12) each
              (block
                (block_parameters PLACEHOLDER_4)
                (body_statement
                  (assignment PLACEHOLDER_5 = "")
                  (assignment PLACEHOLDER_6 = PLACEHOLDER_4)
                  (while
                    (binary PLACEHOLDER_6 >= 0)
                    (do
                      (operator_assignment PLACEHOLDER_5 +=
                        (element_reference (constant PLACEHOLDER_1) PLACEHOLDER_6))
                      (operator_assignment PLACEHOLDER_6 -= 1)))
                  (operator_assignment song +=
                    (string "On the "
                      (interpolation
                        (element_reference (constant PLACEHOLDER_2) PLACEHOLDER_4)) " day of Christmas my true love gave to me: "
                      (interpolation PLACEHOLDER_5) "\n"))
                  (unless
                    (binary PLACEHOLDER_4 == 11)
                    (then
                      (operator_assignment song += "\n")))
                  (operator_assignment PLACEHOLDER_3 += 1)))) song))))))
