(program
  (class Attendee
    (body_statement
      (method initialize
        (method_parameters height)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1) = height)
          (assignment (instance_variable PLACEHOLDER_2) =
            (nil))))
      (method height
        (body_statement (instance_variable PLACEHOLDER_1)))
      (method pass_id
        (body_statement (instance_variable PLACEHOLDER_2)))
      (method issue_pass!
        (method_parameters pass_id)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) = 1)))
      (method revoke_pass!
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) = false))))))
