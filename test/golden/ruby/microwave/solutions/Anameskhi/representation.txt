(file "microwave.rb"
  (program
    (class Microwave
      (body_statement
        (method initialize
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment PLACEHOLDER_2 =
              (binary PLACEHOLDER_1 % 100))
            (assignment (instance_variable PLACEHOLDER_3) =
              (binary
                (binary PLACEHOLDER_1 / 100) +
                (binary PLACEHOLDER_2 / 60)))
            (assignment (instance_variable PLACEHOLDER_4) =
              (binary PLACEHOLDER_2 % 60))))
        (method timer
          (body_statement
            (call format
              (argument_list "%02d:%02d" (instance_variable PLACEHOLDER_3) (instance_variable PLACEHOLDER_4)))))))))
