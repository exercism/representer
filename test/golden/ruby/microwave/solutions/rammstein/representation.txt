(program
  (class Microwave
    (body_statement
      (call attr_reader
        (argument_list :timer))
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2)
            (binary PLACEHOLDER_1 / 100))
          (assignment (instance_variable PLACEHOLDER_3)
            (binary PLACEHOLDER_1 % 100))
          (if
            (binary (instance_variable PLACEHOLDER_3) > 59)
            (then
              (assignment
                (left_assignment_list (instance_variable PLACEHOLDER_2) (instance_variable PLACEHOLDER_3))
                (array
                  (binary (instance_variable PLACEHOLDER_2) + 1)
                  (binary (instance_variable PLACEHOLDER_3) - 60)))))
          (assignment (instance_variable PLACEHOLDER_4)
            (binary "%02d:%02d" %
              (array (instance_variable PLACEHOLDER_2) (instance_variable PLACEHOLDER_3)))))))))
