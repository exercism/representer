(program
  (class Microwave
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method timer
        (body_statement
          (assignment
            (left_assignment_list PLACEHOLDER_3 PLACEHOLDER_4)
            (call (instance_variable PLACEHOLDER_2) divmod
              (argument_list 100)))
          (operator_assignment PLACEHOLDER_3 +=
            (binary PLACEHOLDER_4 / 60))
          (operator_assignment PLACEHOLDER_4 %= 60)
          (binary "%02d:%02d" %
            (array PLACEHOLDER_3 PLACEHOLDER_4)))))))
