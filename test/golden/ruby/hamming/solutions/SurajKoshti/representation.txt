(program
  (class Hamming
    (body_statement
      (singleton_method self compute
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_2)
        (body_statement
          (if
            (binary
              (call PLACEHOLDER_1 length) !=
              (call PLACEHOLDER_2 length))
            (then
              (call raise
                (argument_list ArgumentError "The lengths of the strands must match.")))
            (else
              (assignment
                (left_assignment_list PLACEHOLDER_3 PLACEHOLDER_4) =
                (right_assignment_list 0 0))
              (assignment PLACEHOLDER_1 =
                (call PLACEHOLDER_1 split
                  (argument_list "")))
              (assignment PLACEHOLDER_2 =
                (call PLACEHOLDER_2 split
                  (argument_list "")))
              (while
                (binary PLACEHOLDER_3 <
                  (call PLACEHOLDER_1 length))
                (do
                  (if
                    (binary
                      (element_reference PLACEHOLDER_1 PLACEHOLDER_3) !=
                      (element_reference PLACEHOLDER_2 PLACEHOLDER_3))
                    (then
                      (assignment PLACEHOLDER_4 =
                        (binary PLACEHOLDER_4 + 1))))
                  (assignment PLACEHOLDER_3 =
                    (binary PLACEHOLDER_3 + 1))))
              (return
                (argument_list PLACEHOLDER_4)))))))))
