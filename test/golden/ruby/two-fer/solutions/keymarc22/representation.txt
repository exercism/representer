(file "two_fer.rb"
  (program
    (class TwoFer
      (body_statement
        (singleton_method self two_fer
          (method_parameters
            (splat_parameter * PLACEHOLDER_1))
          (body_statement
            (conditional
              (call PLACEHOLDER_1 empty?)
              (call TwoFer PLACEHOLDER_2)
              (call TwoFer PLACEHOLDER_3
                (argument_list PLACEHOLDER_1)))))
        (singleton_method self PLACEHOLDER_2
          (body_statement "One for you, one for me."))
        (singleton_method self PLACEHOLDER_3
          (method_parameters PLACEHOLDER_4)
          (body_statement
            (string "One for "
              (interpolation
                (element_reference PLACEHOLDER_4 0)) ", one for me.")))))))
