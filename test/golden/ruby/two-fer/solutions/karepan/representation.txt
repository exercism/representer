(program
  (class TwoFer
    (body_statement
      (singleton_method self two_fer
        (method_parameters
          (optional_parameter PLACEHOLDER_1 = 1))
        (body_statement
          (if
            (binary PLACEHOLDER_1 == 1)
            (then
              (return
                (argument_list "One for you, one for me.")))
            (else
              (return
                (argument_list
                  (string "One for "
                    (interpolation PLACEHOLDER_1) ", one for me."))))))))))
