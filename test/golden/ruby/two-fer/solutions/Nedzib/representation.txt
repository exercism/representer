(file "two_fer.rb"
  (program
    (class TwoFer
      (body_statement
        (method initialize)
        (method two_fer
          (method_parameters
            (optional_parameter PLACEHOLDER_1 = "you"))
          (body_statement
            (return
              (argument_list
                (string "One for "
                  (interpolation PLACEHOLDER_1) ", one for me.")))))))
    (assignment TwoFer =
      (call TwoFer new
        (argument_list)))
    (call puts
      (argument_list
        (call TwoFer two_fer)))))
