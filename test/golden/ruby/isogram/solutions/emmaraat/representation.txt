(file "isogram.rb"
  (program
    (class Isogram
      (body_statement
        (singleton_method self isogram?
          (method_parameters input)
          (body_statement
            (assignment PLACEHOLDER_1 =
              (call
                (call input downcase) scan
                (argument_list
                  (regex / [a-z] /))))
            (assignment PLACEHOLDER_2 =
              (call PLACEHOLDER_1 uniq))
            (if
              (binary
                (call PLACEHOLDER_1 length) ==
                (call PLACEHOLDER_2 length))
              (then true))))))))
