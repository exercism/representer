(program
  (class PigLatin
    (body_statement
      (assignment VOWELS
        (regex "a|e|i|o|u"))
      (assignment CONSONANTS
        (regex "[^"
          (interpolation VOWELS) "]"))
      (singleton_method self translate
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (call
            (call
              (call PLACEHOLDER_1 split) map
              (block
                (block_parameters PLACEHOLDER_2)
                (body_statement
                  (call translate_word
                    (argument_list PLACEHOLDER_2))))) join
            (argument_list " "))))
      (singleton_method self translate_word
        (method_parameters PLACEHOLDER_3)
        (body_statement
          (assignment PLACEHOLDER_4
            (call
              (call PLACEHOLDER_3 partition
                (argument_list
                  (regex
                    (interpolation VOWELS) "|"
                    (interpolation CONSONANTS) "*qu|xr|y"))) reject
              (argument_list
                (block_argument :empty?))))
          (if
            (binary
              (binary
                (element_reference PLACEHOLDER_4 0) =~
                (regex "^"
                  (interpolation VOWELS) "|(xr)")) ||
              (binary
                (binary
                  (element_reference PLACEHOLDER_4 0) == "y") &&
                (binary
                  (element_reference PLACEHOLDER_4 1) =~
                  (regex "^"
                    (interpolation CONSONANTS)))))
            (else
              (call PLACEHOLDER_4 rotate!
                (argument_list 1))))
          (binary
            (call PLACEHOLDER_4 join) << "ay"))))))
