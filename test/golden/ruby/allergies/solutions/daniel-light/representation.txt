(program
  (class Allergies
    (body_statement
      (assignment ALLERGEN_SCORES
        (hash
          (pair "eggs" 1)
          (pair "peanuts" 2)
          (pair "shellfish" 4)
          (pair "strawberries" 8)
          (pair "tomatoes" 16)
          (pair "chocolate" 32)
          (pair "pollen" 64)
          (pair "cats" 128)))
      (call attr_reader
        (argument_list (simple_symbol :PLACEHOLDER_1)))
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method allergic_to?
        (method_parameters PLACEHOLDER_3)
        (body_statement
          (call
            (binary PLACEHOLDER_1 &
              (element_reference ALLERGEN_SCORES PLACEHOLDER_3)) nonzero?)))
      (method list
        (body_statement
          (call
            (call ALLERGEN_SCORES keys) select
            (block
              (block_parameters PLACEHOLDER_3)
              (body_statement
                (call allergic_to?
                  (argument_list PLACEHOLDER_3))))))))))
