  (class Allergies
    (body_statement
      (assignment (constant PLACEHOLDER_1) =
        (hash
          (pair "eggs" 1)
          (pair "peanuts" 2)
          (pair "shellfish" 4)
          (pair "strawberries" 8)
          (pair "tomatoes" 16)
          (pair "chocolate" 32)
          (pair "pollen" 64)
          (pair "cats" 128)))

        (method_parameters PLACEHOLDER_2)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_3) = PLACEHOLDER_2)))

        (method_parameters PLACEHOLDER_4)
        (body_statement
          (call
            (binary PLACEHOLDER_2 &
              (element_reference (constant PLACEHOLDER_1) PLACEHOLDER_4)) nonzero?)))

        (body_statement
          (call
            (call (constant PLACEHOLDER_1) keys) select

                (call allergic_to?
                  (argument_list PLACEHOLDER_4))))))))))
