(file "allergies.rb"
  (program
    (class Allergies
      (body_statement
        (assignment (constant PLACEHOLDER_1) =
          (hash
            (pair "eggs" 1)
            (pair "peanuts" 2)
            (pair "shellfish" 4)
            (pair "strawberries" 8)
            (pair "tomatoes" 16)
            (pair "chocolate" 32)
            (pair "pollen" 64)
            (pair "cats" 128)))
        (method initialize
          (method_parameters PLACEHOLDER_2)
          (body_statement
            (assignment (instance_variable PLACEHOLDER_3) = PLACEHOLDER_2)))
        (method allergic_to?
          (method_parameters PLACEHOLDER_4)
          (body_statement
            (binary
              (binary (instance_variable PLACEHOLDER_3) &
                (element_reference (constant PLACEHOLDER_1) PLACEHOLDER_4)) > 0)))
        (method list
          (body_statement
            (call
              (call (constant PLACEHOLDER_1) keys) select
              (block
                (block_parameters PLACEHOLDER_4)
                (body_statement
                  (call allergic_to?
                    (argument_list PLACEHOLDER_4)))))))))))
