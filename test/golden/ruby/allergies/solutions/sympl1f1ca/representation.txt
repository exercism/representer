(program
  (class Allergies
    (body_statement
      (assignment ALLERGIES_HASH
        (hash
          (pair "eggs" 1)
          (pair "peanuts" 2)
          (pair "shellfish" 4)
          (pair "strawberries" 8)
          (pair "tomatoes" 16)
          (pair "chocolate" 32)
          (pair "pollen" 64)
          (pair "cats" 128)))
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method allergic_to?
        (method_parameters PLACEHOLDER_3)
        (body_statement
          (binary
            (binary (instance_variable PLACEHOLDER_2) &
              (element_reference ALLERGIES_HASH PLACEHOLDER_3)) > 0)))
      (method list
        (body_statement
          (call
            (call ALLERGIES_HASH keys) select
            (block
              (block_parameters PLACEHOLDER_3)
              (body_statement
                (call allergic_to?
                  (argument_list PLACEHOLDER_3))))))))))
