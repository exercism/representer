(file "run_length_encoding.rb"
  (program
    (class RunLengthEncoding
      (body_statement
        (singleton_method self encode
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (call PLACEHOLDER_1 gsub
              (argument_list
                (regex / "(.)" "\\1" + /))
              (block
                (block_parameters PLACEHOLDER_2)
                (body_statement
                  (string
                    (interpolation
                      (call PLACEHOLDER_2 length))
                    (interpolation
                      (element_reference PLACEHOLDER_2 0))))))))
        (singleton_method self decode
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (call PLACEHOLDER_1 gsub
              (argument_list
                (regex / "\\d" +. /))
              (block
                (block_parameters PLACEHOLDER_2)
                (body_statement
                  (binary
                    (element_reference PLACEHOLDER_2
                      (unary - 1)) *
                    (call PLACEHOLDER_2 to_i)))))))))))
