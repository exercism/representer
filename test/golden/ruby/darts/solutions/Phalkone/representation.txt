(program
  (class Darts
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_2)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_3)
            (call Math sqrt
              (argument_list
                (binary
                  (binary PLACEHOLDER_1 ** 2) +
                  (binary PLACEHOLDER_2 ** 2)))))))
      (method score
        (body_statement
          (case (instance_variable PLACEHOLDER_3)
            (when
              (pattern
                (range 0 .. 1))
              (then 10))
            (when
              (pattern
                (range 1 .. 5))
              (then 5))
            (when
              (pattern
                (range 5 .. 10))
              (then 1))
            (else 0)))))))
