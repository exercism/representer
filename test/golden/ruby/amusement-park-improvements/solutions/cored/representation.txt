(program
  (class Attendee
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) = PLACEHOLDER_1)))
      (method issue_pass!
        (method_parameters PLACEHOLDER_3)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_4) = PLACEHOLDER_3)))
      (method revoke_pass!
        (body_statement
          (assignment (instance_variable PLACEHOLDER_4) =
            (nil))))
      (method has_pass?
        (body_statement
          (binary (instance_variable PLACEHOLDER_4) !=
            (nil))))
      (method fits_ride?
        (method_parameters PLACEHOLDER_5)
        (body_statement
          (binary (instance_variable PLACEHOLDER_2) >= PLACEHOLDER_5)))
      (method allowed_to_ride?
        (method_parameters PLACEHOLDER_5)
        (body_statement
          (binary
            (call has_pass?) &&
            (call fits_ride?
              (argument_list PLACEHOLDER_5))))))))
