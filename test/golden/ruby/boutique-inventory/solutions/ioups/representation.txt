(program
  (class BoutiqueInventory
    (body_statement
      (method initialize
        (method_parameters items)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1) = items)))
      (method item_names
        (body_statement
          (call
            (call (instance_variable PLACEHOLDER_1) map
              (block
                (block_parameters PLACEHOLDER_2)
                (body_statement
                  (element_reference PLACEHOLDER_2 :name)))) sort)))
      (method cheap
        (body_statement
          (call (instance_variable PLACEHOLDER_1) select
            (block
              (block_parameters PLACEHOLDER_2)
              (body_statement
                (binary
                  (element_reference PLACEHOLDER_2 :price) < 30))))))
      (method out_of_stock
        (body_statement
          (call (instance_variable PLACEHOLDER_1) select
            (block
              (block_parameters PLACEHOLDER_2)
              (body_statement
                (call
                  (element_reference PLACEHOLDER_2 :quantity_by_size) none?))))))
      (method stock_for_item
        (method_parameters name)
        (body_statement
          (element_reference
            (call (instance_variable PLACEHOLDER_1) find
              (block
                (block_parameters PLACEHOLDER_2)
                (body_statement
                  (binary
                    (element_reference PLACEHOLDER_2 :name) == name)))) :quantity_by_size)))
      (method total_stock
        (body_statement
          (call (instance_variable PLACEHOLDER_1) sum
            (block
              (block_parameters PLACEHOLDER_2)
              (body_statement
                (call
                  (call
                    (element_reference PLACEHOLDER_2 :quantity_by_size) values) sum)))))) private
      (call attr_reader
        (argument_list :items)))))
