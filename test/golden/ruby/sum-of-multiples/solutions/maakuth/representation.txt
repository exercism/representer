(program
  (class SumOfMultiples
    (body_statement
      (method initialize
        (method_parameters
          (splat_parameter PLACEHOLDER_1))
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method to
        (method_parameters PLACEHOLDER_3)
        (body_statement
          (assignment PLACEHOLDER_4
            (array))
          (call
            (range 1 ..
              (binary PLACEHOLDER_3 - 1)) each
            (block
              (block_parameters PLACEHOLDER_5)
              (body_statement
                (if
                  (binary
                    (call (instance_variable PLACEHOLDER_2) any?
                      (block
                        (block_parameters PLACEHOLDER_6)
                        (body_statement
                          (binary
                            (binary PLACEHOLDER_5 % PLACEHOLDER_6) == 0)))) &&
                    (unary !
                      (call PLACEHOLDER_4 include?
                        (argument_list PLACEHOLDER_5))))
                  (then
                    (binary PLACEHOLDER_4 << PLACEHOLDER_5))))))
          (binary
            (call PLACEHOLDER_4 reduce
              (argument_list :+)) || 0))))))
