(file "sum_of_multiples.rb"
  (program
    (class SumOfMultiples
      (body_statement
        (method initialize
          (method_parameters
            (splat_parameter * PLACEHOLDER_1))
          (body_statement
            (assignment (instance_variable PLACEHOLDER_2) = PLACEHOLDER_1)))
        (method to
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (call
              (call
                (range 1 ... PLACEHOLDER_3) select
                (block
                  (block_parameters PLACEHOLDER_4)
                  (body_statement
                    (call (instance_variable PLACEHOLDER_2) any?
                      (block
                        (block_parameters PLACEHOLDER_5)
                        (body_statement
                          (binary
                            (binary PLACEHOLDER_4 % PLACEHOLDER_5) == 0))))))) sum)))))))
