(file "lasagna.rb"
  (program
    (class Lasagna
      (body_statement
        (assignment EXPECTED_MINUTES_IN_OVEN = 40)
        (assignment (constant PLACEHOLDER_1) = 2)
        (method remaining_minutes_in_oven
          (method_parameters actual_minutes_in_oven)
          (body_statement
            (call PLACEHOLDER_2
              (argument_list actual_minutes_in_oven))
            (binary EXPECTED_MINUTES_IN_OVEN - actual_minutes_in_oven)))
        (method preparation_time_in_minutes
          (method_parameters PLACEHOLDER_3)
          (body_statement
            (call PLACEHOLDER_2
              (argument_list PLACEHOLDER_3))
            (binary (constant PLACEHOLDER_1) * PLACEHOLDER_3)))
        (method total_time_in_minutes
          (method_parameters
            (keyword_parameter number_of_layers)
            (keyword_parameter actual_minutes_in_oven))
          (body_statement
            (call PLACEHOLDER_2
              (argument_list actual_minutes_in_oven))
            (binary
              (call preparation_time_in_minutes
                (argument_list number_of_layers)) + actual_minutes_in_oven))) private
        (method PLACEHOLDER_2
          (method_parameters PLACEHOLDER_4)
          (body_statement
            (unless
              (call PLACEHOLDER_4 is_a?
                (argument_list Numeric))
              (then
                (call raise
                  (argument_list "minute should be numeric value"))))
            (unless
              (call PLACEHOLDER_4 positive?)
              (then
                (call raise
                  (argument_list "minute should be positive number")))) true))))))
