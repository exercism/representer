(file "lasagna.rb"
  (program
    (class Lasagna
      (body_statement
        (assignment EXPECTED_MINUTES_IN_OVEN = 40)
        (assignment (constant PLACEHOLDER_1) = 2)
        (method remaining_minutes_in_oven
          (method_parameters actual_minutes_in_oven)
          (body_statement
            (assignment remaining_minutes_in_oven =
              (binary EXPECTED_MINUTES_IN_OVEN - actual_minutes_in_oven))
            (return
              (argument_list
                (conditional
                  (binary remaining_minutes_in_oven > 0) remaining_minutes_in_oven 0)))))
        (method preparation_time_in_minutes
          (method_parameters PLACEHOLDER_2)
          (body_statement
            (return
              (argument_list
                (binary PLACEHOLDER_2 * (constant PLACEHOLDER_1))))))
        (method total_time_in_minutes
          (method_parameters
            (keyword_parameter number_of_layers)
            (keyword_parameter actual_minutes_in_oven))
          (body_statement
            (return
              (argument_list
                (binary actual_minutes_in_oven +
                  (call self preparation_time_in_minutes
                    (argument_list number_of_layers)))))))))))
