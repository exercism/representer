(program
  (class Lasagna
    (body_statement
      (assignment EXPECTED_MINUTES_IN_OVEN = 40)
      (method remaining_minutes_in_oven
        (method_parameters actual_minutes_in_oven)
        (body_statement
          (return
            (argument_list
              (binary EXPECTED_MINUTES_IN_OVEN - actual_minutes_in_oven)))
          (call raise
            (argument_list "Please implement the Lasagna#remaining_minutes_in_oven method"))))
      (method preparation_time_in_minutes
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (return
            (argument_list
              (binary PLACEHOLDER_1 * 2)))
          (call raise
            (argument_list "Please implement the Lasagna#preparation_time_in_minutes method"))))
      (method total_time_in_minutes
        (method_parameters
          (keyword_parameter number_of_layers)
          (keyword_parameter actual_minutes_in_oven))
        (body_statement
          (return
            (argument_list
              (binary
                (binary number_of_layers * 2) + actual_minutes_in_oven)))
          (call raise
            (argument_list "Please implement the Lasagna#total_time_in_minutes method"))))))
  (assignment PLACEHOLDER_2 =
    (call Lasagna new))
  (call PLACEHOLDER_2 remaining_minutes_in_oven
    (argument_list 30))
  (call PLACEHOLDER_2 preparation_time_in_minutes
    (argument_list 2))
  (call PLACEHOLDER_2 total_time_in_minutes
    (argument_list
      (pair :number_of_layers 3)
      (pair :actual_minutes_in_oven 20))))
