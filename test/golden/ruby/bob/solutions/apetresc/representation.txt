(program
  (module Bob
    (body_statement
      (singleton_method self hey
        (method_parameters remark)
        (body_statement
          (assignment remark =
            (call remark strip))
          (assignment PLACEHOLDER_1 =
            (call remark end_with?
              (argument_list "?")))
          (assignment PLACEHOLDER_2 =
            (binary
              (binary
                (call remark upcase) == remark) &&
              (binary
                (call remark count
                  (argument_list "A-Z")) > 0)))
          (assignment PLACEHOLDER_3 =
            (call remark empty?))
          (if
            (binary PLACEHOLDER_1 && PLACEHOLDER_2)
            (then "Calm down, I know what I'm doing!")
            (elsif PLACEHOLDER_1
              (then "Sure.")
              (elsif PLACEHOLDER_2
                (then "Whoa, chill out!")
                (elsif PLACEHOLDER_3
                  (then "Fine. Be that way!")
                  (else "Whatever."))))))))))
