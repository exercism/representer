(file "isbn_verifier.rb"
  (program
    (class IsbnVerifier
      (body_statement
        (singleton_method self valid?
          (method_parameters string)
          (body_statement
            (assignment PLACEHOLDER_1 =
              (call string gsub
                (argument_list
                  (regex / - /) "")))
            (if
              (unary !
                (call PLACEHOLDER_1 match?
                  (argument_list
                    (regex / [0-9]{9}[0-9|X] /))))
              (then
                (return
                  (argument_list false))))
            (assignment PLACEHOLDER_1 =
              (call PLACEHOLDER_1 split
                (argument_list "")))
            (assignment PLACEHOLDER_2 = 0)
            (assignment PLACEHOLDER_3 = 10)
            (if
              (binary
                (call PLACEHOLDER_1 size) != 10)
              (then
                (return
                  (argument_list false))))
            (call PLACEHOLDER_1 each
              (block
                (block_parameters PLACEHOLDER_4)
                (body_statement
                  (assignment PLACEHOLDER_5 = PLACEHOLDER_4)
                  (if
                    (binary
                      (call PLACEHOLDER_4 match?
                        (argument_list
                          (regex / [a-zA-Z] /))) &&
                      (binary PLACEHOLDER_4 === "X"))
                    (then
                      (assignment PLACEHOLDER_5 = 10)))
                  (operator_assignment PLACEHOLDER_2 +=
                    (binary
                      (call PLACEHOLDER_5 to_i) * PLACEHOLDER_3))
                  (operator_assignment PLACEHOLDER_3 -= 1))))
            (return
              (argument_list
                (binary
                  (binary PLACEHOLDER_2 % 11) === 0)))))))))
