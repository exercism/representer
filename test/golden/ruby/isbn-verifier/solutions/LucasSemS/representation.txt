(program
  (class IsbnVerifier
    (body_statement
      (singleton_method self valid?
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment PLACEHOLDER_2
            (call PLACEHOLDER_1 delete
              (argument_list "-")))
          (if
            (binary
              (call PLACEHOLDER_2 length) == 10)
            (else
              (return
                (argument_list false))))
          (assignment PLACEHOLDER_3 0)
          (call
            (call PLACEHOLDER_2 chars) each_with_index
            (block
              (block_parameters PLACEHOLDER_4 PLACEHOLDER_5)
              (body_statement
                (assignment PLACEHOLDER_6
                  (if
                    (binary
                      (binary PLACEHOLDER_5 == 9) &&
                      (binary
                        (call PLACEHOLDER_4 upcase) == "X"))
                    (then 10)
                    (else
                      (if
                        (binary PLACEHOLDER_4 =~
                          (regex "\\d"))
                        (then
                          (call PLACEHOLDER_4 to_i))
                        (else
                          (return
                            (argument_list false)))))))
                (operator_assignment PLACEHOLDER_3 +=
                  (binary PLACEHOLDER_6 *
                    (binary 10 - PLACEHOLDER_5))))))
          (binary
            (binary PLACEHOLDER_3 % 11) == 0))))))
