(program
  (module BaseConverter
    (body_statement
      (singleton_method self convert
        (method_parameters input_base digits output_base)
        (body_statement
          (if
            (binary
              (binary
                (binary input_base < 2) ||
                (binary output_base < 2)) ||
              (call digits any?
                (block
                  (body_statement
                    (binary
                      (binary _1 < 0) ||
                      (binary _1 >= input_base))))))
            (then
              (call raise
                (argument_list ArgumentError))))
          (assignment PLACEHOLDER_1 =
            (call digits reduce
              (argument_list 0)
              (block
                (block_parameters PLACEHOLDER_2 PLACEHOLDER_3)
                (body_statement
                  (binary
                    (binary PLACEHOLDER_2 * input_base) + PLACEHOLDER_3)))))
          (assignment converted =
            (array))
          (while
            (call PLACEHOLDER_1 positive?)
            (do
              (call converted append
                (argument_list
                  (binary PLACEHOLDER_1 % output_base)))
              (operator_assignment PLACEHOLDER_1 /= output_base)))
          (conditional
            (call converted empty?)
            (array 0)
            (call converted reverse)))))))
