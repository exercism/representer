(program
  (class Raindrops
    (body_statement
      (assignment PRIME_SOUNDS
        (array
          (hash
            (pair :prime 3)
            (pair :sound "Pling"))
          (hash
            (pair :prime 5)
            (pair :sound "Plang"))
          (hash
            (pair :prime 7)
            (pair :sound "Plong"))))
      (singleton_method self convert
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment PLACEHOLDER_2
            (call prime_sounds
              (argument_list PLACEHOLDER_1)))
          (if
            (call PLACEHOLDER_2 empty?)
            (then
              (call PLACEHOLDER_1 to_s))
            (else PLACEHOLDER_2))))
      (singleton_method self prime_sounds
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (call PRIME_SOUNDS reduce
            (argument_list "")
            (block
              (block_parameters PLACEHOLDER_3 PLACEHOLDER_4)
              (body_statement
                (if
                  (call self has_prime_factor?
                    (argument_list PLACEHOLDER_1
                      (element_reference PLACEHOLDER_4 :prime)))
                  (then
                    (operator_assignment PLACEHOLDER_3 +=
                      (element_reference PLACEHOLDER_4 :sound)))
                  (else PLACEHOLDER_3)))))))
      (singleton_method self has_prime_factor?
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_5)
        (body_statement
          (binary
            (binary PLACEHOLDER_1 % PLACEHOLDER_5) == 0))))))
