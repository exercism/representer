(program
  (class Raindrops
    (body_statement
      (singleton_method self convert
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment PLACEHOLDER_2 "")
          (if
            (binary
              (binary PLACEHOLDER_1 % 3) == 0)
            (then
              (operator_assignment PLACEHOLDER_2 += "Pling")))
          (if
            (binary
              (binary PLACEHOLDER_1 % 5) == 0)
            (then
              (operator_assignment PLACEHOLDER_2 += "Plang")))
          (if
            (binary
              (binary PLACEHOLDER_1 % 7) == 0)
            (then
              (operator_assignment PLACEHOLDER_2 += "Plong")))
          (if
            (call PLACEHOLDER_2 empty?)
            (then
              (call PLACEHOLDER_1 to_s))
            (else PLACEHOLDER_2)))))))
