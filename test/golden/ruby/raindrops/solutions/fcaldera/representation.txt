(file "raindrops.rb"
  (program
    (module Raindrops
      (body_statement
        (singleton_method self convert
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment PLACEHOLDER_2 =
              (binary
                (binary
                  (conditional
                    (binary
                      (binary PLACEHOLDER_1 % 3) == 0) "Pling" "") +
                  (conditional
                    (binary
                      (binary PLACEHOLDER_1 % 5) == 0) "Plang" "")) +
                (conditional
                  (binary
                    (binary PLACEHOLDER_1 % 7) == 0) "Plong" "")))
            (conditional
              (call PLACEHOLDER_2 empty?)
              (call PLACEHOLDER_1 to_s) PLACEHOLDER_2)))))))
