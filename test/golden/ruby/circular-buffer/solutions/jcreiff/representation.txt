(program
  (class CircularBuffer
    (superclass Array)
    (body_statement
      (call attr_reader
        (argument_list :size))
      (method initialize
        (method_parameters size)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1) size)))
      (method read
        (body_statement
          (if
            (call count zero?)
            (then
              (call raise
                (argument_list
                  (scope_resolution CircularBuffer BufferEmptyException))))) shift))
      (method write
        (method_parameters PLACEHOLDER_2)
        (body_statement
          (if
            (binary count == size)
            (then
              (call raise
                (argument_list
                  (scope_resolution CircularBuffer BufferFullException)))))
          (call push
            (argument_list PLACEHOLDER_2))))
      (method write!
        (method_parameters PLACEHOLDER_2)
        (body_statement
          (if
            (binary count == size)
            (then read))
          (call write
            (argument_list PLACEHOLDER_2))))
      (class BufferEmptyException
        (superclass StandardError))
      (class BufferFullException
        (superclass StandardError)))))
