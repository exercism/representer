(file "beer_song.rb"
  (program
    (class BeerSong
      (body_statement
        (singleton_method self recite
          (method_parameters PLACEHOLDER_1 PLACEHOLDER_2)
          (body_statement
            (assignment PLACEHOLDER_3 = "")
            (while
              (binary PLACEHOLDER_2 > 0)
              (do
                (operator_assignment PLACEHOLDER_3 +=
                  (call PLACEHOLDER_4
                    (argument_list PLACEHOLDER_1)))
                (operator_assignment PLACEHOLDER_3 += "\n")
                (operator_assignment PLACEHOLDER_1 -= 1)
                (operator_assignment PLACEHOLDER_2 -= 1)
                (if
                  (binary PLACEHOLDER_2 > 0)
                  (then
                    (operator_assignment PLACEHOLDER_3 += "\n"))))) PLACEHOLDER_3))
        (singleton_method self PLACEHOLDER_4
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (case PLACEHOLDER_1
              (when
                (pattern 0)
                (then
                  (return
                    (argument_list "No more bottles of beer on the wall, no more bottles of beer.\nGo to the store and buy some more, 99 bottles of beer on the wall."))))
              (when
                (pattern 1)
                (then
                  (return
                    (argument_list "1 bottle of beer on the wall, 1 bottle of beer.\nTake it down and pass it around, no more bottles of beer on the wall."))))
              (when
                (pattern 2)
                (then
                  (return
                    (argument_list "2 bottles of beer on the wall, 2 bottles of beer.\nTake one down and pass it around, 1 bottle of beer on the wall."))))
              (else
                (string
                  (interpolation PLACEHOLDER_1) " bottles of beer on the wall, "
                  (interpolation PLACEHOLDER_1) " bottles of beer.\nTake one down and pass it around, "
                  (interpolation
                    (binary PLACEHOLDER_1 - 1)) " bottles of beer on the wall.")))))))))
