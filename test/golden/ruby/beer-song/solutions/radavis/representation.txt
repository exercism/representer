(program
  (module BeerSong
    (body_statement
      (singleton_method self recite
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_2)
        (body_statement
          (assignment PLACEHOLDER_3
            (array))
          (call PLACEHOLDER_2 times
            (block
              (body_statement
                (binary PLACEHOLDER_3 <<
                  (call verse
                    (argument_list PLACEHOLDER_1)))
                (operator_assignment PLACEHOLDER_1 -= 1))))
          (binary
            (call PLACEHOLDER_3 join
              (argument_list "\n\n")) + "\n")))
      (singleton_method self verse
        (method_parameters PLACEHOLDER_4)
        (body_statement
          (assignment PLACEHOLDER_5
            (string
              (interpolation
                (call
                  (call bottles_to_s
                    (argument_list PLACEHOLDER_4)) capitalize)) " of beer on the wall, "
              (interpolation
                (call bottles_to_s
                  (argument_list PLACEHOLDER_4))) " of beer."))
          (assignment PLACEHOLDER_6
            (string "Take "
              (interpolation
                (if
                  (binary PLACEHOLDER_4 > 1)
                  (then "one")
                  (else "it"))) " down and pass it around, "
              (interpolation
                (call bottles_to_s
                  (argument_list
                    (binary PLACEHOLDER_4 - 1)))) " of beer on the wall."))
          (if
            (binary PLACEHOLDER_4 == 0)
            (then
              (assignment PLACEHOLDER_6 "Go to the store and buy some more, 99 bottles of beer on the wall.")))
          (call
            (array PLACEHOLDER_5 PLACEHOLDER_6) join
            (argument_list "\n"))))
      (singleton_method self bottles_to_s
        (method_parameters PLACEHOLDER_4)
        (body_statement
          (if
            (binary PLACEHOLDER_4 > 1)
            (then
              (return
                (argument_list
                  (string
                    (interpolation PLACEHOLDER_4) " bottles")))))
          (if
            (binary PLACEHOLDER_4 == 1)
            (then
              (return
                (argument_list "1 bottle")))) "no more bottles")))))
