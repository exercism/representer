(file "grade_school.rb"
  (program
    (class School
      (body_statement
        (method initialize
          (body_statement
            (assignment (instance_variable PLACEHOLDER_1) =
              (call
                (range 1 .. 12) each_with_object
                (argument_list
                  (hash))
                (block
                  (block_parameters grade roster)
                  (body_statement
                    (assignment
                      (element_reference roster grade) =
                      (array))))))))
        (method add
          (method_parameters PLACEHOLDER_2 grade)
          (body_statement
            (if
              (call roster include?
                (argument_list PLACEHOLDER_2))
              (then
                (return
                  (argument_list false))))
            (binary
              (element_reference (instance_variable PLACEHOLDER_1) grade) << PLACEHOLDER_2)
            (call
              (element_reference (instance_variable PLACEHOLDER_1) grade) sort!) true))
        (method roster
          (body_statement
            (call
              (call (instance_variable PLACEHOLDER_1) values) flatten)))
        (method grade
          (method_parameters grade)
          (body_statement
            (element_reference (instance_variable PLACEHOLDER_1) grade)))))))
