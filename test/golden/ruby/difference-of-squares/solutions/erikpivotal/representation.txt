(file "difference_of_squares.rb"
  (program
    (class Squares
      (body_statement
        (method initialize
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment (instance_variable PLACEHOLDER_2) = PLACEHOLDER_1)))
        (method square_of_sum
          (body_statement
            (binary
              (call
                (range 1 .. (instance_variable PLACEHOLDER_2)) sum) ** 2)))
        (method sum_of_squares
          (body_statement
            (call
              (call
                (range 1 .. (instance_variable PLACEHOLDER_2)) map
                (block
                  (block_parameters PLACEHOLDER_3)
                  (body_statement
                    (binary PLACEHOLDER_3 ** 2)))) sum)))
        (method difference
          (body_statement
            (binary square_of_sum - sum_of_squares)))))))
