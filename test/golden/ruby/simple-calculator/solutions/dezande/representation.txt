(file "simple_calculator.rb"
  (program
    (class SimpleCalculator
      (body_statement
        (class UnsupportedOperation
          (superclass < StandardError))
        (assignment ALLOWED_OPERATIONS =
          (call
            (array "+" "/" "*") freeze))
        (singleton_method self calculate
          (method_parameters PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3)
          (body_statement
            (unless
              (binary
                (call PLACEHOLDER_1 is_a?
                  (argument_list Integer)) &&
                (call PLACEHOLDER_2 is_a?
                  (argument_list Integer)))
              (then
                (call raise
                  (argument_list
                    (call ArgumentError new)))))
            (unless
              (call ALLOWED_OPERATIONS include?
                (argument_list PLACEHOLDER_3))
              (then
                (call raise
                  (argument_list
                    (call UnsupportedOperation new)))))
            (if
              (binary
                (binary PLACEHOLDER_3 == "/") &&
                (binary PLACEHOLDER_2 == 0))
              (then
                (return
                  (argument_list "Division by zero is not allowed."))))
            (string
              (interpolation PLACEHOLDER_1) " "
              (interpolation PLACEHOLDER_3) " "
              (interpolation PLACEHOLDER_2) " = "
              (interpolation
                (call result
                  (argument_list PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3))))))
        (singleton_method self result
          (method_parameters PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3)
          (body_statement
            (case PLACEHOLDER_3
              (when
                (pattern "+")
                (then
                  (binary PLACEHOLDER_1 + PLACEHOLDER_2)))
              (when
                (pattern "/")
                (then
                  (binary PLACEHOLDER_1 / PLACEHOLDER_2)))
              (when
                (pattern "*")
                (then
                  (binary PLACEHOLDER_1 * PLACEHOLDER_2))))))))))
