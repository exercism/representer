(program
  (class SimpleCalculator
    (body_statement
      (assignment UnsupportedOperation = StandardError)
      (assignment ALLOWED_OPERATIONS =
        (call
          (array "+" "/" "*") freeze))
      (singleton_method self calculate
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3)
        (body_statement
          (unless
            (binary
              (call PLACEHOLDER_1 is_a?
                (argument_list Integer)) &&
              (call PLACEHOLDER_2 is_a?
                (argument_list Integer)))
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list))))))
          (assignment PLACEHOLDER_4 =
            (case PLACEHOLDER_3
              (when
                (pattern "+")
                (then
                  (binary PLACEHOLDER_1 + PLACEHOLDER_2)))
              (when
                (pattern "/")
                (then
                  (begin
                    (binary PLACEHOLDER_1 / PLACEHOLDER_2)
                    (rescue
                      (then
                        (return
                          (argument_list "Division by zero is not allowed.")))))))
              (when
                (pattern "*")
                (then
                  (binary PLACEHOLDER_1 * PLACEHOLDER_2)))
              (else
                (call raise
                  (argument_list
                    (call UnsupportedOperation new
                      (argument_list)))))))
          (string
            (interpolation PLACEHOLDER_1) " "
            (interpolation PLACEHOLDER_3) " "
            (interpolation PLACEHOLDER_2) " = "
            (interpolation PLACEHOLDER_4)))))))
