(program
  (class SimpleCalculator
    (body_statement
      (class UnsupportedOperation
        (superclass StandardError))
      (assignment ALLOWED_OPERATIONS
        (call
          (array "+" "/" "*") freeze))
      (singleton_method self calculate
        (method_parameters PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3)
        (body_statement
          (if
            (call ALLOWED_OPERATIONS include?
              (argument_list PLACEHOLDER_3))
            (else
              (call raise
                (argument_list UnsupportedOperation))))
          (if
            (call
              (array PLACEHOLDER_1 PLACEHOLDER_2) all?
              (block
                (body_statement
                  (call _1 is_a?
                    (argument_list Numeric)))))
            (else
              (call raise
                (argument_list ArgumentError))))
          (begin
            (string
              (interpolation
                (call
                  (array PLACEHOLDER_1 PLACEHOLDER_3 PLACEHOLDER_2) join
                  (argument_list " "))) " = "
              (interpolation
                (call PLACEHOLDER_1 public_send
                  (argument_list PLACEHOLDER_3 PLACEHOLDER_2))))
            (rescue
              (exceptions ZeroDivisionError)
              (then "Division by zero is not allowed."))))))))
