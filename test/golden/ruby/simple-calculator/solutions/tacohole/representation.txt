(file "simple_calculator.rb"
  (program
    (class SimpleCalculator
      (body_statement
        (assignment ALLOWED_OPERATIONS =
          (call
            (array "+" "/" "*") freeze))
        (class UnsupportedOperation
          (superclass < StandardError))
        (singleton_method self calculate
          (method_parameters PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3)
          (body_statement
            (if
              (binary
                (binary
                  (call PLACEHOLDER_1 class) != Integer) ||
                (binary
                  (call PLACEHOLDER_2 class) != Integer))
              (then
                (call raise
                  (argument_list ArgumentError))))
            (case PLACEHOLDER_3
              (when
                (pattern "+")
                (then
                  (assignment PLACEHOLDER_4 =
                    (binary PLACEHOLDER_1 + PLACEHOLDER_2))))
              (when
                (pattern "/")
                (then
                  (begin
                    (assignment PLACEHOLDER_4 =
                      (binary PLACEHOLDER_1 / PLACEHOLDER_2))
                    (rescue
                      (exceptions ZeroDivisionError)
                      (exception_variable PLACEHOLDER_5)
                      (then
                        (return
                          (argument_list "Division by zero is not allowed.")))))))
              (when
                (pattern "*")
                (then
                  (assignment PLACEHOLDER_4 =
                    (binary PLACEHOLDER_1 * PLACEHOLDER_2))))
              (else
                (call raise
                  (argument_list
                    (scope_resolution SimpleCalculator :: UnsupportedOperation)))))
            (return
              (argument_list
                (string
                  (interpolation PLACEHOLDER_1) " "
                  (interpolation PLACEHOLDER_3) " "
                  (interpolation PLACEHOLDER_2) " = "
                  (interpolation PLACEHOLDER_4))))))))))
