(file "armstrong_numbers.rb"
  (program
    (class ArmstrongNumbers
      (body_statement
        (singleton_method self include?
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment PLACEHOLDER_2 =
              (call PLACEHOLDER_1 to_s))
            (assignment PLACEHOLDER_3 =
              (call PLACEHOLDER_2 length))
            (assignment PLACEHOLDER_4 =
              (call
                (call PLACEHOLDER_2 each_char) sum
                (block
                  (block_parameters PLACEHOLDER_5)
                  (body_statement
                    (binary
                      (call PLACEHOLDER_5 to_i) ** PLACEHOLDER_3)))))
            (binary PLACEHOLDER_4 == PLACEHOLDER_1)))))))
