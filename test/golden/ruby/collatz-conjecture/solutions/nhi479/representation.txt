    (class CollatzConjecture
      (body_statement
        (singleton_method self steps
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment PLACEHOLDER_2 = 0)
            (if
              (binary PLACEHOLDER_1 <= 0)
              (then
                (call raise

                (operator_assignment PLACEHOLDER_2 += 1)
                (if
                  (binary
                    (binary PLACEHOLDER_1 % 2) == 0)

                    (assignment PLACEHOLDER_1 =
                      (binary PLACEHOLDER_1 / 2)))

                    (assignment PLACEHOLDER_1 =

                        (binary PLACEHOLDER_1 * 3) + 1))))))
