(program
  (class CollatzConjecture
    (body_statement
      (singleton_method self steps
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (unless
            (call PLACEHOLDER_1 positive?)
            (then
              (call raise
                (argument_list ArgumentError))))
          (assignment steps = 0)
          (while
            (binary PLACEHOLDER_1 != 1)
            (do
              (assignment PLACEHOLDER_1 =
                (conditional
                  (call PLACEHOLDER_1 even?)
                  (binary PLACEHOLDER_1 / 2)
                  (binary
                    (binary 3 * PLACEHOLDER_1) + 1)))
              (operator_assignment steps += 1))) steps)))))
