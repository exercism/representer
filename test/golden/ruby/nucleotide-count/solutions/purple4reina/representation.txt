(program
  (class Nucleotide
    (superclass String)
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (if
            (call
              (regex "[^ATCG]") match?
              (argument_list PLACEHOLDER_1))
            (then
              (call raise
                (argument_list ArgumentError))))
          (call super
            (argument_list PLACEHOLDER_1))
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (singleton_method self from_dna
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (call new
            (argument_list PLACEHOLDER_1))))
      (method histogram
        (body_statement
          (hash
            (pair "A"
              (call PLACEHOLDER_1 count
                (argument_list "A")))
            (pair "T"
              (call PLACEHOLDER_1 count
                (argument_list "T")))
            (pair "C"
              (call PLACEHOLDER_1 count
                (argument_list "C")))
            (pair "G"
              (call PLACEHOLDER_1 count
                (argument_list "G")))))) private
      (call attr_accessor
        (argument_list (simple_symbol :PLACEHOLDER_1))))))
