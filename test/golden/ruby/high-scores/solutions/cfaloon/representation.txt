(program
  (class HighScores
    (body_statement
      (call attr_reader
        (argument_list :scores))
      (method initialize
        (method_parameters scores)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_1) scores)))
      (method latest
        (body_statement
          (call scores last)))
      (method personal_best
        (body_statement
          (call scores max)))
      (method personal_top_three
        (body_statement
          (call scores max
            (argument_list 3))))
      (method latest_is_personal_best?
        (body_statement
          (binary latest == personal_best))))))
