(program
  (class LogLineParser
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method message
        (body_statement
          (call
            (element_reference
              (call (instance_variable PLACEHOLDER_2) split
                (argument_list ":")) 1) strip)))
      (method log_level
        (body_statement
          (call
            (call
              (element_reference
                (call (instance_variable PLACEHOLDER_2) split
                  (argument_list ":")) 0) gsub
              (argument_list
                (regex "[\\[\\]]") "")) downcase)))
      (method reformat
        (body_statement
          (string
            (interpolation message) " ("
            (interpolation log_level) ")"))))))
