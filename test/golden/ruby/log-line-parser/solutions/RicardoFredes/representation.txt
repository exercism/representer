(program
  (class LogLineParser
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment PLACEHOLDER_2 =
            (regex / "\\[" "(?<log_level>.*)" "\\]" ": (?<message>.*)" /))
          (assignment PLACEHOLDER_3 =
            (call PLACEHOLDER_1 match
              (argument_list PLACEHOLDER_2)))
          (assignment (instance_variable PLACEHOLDER_4) =
            (call
              (call
                (element_reference PLACEHOLDER_3 :log_level) sub
                (argument_list "[" "")) downcase))
          (assignment (instance_variable PLACEHOLDER_5) =
            (call
              (element_reference PLACEHOLDER_3 :message) strip))))
      (method message
        (body_statement (instance_variable PLACEHOLDER_5)))
      (method log_level
        (body_statement (instance_variable PLACEHOLDER_4)))
      (method reformat
        (body_statement
          (string
            (interpolation message) " ("
            (interpolation log_level) ")"))))))
