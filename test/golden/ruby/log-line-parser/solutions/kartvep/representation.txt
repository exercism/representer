(program
  (class LogLineParser
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) = PLACEHOLDER_1)))
      (method message
        (body_statement
          (return
            (argument_list
              (call
                (element_reference (instance_variable PLACEHOLDER_2)
                  (binary
                    (call (instance_variable PLACEHOLDER_2) index
                      (argument_list ": ")) + 2)
                  (call (instance_variable PLACEHOLDER_2) size)) strip)))))
      (method log_level
        (body_statement
          (return
            (argument_list
              (call
                (element_reference (instance_variable PLACEHOLDER_2) 1
                  (binary
                    (call (instance_variable PLACEHOLDER_2) index
                      (argument_list ":")) - 2)) downcase)))))
      (method reformat
        (body_statement
          (return
            (argument_list
              (string
                (interpolation
                  (call self message)) " ("
                (interpolation log_level) ")"))))))))
