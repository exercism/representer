(file "log_line_parser.rb"
  (program
    (class LogLineParser
      (body_statement
        (call attr_reader
          (argument_list :log_level :message))
        (method initialize
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (assignment PLACEHOLDER_2 =
              (call PLACEHOLDER_1 split
                (argument_list ":")))
            (assignment (instance_variable PLACEHOLDER_3) =
              (call
                (call
                  (call PLACEHOLDER_2 first) gsub
                  (argument_list
                    (regex / [ "\\[" "\\]" "\\s" ] /) "")) downcase))
            (assignment (instance_variable PLACEHOLDER_4) =
              (call
                (call PLACEHOLDER_2 last) strip))))
        (method reformat
          (body_statement
            (string
              (interpolation message) " ("
              (interpolation log_level) ")")))))))
