(program
  (class AssemblyLine
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) = PLACEHOLDER_1)))
      (method PLACEHOLDER_3
        (body_statement
          (binary (instance_variable PLACEHOLDER_2) * 221.0)))
      (method PLACEHOLDER_4
        (body_statement
          (case (instance_variable PLACEHOLDER_2)
            (when
              (pattern
                (range 1 .. 4))
              (then 1.0))
            (when
              (pattern
                (range 5 .. 8))
              (then 0.9))
            (when
              (pattern 9)
              (then 0.8))
            (when
              (pattern 10)
              (then 0.77))
            (else
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list
                      (string "Expected a speed value of 1-10, got "
                        (interpolation PLACEHOLDER_1))))))))))
      (method production_rate_per_hour
        (body_statement
          (call
            (binary PLACEHOLDER_4 * PLACEHOLDER_3) round
            (argument_list 1))))
      (method working_items_per_minute
        (body_statement
          (call
            (binary production_rate_per_hour / 60) to_i))))))
