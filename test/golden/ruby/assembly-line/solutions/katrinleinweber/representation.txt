(file "assembly_line.rb"
  (program
    (class AssemblyLine
      (body_statement
        (call attr_reader
          (argument_list (simple_symbol :PLACEHOLDER_1)))
        (method initialize
          (method_parameters PLACEHOLDER_1)
          (body_statement
            (unless
              (call PLACEHOLDER_1 between?
                (argument_list 1 10))
              (then
                (call raise
                  (argument_list ArgumentError))))
            (assignment (instance_variable PLACEHOLDER_2) =
              (call PLACEHOLDER_1 to_i))))
        (assignment (constant PLACEHOLDER_3) =
          (call 221 to_f))
        (method PLACEHOLDER_4
          (body_statement
            (case PLACEHOLDER_1
              (when
                (pattern
                  (range 1 .. 4))
                (then 1.0))
              (when
                (pattern
                  (range 5 .. 8))
                (then 0.9))
              (when
                (pattern 9)
                (then 0.8))
              (when
                (pattern 10)
                (then 0.77)))))
        (method production_rate_per_hour
          (body_statement
            (binary
              (binary PLACEHOLDER_1 * (constant PLACEHOLDER_3)) * PLACEHOLDER_4)))
        (method working_items_per_minute
          (body_statement
            (call
              (binary production_rate_per_hour /
                (call 60 to_f)) floor)))))))
