(program
  (class AssemblyLine
    (body_statement
      (assignment NUMBER_OF_CARS 221)
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method PLACEHOLDER_3
        (body_statement
          (case (instance_variable PLACEHOLDER_2)
            (when
              (pattern 10)
              (then
                (return
                  (argument_list 0.77))))
            (when
              (pattern 9)
              (then
                (return
                  (argument_list 0.8))))
            (when
              (pattern
                (range 5 .. 8))
              (then
                (return
                  (argument_list 0.9))))
            (else
              (return
                (argument_list 1))))))
      (method production_rate_per_hour
        (body_statement
          (binary
            (binary NUMBER_OF_CARS * (instance_variable PLACEHOLDER_2)) * PLACEHOLDER_3)))
      (method working_items_per_minute
        (body_statement
          (call
            (binary production_rate_per_hour / 60) to_i))))))
