(program
  (class NotMovieClubMemberError
    (superclass RuntimeError))
  (class Moviegoer
    (body_statement
      (method initialize
        (method_parameters PLACEHOLDER_1
          (keyword_parameter member false))
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)
          (assignment (instance_variable PLACEHOLDER_3) member)))
      (method ticket_price
        (body_statement
          (if
            (binary (instance_variable PLACEHOLDER_2) >= 60)
            (then 10)
            (else 15))))
      (method watch_scary_movie?
        (body_statement
          (if
            (binary (instance_variable PLACEHOLDER_2) >= 18)
            (then true)
            (else false))))
      (method claim_free_popcorn!
        (body_statement
          (if
            (binary (instance_variable PLACEHOLDER_3) == false)
            (then
              (call raise
                (argument_list
                  (call NotMovieClubMemberError new
                    (argument_list "Exception was raised!")))))
            (else "\u{1F37F}")))))))
