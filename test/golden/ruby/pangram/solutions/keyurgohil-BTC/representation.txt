(program
  (class Pangram
    (body_statement
      (singleton_method self pangram?
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment PLACEHOLDER_2 0)
          (call
            (range 0 .. 26) each
            (block
              (block_parameters PLACEHOLDER_3)
              (body_statement
                (call PLACEHOLDER_1 each_byte
                  (block
                    (block_parameters PLACEHOLDER_4)
                    (body_statement
                      (if
                        (binary
                          (binary
                            (binary PLACEHOLDER_3 + 65) == PLACEHOLDER_4) ||
                          (binary
                            (binary PLACEHOLDER_3 + 97) == PLACEHOLDER_4))
                        (then
                          (operator_assignment PLACEHOLDER_2 += 1)
                          (break)))))))))
          (if
            (binary PLACEHOLDER_2 == 26)
            (then
              (string
                (interpolation PLACEHOLDER_1) " IS a pangram"))
            (else
              (call puts
                (argument_list
                  (string
                    (interpolation PLACEHOLDER_1) " is NOT a pangram")))))))))
  (module (constant PLACEHOLDER_5)
    (body_statement
      (assignment VERSION 3))))
