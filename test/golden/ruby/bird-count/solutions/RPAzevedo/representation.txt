(program
  (class BirdCount
    (body_statement
      (singleton_method self last_week
        (body_statement
          (array 0 2 5 3 7 8 4)))
      (method initialize
        (method_parameters PLACEHOLDER_1)
        (body_statement
          (assignment (instance_variable PLACEHOLDER_2) PLACEHOLDER_1)))
      (method yesterday
        (body_statement
          (element_reference (instance_variable PLACEHOLDER_2)
            (unary - 2))))
      (method total
        (body_statement
          (call (instance_variable PLACEHOLDER_2) sum)))
      (method busy_days
        (body_statement
          (call (instance_variable PLACEHOLDER_2) count
            (block
              (block_parameters PLACEHOLDER_3)
              (body_statement
                (binary PLACEHOLDER_3 >= 5))))))
      (method day_without_birds?
        (body_statement
          (call (instance_variable PLACEHOLDER_2) any?
            (block
              (block_parameters PLACEHOLDER_3)
              (body_statement
                (binary PLACEHOLDER_3 == 0)))))))))
