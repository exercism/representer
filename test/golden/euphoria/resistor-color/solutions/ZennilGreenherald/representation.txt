sequence all_colors = {
"\"black\","
"\"brown\","
"\"red\","
"\"orange\","
"\"yellow\","
"\"green\","
"\"blue\","
"\"violet\","
"\"grey\","
"\"white\""
}
public function colorCode(sequence color)
return find(color, all_colors) - 1
end function
public function colors()
return all_colors
end function
