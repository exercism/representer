public function tick(sequence m)
if equal(m,{}) then return m end if
integer w = length(m[1]), h = length(m)
sequence n = repeat(repeat(0,w),h)
for x=1 to w do
for y=1 to h do
integer mxy = m[x][y], nxy = -mxy
for i=x-(x>1) to x+(x<w) do
for j=y-(y>1) to y+(y<h) do
nxy += m[i][j]
end for
end for
n[x][y] = nxy=3 or (mxy and nxy=2)
end for
end for
return n
end function
