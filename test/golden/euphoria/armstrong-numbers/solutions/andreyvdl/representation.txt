include std/math.e
public function is_armstrong_number(integer number)
integer nbr_of_digits = 0
integer number_copy = number
while number_copy != 0 do
number_copy = floor(number_copy/10)
nbr_of_digits += 1
end while
number_copy = number
integer total = 0
while number != 0 do
total += power(mod(number, 10), nbr_of_digits)
number = floor(number/10)
end while
return equal(total, number_copy)
end function
