include std/math.e
public function is_armstrong_number(integer number)
-- guard against log10(0), an error
if number = 0 then
return 1
end if
integer number_length = floor(log10(number) + 1)
integer digit
integer sum = 0
integer n = number
while n > 0 do
digit = remainder(n, 10)
sum += power(digit, number_length)
n = floor(n / 10)
end while
return sum = number
end function
