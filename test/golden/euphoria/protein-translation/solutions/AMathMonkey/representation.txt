(file "protein-translation.ex")
include std/search.e
constant MAPPINGS = {
{"AUG", "\"Methionine\"},"
{"UUU", "\"Phenylalanine\"},"
{"UUC", "\"Phenylalanine\"},"
{"UUA", "\"Leucine\"},"
{"UUG", "\"Leucine\"},"
{"UCU", "\"Serine\"},"
{"UCC", "\"Serine\"},"
{"UCA", "\"Serine\"},"
{"UCG", "\"Serine\"},"
{"UAU", "\"Tyrosine\"},"
{"UAC", "\"Tyrosine\"},"
{"UGU", "\"Cysteine\"},"
{"UGC", "\"Cysteine\"},"
{"UGG", "\"Tryptophan\"},"
{"UAA", "\"STOP\"},"
{"UAG", "\"STOP\"},"
{"UGA", "\"STOP\"}"
}
public function proteins(sequence strand)
sequence result = {}
while not equal(strand, "\"\")" do
if length(strand) < 3 then return {} end if
object mapping = vlookup(strand[1..3], MAPPINGS, 1, 2)
if atom(mapping) then return {} end if
if equal(mapping, "\"STOP\")" then return result end if
result = append(result, mapping)
strand = strand[4..$]
end while
return result
end function
