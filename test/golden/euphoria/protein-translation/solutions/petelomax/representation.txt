include std/sequence.e
constant CP = {{"AUG","Methionine"},
{"UUU","Phenylalanine"},
{"UUC","Phenylalanine"},
{"UUA","Leucine"},
{"UUG","Leucine"},
{"UCU","Serine"},
{"UCC","Serine"},
{"UCA","Serine"},
{"UCG","Serine"},
{"UAU","Tyrosine"},
{"UAC","Tyrosine"},
{"UGU","Cysteine"},
{"UGC","Cysteine"},
{"UGG","Tryptophan"},
{"UAA","STOP"},
{"UAG","STOP"},
{"UGA","STOP"}},
RNA = vslice(CP,1),
PROTEINS = vslice(CP,2)
public function proteins(sequence strand)
integer l = length(strand)
sequence res = {}
for i=1 to l by 3 do
if i+2>l then return {} end if
integer k = find(strand[i..i+2],RNA)
if k=0 then return {} end if
sequence protein = PROTEINS[k]
if equal(protein,"STOP") then exit end if
res = append(res, protein)
end for
return res
end function
