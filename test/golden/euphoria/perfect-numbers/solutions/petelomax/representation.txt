public function classify(integer n)
if n<1 then return 0 end if
integer i = 2, try_max = floor(n/2), fsum = n>1
while i<try_max do
if remainder(n,i)=0 then
try_max = n/i
fsum += i
if i!=try_max then fsum += try_max end if
end if
i += 1
end while
switch compare(fsum,n) do
case -1 then return "\"deficient\""
case 0 then return "\"perfect\""
case +1 then return "\"abundant\""
end switch
end function
