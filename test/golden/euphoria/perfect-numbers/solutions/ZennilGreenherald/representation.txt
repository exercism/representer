(file "perfect-numbers.ex")
public function classify(integer n)
integer asum = 0
for d = 1 to floor(n / 2) do
if remainder(n, d) = 0 then
asum += d
end if
end for
if n < 1 then
return 0
elsif asum = n then
return "\"perfect\""
elsif asum > n then
return "\"abundant\""
else
return "\"deficient\""
end if
end function
