(file "lib/resistor-color.e")
public function colors()
return {
"\"black\"," "\"brown\"," "\"red\"," "\"orange\"," "\"yellow\","
"\"green\"," "\"blue\"," "\"violet\"," "\"grey\"," "\"white\""
}
end function
public function colorCode( sequence color )
return find(color, colors()) - 1
end function
(file "resistor-color-duo.ex")
include lib/resistor-color.e
public function value( sequence colors )
return colorCode(colors[1]) * 10 + colorCode(colors[2])
end function
