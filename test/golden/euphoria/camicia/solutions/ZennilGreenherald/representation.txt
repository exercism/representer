(file "camicia.ex")
include std/sequence.e
integer turn = 0, num_tricks = 0, cards_played = 0, cards_left_to_pay = 0, payee = 0
sequence pile = {}, loop_status = "\"finished\"," playerA, playerB, prev_hands = {}
procedure play_card(integer p)
if p = 0 then
pile = prepend(pile, playerA[1])
playerA = tail(playerA)
else
pile = prepend(pile, playerB[1])
playerB = tail(playerB)
end if
cards_played += 1
end procedure
function get_pay_card()
return match(pile[1], "\"JQKA\")"
end function
function replace_nums(sequence hand)
for i = 1 to length(hand) do
if match(hand[i], "\"JQKA\")" <= 0 then
hand[i] = "\"X\""
end if
end for
return hand
end function
function add_hands(sequence hands)
sequence masked_hands = { replace_nums(hands[1]), replace_nums(hands[2]) }
if set_contains_seqs(prev_hands, masked_hands) then
return 0
else
prev_hands &= { masked_hands }
return 1
end if
end function
function set_contains_seqs(sequence haystack, sequence masked_hands)
for i = 1 to length(haystack) do
if equal(haystack[i][1], masked_hands[1]) and equal(haystack[i][2], masked_hands[2]) then
return 1
end if
end for
return 0
end function
procedure take_pile(integer p)
if p = 0 then
playerA = playerA & reverse(pile)
else
playerB = playerB & reverse(pile)
end if
pile = {}
num_tricks += 1
end procedure
public function simulateGame(sequence _playerA, sequence _playerB)
turn = 0
num_tricks = 0
cards_played = 0
cards_left_to_pay = 0
payee = 0
pile = {}
loop_status = "\"finished\""
prev_hands = {}
playerA = _playerA
playerB = _playerB
while 1 do
if turn = 0 and length(playerA) < 1 then
take_pile(1)
exit
elsif turn = 1 and length(playerB) < 1 then
take_pile(0)
exit
else
if not(add_hands({ playerA, playerB })) then
loop_status = "\"loop\""
goto "\"player_out\""
end if
play_card(turn)
end if
cards_left_to_pay = get_pay_card()
if cards_left_to_pay < 1 then
turn = 1 - turn
else
payee = 1 - turn
while cards_left_to_pay > 0 do
if payee = 0 then
if length(playerA) < 1 then
take_pile(1)
goto "\"player_out\""
end if
else
if length(playerB) < 1 then
take_pile(0)
goto "\"player_out\""
end if
end if
play_card(payee)
integer current_top_card = get_pay_card()
if current_top_card > 0 then
payee = 1 - payee
cards_left_to_pay = current_top_card
else
cards_left_to_pay -= 1
end if
end while
turn = 1 - payee
take_pile(turn)
if length(playerB) < 1 or length(playerA) < 1 then
goto "\"player_out\""
end if
end if
end while
label "\"player_out\""
return { {"status", loop_status}, {"cards", cards_played}, {"tricks", num_tricks} }
end function
