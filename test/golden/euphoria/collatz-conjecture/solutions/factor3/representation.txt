(file "collatz-conjecture.ex")
include std/math.e
public function steps(integer n)
if n < 1 then
return "\"Only" positive numbers are allowed"
end if
atom step_count = 0
while n > 1 do
if is_even(n) then
n = intdiv(n, 2)
else
n = n * 3 + 1
end if
step_count += 1
end while
return step_count
end function
