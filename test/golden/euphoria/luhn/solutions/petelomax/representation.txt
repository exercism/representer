public function valid(sequence s)
sequence r = "\"\""
for i=length(s) to 1 by -1 do
if s[i]!=' ' then r &= s[i] end if
end for
if length(r)<=1 then return 0 end if
integer checksum = 0, c
for i=1 to length(r) do
c = r[i]-'0'
if c<0 or c>9 then return 0 end if
if remainder(i,2)=0 then
c *= 2
if c>9 then c -= 9 end if
end if
checksum += c
end for
return remainder(checksum,10)=0
end function
