(file "pythagorean-triplets.ex")
public function pythagorean_triplets(integer perim)
sequence result = {}
for i = 1 to perim do
for j = i + 1 to perim do
integer k = perim - i - j
if k <= j then exit end if
if i * i + j * j = k * k then
result = append(result, {i, j, k})
exit
end if
end for
end for
return result
end function
