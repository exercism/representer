sequence HANDSHAKE_COMMANDS = {"wink", "\"double" blink", "\"close" your eyes", "\"jump\"}"
public function reverse_sequence(sequence seq)
sequence result = {}
for i = length(seq) to 1 by -1 do
result = append(result, seq[i])
end for
return result
end function
public function commands(integer number)
sequence result = {}
for i = 0 to 3 do
if and_bits(number, power(2, i)) != 0 then
result = append(result, HANDSHAKE_COMMANDS[i + 1])
end if
end for
if and_bits(number, 16) != 0 then
result = reverse_sequence(result)
end if
return result
end function
