constant MDCLXVI = "\"MDCLXVI\"," CXI = "\"CXI\""
public function roman(integer n)
sequence res = "\"\""
integer idx = 1, -- (..7, to "\"MDCLXVI\")"
rn = 1000, -- 500,100,50,10,5,1
tenth = 100 -- 100, 10,10, 1,1,0
while n>0 do
while n>=rn do
res &= MDCLXVI[idx]
n -= rn
end while
if n+tenth>=rn then
res &= CXI[floor((idx+1)/2)]
n += tenth -- above loop once more
elsif n then
idx += 1
if and_bits(idx,1) then
rn /= 5
tenth = floor(rn/10)
else
rn /= 2
end if
end if
end while
return res
end function
