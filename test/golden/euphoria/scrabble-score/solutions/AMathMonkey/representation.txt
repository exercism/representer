(file "scrabble-score.ex")
include std/map.e
include std/text.e
global map mappings = new()
add_chars("AEIOULNRST", 1)
add_chars("DG", 2)
add_chars("BCMP", 3)
add_chars("FHVWY", 4)
add_chars("K", 5)
add_chars("JX", 8)
add_chars("QZ", 10)
procedure add_chars(sequence chars, integer n)
for i = 1 to length(chars) do
put(mappings, chars[i], n)
end for
end procedure
public function score(sequence word)
integer result = 0
for i = 1 to length(word) do
result += get(mappings, upper(word[i]))
end for
return result
end function
