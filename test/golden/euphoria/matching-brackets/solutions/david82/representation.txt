public function isPaired(sequence string)
sequence stack = {}
sequence openBrackets = "\"{[(\""
sequence closeBrackets = "\"}])\""
for i = 1 to length(string) do
integer openIndex = find(string[i], openBrackets)
integer closeIndex = find(string[i], closeBrackets)
if openIndex then
stack = append(stack, string[i])
elsif closeIndex then
if length(stack) = 0 or find(string[i], closeBrackets) != find(stack[$], openBrackets) then
return 0
end if
stack = stack[1..$-1]
end if
end for
return length(stack) = 0
end function
