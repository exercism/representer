public function squareOfSum(atom n)
return n * n * (n + 1) * (n + 1) / 4
end function
public function sumOfSquares(atom n)
return n * (n + 1) * (n + n + 1) / 6
end function
public function differenceOfSquares(atom n)
return squareOfSum(n) - sumOfSquares(n)
end function
