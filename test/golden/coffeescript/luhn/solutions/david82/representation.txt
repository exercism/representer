class Luhn
  constructor : ( @ number ) ->
  valid : ->
    number = @ number . replace ( /\s/g , '' )
    if number . length <= 1 or not /^[0-9]+$/ . test ( number )
      return false
    digits = number . split ( '' ). reverse (). map ( Number )
    for i in [ 1 ... digits . length ] by 2
      digits [ i ] *= 2
      digits [ i ] -= 9 if digits [ i ] > 9
    sum = digits . reduce (( a , b ) -> a + b )
    sum % 10 == 0
module . exports = Luhn
