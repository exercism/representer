(file "luhn.coffee")
module . exports = class Luhn
  constructor : ( num ) ->
    @ num = if "/[ \\d]/" . test num then num . replace ( "/ /g" , '' ) else ''
  valid : ->
    sum = 0
    sum -= - d - ( d - ( d > 4 ) * 9 ) * (( @ num . length - 1 - i ) & 1 ) for d , i in @ num
    sum % 10 is 0 and @ num . length > 1
