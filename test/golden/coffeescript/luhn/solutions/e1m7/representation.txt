doubleOddPositions = ( n , i ) ->
  if i % 2 == 1
    return n * 2
  return n
subNineIfNeeded = ( n ) ->
  if n > 9
    return n - 9
  return n
add = ( a , b ) -> a + b
class Luhn
  constructor : ( num ) ->
    num = num . replace ( /\s/g , '' )
    if num . length <= 1
      @ _valid = false
    else
      @ _valid = num . split ( '' )
        . reverse ()
        . map ( Number )
        . map ( doubleOddPositions )
        . map ( subNineIfNeeded )
        . reduce ( add ) % 10 == 0
  valid : -> @ _valid
module . exports = Luhn
