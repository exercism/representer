class ScrabbleScore
  @ score : ( word ) ->
    values = [
      [ "\"AEIOULNRST\"" , 1 ]
      [ "\"DG\"" , 2 ]
      [ "\"BCMP\"" , 3 ]
      [ "\"FHVWY\"" , 4 ]
      [ "\"K\"" , 5 ]
      [ "\"JX\"" , 8 ]
      [ "\"QZ\"" , 10 ]
    ]
    value_map = {}
    values . forEach ( x ) ->
      [ x [ 0 ]...]. forEach ( c ) ->
        value_map [ c ] = x [ 1 ]
    return [ word . toUpperCase ()...]. reduce (
      (( a , c ) -> a + value_map [ c ]), 0
    )
module . exports = ScrabbleScore
