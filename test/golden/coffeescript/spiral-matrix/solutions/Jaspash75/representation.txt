class SpiralMatrix
  @ spiralMatrix : ( size ) ->
    return [] if size is 0
    matrix = Array . from ({ length : size }, -> Array ( size ). fill ( 0 ))
    dr = [ 0 , 1 , 0 , - 1 ]
    dc = [ 1 , 0 , - 1 , 0 ]
    r = 0
    c = 0
    dir = 0
    for val in [ 1 ..( size * size )]
      matrix [ r ][ c ] = val
      nextR = r + dr [ dir ]
      nextC = c + dc [ dir ]
      if nextR < 0 or nextR >= size or nextC < 0 or nextC >= size or matrix [ nextR ][ nextC ] != 0
        dir = ( dir + 1 ) % 4
        nextR = r + dr [ dir ]
        nextC = c + dc [ dir ]
      r = nextR
      c = nextC
    matrix
module . exports = SpiralMatrix
