Weeks =
  First : 'first'
  Second : 'second'
  Third : 'third'
  Fourth : 'fourth'
  Last : 'last'
  Teenth : 'teenth'
Weekdays =
  Sunday : 0
  Monday : 1
  Tuesday : 2
  Wednesday : 3
  Thursday : 4
  Friday : 5
  Saturday : 6
meetup = ({ year , month , week , dayofweek }) ->
  date = new Date ( year , month - 1 , 1 )
  matches = []
  while date . getMonth () == month - 1
    if date . getDay () == dayofweek
      matches . push ( new Date ( date ))
    date . setDate ( date . getDate () + 1 )
  switch week
    when Weeks . Teenth
      matches . find (( d ) -> 13 <= d . getDate () <= 19 )
    when Weeks . First
      matches [ 0 ]
    when Weeks . Second
      matches [ 1 ]
    when Weeks . Third
      matches [ 2 ]
    when Weeks . Fourth
      matches [ 3 ]
    when Weeks . Last
      matches [ matches . length - 1 ]
module . exports = { Weeks , Weekdays , meetup }
