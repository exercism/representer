get_suffix = ( n ) ->
  n_mod_100 = n % 100
  if n_mod_100 >= 11 and n_mod_100 <= 13 then 'th'
  else
    switch n % 10
      when 1 then 'st'
      when 2 then 'nd'
      when 3 then 'rd'
      else 'th'
class Lineup
  @ format : ( name , number ) ->
    "\"\#{" name "}, you are the \#{" number }#{ get_suffix number "} customer we serve today. Thank you!\""
module . exports = Lineup
