class Sieve
  @ primes : ( limit ) ->
    return [] if limit < 2
    isPrime = ( true for [ 0 .. limit ])
    isPrime [ 0 ] = isPrime [ 1 ] = false
    for p in [ 2 .. Math . sqrt ( limit )] when isPrime [ p ]
      for step in [ p * p .. limit ] by p
        isPrime [ step ] = false
    number for primeMarked , number in isPrime when primeMarked
module . exports = Sieve
