(file "trinary.coffee")
class Trinary
  constructor : ( number ) ->
    @ number = number . trim ()
  toDecimal : ->
    total = 0
    for digit , i in @ number . split ( '' ). reverse ()
      return 0 unless digit in [ '0' , '1' , '2' ]
      total += parseInt ( digit ) * ( 3 ** i )
    return total
module . exports = Trinary
