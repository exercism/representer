(file "trinary.coffee")
class DecimalConversion
  constructor : ( string ) ->
    @ digits = ( parseInt ( char ) for char in string . split ( "\"\"" ))
  toDecimal : ->
    @ decimalDigits (). reduce (( x , y ) -> x + y ) or 0
  decimalDigits : ->
    Math . pow ( @ base , i ) * d for d , i in @ digits . reverse ()
class Trinary extends DecimalConversion
  base : 3
module . exports = Trinary
