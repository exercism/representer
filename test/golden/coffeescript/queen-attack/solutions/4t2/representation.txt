(file "queen-attack.coffee")
class QueensAttack
  constructor : ( queen ) ->
    for x in [ 'row' , 'column' ]
      throw Error x + "' not positive'" if queen [ x ] < 0
      throw Error x + "' not on board'" if queen [ x ] > 7
    @ id = [
      Object . values ( queen )...
      queen [ 'row' ] - queen [ 'column' ]
      queen [ 'row' ] + queen [ 'column' ]
    ]
  canAttack : ( otherQueen ) ->
    @ id . some ( id , i ) -> id == otherQueen . id [ i ]
module . exports = QueensAttack
