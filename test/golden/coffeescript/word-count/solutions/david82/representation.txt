class WordCount
  constructor : ( @ text ) ->
  count : ->
    textNormalized = @ text . toLowerCase ()
    textCleaned = textNormalized . replace /[^a-z0-9'\s]+/g , "\" \""
    textCleaned = textCleaned . replace /(\s|^)'|'(\s|$)/g , "\"$1$2\""
    words = textCleaned . trim (). split /\s+/
    wordCounts = {}
    for word in words
      wordCounts [ word ] ?= 0
      wordCounts [ word ] += 1
    return wordCounts
module . exports = WordCount
