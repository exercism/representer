ResistorColor = require './resistor-color'
Array . prototype . take = ( n ) -> @ slice ( 0 , n )
class ResistorColorDuo
  @ value : ( colors ) ->
    colors
      . take ( 2 )
      . map ( ResistorColor . colorCode )
      . reduce (( val , code ) -> val * 10 + code )
module . exports = ResistorColorDuo
colors = [ 'black' , 'brown' , 'red' , 'orange' , 'yellow' ,
  'green' , 'blue' , 'violet' , 'grey' , 'white' ]
class ResistorColor
  @ colorCode : ( color ) -> colors . indexOf ( color )
  @ colors : () -> colors
module . exports = ResistorColor
