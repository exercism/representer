(file "hexadecimal.coffee")
class hexadecimal
  constructor : ( string ) ->
    @ string = string . split ( '' ). reverse ()
  parseHex = ( str ) ->
    switch str
      when 'a' then 10
      when 'b' then 11
      when 'c' then 12
      when 'd' then 13
      when 'e' then 14
      when 'f' then 15
      else parseInt ( str )
  toDecimal : ->
    for char in @ string
      if '0123456789abcdef' . indexOf ( char ) ==- 1
        @ string = '0'
        break
    if @ string . length == 1
      parseHex ( @ string [ 0 ])
    else
      @ string . reduce ( x , y , i ) ->
        Math . pow ( 16 , i ) * parseHex ( y ) + parseHex ( x )
module . exports = hexadecimal
