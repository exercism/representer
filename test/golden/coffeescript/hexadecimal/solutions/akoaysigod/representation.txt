(file "hexadecimal.coffee")
class Hexadecimal
  constructor : ( @ b ) ->
  toDecimal : ->
    v = @ b . length
    t = @ b . split ( "\"\"" )
    a = t . map ( b ) ->
      switch
        when b is 'a' then b = 10
        when b is 'b' then b = 11
        when b is 'c' then b = 12
        when b is 'd' then b = 13
        when b is 'e' then b = 14
        when b is 'f' then b = 15
      v -= 1
      b * 16 ** v
    if /[g-z]/ . test ( @ b )
      0
    else
      a . reduce ( x = 0 , y ) -> x + y
module . exports = Hexadecimal
