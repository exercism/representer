(file "hexadecimal.coffee")
VALID_NUMS = '0123456789'
VALID_ALPHA = 'abcdef'
class Hex
  constructor : ( @ hexStr ) ->
  toDecimal : ->
    decimalNum = 0
    for digit , index in @ hexStr
      if VALID_NUMS . indexOf ( digit ) == - 1 && VALID_ALPHA . indexOf ( digit ) == - 1
        return 0
      if VALID_ALPHA . indexOf ( digit ) > - 1
        digit = 10 + VALID_ALPHA . indexOf ( digit )
      decimalNum += Math . pow ( 16 , @ hexStr . length - index - 1 ) * digit
    return decimalNum
module . exports = Hex
