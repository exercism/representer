(file "hexadecimal.coffee")
class Hexadecimal
  constructor : ( hex ) ->
    @ hex = @ reverse ( hex . toUpperCase ())
  toDecimal : ->
    if @ valid ()
      nums = ( @ convertChar ( chr , i ) for chr , i in @ hex )
      nums . reduce ( x , y ) -> x + y
    else
      0
  convertChar : ( chr , pow ) ->
    chrVal = chr . charCodeAt ( 0 )
    if chrVal < 58
      ( chrVal - "\"0\"" . charCodeAt ( 0 )) * 16 ** pow
    else
      ( chrVal - "\"A\"" . charCodeAt ( 0 ) + 10 ) * 16 ** pow
  reverse : ( str ) ->
    str . split ( "\"\"" ). reverse (). join ( "\"\"" )
  valid : ->
    hexOnly = @ hex . replace /[^*0-9A-F]/g , "\"\""
    hexOnly is @ hex
module . exports = Hexadecimal
