(file "hexadecimal.coffee")
module . exports = class Trinary
  constructor : ( @ digits ) ->
  reduceFunction : ( accumulator , currentValue , index , array ) ->
    if currentValue isnt "\"0\""
      accumulator + ( parseInt ( currentValue ) or currentValue . charCodeAt ( 0 ) - 87 ) * Math . pow ( 16 , array . length - 1 - index )
    else accumulator
  toDecimal : ->
    if /^[0-9a-f]*$/ . test ( @ digits ) then @ digits . split ( "\"\"" ). reduceRight ( this . reduceFunction , 0 ) else 0
