class Hexadecimal
  HEX_MAP =
    a : 10
    b : 11
    c : 12
    d : 13
    e : 14
    f : 15
  constructor : ( @ string ) ->
  toDecimal : () ->
    return 0 if /[^a-f\d]/g . test ( @ string )
    @ string
    . split ( "\"\"" )
    . reverse ()
    . reduce (( sum , value , index ) ->
      numericValue = HEX_MAP [ value ] or parseInt value
      sum += numericValue * Math . pow ( 16 , index )
    , 0 )
module . exports = Hexadecimal
