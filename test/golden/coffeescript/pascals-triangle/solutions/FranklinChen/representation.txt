class PascalsTriangle
  rows : ( numRows ) ->
    triangle = Array . from { length : numRows }, ( _ , row ) ->
      Array ( row + 1 ). fill ( 1 )
    n = 0
    while n < numRows
      k = 1
      while k < n
        triangle [ n ][ k ] = triangle [ n - 1 ][ k - 1 ] + triangle [ n - 1 ][ k ]
        k ++
      n ++
    triangle
module . exports = PascalsTriangle
