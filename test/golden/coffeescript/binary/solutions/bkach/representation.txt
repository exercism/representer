class Binary
  constructor : ( input ) ->
    @ dec = 0
    @ convert ( @ parsed input ) if /^[01]+$/ . test ( input )
  parsed : ( string ) ->
    string . split ( '' ). map (( n ) -> parseInt ( n ))
  convert : ( bin ) ->
    bin = bin . reverse ()
    for i in [ 0 ... bin . length ]
      @ dec += bin [ i ] * ( 2 ** i )
  toDecimal : () ->
    @ dec
module . exports = Binary
