  (source
    (function_definition is_pangram
      (parameters (local PLACEHOLDER_1)) :
      (body
        (variable_statement (name PLACEHOLDER_2) = "ABCDEFGHIJKLMNOPQRSTUVWXYZ")
        (variable_statement (name PLACEHOLDER_3) =
          (attribute PLACEHOLDER_1 .
            (attribute_call to_upper
              (arguments))))
        (for_statement (local PLACEHOLDER_4) in PLACEHOLDER_3 :
          (body
            (expression_statement
              (assignment PLACEHOLDER_2 =
                (attribute PLACEHOLDER_2 .
                  (attribute_call remove_chars
                    (arguments PLACEHOLDER_4)))))
            (if_statement
              (attribute PLACEHOLDER_2 .
                (attribute_call is_empty
                  (arguments))) :
              (body
                (return_statement true)))))
        (return_statement false)))))
