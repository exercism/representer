(file "resistor_color.asm")
default rel
section .rodata
black: db "\"black\"" , 0
brown: db "\"" brown "\", 0\nred: db \"" red "\"" , 0
orange: db "\"" orange "\", 0\nyellow: db \"" yellow "\"" , 0
green: db "\"" green "\", 0\nblue: db \"" bl ue "\", 0\nviolet: db \"" violet "\", 0\ngrey: db \"" grey "\", 0\nwhite: db \"" white "\"" , 0
section .data
color_array:
dq bl ack
dq brown
dq red
dq orange
dq yellow
dq green
dq bl ue
dq violet
dq grey
dq white
dq 0
section .text
global color_code
color_code:
xor eax , eax
lea rcx , [ color_array ]
mov rdx , [ rcx + rax * 8 ]
.arr_loop_start:
mov rsi , rdi
mov r8b , byte [ rdx ]
cmp r8b , byte [ rsi ]
jne .str_loop_end
.str_loop_start:
inc rdx
inc rsi
mov r8b , byte [ rdx ]
cmp r8b , byte [ rsi ]
jne .str_loop_end
test r8b , r8b
jne .str_loop_start
.str_loop_end:
cmp r8b , byte [ rsi ]
je .return
inc eax
mov rdx , [ rcx + rax * 8 ]
test rdx , rdx
jne .arr_loop_start
mov eax , - 1
.return:
ret
global colors
colors:
lea rax , [ color_array ]
ret
