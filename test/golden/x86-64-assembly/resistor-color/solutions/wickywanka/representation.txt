(file "resistor_color.asm")
default rel
section .rodata
black: db "\"black\"" , 0
brown: db "\"" brown "\", 0\nred: db \"" red "\"" , 0
orange: db "\"" orange "\", 0\nyellow: db \"" yellow "\"" , 0
green: db "\"" green "\", 0\nblue: db \"" bl ue "\", 0\nviolet: db \"" violet "\", 0\ngrey: db \"" grey "\", 0\nwhite: db \"" white "\"" , 0
section .data
all_colors: dq bl ack , brown , red , orange , yellow , green , bl ue , violet , grey , white , 0
section .text
global color_code
color_code:
lea rdx , [ al l_colors ]
xor rcx , rcx
_while:
mov rsi , [ rdx + rcx * 8 ]
call _strcmp
test rax , rax
jnz _end
inc rcx
jmp _while
_end:
mov rax , rcx
ret
_strcmp:
mov al , byte [ rsi ]
cmp al , byte [ rdi ]
jnz _strcmp_false
_strcmp_while:
cmp [ rdi ], byte 0
je _strcmp_true
inc rdi
inc rsi
mov al , byte [ rsi ]
cmp al , byte [ rdi ]
je _strcmp_while
_strcmp_false:
xor rax , rax
ret
_strcmp_true:
mov rax , 1
ret
global colors
colors:
lea rax , [ al l_colors ]
ret
