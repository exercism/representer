section .text
global steps
steps:
push rbp
setup:
cmp edi , 1
jl illegal
mov ecx , 0
mov esi , 2
mov r8d , 3
mov eax , edi
check_one:
cmp eax , 1
je done
step:
inc ecx
xor edx , edx
div esi
cmp edx , 0
jne is_odd
mov edi , eax
jmp ch eck_one
is_odd:
imul edi , r8d
inc edi
mov eax , edi
jmp ch eck_one
illegal:
mov ecx , - 1
done:
mov eax , ecx
pop rbp
ret
