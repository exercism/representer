section .text
global steps
steps:
xor rax , rax
test edi , edi
jng error
loop:
cmp rdi , 1
je complete
inc rax
bt rdi , 0
jc isodd
iseven:
shr rdi , 1
jmp loop
isodd:
lea rdi , [ rdi * 3 + 1 ]
jmp loop
complete:
ret
error:
mov rax , - 1
ret
