(file "inventory_management.asm")
WEIGHT_OF_EMPTY_BOX equ 500
TRUCK_HEIGHT equ 300
PAY_PER_BOX equ 5
PAY_PER_TRUCK_TRIP equ 220
section .text
global get_box_weight
get_box_weight:
movzx edi , di
movzx esi , si
imul edi , esi
movzx edx , dx
movzx ecx , cx
imul edx , ecx
lea eax , [ rdx + rdi + WEIGHT_OF_EMPTY_BOX ]
ret
global max_number_of_boxes
max_number_of_boxes:
xor edx , edx
movzx eax , al
mov eax , TRUCK_HEIGHT
div edi
ret
global items_to_be_moved
items_to_be_moved:
mov eax , edi
sub eax , esi
ret
global calculate_payment
calculate_payment:
mov esi , esi
imul rax , rsi , PAY_PER_BOX
mov edx , edx
imul rdx , rdx , PAY_PER_TRUCK_TRIP
add rax , rdx
mov ecx , ecx
imul rcx , r8
sub rax , rdi
sub rax , rcx
movzx r9 , r9b
lea rsi , [ r9 + 1 ]
cqo
idiv rsi
add rax , rdx
ret
