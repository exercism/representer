section .text
global leap_year
leap_year:
mov eax , edi
test eax , 3
jnz .nope
mov ecx , 100
cdq
div ecx
test edx , edx
jnz .yup
mov eax , edi
mov ecx , 400
cdq
div ecx
test edx , edx
jz .yup
.nope:
mov eax , 0
ret
.yup:
mov eax , 1
ret
