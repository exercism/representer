section .text
global square_of_sum
square_of_sum:
xor rax , rax
loop_square_of_sum:
test rdi , rdi
jz sum_done
add rax , rdi
dec rdi
jmp loop_square_of_sum
sum_done:
imul rax , rax
ret
global sum_of_squares
sum_of_squares:
xor rax , rax
loop_sum_of_squares:
test rdi , rdi
jz end
mov rdx , rdi
imul rdx , rdi
add rax , rdx
dec rdi
jmp loop_sum_of_squares
global di fference_of_squares
difference_of_squares:
push rbx
push rdi
call sum_of_squares
mov rbx , rax
pop rdi
call square_of_sum
sub rax , rbx
pop rbx
end:
ret
