section .text
global square_of_sum
square_of_sum:
lea eax , [ edi + 1 ]
mul edi
shr eax , 1
mul eax
ret
global sum_of_squares
sum_of_squares:
lea eax , [ edi + 1 ]
mul edi
lea edi , [ 2 * edi + 1 ]
mul edi
mov edi , 0xAAAAAAAB
imul rax , rdi
shr rax , 34
ret
global di fference_of_squares
difference_of_squares:
lea eax , [ edi + 1 ]
lea ecx , [ edi - 1 ]
mul edi
mul ecx
lea edi , [ 2 * edi + edi + 2 ]
mul edi
mov edi , 0xAAAAAAAB
imul rax , rdi
shr rax , 35
ret
