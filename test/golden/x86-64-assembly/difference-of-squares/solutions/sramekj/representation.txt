(file "difference_of_squares.asm")
section .text
global square_of_sum
square_of_sum:
xor rax , rax
loop:
add rax , rdi
dec rdi
jnz loop
mul rax
ret
global sum_of_squares
sum_of_squares:
xor rcx , rcx
loop2:
mov rax , rdi
mul rdi
lea rcx , [ rax + rcx ]
dec rdi
jnz loop2
mov rax , rcx
ret
global di fference_of_squares
difference_of_squares:
push rdi
call sum_of_squares
mov rbx , rax
pop rdi
call square_of_sum
sub rax , rbx
ret
