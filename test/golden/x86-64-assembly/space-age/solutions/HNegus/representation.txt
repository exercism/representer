[ default rel ]
section .text
global age
age:
lea rax , [ mercury_year ]
movss xmm1 , [ rax + rdi * 4 ]
mulss xmm1 , [ earth_seconds ]
cvtsi2ss xmm0 , qword rsi
divss xmm0 , xmm1
ret
