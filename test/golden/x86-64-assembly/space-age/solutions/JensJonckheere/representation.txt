(file "space_age.asm")
section .text
global age
age:
lea rax , [ rel orbit_array ]
movss xmm0 , [ rax + rdi * 4 ]
movss xmm1 , [ rel earth_orbit_in_seconds ]
mulss xmm1 , xmm0
cvtsi2ss xmm0 , rsi
divss xmm0 , xmm1
ret
earth_orbit_in_seconds: dd 31557600.0
orbit_array: dd 0.2408467 , 0.61519726 , 1.0 , \
1 .8808158 , 11.862615 , 29.447498 , \
84 .016846 , 164.79132

%endif
