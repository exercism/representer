default rel
section .text
planets equ 8
global age
age:
cmp edi , planets
mov ecx , edi
lea rax , [ seconds_per_year ]
cvtsi2ss xmm0 , esi
divss xmm0 , [ rax + 4 * rcx ]
ret
section .rodata
seconds_per_year:
dd 7600543.81992
dd 19414149.052176
dd 31557600.0
dd 59354032.69008
dd 374355659.124
dd 929292362.8848
dd 2651370019.3296
dd 5200418560.032
