(file "space_age.asm")
default rel
section .rodata
planets:
mercury: dd 0.2408467
venus: dd 0.61519726
earth: dd 1.0
mars: dd 1.8808158
jupter: dd 11.862615
saturn: dd 29.447498
uranus: dd 84.016846
neptune: dd 164.79132
earthorbitalyear dd 31557600.0
section .text
global age
age:
lea r8 , [ planets ]
movss xmm0 , [ earthorbitalyear ]
movss xmm1 , [ r8 + rdi * 4 ]
mulss xmm1 , xmm0
cvtsi2ss xmm0 , rsi
divss xmm0 , xmm1
ret
