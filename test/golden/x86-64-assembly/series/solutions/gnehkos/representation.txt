(file "series.asm")
EMPTY_SERIES equ - 1
ZERO_LENGTH equ - 2
NEGATIVE_LENGTH equ - 3
EXCESSIVE_LENGTH equ - 4
SLICE_SIZE equ 20
section .text
global slices
slices:
cmp rdx , 0
je zero_length
jl negative_length
mov r8 , rsi
mov rcx , - 1
count_input:
inc rcx
lodsb
cmp al , 0
jne count_input
cmp rcx , 0
je empty_series
cmp rcx , rdx
jl excessive_length
mov r9 , rdi
mov r10 , rdx
sub rcx , rdx
inc rcx
mov rdx , rcx
mov r11 , rcx
add_slices:
dec rdx
mov rsi , r8
inc r8
mov rdi , r9
add r9 , SLICE_SIZE
mov rcx , r10
rep movsb
mov al , 0
stosb
cmp rdx , 0
jne add_slices
mov rax , r11
ret
empty_series:
mov rax , EMPTY_SERIES
ret
zero_length:
mov rax , ZERO_LENGTH
ret
negative_length:
mov rax , NEGATIVE_LENGTH
ret
excessive_length:
mov rax , EXCESSIVE_LENGTH
ret
%ifidn __OUTPUT_FORMAT__,elf64
section .note.GNU - stack noalloc noexec nowrite progbits
%endif
