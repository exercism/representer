EMPTY_SERIES equ - 1
ZERO_LENGTH equ - 2
NEGATIVE_LENGTH equ - 3
EXCESSIVE_LENGTH equ - 4
SLICE_SIZE equ 20
section .text
global slices
slices:
push rbx
push r12
push r13
push r14
push r15
mov r12 , rdi
mov r13 , rsi
mov r14 , rdx
test r14 , r14
jz .err_zero_len
cmp r14 , 0
jl .err_neg_len
xor r15 , r15
.len_loop:
movzx eax , byte [ r13 + r15 ]
test al , al
jz .len_done
inc r15
jmp .len_loop
.len_done:
test r15 , r15
jz .err_empty_series
cmp r14 , r15
ja .err_excessive_len
mov rax , r15
sub rax , r14
inc rax
mov rbx , rax
xor rcx , rcx
.outer_loop:
cmp rcx , rbx
jae .success_exit
imul rax , rcx , SLICE_SIZE
lea rdi , [ r12 + rax ]
xor rdx , rdx
.inner_loop:
cmp rdx , r14
jae .inner_done
mov r8 , rcx
add r8 , rdx
mov al , [ r13 + r8 ]
mov [ rdi + rdx ], al
inc rdx
jmp .inner_loop
.inner_done:
mov byte [ rdi + rdx ], 0
inc rcx
jmp .outer_loop
.success_exit:
mov rax , rbx
jmp .exit
.err_empty_series:
mov rax , EMPTY_SERIES
jmp .exit
.err_zero_len:
mov rax , ZERO_LENGTH
jmp .exit
.err_neg_len:
mov rax , NEGATIVE_LENGTH
jmp .exit
.err_excessive_len:
mov rax , EXCESSIVE_LENGTH
.exit:
pop r15
pop r14
pop r13
pop r12
pop rbx
ret
