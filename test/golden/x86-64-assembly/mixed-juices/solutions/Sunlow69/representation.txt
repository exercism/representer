section .rodata
juice_time db 1 , 3 , 3 , 4 , 5 , 4 , 7 , 10
section .text
global time_to_make_juice
time_to_make_juice:
lea r9 , [ rel juice_time ]
movzx eax , byte [ r9 + rdi - 1 ]
ret
global time_to_prepare
time_to_prepare:
mov r8 , rdi
xor edx , edx
.loop:
test esi , esi
jz .ret
mov edi , [ r8 ]
call time_to_make_juice
add edx , eax
add r8 , 4
dec esi
jmp .loop
.ret:
mov rax , rdx
ret
global limes_to_cut
limes_to_cut:
xor eax , eax
xor r8d , r8d
.loop:
cmp eax , edx
je .ret
cmp edi , r8d
jbe .ret
sub edi , r8d
mov cl , [ rsi + rax ]
inc eax
mov r8b , 6
cmp cl , 'S'
je .loop
add r8b , 2
cmp cl , 'M'
je .loop
add r8b , 2
jmp .loop
.ret:
ret
global remaining_orders
remaining_orders:
mov ecx , edi
xor edx , edx
.loop:
mov edi , [ rsi + 4 * rdx ]
inc edx
call time_to_make_juice
sub ecx , eax
ja .loop
.ret:
mov eax , edx
ret
