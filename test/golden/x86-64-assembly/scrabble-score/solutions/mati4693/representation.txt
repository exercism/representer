default rel
section .rodata
one_point db "\"AEIOULNRST\""
one_len dq $ - one_point
two_points db "\"DG\""
two_len dq $ - two_points
three_points db "\"BCMP\""
three_len dq $ - three_points
four_points db "\"FHVWY\""
four_len dq $ - four_points
five_points db "\"K\""
five_len dq $ - five_points
eight_points db "\"JX\""
eight_len dq $ - eight_points
ten_points db "\"QZ\""
ten_len dq $ - ten_points
section .text
global score
score:
mov r9 , rdi
mov r10 , 0
mov r11 , 0
.score_loop:
movzx rax , byte [ r9 + r11 ]
cmp al , 0x00
jz .done
movzx rdi , al
call upper_string
lea rdi , [ one_point ]
mov rcx , [ one_len ]
mov r12 , 1
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ two_points ]
mov rcx , [ two_len ]
mov r12 , 2
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ three_points ]
mov rcx , [ three_len ]
mov r12 , 3
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ four_points ]
mov rcx , [ four_len ]
mov r12 , 4
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ five_points ]
mov rcx , [ five_len ]
mov r12 , 5
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ eight_points ]
mov rcx , [ eight_len ]
mov r12 , 8
repne scasb
cmove rdx , r12
je .add_point
lea rdi , [ ten_points ]
mov rcx , [ ten_len ]
mov r12 , 10
repne scasb
cmove rdx , r12
.add_point:
add r10 , rdx
inc r11
jmp .score_loop
.done:
mov rax , r10
ret
global upper_string
upper_string:
cmp di l , 0x61
jl .upper_skip
sub di l , 32
.upper_skip:
mov al , di l
ret
