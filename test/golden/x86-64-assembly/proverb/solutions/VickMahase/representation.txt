section .text
global recite
copy_string:
push rbx
mov rbx , rdi
.loop:
movzx eax , byte [ rsi ]
test al , al
jz .done
mov [ rbx ], al
inc rbx
inc rsi
jmp .loop
.done:
mov rax , rbx
pop rbx
ret
recite:
push rbp
mov rbp , rsp
push rbx
push r12
push r13
push r14
push r15
mov r12 , rdi
mov r13 , rsi
mov r14 , r12

jz .single_element
mov r15 , [ r13 ]
xor ebx , ebx
.line_loop:
mov rsi , [ r13 + rbx * 8 ]
test rsi , rsi
jz .last_element
mov rsi , [ r13 + rbx * 8 + 8 ]
test rsi , rsi
jz .last_element
lea rsi , [ prefix ]
mov rdi , r14
call copy_string
mov r14 , rax

mov rdi , r14
call copy_string
mov r14 , rax

mov rdi , r14
call copy_string
mov r14 , rax
mov rsi , [ r13 + rbx * 8 + 8 ]
mov rdi , r14
call copy_string
mov r14 , rax
lea rsi , [ suffix ]
mov rdi , r14
call copy_string
mov r14 , rax
inc rbx
jmp .line_loop
.last_element:
lea rsi , [ last_line ]
mov rdi , r14
call copy_string
mov r14 , rax
mov rsi , r15
mov rdi , r14
call copy_string

mov byte [ r14 ], 10

.single_element:

call copy_string
mov r14 , rax
mov rsi , [ r13 ]
mov rdi , r14
call copy_string
mov r14 , rax
mov byte [ r14 ], '.'
inc r14
mov byte [ r14 ], 10
inc r14
mov byte [ r14 ], 0
jmp .done
.empty_list:
mov byte [ r12 ], 0
.done:
pop r15
pop r14
pop r13
pop r12
pop rbx
pop rbp
ret
