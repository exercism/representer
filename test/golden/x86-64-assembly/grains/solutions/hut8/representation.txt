(file "grains.asm")
section .text
global square
square:
cmp rdi , 0
jle .error
cmp rdi , 64
jg .error
mov cl , di l
dec cl
mov rax , 1
shl rax , cl
ret
.error:
xor rax , rax
ret
global total
total:
xor rcx , rcx
xor rbx , rbx
.loop:
inc rcx
cmp rcx , 64
jg .done
mov rdi , rcx
push rcx
push rbx
call square
pop rbx
pop rcx
add rbx , rax
jmp .loop
.done:
mov rax , rbx
ret
