section .text
global square
square:
movsx rdi , edi
cmp rdi , 0
jle .return_zero
mov rax , rdi
cmp rdi , 2
jle .end
mov rax , 2
.loop:
dec rdi
imul rax , 2
cmp rdi , 2
jne .loop
.end:
ret
.return_zero:
mov rax , 0
ret
