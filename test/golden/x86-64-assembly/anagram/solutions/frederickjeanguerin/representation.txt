bits 64
section .text
global find_anagrams
find_anagrams:
mov r10 , rdi
mov r11 , rsi
.for_each_candidate:
dec edx
js .done
mov rdi , [ r11 + rdx * 8 ]
mov rsi , rcx
call are_unequal_anagrams
mov [ r10 + rdx * 4 ], eax
jmp .for_each_candidate
.done:
vzeroupper
ret
are_unequal_anagrams:
push rdi
push rsi
call are_equals
pop rsi
pop rdi
test eax , eax
jnz .false
jmp are_anagrams
.false:
xor eax , eax
ret
are_anagrams:
mov r8 , rdi
mov r9 , rsi
sub rsp , 32
mov rdi , rsp
mov rsi , r8
call count_letters
mov r8 , rdi
sub rsp , 32
mov rdi , rsp
mov rsi , r9
call count_letters
mov r9 , rdi
vmovdqu ymm0 , [ r8 ]
vmovdqu ymm1 , [ r9 ]
vpcmpeqb ymm0 , ymm0 , ymm1
vpmovmskb eax , ymm0
cmp eax , 0xFFFFFFFF
sete al
movzx eax , al
add rsp , 64
ret
count_letters:
vpxor ymm0 , ymm0
vmovdqu [ rdi ], ymm0
xor eax , eax
.for_each_char:
mov al , [ rsi ]
test al , al
jz .done
and al , ~ ( 'a' - 'A' )
sub al , 'A'
and al , 0b 11111
inc byte [ rdi + rax ]
inc rsi
jmp .for_each_char
.done:
ret
are_equals:
mov eax , 1
.for_each_char_pairs:
mov r8b , [ rdi ]
and r8b , ~ ( 'a' - 'A' )
mov r9b , [ rsi ]
and r9b , ~ ( 'a' - 'A' )
cmp r8b , r9b
jne .false
test r8b , r8b
jz .true
inc rdi
inc rsi
jmp .for_each_char_pairs
.false:
xor eax , eax
.true:
ret
