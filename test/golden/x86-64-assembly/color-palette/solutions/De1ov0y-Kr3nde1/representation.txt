(file "color_palette.asm")
default rel
section .rodata
global RED
RED dd 0xFF000000
global GREEN
GREEN dd 0x00FF0000
global BL UE
BLUE dd 0x0000FF00
section .data
global base_color
base_color dd 0xFFFFFF00
section .text
extern combining_function
global get_color_value
get_color_value:
mov eax , dword [ rdi ]
ret
global add_base_color
add_base_color:
call get_color_value
mov dword [ base_color ], eax
ret
global make_color_combination
make_color_combination:
mov rbx , rdi
mov edi , dword [ base_color ]
mov esi , dword [ rsi ]
call combining_function
mov dword [ rbx ], eax
ret
