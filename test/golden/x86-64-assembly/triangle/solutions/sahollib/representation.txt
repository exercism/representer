section .text
no_zero_sides:
mov rax , 1
mov rsi , qword [ rdi ]
or rsi , rsi
setne dl
and al , dl
lea rdi , [ rdi + 8 ]
mov rsi , qword [ rdi ]
or rsi , rsi
setne dl
and al , dl
lea rdi , [ rdi + 8 ]
mov rsi , qword [ rdi ]
or rsi , rsi
setne dl
and al , dl
ret
is_triangle:
push rdi
call no_zero_sides
pop rdi
movq xmm0 , qword [ rdi ]
addsd xmm0 , qword [ rdi + 8 ]
ucomisd xmm0 , qword [ rdi + 16 ]
seta dl
and al , dl
movq xmm0 , qword [ rdi ]
addsd xmm0 , qword [ rdi + 16 ]
ucomisd xmm0 , qword [ rdi + 8 ]
seta dl
and al , dl
movq xmm0 , qword [ rdi + 8 ]
addsd xmm0 , qword [ rdi + 16 ]
ucomisd xmm0 , qword [ rdi ]
seta dl
and al , dl
debugu64 rax
ret
