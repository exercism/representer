section .text
global value_of_card
value_of_card:
cmp di l , C10
jg .face_card
mov al , di l
ret
.face_card:
cmp di l , CA
je .ace
mov al , 10
ret
.ace:
mov al , 1
ret
get_card_pair_values:
call value_of_card
mov bl , al
mov di l , si l
call value_of_card
mov dl , al
mov al , bl
ret
global higher_card
higher_card:
mov r12b , di l
mov r13b , si l
call get_card_pair_values
mov bl , dl
xor rdx , rdx
cmp al , bl
jz .equal
jl .less
.greater:
mov al , r12b
ret
.less:
mov al , r13b
ret
.equal:
mov al , r12b
mov dl , r13b
ret
global value_of_ace
value_of_ace:
cmp di l , CA
jz .has_ace
cmp si l , CA
jz .has_ace
call get_card_pair_values
add al , dl
cmp al , C10
ja .no_11
mov al , 11
ret
.has_ace:
.no_11:
mov al , 1
ret
global is_blackjack
is_blackjack:
.check_first_is_ace:
cmp di l , CA
jz .first_is_ace
.check_second_is_ace:
cmp si l , CA
jz .second_is_ace
jmp .no
.first_is_ace:
cmp si l , C9
jna .no
cmp si l , CA
jz .no
jmp .yes
.second_is_ace:
cmp di l , C9
jna .no
jmp .yes
.no:
mov al , FALSE
ret
.yes:
mov al , TRUE
ret
ret
global can_split_pairs
can_split_pairs:
call get_card_pair_values
cmp al , dl
jz .can_split
.cant:
mov rax , FALSE
ret
.can_split:
mov rax , TRUE
ret
global can_double_down
can_double_down:
call get_card_pair_values
add al , dl
cmp al , 8
jna .no_double_down
cmp al , 12
jnb .no_double_down
.double_down:
mov rax , TRUE
ret
.no_double_down:
mov rax , FALSE
ret
