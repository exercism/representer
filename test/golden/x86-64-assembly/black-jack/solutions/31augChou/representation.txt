section .text
global value_of_card
value_of_card:
cmp rdi , 10
jle numeric_card
cmp rdi , 14
je ace
mov rax , 10
jmp end
ace:
mov rax , 1
jmp end
numeric_card:
mov rax , rdi
end:
ret
global higher_card
higher_card:
call value_of_card
mov rbx , rax
mov r8 , rdi
mov rdi , rsi
call value_of_card
cmp rax , rbx
jg second_card
je equ al s
first_card:
mov rax , r8
mov rdx , 0
ret
second_card:
mov rax , rsi
mov rdx , 0
ret
equ al s :
mov rax , r8
mov rdx , rsi
ret
ret
global value_of_ace
value_of_ace:
cmp rdi , 14
je ace_is_one
call value_of_card
mov r8 , rax
mov rdi , rsi
cmp rdi , 14
je ace_is_one
call value_of_card
add r8 , rax
cmp r8 , 10
jg ace_is_one
mov rax , 11
ret
ace_is_one:
mov rax , 1
ret
global is_blackjack
is_blackjack:
mov r8 , rdi
mov r9 , rsi
call higher_card
mov rdi , rax
call value_of_card
cmp rax , 10
je ch eck_ace_card
mov rax , FALSE
ret
check_ace_card:
cmp r8 , CA
je match
cmp r9 , CA
je match
mov rax , FALSE
ret
match:
mov rax , TRUE
ret
global can_split_pairs
can_split_pairs:
call value_of_card
mov rbx , rax
mov rdi , rsi
call value_of_card
cmp rax , rbx
je can_split
mov rax , FALSE
ret
can_split:
mov rax , TRUE
ret
global can_double_down
can_double_down:
call value_of_card
mov rbx , rax
mov rdi , rsi
call value_of_card
add rbx , rax
cmp rbx , 9
je can_d
cmp rbx , 10
je can_d
cmp rbx , 11
je can_d
mov rax , FALSE
ret
can_d:
mov rax , TRUE
ret
