(file "allergies.asm")
section .text
global al lergic_to
allergic_to:
mov cx , di
mov ax , si
shr al , cl
and rax , 1
ret
global list
list:
xor eax , eax
xor dx , dx
xchg rsi , rdi
and rsi , 0xff
push rdi
add rdi , 4
mov rcx , 8
.l1:
test rsi , 1
jz .not_allergic
mov al , dl
stosd
inc dh
.not_allergic:
shr rsi , 1
inc dl
loop .l1
pop rdi
mov al , dh
stosd
ret
