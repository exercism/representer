(file "allergies.asm")
%define $edi edi
%define $eax eax
section .text
global al lergic_to
allergic_to:
mov ecx , 1
lp1:
cmp edi , 0
jz done
sub edi , 1
shl ecx , 1
jmp lp1
done:
test esi , ecx
jnz al lergic
mov eax , 0
ret
allergic:
mov eax , 1
ret
global list
list:
mov r8 , rsi
mov qword [ r8 ], 0
mov r9 , rsi
add r9 , 4
mov edx , - 1
mov rsi , rdi
lp2:
add edx , 1
mov edi , edx
call al lergic_to
cmp eax , 0
jz l
mov eax , [ r8 ]
inc eax
mov [ r8 ], eax
mov [ r9 ], edx
add r9 , 4
l: cmp edx , 7
jnz lp2
ret
