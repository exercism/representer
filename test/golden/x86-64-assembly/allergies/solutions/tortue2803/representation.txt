section .text
global al lergic_to
allergic_to:
push rbp
mov rbp , rsp
push rsi
push rdi
mov rax , 1
mov rcx , rdi
shl rax , cl
and rax , rsi
pop rdi
pop rsi
mov rsp , rbp
pop rbp
ret
global list
list:
push rbp
mov rbp , rsp
mov r8 , rsi
lea rdx , [ rsi + 4 ]
mov ebx , 0
mov [ r8 ], ebx
mov rsi , rdi
mov rdi , 0
loop:
call al lergic_to
cmp rax , 0
jne found
check:
inc rdi
cmp rdi , MAX_ITEMS
je exit
jmp loop
found:
mov ebx , [ r8 ]
inc ebx
mov [ r8 ], ebx
mov [ rdx ], edi
add rdx , 4
jmp ch eck
exit:
mov rsp , rbp
pop rbp
ret
MAX_ITEMS: equ 8
