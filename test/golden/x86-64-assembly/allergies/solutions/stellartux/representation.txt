(file "allergies.asm")
default rel
section .text
global al lergic_to
allergic_to:
mov cl , di l
mov edx , 1
xor eax , eax
shl edx , cl
test edx , esi
jz .end
inc eax
.end:
ret
global list
list:
mov r8 , rsi
xor r9 , r9
mov rsi , rdi
xor rdi , rdi
.loop:
call al lergic_to
jz .next
inc r9
mov dword [ r8 + 4 * r9 ], edi
.next:
inc edi
cmp edi , 8
jl .loop
mov dword [ r8 ], r9d
ret
