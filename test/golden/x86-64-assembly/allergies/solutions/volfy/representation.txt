(file "allergies.asm")
section .text
global al lergic_to
allergic_to:
xor rax , rax
bt esi , edi
setc al
ret
global list
list:
mov rcx , rsi
xor rdx , rdx
mov rsi , rdi
xor rdi , rdi
.loop:
call al lergic_to
test al , al
jz .inc_ops
mov [ rcx + rdx * 4 + 4 ], edi
inc edx
.inc_ops:
inc edi
cmp edi , 8
jl .loop
.endlist:
mov [ rcx ], edx
ret

%endif
