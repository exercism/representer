(file "allergies.asm")
section .text
global al lergic_to
allergic_to:
xor eax , eax
bts esi , edi
setc al
ret
global list
list:
xor eax , eax
xor ecx , ecx
%assign i 0
%rep 8
xor edx , edx
mov dword [ rsi + 4 + 4 * rax ], i
test edi , 1 << i
setnz dl
add eax , edx
%assign i i + 1
%endrep
mov [ rsi ], eax
ret
list_looping:
xor eax , eax
xor ecx , ecx
.loop:
test edi , 1
jz .next
mov [ rsi + 4 + 4 * rax ], ecx
inc eax
.next:
shr edi , 1
inc ecx
cmp ecx , 8
jne .loop
mov [ rsi ], eax
ret
