global exchange_rate
exchange_rate:
imul rdi , 8
add rdi , rdx
movsd xmm1 , [ rdi ]
imul rsi , 8
add rsi , rdx
movsd xmm0 , [ rsi ]
divsd xmm0 , xmm1

global get_value_of_bills
get_value_of_bills:
movsx rax , si
imul rax , rdi
ret
global get_number_of_bills
get_number_of_bills:

divss xmm0 , xmm1
roundss xmm0 , xmm0 , 1
cvtss2si eax , xmm0
ret

exchangeable_value:

add rdi , 100

divss xmm0 , xmm2
mov rax , 100
cvtsi2ss xmm3 , rax
mulss xmm0 , xmm3

call get_number_of_bills
