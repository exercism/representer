section .text
global find
find:
push rbx
test rdi , rdi
jz fail
test esi , esi
jz fail
xor ecx , ecx
sub esi , 1
bog:
mov eax , ecx
add eax , esi
shr eax , 1
mov r8d , [ rdi + rax * 4 ]
cmp edx , r8d
je end
jl less
cmp ecx , esi
je fail
more:
mov ecx , eax
inc ecx
jmp afterless
less:
cmp eax , ecx
je fail
mov esi , eax
dec esi
afterless:
jmp bog
fail:
mov eax , - 1
end:
pop rbx
ret
