%include "\"debug.mac\""
default rel
%ifndef MAGAZINE_LAYOUT_ASM
%define MAGAZINE_LAYOUT_ASM
%define COLUMN_COUNT 4
%define COLUMN_STRIDE 18
%define SPACE 0x20
%define RULE 0x3D
%define RULE_WIDTH 4
%define PLAIN 0x2E
%define RULED 0x2D
%define BOXED 0x23
%define COUNTER rcx
%macro fill_run 3
push rbx
mov COUNTER , % 2
test COUNTER , COUNTER
jz %% fill_loop_end
lea rbx , [ % 1 ]
%%fill_loop:
mov byte [ rbx ], % 3
inc rbx
dec COUNTER
jnz %% fill_loop
%%fill_loop_end:
pop rbx
%endmacro
%macro lay_columns 2
push rax
push rbx
push rcx
push rdx
push rdi
push rsi
push r9
push r10
mov COUNTER , COLUMN_COUNT
mov rdi , % 1
mov rsi , % 2
xor r9 , r9
%%lay_columns_loop:
lea r10 , [ rdi + r9 * 8 ]
mov rax , r9
mov rbx , COLUMN_STRIDE
mul rbx
add rax , rsi
mov [ r10 ], rax
inc r9
dec COUNTER
jnz %% lay_columns_loop
pop r10
pop r9
pop rsi
pop rdi
pop rdx
pop rcx
pop rbx
pop rax
%endmacro
%macro set_header 2
%if %2 == 0
mov byte [ % 1 ], PLAIN
%elif %2 == 1
mov byte [ % 1 ], RULED
%else
mov byte [ % 1 ], BOXED
%endif
%endmacro
%endif
