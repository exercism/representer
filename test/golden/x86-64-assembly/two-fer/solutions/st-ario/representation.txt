(file "two_fer.asm")
default rel
section .rodata
text1 db "\"" One for "\""
text2 db "\"" , one for me. "\"" , 0
default_name db "\"you\"" , 0
section .text
global two_fer
two_fer:
mov rdx , rsi
mov rcx , [ text1 ]
mov [ rdx ], rcx
add rdx , 8
test rdi , rdi
jnz name_given
mov ecx , [ default_name ]
mov [ rdx ], ecx
add rdx , 3
jmp last
name_given:
mov rax , rdi
mov cl , byte [ rax ]
loop:
mov byte [ rdx ], cl
inc rdx
inc rax
mov cl , byte [ rax ]
cmp cl , 0
jne loop
last:
mov rcx , [ text2 ]
mov [ rdx ], rcx
mov rcx , [ text2 + 8 ]
mov [ rdx + 8 ], rcx
ret
