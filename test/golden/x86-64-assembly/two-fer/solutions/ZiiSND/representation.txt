default rel
section .text
global two_fer
two_fer:
lea rax , [ prefix ]
call load_into_buffer
lea rax , [ rdi ]
cmp rax , 0
jne .not_null
lea rax , [ defName ]
.not_null:
call load_into_buffer
lea rax , [ postfix ]
call load_into_buffer
ret
load_into_buffer:
.loop:
mov dl , byte [ rax ]
mov byte [ rsi ], dl
inc rax
inc rsi
cmp dl , 0
jnz .loop
dec rsi
ret
section .rodata
prefix: db "\"" One for "\", 0\ndefName: db \"" you "\", 0\npostfix: db \"" , one for me. "\"" , 0
