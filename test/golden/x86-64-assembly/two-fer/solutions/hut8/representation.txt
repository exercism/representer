(file "two_fer.asm")
default rel
section .text
global two_fer
extern strlen
two_fer:
mov r8 , rdi
mov r9 , rsi
cld
lea rsi , [ msg_pre ]
mov rdi , r9
mov rcx , msg_pre_len
rep movsb
mov rdi , r8
call _namelen
mov r10 , rcx
mov rdi , r9
add rdi , msg_pre_len
test rcx , rcx
jnz _string_present
lea rsi , [ msg_name_default ]
mov rcx , msg_name_default_len
mov r10 , rcx
jmp _copy_name
_string_present:
mov rsi , r8
_copy_name:
rep movsb
_copy_post:
push rbp
mov rbp , rsp
lea rsi , [ msg_post ]
mov rdi , r9
add rdi , msg_pre_len
add rdi , r10
mov rcx , msg_post_len
rep movsb
pop rbp
ret
_namelen:
xor rcx , rcx
test rdi , rdi
jz __namelen_done
__namelen_loop:
cmp byte [ rdi ], 0
jz __namelen_done
inc rcx
inc rdi
jmp __namelen_loop
__namelen_done:
ret
section .data
msg_pre: db "\"" One for "\""
msg_pre_len: equ $ - msg_pre
msg_post: db "\"" , one for me. "\"" , 0
msg_post_len: equ $ - msg_post
msg_name_default: db "\"you\""
msg_name_default_len: equ $ - msg_name_default
