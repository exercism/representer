default rel
section .rodata
msg_begin: db "\"" One for "\"" , 0
msg_name_def: db "\"you\"" , 0
msg_end: db "\"" , one for me. "\"" , 0
%macro WriteToBuf 1
cmp byte [ rcx ], 0
je %% end_write
%%write_loop:
mov al , [ rcx ]
mov [ % 1 ], al
inc % 1
inc rcx
cmp byte [ rcx ], 0
jne %% write_loop
%%end_write:
%endmacro
section .text
global two_fer
two_fer:
lea rcx , [ msg_begin ]
WriteToBuf rsi
lea rcx , [ msg_name_def ]
test rdi , rdi
cmovne rcx , rdi
WriteToBuf rsi
lea rcx , [ msg_end ]
WriteToBuf rsi
mov byte [ rsi ], 0
ret
