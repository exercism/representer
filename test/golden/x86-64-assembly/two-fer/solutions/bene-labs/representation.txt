(file "two_fer.asm")
default rel
section .rodata
pre_name: db "\"" One for "\"" , 0
default_name: db "\"you\"" , 0
after_name: db "\"" , one for me. "\"" , 0
section .text
global two_fer
two_fer:
mov r10 , rsi
xor rcx , rcx
lea r12 , [ pre_name ]
.ADD_PRE_NAME_LOOP:
mov bl , BYTE [ r12 + rcx ]
mov BYTE [ r10 + rcx ], bl
cmp bl , 0
je .SETUP_SET_NAME_LOOP
inc rcx
jmp .ADD_PRE_NAME_LOOP
.SETUP_SET_NAME_LOOP:
add r10 , rcx
xor rcx , rcx
cmp rdi , 0
je .ON_EMPTY_NAME
mov al , BYTE [ rdi ]
cmp al , 0
je .ON_EMPTY_NAME
jmp .SET_NAME_LOOP
.ON_EMPTY_NAME:
lea rdi , [ default_name ]
.SET_NAME_LOOP:
mov bl , BYTE [ rdi + rcx ]
mov BYTE [ r10 + rcx ], bl
cmp bl , 0
je .SETUP_AFER_NAME_LOOP
inc rcx
jmp .SET_NAME_LOOP
.SETUP_AFER_NAME_LOOP:
add r10 , rcx
xor rcx , rcx
lea r12 , [ after_name ]
.ADD_AFTER_NAME_LOOP:
mov bl , BYTE [ r12 + rcx ]
mov BYTE [ r10 + rcx ], bl
cmp bl , 0
je .EXIT
inc rcx
jmp .ADD_AFTER_NAME_LOOP
.EXIT:
ret
