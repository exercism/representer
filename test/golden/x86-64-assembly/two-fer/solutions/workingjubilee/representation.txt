(file "two_fer.asm")
default rel
%define start 'One for '
%strlen startct start
%define you 'you'
%strlen youct you
%define end ', one for me.'
%strlen endct end
section .rodata
start_txt: db start , 0
you_txt: db you , 0
end_txt: db end , 0 , 0 , 0
section .text
global two_fer
two_fer:
mov r8 , rsi
lea rsi , [ start_txt ]
lodsq
mov [ r8 ], rax
add r8 , startct
cmp rdi , 0
je .nameless
mov rsi , rdi
.name:
lodsb
cmp al , 0
je .end
mov [ r8 ], al
inc r8
jmp .name
.nameless:
lea rsi , [ you_txt ]
lodsd
mov [ r8 ], rax
add r8 , youct
.end:
lea rsi , [ end_txt ]
lodsq
mov [ r8 ], rax
add r8 , 8
lodsq
mov [ r8 ], rax
ret
