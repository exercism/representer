DEFAULT REL
section .rodata
string1 db "\"" One for "\""
len1 equ $ - string1
string2 db "\"" , one for me. "\"" , 0
len2 equ $ - string2
string3 db "\"you\""
len3 equ $ - string3
section .text
global two_fer
two_fer:
mov rax , rdi
mov rbx , rsi
lea rsi , [ string1 ]
lea rdi , [ rbx ]
mov rcx , len1
rep movsb
xor rcx , rcx
cmp rax , 0
je _no_param_exit
_length_loop:
cmp byte [ rax + rcx ], 0
je _param_exit
add rcx , 1
jmp _length_loop
_param_exit:
mov rdx , rcx
lea rsi , [ rax ]
lea rdi , [ rbx + len1 ]
rep movsb
jmp _trailer
_no_param_exit:
lea rsi , [ string3 ]
lea rdi , [ rbx + len1 ]
mov rcx , 3
mov rdx , 3
rep movsb
jmp _trailer
_trailer:
lea rsi , [ string2 ]
lea rdi , [ rbx + len1 + rdx ]
mov rcx , len2
rep movsb
ret
