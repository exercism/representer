(file "two_fer.asm")
section .rodata
msg_start: db "\"" One for "\"" , 0
msg_end: db "\"" , one for me. "\", 0\nmsg_you:    db \"" you "\"" , 0
section .text
global two_fer
two_fer:
mov rax , rdi
mov rdi , rsi
lea rsi , [ rel msg_start ]
call append_string
cmp rax , 0
je add_default
mov rsi , rax
jmp continue
add_default:
lea rsi , [ rel msg_you ]
continue:
call append_string
lea rsi , [ rel msg_end ]
call append_string
mov byte [ rdi ], 0
ret
append_string:
movsb
cmp byte [ rsi ], 0
jne append_string
ret
