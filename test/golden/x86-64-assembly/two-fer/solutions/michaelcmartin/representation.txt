(file "two_fer.asm")
default rel
section .text
global two_fer
mystrcat:
cld
xchg rdi , rsi
.suflp: lodsb
or al , al
jnz .suflp
xchg rdi , rsi
dec rdi
.cpylp: lodsb
stosb
or al , al
jnz .cpylp
ret
two_fer:
push rbp
mov rbp , rsp
push r12
push r13
mov r12 , rdi
mov r13 , rsi
or r12 , r12
jnz .ok
lea r12 , [ def ]
.ok: mov byte [ r13 ], 0
mov rdi , r13
lea rsi , [ pre ]
call mystrcat
mov rdi , r13
mov rsi , r12
call mystrcat
mov rdi , r13
lea rsi , [ post ]
call mystrcat
pop r13
pop r12
pop rbp
ret
section .rodata
pre: db "\"" One for "\",0\ndef:    db \"" you "\",0\npost:   db \"" , one for me. "\"" , 0
