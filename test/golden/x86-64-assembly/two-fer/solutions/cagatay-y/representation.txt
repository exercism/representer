(file "two_fer.asm")
default rel
section .rodata
prefix: db "\"" One for "\", 0\n    suffix: db \"" , one for me. "\"" , 0
default_name: db "\"you\"" , 0
section .text
global two_fer
two_fer:
push rbp
mov rbp , rsp
lea rcx , [ suffix ]
push rcx
cmp rdi , 0
jne .push_name
lea rdi , [ default_name ]
.push_name:
push rdi
lea rcx , [ prefix ]
push rcx
mov rdi , rsi
.stack_loop:
cmp rsp , rbp
je .return
pop rsi
.string_loop:
mov cl , [ rsi ]
cmp cl , BYTE 0
je .stack_loop
mov [ rdi ], cl
inc rdi
inc rsi
jmp .string_loop
.return:
mov [ rdi ], BYTE 0
pop rbp
ret
%ifidn __OUTPUT_FORMAT__,elf64
section .note.GNU - stack noalloc noexec nowrite progbits
%endif
