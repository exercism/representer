section .text
global two_fer
two_fer:
mov rax , rdi
mov rdi , rsi
lea rsi , [ rel part1 ]
mov rcx , part1len
rep movsb
cmp rax , 0
je .no_name
mov r8 , rax
mov rcx , 0
.use_name:
inc cx
inc r8
cmp byte [ r8 - 1 ], 0
jne .use_name
dec cx
mov rsi , rax
rep movsb
jmp .end
.no_name:
lea rsi , [ rel part2 ]
mov rcx , part2len
rep movsb
.end:
lea rsi , [ rel part3 ]
mov rcx , part3len
rep movsb
mov byte [ rdi ], 0
ret
section .rodata
part1: db "\"" One for "\""
part1len equ $ - part1
part2: db "\"you\""
part2len equ $ - part2
part3: db "\"" , one for me. "\""
part3len equ $ - part3
