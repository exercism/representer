(file "two_fer.asm")
section .text
extern sp rintf
global two_fer
two_fer:
cmp rdi , 0
jne name
you:
lea rdx , [ rel youstr ]
jmp dowork
name:
mov rdx , rdi
dowork:
mov rdi , rsi
lea rsi , [ rel format ]
sub rsp , 8
call sp rintf WRT ..plt
add rsp , 8
ret
section .data
format: db "\"" One for % s , one for me. "\", 0\nyoustr:     db      \"" you "\"" , 0
