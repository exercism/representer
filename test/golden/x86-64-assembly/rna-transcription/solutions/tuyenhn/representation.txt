(file "rna_transcription.asm")
section .text
global to_rna
to_rna:
push rsi
.L0:
cmp byte [ rdi ], 0
je func_ret
cmp byte [ rdi ], "\"A\""
je .L1
cmp byte [ rdi ], "\"G\""
je .L2
cmp byte [ rdi ], "\"C\""
je .L3
cmp byte [ rdi ], "\"T\""
je .L4
.L1:
mov byte [ rsi ], "\"U\""
jmp .L5
.L2:
mov byte [ rsi ], "\"C\""
jmp .L5
.L3:
mov byte [ rsi ], "\"G\""
jmp .L5
.L4:
mov byte [ rsi ], "\"A\""
jmp .L5
.L5:
inc rdi
inc rsi
jmp .L0
func_ret:
mov byte [ rsi ], 0
pop rsi
mov rax , rsi
ret
