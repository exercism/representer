(file "rna_transcription.asm")
section .text
global to_rna
to_rna:
lea rax , [ rdi ]
dec rsi
.top:
inc rsi
mov dl , byte [ rax ]
inc rax
cmp dl , 0x41
je .A
cmp dl , 0x43
je .C
cmp dl , 0x47
je .G
cmp dl , 0x54
je .T
mov byte [ rsi ], 0x00
ret
.A:
mov byte [ rsi ], 0x55
jmp .top
.C:
mov byte [ rsi ], 0x47
jmp .top
.G:
mov byte [ rsi ], 0x43
jmp .top
.T:
mov byte [ rsi ], 0x41
jmp .top
