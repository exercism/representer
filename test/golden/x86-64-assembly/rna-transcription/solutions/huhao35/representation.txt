section .text
global to_rna
to_rna:
xor ax , ax
.loop:
mov al , byte [ rdi ]
test al , al
jz .r
cmp al , 'G'
je .c
cmp al , 'C'
je .g
cmp al , 'T'
je .a
jmp .u
.c:
mov ah , 'C'
jmp .end
.g:
mov ah , 'G'
jmp .end
.a:
mov ah , 'A'
jmp .end
.u:
mov ah , 'U'
.end:
mov [ rsi ], ah
inc rdi
inc rsi
jmp .loop
.r:
mov byte [ rsi ], 0
ret

section .note.GNU - stack noalloc noexec nowrite progbits
%endif
