default rel
section .rodata
dna_2_rna: db "\"U-G---C------------A\""
dna_2_rna_len equ $ - dna_2_rna
section .text
global to_rna
to_rna:
mov r10 , rsi
push rdi
pop rsi
mov rdi , r10
lea r10 , [ dna_2_rna ]
convert_dna_2_rna:
xor rax , rax
xor rbx , rbx
mov byte al , [ rsi ]
cmp al , 0
je convert_done
sub al , 65
jge ch eck_upper_index
jmp err_index
check_upper_index:
cmp al , dna_2_rna_len
jge err_index
mov byte bl , [ r10 + rax ]
mov byte [ rdi ], bl
inc rsi
inc rdi
jmp convert_dna_2_rna
convert_done:
err_index:
exit:
mov byte [ rdi ], 0
ret
