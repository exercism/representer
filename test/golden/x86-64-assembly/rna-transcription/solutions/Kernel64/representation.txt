(file "rna_transcription.asm")
section .text
global to_rna
to_rna:
mov rcx , - 1
loop:
inc rcx
mov al , [ rdi + rcx ]
cmp al , 0
je done
cmp al , "\"A\""
je put_U
cmp al , "\"T\""
je put_A
cmp al , "\"G\""
je put_C
cmp al , "\"C\""
je put_G
put_A:
mov byte [ rsi + rcx ], "\"A\""
jmp loop
put_U:
mov byte [ rsi + rcx ], "\"U\""
jmp loop
put_C:
mov byte [ rsi + rcx ], "\"C\""
jmp loop
put_G:
mov byte [ rsi + rcx ], "\"G\""
jmp loop
done:
mov byte [ rsi + rcx ], 0
ret
