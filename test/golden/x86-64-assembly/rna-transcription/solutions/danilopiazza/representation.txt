default rel
section .rodata
complements:
times 'A' db 0
db 'U'
times 'C' - 'A' - 1 db 0
db 'G'
times 'G' - 'C' - 1 db 0
db 'C'
times 'T' - 'G' - 1 db 0
db 'A'
times 256 - 'T' - 1 db 0
section .text
global to_rna
to_rna:
lea rbx , [ complements ]
.loop:
movzx rdx , byte [ rdi ]
cmp rdx , 0
je .end
mov rax , rbx
add rax , rdx
mov cl , [ rax ]
mov [ rsi ], cl
inc rdi
inc rsi
jmp .loop
.end:
mov byte [ rsi ], 0
ret
