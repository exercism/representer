(file "hamming.asm")
section .text
global di stance
distance:
xor rax , rax
.compare_loop:
mov dl , byte [ rsi ]
test dl , dl
jz .end_str
mov dl , byte [ rdi ]
test dl , dl
jz .end_str
xor rbx , rbx
cmp dl , byte [ rsi ]
setne bl
add rax , rbx
inc rsi
inc rdi
jmp .compare_loop
.end_str:
mov bl , byte [ rsi ]
cmp bl , byte [ rdi ]
je .done
mov rax , - 1
.done:
ret
