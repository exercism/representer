(file "hamming.asm")
section .text
global di stance
%define StrandA rdi
%define StrandB rsi
align 16
times - ( di stance.loop - di stance ) % 16 nop
distance:
test StrandA , StrandB
jz .fail
xor eax , eax
.loop:
mov dl , byte [ StrandA ]
mov cl , byte [ StrandB ]
inc StrandA
inc StrandB
cmp dl , cl
je .equal
inc eax
test cl , dl
jnz .loop
.fail:
mov eax , - 1
ret
.alignEqual:
align 16
.equal:
test dl , cl
jnz .loop
ret
