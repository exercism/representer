mov rsi , rsp

jne init

jz ch eck
cmp al , 'A'
jl cont

jg cont
cmp al , 'Z'
jle low

jl cont
sub rax , 'a'
jmp record
low: sub rax , 'A'
record: add rax , rsp
mov byte [ rax ], 1
cont: inc rdi
jmp loop
check: mov rax , 1
loop2: mov dl , [ rsp ]
and al , dl
jz out
inc rsp
cmp rsp , rcx
jne loop2
out: mov rsp , rcx
ret
