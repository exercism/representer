(file "grade-school.v")
module main
import arrays { flatten, lower_bound, binary_search }
struct SchoolRooster {
data map[int][]string
}
fn add_student(roster SchoolRooster, name string, grade int) SchoolRooster {
mut ans := roster.data.clone()
if !(name in flatten(ans.values())) {
mut grd_lst := ans[grade]
insort_left(mut grd_lst, name) or {
grd_lst << name
}
ans[grade] = grd_lst
}
return SchoolRooster{ans}
}
fn get_students_in_grade(roster SchoolRooster, grade int) []string {
return roster.data[grade]
}
fn get_all_students(roster SchoolRooster) []string {
mut ans := []string{}
mut grades := roster.data.keys()
if grades == []int{} { return ans }
grades.sort()
for grade in grades {
ans << roster.data[grade]
}
return ans
}
fn create_new_roster() SchoolRooster {
return SchoolRooster{map[int][]string{}}
}
// requires sorted array, returns an error, or returns nothing
fn insort_left[T](mut arr []T, val T) !{
arr.insert(binary_search(arr, lower_bound(arr, val)!)!, val)
}
