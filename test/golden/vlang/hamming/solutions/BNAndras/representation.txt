module main
struct Comparison {
left rune
right rune
}
fn (c Comparison) ok() bool {
return c.left == c.right
}
fn zip(a string, b string) []Comparison {
mut results := []Comparison{len: a.len, cap: a.len}
for i in 0 .. a.len {
results << Comparison{left: a[i], right: b[i]}
}
return results
}
fn distance(a string, b string) !int {
if a.len != b.len { return error('lengths must match!') }
mut count := 0
for comparison in zip(a, b) {
if !comparison.ok() {
count += 1
}
}
return count
}
