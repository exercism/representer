module main
import regex
import datatypes { Set }
fn is_isogram(word string) bool {
mut word_ := word.to_lower()
remove := r'[\-\s]+'
mut re := regex.regex_opt(remove) or { panic(err) }
word_ = re.replace(word_, '')
if word_.len == 0 {
return true
}
mut char_set := Set[u8]{}
for c in word_ {
if char_set.exists(c) {
return false
}
char_set.add(c)
}
return true
}
