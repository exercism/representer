module main
fn collatz(number int) !int {
return collatz_recursive(number, 0)
}
fn collatz_recursive(number int, steps int) !int {
if number <= 0 {
return error("Invalid number")
}
else if number == 1 {
return steps
}
else if number % 2 == 0 {
return collatz_recursive(number / 2, steps + 1)
} else {
return collatz_recursive(number * 3 + 1, steps + 1)
}
}
