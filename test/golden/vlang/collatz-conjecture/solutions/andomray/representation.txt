(file "collatz-conjecture.v")
module main
fn collatz(number int) !int {
mut n := number
if n == 1 {
return 0
} else if n > 1 {
mut count := 0
for n > 1 {
if n % 2 == 0 {
n = n / 2
} else {
n = (n * 3) + 1
}
count++
}
return count
} else {
return error('collatz(${number} should return an error')
}
}
