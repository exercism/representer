module main
fn is_isosceles(a f64, b f64, c f64) bool {
vertex := sum_vertex(a,b,c)
if vertex == 1 || vertex == 3{
return true
}else{
return false
}
}
fn is_equilateral(a f64, b f64, c f64) bool {
vertex := sum_vertex(a,b,c)
if vertex == 3{
return true
}else{
return false
}
}
fn is_scalene(a f64, b f64, c f64) bool {
vertex := sum_vertex(a,b,c)
if vertex != 1 && vertex != 3 && vertex == 0 {
return true
}else{
return false
}
}
fn sum_vertex(a f64, b f64, c f64) f64{
mut count := 0
mut vertex := 0
if a == b {
count = count + 1
}
if b == c {
count = count + 1
}
if a == c {
count = count + 1
}
if a+b>=c && b+c>=a && a+c>=b && a+b+c != 0{
vertex = count
}else{
vertex = -1
}
return vertex
}
