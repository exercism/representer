(file "clock.v")
module main
struct Clock {
hour int
minute int
}
fn new_clock(hour int, minute int) Clock {
mut c := Clock{hour, minute}
c.add_time(0)
c.subtract_time(0) //normalise
return c
}
fn (mut c Clock) add_time(minute int) {
hour := (c.hour + (c.minute + minute) / 60) % 24
c = Clock{hour, (c.minute + minute) % 60}
}
fn (mut c Clock) subtract_time(minute int) {
mut hour := c.hour + (c.minute - minute - 59) / 60
for hour < 0 { hour += 24 }
c = Clock{hour, (60 + c.minute - (minute%60)) % 60}
}
fn (c Clock) string() string {
return "\"${c.hour:02}:${c.minute:02}\""
}
