module main
fn response(hey_bob string) string {
trimmed := hey_bob.trim_space()
// Silence: nothing but optional spaces
if trimmed.len == 0 {
return 'Fine. Be that way!'
}
// Check if the string has uppercase and lowercase letters
mut has_upper := false
mut has_lower := false
for c in trimmed {
if c >= `A` && c <= `Z` {
has_upper = true
}
if c >= `a` && c <= `z` {
has_lower = true
}
}
// Check if the string ends with a question mark
is_question := trimmed.ends_with('?')
is_yelling := has_upper && !has_lower
if is_question {
if is_yelling {
return 'Calm down, I know what I\'m doing!'
} else {
return 'Sure.'
}
} else if is_yelling {
return 'Whoa, chill out!'
} else {
return 'Whatever.'
}
}
