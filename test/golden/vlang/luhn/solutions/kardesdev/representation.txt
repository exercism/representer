module main
fn valid(value string) bool {
str_digits := value.split('').filter(it != ' ')
if str_digits.len <= 1 {
return false
}
if str_digits.any(!(it >= '0' && it <= '9')) {
return false
}
mut digits := str_digits.map(it.int()).reverse()
mut sum := 0
for i, digit in digits {
if i % 2 == 1 {
sum += if digit * 2 > 9 { digit * 2 - 9 } else { digit * 2 }
continue
}
sum += digit
}
return sum % 10 == 0
}
