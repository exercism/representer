const board_cap = 64
const board_sequence = []int{len: board_cap, init: index + 1}
const grains = generate_board(board_sequence)
fn grains_on_square(square int) !u64 {
if square < 1 || square > 64 {
return error('Invalid square number: $square')
}
return grains[square]
}
fn total_grains_on_board() u64 {
return grains[board_cap] * 2 - 1
}
fn generate_board(sequence []int) map[int]u64 {
mut m := map[int]u64{}
for i in sequence {
if i == 1 {
m[i] = u64(1)
} else {
m[i] = u64(m[i - 1] * 2)
}
}
return m
}
