module main
fn grains_on_square(square int) !u64 {
match true {
square <= 0, square > 64 { return error('square must be between 1 and 64') }
else { }
}
return u64(1) << (square - 1)
}
fn total_grains_on_board() u64 {
mut sum := u64(0)
for i in 1..64+1 {
sum += grains_on_square(i) or {0}
}
return sum
}
