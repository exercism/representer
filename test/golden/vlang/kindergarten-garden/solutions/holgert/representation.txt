module main
fn get_plant(c string) []string {
return match c {
'G' { ['grass'] }
'C' { ['clover'] }
'R' { ['radishes'] }
'V' { ['violets'] }
else { [''] }
}
}
fn plants(diagram string, student string) []string {
mut children := {
'Alice': 1
'Bob': 3
'Charlie': 5
'David': 7
'Eve': 9
'Fred': 11
'Ginny': 13
'Harriet': 15
'Ileana': 17
'Joseph': 19
'Kincaid': 21
'Larry': 23
}
d := diagram.split('\n')
d1 := d[0].split('')
d2 := d[1].split('')
mut result := []string{}
result << get_plant(d1[ children[student] - 1 ])
result << get_plant(d1[ children[student] ])
result << get_plant(d2[ children[student] - 1 ])
result << get_plant(d2[ children[student] ])
return result
}
