module main
const plants_map = {
`G`: 'grass'
`C`: 'clover'
`R`: 'radishes'
`V`: 'violets'
}
fn plants(diagram string, student string) []string {
d_spl := diagram.split('\n')
s_idx := (student[0] - `A`) * 2
s_plants := [d_spl[0][s_idx], d_spl[0][s_idx + 1], d_spl[1][s_idx], d_spl[1][s_idx + 1]]
return [plants_map[s_plants[0]], plants_map[s_plants[1]], plants_map[s_plants[2]], plants_map[s_plants[3]]]
}
