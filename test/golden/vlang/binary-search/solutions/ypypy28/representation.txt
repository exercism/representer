(file "binary-search.v")
module main
pub fn find(array []int, value int) !int {
mut l, mut middle, mut r := 0, 0, array.len - 1
for l <= r {
middle = l + ((r - l) >> 1)
if array[middle] == value {
return middle
} else if array[middle] < value {
l = middle + 1
} else {
r = middle - 1
}
}
return error("value not in array")
}
