(file "binary-search.v")
module main
pub fn find(array []int, value int) !int {
mut low := 0 // inclusive
mut high := array.len // exclusive
for low != high {
i := (low + high) / 2
if array[i] == value { return i }
else if array[i] < value { low = i+1 }
else { high = i }
}
return error("value not in array")
}
