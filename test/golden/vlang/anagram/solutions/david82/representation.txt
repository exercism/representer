module main
fn find_anagrams(subject string, candidates []string) []string {
mut subject_lower := subject.to_lower()
subject_sorted := sort_string(subject_lower)
mut anagrams := []string{}
for candidate in candidates {
candidate_lower := candidate.to_lower()
if candidate_lower != subject_lower && sort_string(candidate_lower) == subject_sorted {
anagrams << candidate
}
}
return anagrams
}
fn sort_string(s string) string {
mut chars := s.runes()
chars.sort()
return chars.map(it.str()).join('')
}
