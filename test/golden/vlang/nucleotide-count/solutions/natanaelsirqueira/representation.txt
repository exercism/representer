module main
fn count_nucleotides(strand string) !map[string]int {
mut map_of_counts := {
'A': 0
'C': 0
'G': 0
'T': 0
}
for nucleotide in strand.split('') {
if nucleotide !in map_of_counts.keys() {
return error('${nucleotide} is not a valid nucleotide!')
}
map_of_counts[nucleotide] = map_of_counts[nucleotide] + 1
}
return map_of_counts
}
