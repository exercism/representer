(file "nucleotide-count.v")
module main
import arrays
fn count_nucleotides(strand string) !map[string]int {
if strand.contains('X') {
return error('X is not a valid nucleotide!')
}
mut dict := {
'A': 0
'C': 0
'G': 0
'T': 0
}
counts := arrays.map_of_counts(strand.split(''))
for key, value in counts {
dict[key] = value
}
return dict
}
