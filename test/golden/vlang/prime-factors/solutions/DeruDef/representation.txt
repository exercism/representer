module main
import math {sqrti}
fn prime_factors(n i64) []i64 {
mut result := []i64{}
mut number := n
for number % 2 == 0 {
result << 2
number /= 2
}
for i := 3; i <= sqrti(number); i += 2 {
for number % i == 0 {
result << i
number /= i
}
}
if number > 2 {
result << number
}
return result
}
