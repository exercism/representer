module main
import math
fn prime_factors(n i64) []i64 {
mut factors := []i64{}
mut n_ := n
if n < 2 {
return factors
}
// special case 2 / even factors
for n_ % 2 == 0 {
factors << 2
n_ /= 2
}
// odd factors
for f := 3; f < math.sqrti(n) + 1; f += 2 {
for n_ % f == 0 {
factors << f
n_ /= f
}
}
if n_ > 2 {
factors << n_
}
return factors
}
