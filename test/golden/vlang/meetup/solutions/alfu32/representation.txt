module main
import regex
import time
const teenths=[
'sun'
'mon'
'tues'
'wednes'
'thurs'
'fri'
'satur'
]
const nums=[
'first'
'second'
'third'
'fourth'
'last'
]
pub const matcher=regex.regex_base('(The) (first|second|third|fourth|last|((sun|mon|tues|wednes|thurs|fri|satur)teenth)) of (January|February|March|April|May|June|July|August|September|October|November|December) \\d{4}')
fn get_date_from_num_day(num string,day string,month string,year string) string {
mut last:=time.Time{}
mut count_days_of_week:=0
for d in 1..time.month_days[time.long_months.index(month)]+1+if month=='February' && time.is_leap_year(year.u16()){1}else{0} {
dd:=time.Time{
year: year.u16()
month: time.long_months.index(month)+1
day: d
hour: 12
minute: 0
second: 0
microsecond: 0
}
dow:=time.day_of_week(year.u16(),time.long_months.index(month)+1,d)%7
if dow == (time.long_days.index(day)+1)%7{
if nums[count_days_of_week]==num {
return '$year/${time.long_months.index(month)+1}/${dd.day}'
}
count_days_of_week++
last=dd
println(dd)
}
}
return '$year/${time.long_months.index(month)+1}/${last.day}'
}
fn get_date_teenth(a string,month string,year string) string {
mut last:=time.Time{}
for d in 13..20 {
dd:=time.Time{
year: year.u16()
month: time.long_months.index(month)+1
day: d
hour: 12
minute: 0
second: 0
microsecond: 0
}
dow:=time.day_of_week(year.u16(),time.long_months.index(month)+1,d)%7
if dow == teenths.index(a){
return '$year/${time.long_months.index(month)+1}/${dd.day}'
}
println('$dd,$dow,$a,${teenths.index(a)}')
}
return '$year/${time.long_months.index(month)+1}/${last.day}'
}
fn date(phrase string) string {
tk:=phrase.replace('teenth',' teenth').split(' ')
a:=tk[1]
b:=tk[2]
month:=tk[4]
year:=tk[5]
println(tk)
if a in nums {
return get_date_from_num_day(a,b,month,year)
}
if a in teenths {
return get_date_teenth(a,month,year)
}
return '$year/${time.long_months.index(month)+1}/1'
}
