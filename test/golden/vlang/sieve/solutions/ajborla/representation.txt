module main
fn sieve(limit int) []int {
if limit < 2 { return [] }
if limit < 3 { return [2] }
mut primes := []int{}
mut sieve := []bool{len: limit + 1, init:true }
for p := 2; (p * p) <= limit; p++ {
if sieve[p] {
for q := (p * p); q <= limit; q += p {
sieve[q] = false
}
}
}
for p := 2; p <= limit; p++ {
if sieve[p] { primes << p }
}
return primes
}
