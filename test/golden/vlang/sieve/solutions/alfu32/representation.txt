(file "sieve.v")
module main
fn sieve(limit int) []int {
match limit {
1 { return [] }
2 { return [2] }
3 { return [2,3] }
else {
mut s0:=[]int{}
for l in 2 .. limit+1{
s0<<l
}
for divisor in 2 .. ((limit>>1)+1) {
s0=s0.filter(fn[divisor](k int)bool{return k==2 || k%divisor!=0 || k==divisor})
}
return s0
}
}
}
