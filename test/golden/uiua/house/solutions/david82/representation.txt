# Define all verse components with their actions combined
Subjects ← ⨬(
"\"\"" # Base case
| "\"house" that Jack built." # Verse 1
| "\"malt" that lay in" # Verse 2
| "\"rat" that ate" # Verse 3
| "\"cat" that killed" # Verse 4
| "\"dog" that worried" # Verse 5
| "\"cow" with the crumpled horn that tossed" # Verse 6
| "\"maiden" all forlorn that milked" # Verse 7
| "\"man" all tattered and torn that kissed" # Verse 8
| "\"priest" all shaven and shorn that married" # Verse 9
| "\"rooster" that crowed in the morn that woke" # Verse 10
| "\"farmer" sowing his corn that kept" # Verse 11
| "\"horse" and the hound and the horn that belonged to") # Verse 12
# Build a single verse
# Takes verse number, returns formatted verse string
Verse ← |1 (
+1⇡ # Generate range from 1 to verse number
≡(□Subjects) # Box each subject for the verse
⇌ # Reverse to get correct order
/$"_ the _" # Join with "\"" the "\"" between parts
□$"This is the _" # Add prefix "\"This" is the"
)
# Build multiple verses
# Takes start and end verse numbers
Recite ← |2 (
⍜-⇡ ⊙(+1) # Generate range and adjust indexing in one step
≡Verse # Build each verse
)
