abbreviate ( Sentence , Acronym ) :-
split_string (
Sentence ,
"\" -\"" ,
"\" -_\"" ,
Words
),
maplist (
take_first ,
Words ,
AcronymChars
),
string_chars (
AcronymString ,
AcronymChars
),
string_upper ( AcronymString , Acronym ).
take_first ( Word , FirstChar ) :-
string_chars ( Word , [ FirstChar | _ ]).
