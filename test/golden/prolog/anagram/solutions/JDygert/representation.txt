anagram ( Word , Options , Matching ) :-
include ( is_anagram_of ( Word ), Options , Matching ).
is_anagram_of ( Word , Option ) :-
string_lower ( Word , WordLower ),
string_lower ( Option , OptionLower ),
WordLower \ = OptionLower ,
string_chars ( WordLower , WordChars ),
string_chars ( OptionLower , OptionChars ),
msort ( WordChars , Sorted ),
msort ( OptionChars , Sorted ).
