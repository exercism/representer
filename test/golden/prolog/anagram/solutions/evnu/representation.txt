anagram ( _ , [], []).
anagram ( Word , [ Candidate | Candidates ], [ Candidate | Anagrams ]) :-
is_anagram ( Word , Candidate ),
anagram ( Word , Candidates , Anagrams ).
anagram ( Word , [ Candidate | Candidates ], Anagrams ) :-
not_is_anagram ( Word , Candidate ),
anagram ( Word , Candidates , Anagrams ).
is_anagram ( Word , Candidate ) :-
string_lower ( Word , LowerWord ),
string_lower ( Candidate , LowerCandidate ),
LowerWord \ = LowerCandidate ,
canonic_form ( Word , CanonicWord ),
canonic_form ( Candidate , CanonicWord ).
canonic_form ( Word , Canonic ) :-
string_lower ( Word , Lower ),
string_chars ( Lower , Chars ),
msort ( Chars , Canonic ).
not_is_anagram ( Word , Candidate ) :-
string_lower ( Word , LowerWord ),
string_lower ( Candidate , LowerCandidate ),
LowerWord = LowerCandidate .
not_is_anagram ( Word , Candidate ) :-
canonic_form ( Word , CanonicWord ),
canonic_form ( Candidate , CanonicCandidate ),
CanonicWord \ = CanonicCandidate .
