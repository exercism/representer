sorted_string ( X , Y ) :-
string_chars ( X , Chars ),
msort ( Chars , Sorted ),
string_chars ( Y , Sorted ).
anagram_pair ( X , Y ) :-
string_lower ( X , X1 ),
string_lower ( Y , Y1 ),
X1 \ = Y1 ,
sorted_string ( X1 , X2 ),
sorted_string ( Y1 , Y2 ),
X2 == Y2 .
anagram ( Word , Options , Anagrams ) :-
include ( anagram_pair ( Word ), Options , Anagrams ).
