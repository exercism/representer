(file "sum_of_multiples.pl")
sum_of_multiples ( _ , 1 , 0 ) :- !.
sum_of_multiples ( Factors , Limit , Sum ) :-
NewLimit is Limit - 1 ,
sum_of_multiples ( Factors , NewLimit , NewSum ),
( is_multiple ( Factors , NewLimit ) ->
Sum is NewSum + NewLimit ;
Sum is NewSum ).
is_multiple ([], _ ) :- !, fail .
is_multiple ([ H | T ], Number ) :-
0 is Number rem H , !;
is_multiple ( T , Number ).
