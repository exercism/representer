sum_of_multiples ( Factors , Limit , Sum ) :-
elab ( Factors , Limit , List ),
sort ( List , NList ),
fsum ( NList , Sum ).
get_mult ( Number , Limit , L ) :-
get_mult ( Number , Limit , 0 , L , []).
get_mult ( Number , Limit , Acc , From , From ) :-
NewAcc is Number + Acc ,
( NewAcc >= Limit ; Number >= Limit ), !.
get_mult ( Number , Limit , Acc , L , From ) :-
NewAcc is Number + Acc ,
NewAcc < Limit ,
get_mult ( Number , Limit , NewAcc , L , [ NewAcc | From ]).
elab ( Factors , Limit , List ) :-
elab ( Factors , Limit , List , []).
elab ([], _ , List , List ).
elab ([ F | Factors ], Limit , List , From ) :-
get_mult ( F , Limit , L1 ),
append ( L1 , From , NewList ),
elab ( Factors , Limit , List , NewList ).
fsum ([], 0 ).
fsum ([ H | T ], Sum ) :-
fsum ( T , Rest ),
Sum is H + Rest .
