convert ( N , Sounds ) :-
bagof ( Sound , match_sounds ( N , Sound ), SoundsList )
-> atomics_to_string ( SoundsList , Sounds )
; number_string ( N , Sounds ).
match_sounds ( N , Sound ) :-
0 is N mod 3 , Sound = "\"Pling\"" ;
0 is N mod 5 , Sound = "\"Plang\"" ;
0 is N mod 7 , Sound = "\"Plong\"" .
