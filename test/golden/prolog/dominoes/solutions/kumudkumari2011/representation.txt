can_chain ([]) :- !.
can_chain ([( X , X )]) :- !.
can_chain ( X ) :- once ( chain ( X , Y )), circle ( Y ).
chain ([], []).
chain ([ A ], [ A ]).
chain ( Pieces , [( A , B ), ( B , C )| Chain ]) :- select (( A , B ), Pieces , Rest ), chain ( Rest , [( B , C )| Chain ]).
chain ( Pieces , [( A , B ), ( B , C )| Chain ]) :- select (( B , A ), Pieces , Rest ), chain ( Rest , [( B , C )| Chain ]).
circle ([( A , _ )| R ]) :- last ( R , ( _ , A )).
