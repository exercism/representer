recite ( Day , [ Verse ]) :-
verse ( Day , Verse ).
recite ( Start , End , Verses ) :-
numlist ( Start , End , Days ),
maplist ( verse , Days , Verses ).
verse ( Day , Verse ) :-
ordinal ( Day , Ordinal ),
gifts_text ( Day , Gifts ),
atomics_to_string ([ "\"On the \"" , Ordinal , "\" day of Christmas my true love gave to me: \"" , Gifts , "\".\"" ], Verse ).
gifts_text ( Day , Text ) :-
numlist ( 1 , Day , Days ),
reverse ( Days , ReverseDays ),
maplist ( gift , ReverseDays , GiftList ),
join_gifts ( GiftList , Text ).
join_gifts ([ Single ], Single ).
join_gifts ([ Last , First ], Text ) :-
atomics_to_string ([ Last , "\", and \"" , First ], Text ).
join_gifts ([ H | T ], Text ) :-
join_gifts ( T , Rest ),
atomics_to_string ([ H , "\", \"" , Rest ], Text ).
