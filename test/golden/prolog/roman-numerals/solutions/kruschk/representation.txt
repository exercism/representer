string_codes ( RomanNumerals , Result ).

convert ( Number , PowerOfTen , Accumulator , Result ) :-
Remainder is Number mod PowerOfTen ,
NewNumber is Number - Remainder ,
NewPowerOfTen is 10 * PowerOfTen ,
translate ( Remainder , Glyphs ),
append ( Glyphs , Accumulator , NewAccumulator ),
convert ( NewNumber , NewPowerOfTen , NewAccumulator , Result ).
