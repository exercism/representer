convert ( 0 , '' ) :- !.
convert ( Arabic , Roman ) :-
roman_numeral ( Value , Symbol ),
Arabic >= Value ,
NewArabic is Arabic - Value ,
convert ( NewArabic , Rest ),
string_concat ( Symbol , Rest , Roman ).
