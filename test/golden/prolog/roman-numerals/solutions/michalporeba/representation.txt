log10 ( X , Log10X ) :-
Log10X is log ( X ) / log ( 10 ).
highest_digit_value ( Number , HighestDigit ) :-
log10 ( Number , Log10Number ),
Value is ( 10 ** floor ( Log10Number )),
HighestDigit is div ( Number , Value ) * Value .

convert ( N , Numeral ) :-
N > 0 ,
highest_digit_value ( N , HighestDigitValue ),
Reminder is N - HighestDigitValue ,
convert ( HighestDigitValue , ThisPartial ),
convert ( Reminder , OtherPartial ),
atom_concat ( ThisPartial , OtherPartial , NumeralAtom ),
atom_string ( NumeralAtom , Numeral ).
