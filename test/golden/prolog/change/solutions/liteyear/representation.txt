fewest_coins ( _ , 0 , Change ) :- Change = [], !.
fewest_coins ( Coins , Target , Change ) :-
sort ( Coins , SCoins ),
[ SmallestCoin | _ ] = SCoins ,
SmallestCoin =< Target ,
reverse ( SCoins , RCoins ),
MaxCoins is Target div SmallestCoin ,
between ( 1 , MaxCoins , MaxDepth ),
candidate ( RCoins , Target , RChange , 0 , MaxDepth ), !,
msort ( RChange , Change ).
candidate ( Coins , Target , Change , _ , _ ) :-
nth0 ( _ , Coins , Target ),
Change = [ Target ].
candidate ( Coins , Target , Change , Depth , MaxDepth ) :-
Depth < MaxDepth ,
nth0 ( 0 , Coins , BiggestCoin ),
BiggestCoin * ( MaxDepth - Depth ) >= Target ,
select ( Coin , Coins , _ ),
RTarget is Target - Coin ,
RTarget > 0 ,
NDepth is Depth + 1 ,
candidate ( Coins , RTarget , RChange , NDepth , MaxDepth ),
Change = [ Coin | RChange ].
