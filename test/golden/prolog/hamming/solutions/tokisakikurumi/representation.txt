dif_list ([], [], 0 ).
dif_list ([ X | XS ], [ Y | YS ], Dif ) :-
dif_list ( XS , YS , Sub_dif ),
(( X = Y , Dif is Sub_dif ,!);
( X \ = Y , Dif is Sub_dif + 1 )).
hamming_distance ( Str1 , Str2 , Dist ) :-
string_length ( Str1 , Len_str1 ),
string_length ( Str2 , Len_str2 ),
Len_str1 =:= Len_str2 ,
string_chars ( Str1 , List1 ),
string_chars ( Str2 , List2 ),
dif_list ( List1 , List2 , Dist ).
