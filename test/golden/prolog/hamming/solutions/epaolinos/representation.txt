hamming_distance ( Str1 , Str2 , Dist ) :-
string_to_list ( Str1 , List1 ),
string_to_list ( Str2 , List2 ),
equal_length ( List1 , List2 ),
is_valid ( List1 ), is_valid ( List2 ),
distance_count ( List1 , List2 , Dist ).
distance_count ([], [], 0 ).
distance_count ([ Head1 | Tail1 ], [ Head2 | Tail2 ], Sum ) :-
distance_count ( Tail1 , Tail2 , LastSum ),
(( Head1 =:= Head2 , Sum is LastSum ) ;
( Head1 = \ = Head2 , Sum is LastSum + 1 )).
equal_length ( List1 , List2 ) :-
length ( List1 , Length1 ),
length ( List2 , Lenth2 ),
Length1 =:= Lenth2 .
is_valid ([]).
is_valid ([ Head | Tail ]) :-
( Head =:= 65 ;
Head =:= 67 ;
Head =:= 71 ;
Head =:= 84 ),
is_valid ( Tail ).
