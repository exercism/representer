(file "hamming.pl")
hamming_distance ( Str1 , Str2 , Dist ) :-
atom_chars ( Str1 , List1 ),
atom_chars ( Str2 , List2 ),
dist ( List1 , List2 , Dist ).
dist ([], [], 0 ).
dist ([ H | T1 ], [ H | T2 ], Dist ) :- !, dist ( T1 , T2 , Dist ).
dist ([ _ | T1 ], [ _ | T2 ], Dist ) :- dist ( T1 , T2 , NewDist ), Dist is NewDist + 1 .
