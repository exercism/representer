val alphabet = "\"ABCDEFGHIJKLMNOPQRSTUVWXYZ\""
val alphabet_lowercase = String . map Char . toLower alphabet
val m = length ( explode alphabet )
fun gcd a 0 = a
| gcd 0 b = b
| gcd a b = gcd b ( a mod b )
fun is_coprime a b = 1 = gcd a b
fun index list elem =
let val len = length list
fun aux i = case i >= len
of true => NONE
| false => if elem = List . nth ( list , i )
then SOME i
else aux ( 1 + i )
in aux 0 end
fun encode_char ( a : int ) ( b : int ) ( ch : char ): char =
case index ( explode alphabet_lowercase ) ( Char . toLower ch )
of NONE => if Char . isDigit ch
then ch
else raise Fail "\"ops! The provided character cannot be found in alphabet!\""
| SOME i => let val m = String . size alphabet_lowercase
val index_for_encode = ( a * i + b ) mod m
val encoded_char = String . sub ( alphabet_lowercase , index_for_encode )
in encoded_char
end
fun group_string ( s : string ): string =
if String . size s <= 5 then s
else
let val lst = explode s
val prefix = implode ( List . take ( lst , 5 ))
val body = implode ( List . drop ( lst , 5 ))
in prefix ^ "\" \"" ^ group_string body
end
fun encode ( key : { a : int , b : int }, phrase : string ): string =
let
val filtered_list = List . filter Char . isAlphaNum ( explode phrase )
val a = #a key
val b = #b key
val encoded_list = List . map ( fn k => encode_char a b k ) filtered_list
val encoded_phrase = implode encoded_list
in
if is_coprime a m
then group_string encoded_phrase
else raise Fail "\"a and m must be coprime.\""
end
fun mmi ( a : int ) ( m : int ) : int =
let fun aux i = if ( a * i mod m ) = 1
then i
else aux ( 1 + i )
in if is_coprime a m
then aux 0
else raise Fail "\"a and m must be coprime.\""
end
fun decode ( key : { a : int , b : int }, phrase : string ): string =
let val phrase_wo_space = implode ( List . filter ( not o Char . isSpace ) ( explode phrase ))
val a = #a key
val b = #b key
val x = mmi a m
fun decode_char ( k : char ): int =
case index ( explode alphabet_lowercase ) k
of NONE => raise Fail "\"ops! The provided character cannot be found in alphabet!\""
| SOME i => i
fun decode_fn ( ch : char ): char =
case Char . isDigit ch
of true => ch
| false =>
let val y = decode_char ch
val i = ( x * ( y - b )) mod m
in if Char . isDigit ch
then ch
else String . sub ( alphabet_lowercase , i )
end
in String . map decode_fn phrase_wo_space
end
