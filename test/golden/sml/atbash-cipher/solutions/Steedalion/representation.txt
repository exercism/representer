exception notFound ;
val latin = "\"abcdefghijklmnopqrstuvwxyz\""
val plain = String . explode latin
val cipher = List . rev plain
fun notStrip c = not ( Char . isSpace c orelse Char . isPunct c )
fun stripLs ls = List . filter notStrip ls
fun get n ( x::xs ) = if n = 0 then x else get ( n- 1 ) xs
| get n [] = raise notFound
fun index n c []= raise notFound
| index n c ls = if c = ( get n ls ) then n else index ( n+ 1 ) c ls
fun convert from to c = if ( Char . isDigit c ) then c else get ( index 0 ( Char . toLower c ) from ) to
fun encodeChar c = convert plain cipher c
fun decodeChar c = convert cipher plain c
fun insertPeriodicSpace ls =
let
fun insertSpace n ( x::xs ) = if n = 0 then insertSpace 6 (( "#\" \"" ) ::x::xs ) else x:: ( insertSpace ( n- 1 ) xs )
| insertSpace n [] = []
in
insertSpace 5 ls
end
fun decode ( phrase : string ): string =
( String . implode o ( List . map encodeChar ) o stripLs o String . explode ) phrase
fun encode ( phrase : string ): string =
( String . implode o insertPeriodicSpace o ( List . map decodeChar ) o stripLs o String . explode ) phrase
