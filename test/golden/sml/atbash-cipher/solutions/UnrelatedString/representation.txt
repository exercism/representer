fun decode ( phrase : string ): string =
let fun d [] a = List . rev a
| d ( h::t ) a =
if Char . isDigit h
then d t ( h::a )
else
if Char . isAlpha h
then d t (( Char . chr (( 122 - Char . ord ( Char . toLower h )) + 97 )) ::a )
else d t a
in String . implode ( d ( String . explode phrase ) [])
end
fun encode ( phrase : string ): string =
let fun d [] ( "#\" \"" ::a ) _ = List . rev a
| d [] a _ = List . rev a
| d s a 5 = d s ( "#\" \"" ::a ) 0
| d ( h::t ) a n =
if Char . isDigit h
then d t ( h::a ) ( n+ 1 )
else
if Char . isAlpha h
then d t (( Char . chr (( 122 - Char . ord ( Char . toLower h )) + 97 )) ::a ) ( n+ 1 )
else d t a n
in String . implode ( d ( String . explode phrase ) [] 0 )
end
