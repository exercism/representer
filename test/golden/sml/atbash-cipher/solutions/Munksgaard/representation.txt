(file "atbash-cipher.sml")
fun decode ( phrase : string ): string =
implode ( map ( fn c => if Char . isAlpha c then chr ( 122 - ord c + 97 ) else c )
( List . filter Char . isAlphaNum ( explode ( String . map Char . toLower phrase ))))
fun groupByN acc n i [] = rev acc
| groupByN acc n 0 xs = groupByN ( "#\" \"" :: acc ) n n xs
| groupByN acc n i ( x :: xs ) = groupByN ( x :: acc ) n ( i - 1 ) xs
fun encode ( phrase : string ): string =
implode ( groupByN [] 5 5
( map ( fn c => if Char . isAlpha c then chr ( 122 - ord c + 97 ) else c )
( List . filter Char . isAlphaNum ( explode ( String . map Char . toLower phrase )))))
