datatype responseType =
Sure |
Whoa |
CalmDown |
Fine |
Whatever
fun trim ( message : string ): string =
let
fun index ([], depth ) = ~1
| index ( chr::rest , depth ) =
if ( Char . ord chr ) > 32 then
depth
else
index ( rest , depth + 1 )
val chars = ( explode message )
val left = index ( chars , 0 )
val right = index (( List . rev chars ), 0 )
val i = ( if left = ~1 then 0 else left )
val j = (( size message ) - ( if right = ~1 then 0 else right )) - i
in
if left = ~1 then
"\"\""
else
String . substring ( message , i , j )
end
fun isQuestion ( message : string ): bool =
let
val trimmed = ( trim message )
val i =
if ( size trimmed ) > 0
then ( size trimmed ) - 1
else 0
in
String . extract ( trimmed , i , NONE ) = "\"?\""
end
fun isAllCaps ( message : string ): bool =
let
fun isChar ( chr : char ): bool =
let
val code = ( Char . ord chr )
in
( code > 64 andalso code < 91 ) orelse
( code > 97 andalso code < 123 )
end
val a = List . filter isChar ( explode ( trim message ))
val b = List . map Char . toUpper a
in
a = b andalso ( length a ) > 0
end
fun isSilence ( message : string ): bool =
( size ( trim message )) = 0
fun getResponseType ( message : string ): responseType =
if ( isAllCaps message ) andalso ( isQuestion message ) then
CalmDown
else if ( isAllCaps message ) then
Whoa
else if ( isQuestion message ) then
Sure
else if ( isSilence message ) then
Fine
else
Whatever
fun response ( message : string ): string =
case ( getResponseType message ) of
Sure => "\"Sure.\""
| Whoa => "\"Whoa, chill out!\""
| CalmDown => "\"Calm down, I know what I'm doing!\""
| Fine => "\"Fine. Be that way!\""
| Whatever => "\"Whatever.\""
