fun onlySpaces [] = true
| onlySpaces ( c :: cs ) = Char . isSpace c andalso onlySpaces cs ;
fun isQuestion [] = false
| isQuestion ( c::cs ) = ( c = #"?" andalso onlySpaces cs ) orelse isQuestion cs ;
fun allUpper cs = List . all ( fn c => not ( Char . isAlpha c ) orelse Char . isUpper c ) cs ;
fun hasAlpha cs = List . exists Char . isAlpha cs ;
fun isYelled cs = hasAlpha cs andalso allUpper cs ;
fun response s =
let
val cs = String . explode s
in
if onlySpaces cs then "\"Fine. Be that way!\""
else if isQuestion cs then
if isYelled cs then "\"Calm down, I know what I'm doing!\""
else "\"Sure.\""
else if isYelled cs then "\"Whoa, chill out!\""
else "\"Whatever.\""
end ;
