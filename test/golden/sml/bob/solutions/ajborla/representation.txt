fun response ( s : string ) =
let
val s_as_list = List . filter ( fn c => ( not ( Char . isSpace ( c )))) ( explode ( s ))
val lc_count = List . length ( List . filter Char . isLower s_as_list )
val uc_count = List . length ( List . filter Char . isUpper s_as_list )
val digit_count = List . length ( List . filter Char . isDigit s_as_list )
val last_char = ( if List . null ( s_as_list ) then "\"\"" else str ( List . last ( s_as_list )))
val sum_counts = lc_count + uc_count + digit_count
val is_only_silence = s = "\"\""
val is_shout = uc_count > lc_count
val is_question = last_char = "\"?\""
val has_no_alphanums = sum_counts = 0
in
if is_shout then
if is_question then
"\"Calm down, I know what I'm doing!\""
else
"\"Whoa, chill out!\""
else
if is_question then
"\"Sure.\""
else
if has_no_alphanums orelse is_only_silence then
"\"Fine. Be that way!\""
else
"\"Whatever.\""
end
