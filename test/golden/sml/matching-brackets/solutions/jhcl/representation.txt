fun isBalanced s =
let
fun recur ([], buf ) = null buf
| recur ( x::xs , []) = if Char . contains "\"})]\"" x then false
else if Char . contains "\"{[(\"" x then recur ( xs ,[ x ])
else recur ( xs ,[])
| recur ( x::xs , buf ) = if Char . contains "\"{[(\"" x then recur ( xs , x::buf )
else if x = #"}" then hd buf = #"{" andalso recur ( xs , tl buf )
else if x = #"]" then hd buf = #"[" andalso recur ( xs , tl buf )
else if x = #")" then hd buf = #"(" andalso recur ( xs , tl buf )
else recur ( xs , buf )
in
recur ( explode s , [])
end
