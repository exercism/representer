(file "matching-brackets.sml")
fun isEmpty ( xs : 'a list ) =
case xs of
[] => true
| _ => false
fun isBalancedPrime ( s : char , xs : char list ) =
case ( s , xs ) of
( #"[" , _) => [ #"[" ] @ xs
| ( #"(" , _) => [ #"(" ] @ xs
| ( #"{" , _) => [ #"{" ] @ xs
| ( #"]" , #"[" ::rest ) => rest
| ( #")" , #"(" ::rest ) => rest
| ( #"}" , #"{" ::rest ) => rest
| ( #"]" , xs ) => [ #"]" ] @ xs
| ( #")" , xs ) => [ #")" ] @ xs
| ( #"}" , xs ) => [ #"}" ] @ xs
| _ => xs
fun isBalanced s =
( isEmpty o ( foldl ( isBalancedPrime ) []) o String . explode )( s )
