datatype 'a tree = Empty | Elem of 'a | List of 'a tree list
fun flatten tree =
let
fun flattenHelper Empty = []
| flattenHelper ( Elem x ) = [ x ]
| flattenHelper ( List xs ) =
List . foldr ( fn ( subtree , acc ) =>
case subtree of
Empty => acc
| _ => flattenHelper subtree @ acc
) [] xs
in
flattenHelper tree
end
