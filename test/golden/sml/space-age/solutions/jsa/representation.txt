datatype planet = Mercury | Venus | Earth | Mars
| Jupiter | Saturn | Uranus | Neptune
fun age_on planet seconds =
let
val s = real ( seconds )
val e = 365.25 * real ( 24 * 60 * 60 )
val se = s/e
in
case planet of
Mercury => se / ( 0.2408467 )
| Venus => se / ( 0.61519726 )
| Earth => se
| Mars => se / ( 1.8808158 )
| Jupiter => se / ( 11.862615 )
| Saturn => se / ( 29.447498 )
| Uranus => se / ( 84.016846 )
| Neptune => se / ( 164.79132 )
end
