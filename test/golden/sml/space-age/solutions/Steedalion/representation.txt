datatype planet = Mercury | Venus | Earth | Mars
| Jupiter | Saturn | Neptune | Uranus
fun age_on planet seconds =
let
fun earthYears seconds = Real . fromInt ( seconds ) / 31557600.0
fun relativeEarthYears planet years =
let val denom = case planet of
Earth => 1.0
| Mercury => 0.2408467
| Venus => 0.61519726
| Mars => 1.8808158
| Jupiter => 11.862615
| Saturn => 29.447498
| Uranus => 84.016846
| Neptune => 164.79132
in
years / denom
end
in
relativeEarthYears planet ( earthYears seconds )
end
