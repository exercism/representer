(file "space-age.sml")
datatype planet = Mercury | Venus | Earth | Mars
| Jupiter | Saturn | Neptune | Uranus
val seconds_per_earth_year = 31557600.0
fun age_on Mercury seconds = Real . fromInt seconds / 0.2408467 / seconds_per_earth_year
| age_on Venus seconds = Real . fromInt seconds / 0.61519726 / seconds_per_earth_year
| age_on Earth seconds = Real . fromInt seconds / seconds_per_earth_year
| age_on Mars seconds = Real . fromInt seconds / 1.8808158 / seconds_per_earth_year
| age_on Jupiter seconds = Real . fromInt seconds / 11.862615 / seconds_per_earth_year
| age_on Saturn seconds = Real . fromInt seconds / 29.447498 / seconds_per_earth_year
| age_on Uranus seconds = Real . fromInt seconds / 84.016846 / seconds_per_earth_year
| age_on Neptune seconds = Real . fromInt seconds / 164.79132 / seconds_per_earth_year
