datatype planet = Mercury | Venus | Earth | Mars
| Jupiter | Saturn | Uranus | Neptune
fun age_on planet seconds =
let
val seconds = real seconds
in
case planet of
Mercury => seconds / ( 31557600.0 * 0.2408467 )
| Venus => seconds / ( 31557600.0 * 0.61519726 )
| Earth => seconds / 31557600.0
| Mars => seconds / ( 31557600.0 * 1.8808158 )
| Jupiter => seconds / ( 31557600.0 * 11.862615 )
| Saturn => seconds / ( 31557600.0 * 29.447498 )
| Uranus => seconds / ( 31557600.0 * 84.016846 )
| Neptune => seconds / ( 31557600.0 * 164.79132 )
end
