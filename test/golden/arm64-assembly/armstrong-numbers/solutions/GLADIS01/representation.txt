// AArch64 (ARM64) System V ABI
// int is_armstrong_number(int n)
//
// Returns 1 if n is an Armstrong number, else 0.
.text
.global is_armstrong_number
// x0 = n (input)
// return in w0 (int)
is_armstrong_number:
mov x19, x0 // save n
// prepare divisor 10 in x22
mov x22, #10
// count digits
mov x1, x19 // temp = n
mov x2, #0 // digit count
count_loop:
cbz x1, count_done
udiv x3, x1, x22 // x3 = x1 / 10
msub x4, x3, x22, x1 // remainder = x1 - x3*10
mov x1, x3
add x2, x2, #1
b count_loop
count_done:
mov x20, x2 // x20 = number of digits
// compute sum of digits^count
mov x1, x19 // temp = n
mov x21, #0 // sum
sum_loop:
cbz x1, sum_done
udiv x3, x1, x22
msub x4, x3, x22, x1 // digit
mov x1, x3
// pow(digit, count)
mov x5, #1
mov x6, x20
pow_loop:
cbz x6, pow_done
mul x5, x5, x4
sub x6, x6, #1
b pow_loop
pow_done:
add x21, x21, x5
b sum_loop
sum_done:
cmp x21, x19
cset w0, eq
ret
