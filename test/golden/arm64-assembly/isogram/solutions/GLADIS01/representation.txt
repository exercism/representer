// AArch64 (ARM64) System V ABI
// int is_isogram(const char *text)
//
// Returns 1 if text is an isogram, else 0.
// Rules: ignore spaces and hyphens, case-insensitive.
.text
.global is_isogram
// x0 = text (const char*)
// return in w0 (int)
is_isogram:
// zero out seen[26]
adrp x1, seen
add x1, x1, :lo12:seen
mov x2, #26
mov x3, #0
clear_loop:
cbz x2, clear_done
strb w3, [x1], #1
sub x2, x2, #1
b clear_loop
clear_done:
// iterate over string
mov x4, x0 // pointer to text
next_char:
ldrb w5, [x4]
cbz w5, done // end of string
add x4, x4, #1
// ignore space (0x20) and hyphen (0x2D)
cmp w5, #' '
beq next_char
cmp w5, #'-'
beq next_char
// convert to lowercase if uppercase
cmp w5, #'A'
blt check_alpha
cmp w5, #'Z'
bgt check_alpha
add w5, w5, #32 // tolower
check_alpha:
// check if 'a' <= w5 <= 'z'
cmp w5, #'a'
blt next_char
cmp w5, #'z'
bgt next_char
// index = w5 - 'a'
sub w6, w5, #'a'
// load seen[index]
adrp x1, seen
add x1, x1, :lo12:seen
add x1, x1, x6
ldrb w7, [x1]
cbnz w7, not_isogram // already seen
// mark seen
mov w7, #1
strb w7, [x1]
b next_char
not_isogram:
mov w0, #0
ret
done:
mov w0, #1
ret
.bss
.balign 16
seen:
.space 26
