.text
.globl value
// int value(const char **colors);
// colors[0] = first band, colors[1] = second band.
// Returns 10 * first_digit + second_digit.
value:
stp x19, x30, [sp, #-16]! // save x19 (callee-saved) and x30
mov x19, x0 // x19 = colors array (preserved across bl)
ldr x0, [x19] // x0 = colors[0]
bl color_to_digit // w0 = first digit
mov w20, w0 // w20 = first digit (callee-saved)
ldr x0, [x19, #8] // x0 = colors[1] (reload from saved x19)
bl color_to_digit // w0 = second digit
add w0, w0, w20, lsl #3 // w0 = w0 + w20*8
add w0, w0, w20, lsl #1 // w0 = w0 + w20*2 => 10*first + second
ldp x19, x30, [sp], #16
ret
// color_to_digit: x0 -> color name. Returns 0..9 in w0.
// Leaf function: no bl, no frame needed.
// First letter narrows to <= 3 candidates; one or two more byte loads
// resolve the rest.
color_to_digit:
ldrb w1, [x0]
cmp w1, #'b'
b.eq L_b
cmp w1, #'r'
b.eq L_red
cmp w1, #'o'
b.eq L_orange
cmp w1, #'y'
b.eq L_yellow
cmp w1, #'g'
b.eq L_g
cmp w1, #'v'
b.eq L_violet
cmp w1, #'w'
b.eq L_white
mov w0, #0 // unknown -> black
ret
L_b: // black, blue, brown
ldrb w2, [x0, #1]
cmp w2, #'l'
b.ne L_brown // "\"brown\"" : b,r
ldrb w2, [x0, #2]
cmp w2, #'a'
b.eq L_black // "\"black\"" : b,l,a
mov w0, #6 // "\"blue\"" : b,l,u
ret
L_brown:
mov w0, #1
ret
L_black:
mov w0, #0
ret
L_red:
mov w0, #2
ret
L_orange:
mov w0, #3
ret
L_yellow:
mov w0, #4
ret
L_g: // green, grey share 'g','r','e' -> check 4th char
ldrb w2, [x0, #3]
cmp w2, #'y'
b.eq L_grey
mov w0, #5 // "\"green\""
ret
L_grey:
mov w0, #8
ret
L_violet:
mov w0, #7
ret
L_white:
mov w0, #9
ret
