(file "reverse_string.s")
.text
.globl reverse
reverse:
mov x1, x0
mov x2, x0
find_end:
ldrb w3, [x2]
cbz w3, reverse_loop_setup
add x2, x2, 1
b find_end
reverse_loop_setup:
sub x2, x2, 1
reverse_loop:
cmp x1, x2
bge reverse_done
ldrb w3, [x1]
ldrb w4, [x2]
strb w3, [x2]
strb w4, [x1]
add x1, x1, 1
sub x2, x2, 1
b reverse_loop
reverse_done:
ret
