(file "micro_blog.s")
.text
.globl truncate
// void truncate(char *buffer, const char *phrase)
// x0 = buffer (destination)
// x1 = phrase (source)
//
// UTF-8 codepoint byte lengths are determined from the leading byte:
// 0xxxxxxx (0x00-0x7F) -> 1 byte (ASCII)
// 110xxxxx (0xC0-0xDF) -> 2 bytes
// 1110xxxx (0xE0-0xEF) -> 3 bytes
// 11110xxx (0xF0-0xF7) -> 4 bytes
//
// Algorithm: walk the source, determine each codepoint's length from its
// leading byte, copy that many bytes to the buffer, and count codepoints.
// Stop after 5 codepoints (or when the null terminator is reached).
truncate:
mov x2, x0 // x2 = buffer (destination)
mov x3, x1 // x3 = phrase (source)
mov w4, #0 // w4 = codepoint counter
.Lcopy_loop:
ldrb w5, [x3] // load current byte from source
cbz w5, .Ldone // null terminator -> copy it and return
// Already have 5 codepoints? Truncate.
cmp w4, #5
b.ge .Lnull_terminate
// Determine byte length from leading byte
cmp w5, #0x80
b.lo .Lone_byte
cmp w5, #0xE0
b.lo .Ltwo_bytes
cmp w5, #0xF0
b.lo .Lthree_bytes
b .Lfour_bytes
.Lone_byte:
mov w6, #1
b .Lcopy_codepoint
.Ltwo_bytes:
mov w6, #2
b .Lcopy_codepoint
.Lthree_bytes:
mov w6, #3
b .Lcopy_codepoint
.Lfour_bytes:
mov w6, #4
.Lcopy_codepoint:
// Copy w6 bytes from source (x3) to destination (x2)
cbz w6, .Ladvance
.Lcopy_bytes:
ldrb w7, [x3], #1 // load byte, advance source
strb w7, [x2], #1 // store byte, advance dest
sub w6, w6, #1
cbnz w6, .Lcopy_bytes
.Ladvance:
add w4, w4, #1 // count one more codepoint
b .Lcopy_loop
.Lnull_terminate:
strb wzr, [x2] // write null terminator
ret
.Ldone:
strb wzr, [x2] // copy the null terminator
ret
