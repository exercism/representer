(file "phone_number.s")
.text
.globl clean, _clean
clean:
_clean:
mov x10, x0 // x10 = *str
mov x11, x0 // x11 = working reg
mov x5, xzr // x5 = length of clean phone number (= 10)
sub sp, sp, #16
mov x12, sp // x12 = char *array[16]
mov x13, x12 // x13 = working reg
ldrb w2, [x11] // w2 = current character
cmp x2, #'+'
bne .skip_entry
add X11, x11, #1
ldrb w2, [x11]
.skip_entry:
cmp x2, #'1'
bne .no_country_code
add x11, x11, #1
.clean_loop:
ldrb w2, [x11]
.no_country_code:
cmp w2, #0
beq .end_loop
cmp w2, #'('
bne .1$
add x11, x11, #1
b .clean_loop
.1$:
cmp w2, #')'
bne .2$
add x11, x11, #1
b .clean_loop
.2$:
cmp w2, #'.'
bne .3$
add x11, x11, #1
b .clean_loop
.3$:
cmp w2, #'-'
bne .4$
add x11, x11, #1
b .clean_loop
.4$:
cmp w2, #' '
bne .valid_number
add x11, x11, #1
b .clean_loop
.valid_number:
cmp w2, #'9'
bhi .end_clean
add x5, x5, #1
add x11, x11, #1
strb w2, [x13]
add x13, x13, #1
b .clean_loop
.end_loop:
strb wzr, [x13]
cmp x5, #11
bhs .end_clean
cmp x5, #9
blo .end_clean
ldrb w2, [x12]
cmp w2, #'0'
bne .correct_area_code2
b .end_clean
.correct_area_code2:
cmp w2, #'1'
bne .correct_local_number1
b .end_clean
.correct_local_number1:
ldrb w2, [x12, #3]
cmp w2, #'0'
bne .correct_local_number2
b .end_clean
.correct_local_number2:
cmp w2, #'1'
beq .end_clean
mov x3, xzr
.tmp_to_str:
ldrb w2, [x12]
strb w2, [x0]
add x0, x0, #1
add x3, x3, #1
add x12, x12, #1
cmp x3, x5
blo .tmp_to_str
.end_clean:
strb wzr, [x0]
add sp, sp, #16
ret
