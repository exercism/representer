(file "collatz_conjecture.s")
.equ INVALID_NUMBER, -1
.text
.globl steps
steps:
cmp x0, #0
b.le invalid
mov x1, #0
collatz_loop:
cmp x0, #1
b.eq done
and x2, x0, #1
cmp x2, #0
b.eq handle_even
handle_odd:
add x0, x0, x0, lsl #1
add x0, x0, #1
b increment_counter
handle_even:
lsr x0, x0, #1
increment_counter:
add x1, x1, #1
b collatz_loop
invalid:
mov x0, INVALID_NUMBER
ret
done:
mov x0, x1
ret
