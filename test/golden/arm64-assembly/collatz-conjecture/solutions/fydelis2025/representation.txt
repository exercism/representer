.equ INVALID_NUMBER, -1
.text
.global steps
// Entrada: x0 = número inteiro positivo
// Saída: x0 = quantidade de passos até chegar a 1, ou -1 se inválido
steps:
cmp x0, 1
b.lt numero_invalido
b.eq retorna_zero
mov x1, 0
loop:
tst x0, 1
b.ne caso_impar
caso_par:
lsr x0, x0, 1
b conta_passo
caso_impar:
mov x2, 3
mul x0, x0, x2
add x0, x0, 1
conta_passo:
add x1, x1, 1
cmp x0, 1
b.ne loop
mov x0, x1
ret
retorna_zero:
mov x0, 0
ret
numero_invalido:
mov x0, INVALID_NUMBER
ret
