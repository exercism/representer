.equ INVALID_NUMBER, -1
.text
.globl steps
steps:
// Is the starting number valid?
cmp x0, #0
ble invalid
mov x1, #0 // step_count = zero
loop:
// Have we reached one?
cmp x0, #1
beq done
// Increment step count.
add x1, x1, #1
// Check if even: check LSB.
and x2, x0, #1
cbz x2, is_even
// Otherwise, odd: multiply by three, add 1, and continue.
add x0, x0, x0, lsl #1
add x0, x0, #1
b loop
is_even:
// Divide by 2 and continue. Even, so just shift right.
lsr x0, x0, #1
b loop
done:
mov x0, x1 // store result in return register
ret
invalid:
mov x0, #INVALID_NUMBER
ret
