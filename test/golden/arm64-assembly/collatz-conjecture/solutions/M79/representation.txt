.equ INVALID_NUMBER, -1
.text
.globl steps
steps:
// Verifica se x0 < 1
cmp x0, 1
blt .invalid
mov x1, x0 // x1 = n (número atual)
mov x2, 0 // x2 = contador de passos
.loop:
cmp x1, 1
beq .done // Se n == 1, fim do loop
// Testa se número é par: n % 2 == 0
and x3, x1, 1
cbz x3, .even // Se bit mais baixo for 0, é par
.odd:
// n = 3 * n + 1
mov x4, x1
lsl x4, x4, 1 // x4 = n * 2
add x1, x1, x4 // x1 = n + 2n = 3n
add x1, x1, 1 // x1 = 3n + 1
b .incr
.even:
// n = n / 2
lsr x1, x1, 1
.incr:
add x2, x2, 1 // passos += 1
b .loop
.done:
mov x0, x2 // retorna passos em x0
ret
.invalid:
mov x0, INVALID_NUMBER
ret
