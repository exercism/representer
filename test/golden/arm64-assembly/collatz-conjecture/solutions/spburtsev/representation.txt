.globl steps
.equ INVALID_NUMBER, -1
.text
steps:
cmp x0, #0
ble .steps_exit_invalid
mov x1, x0
mov x0, #0
.step:
cmp x1, #1
beq .steps_exit
tst x1, #1
beq .was_even
b .was_odd
.steps_exit:
ret
.was_even:
add x0, x0, #1
mov x3, #2
udiv x1, x1, x3
b .step
.was_odd:
add x0, x0, #1
mov x4, #3
mul x1, x1, x4
add x1, x1, #1
b .step
.steps_exit_invalid:
mov x0, INVALID_NUMBER
b .steps_exit
