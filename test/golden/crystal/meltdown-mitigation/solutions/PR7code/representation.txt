(class_def Reactor
  (expressions
    (method_def self reactor_efficiency
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2)
        (param PLACEHOLDER_3))
      (expressions
        (assign PLACEHOLDER_4 =
          (call PLACEHOLDER_1 *
            (argument_list PLACEHOLDER_2)))
        (assign PLACEHOLDER_5 =
          (call
            (call PLACEHOLDER_4 /
              (argument_list PLACEHOLDER_3)) *
            (argument_list 100)))
        (if
          (call PLACEHOLDER_5 <
            (argument_list 30))
          (then "black")
          (elsif
            (call PLACEHOLDER_5 <
              (argument_list 60))
            (then "red")
            (elsif
              (call PLACEHOLDER_5 <
                (argument_list 80))
              (then "orange")
              (else
                (expressions "green")))))))
    (method_def self fail_safe
      (param_list
        (param PLACEHOLDER_6)
        (param PLACEHOLDER_7)
        (param PLACEHOLDER_8))
      (expressions
        (assign PLACEHOLDER_9 =
          (call PLACEHOLDER_6 *
            (argument_list PLACEHOLDER_7)))
        (if
          (call PLACEHOLDER_9 <
            (argument_list
              (call PLACEHOLDER_8 *
                (argument_list 0.9))))
          (then "LOW")
          (elsif
            (call PLACEHOLDER_9 <
              (argument_list
                (call PLACEHOLDER_8 *
                  (argument_list 1.1))))
            (then "NORMAL")
            (else
              (expressions "DANGER"))))))
    (method_def self criticality_balanced?
      (param_list
        (param PLACEHOLDER_6)
        (param PLACEHOLDER_10))
      (expressions
        (and
          (and
            (call PLACEHOLDER_6 <
              (argument_list 800)) &&
            (call PLACEHOLDER_10 >
              (argument_list 500))) &&
          (call
            (call PLACEHOLDER_6 *
              (argument_list PLACEHOLDER_10)) <
            (argument_list 500000)))))))
