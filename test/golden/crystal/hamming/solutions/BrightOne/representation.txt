(module_def Hamming
  (expressions
    (extend self)
    (method_def distance
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2))
      (expressions
        (if
          (call
            (call PLACEHOLDER_1 size) !=
            (argument_list
              (call PLACEHOLDER_2 size)))
          (then
            (call raise
              (argument_list
                (call ArgumentError new
                  (argument_list "strands are of different lengths"))))))
        (call
          (call
            (call PLACEHOLDER_1 chars) zip
            (argument_list
              (call PLACEHOLDER_2 chars))) count
          (block
            (param_list
              (param PLACEHOLDER_3)
              (param PLACEHOLDER_4))
            (expressions
              (call PLACEHOLDER_3 !=
                (argument_list PLACEHOLDER_4)))))))))
