(module_def Blackjack
  (expressions
    (method_def self first_turn
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2)
        (param PLACEHOLDER_3))
      (expressions
        (case
          (call self card_range
            (argument_list PLACEHOLDER_1 PLACEHOLDER_2))
          (when "blackjack"
            (expressions
              (case PLACEHOLDER_3
                (when "ace" "ten" "jack" "queen" "king"
                  (expressions "S"))
                (else
                  (expressions "W")))))
          (when "doubleace"
            (expressions "P"))
          (when "high"
            (expressions "S"))
          (when "mid"
            (expressions
              (unless
                (call
                  (call self parse_card
                    (argument_list PLACEHOLDER_3)) >=
                  (argument_list 7))
                (then
                  (return
                    (argument_list "S")))) "H"))
          (when "low"
            (expressions "H")))))
    (method_def self card_range
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2))
      (expressions
        (assign PLACEHOLDER_4 =
          (call
            (call self parse_card
              (argument_list PLACEHOLDER_1)) +
            (argument_list
              (call self parse_card
                (argument_list PLACEHOLDER_2)))))
        (case PLACEHOLDER_4
          (when
            (range 4 .. 11)
            (expressions "low"))
          (when
            (range 12 .. 16)
            (expressions "mid"))
          (when
            (range 17 .. 20)
            (expressions "high"))
          (when 21
            (expressions "blackjack"))
          (else
            (expressions "doubleace")))))
    (method_def self parse_card
      (param_list
        (param PLACEHOLDER_5))
      (expressions
        (case PLACEHOLDER_5
          (when "ace"
            (expressions 11))
          (when "two"
            (expressions 2))
          (when "three"
            (expressions 3))
          (when "four"
            (expressions 4))
          (when "five"
            (expressions 5))
          (when "six"
            (expressions 6))
          (when "seven"
            (expressions 7))
          (when "eight"
            (expressions 8))
          (when "nine"
            (expressions 9))
          (when "ten" "jack" "queen" "king"
            (expressions 10))
          (else
            (expressions 0)))))))
