(module_def Proverb
  (expressions
    (extend self)
    (method_def recite
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (assign PLACEHOLDER_2 = 1)
        (assign PLACEHOLDER_3 =
          (array of String))
        (if
          (call
            (call PLACEHOLDER_1 size) >
            (argument_list 0))
          (then
            (if
              (call
                (call PLACEHOLDER_1 size) >
                (argument_list 1))
              (then
                (while
                  (call PLACEHOLDER_2 <
                    (argument_list
                      (call PLACEHOLDER_1 size)))
                  (expressions
                    (call PLACEHOLDER_3 <<
                      (argument_list
                        (string "For want of a "
                          (interpolation #{
                            (index_call PLACEHOLDER_1 [
                              (argument_list
                                (call PLACEHOLDER_2 -
                                  (argument_list 1))))) " the "
                          (interpolation #{
                            (index_call PLACEHOLDER_1 [
                              (argument_list PLACEHOLDER_2))) " was lost.")))
                    (op_assign PLACEHOLDER_2 += 1)))))
            (call PLACEHOLDER_3 <<
              (argument_list
                (string "And all for the want of a "
                  (interpolation #{
                    (index_call PLACEHOLDER_1 [
                      (argument_list 0))) ".")))))
        (return
          (argument_list PLACEHOLDER_3))))))
