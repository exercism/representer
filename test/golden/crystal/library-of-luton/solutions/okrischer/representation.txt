(class_def Library
  (expressions
    (method_def self decrypt_character
      (param_list
        (param PLACEHOLDER_1 Char)) Char
      (expressions
        (case PLACEHOLDER_1
          (when
            (char ' A ')
            (expressions
              (char ' Z ')))
          (when
            (char ' a ')
            (expressions
              (char ' z ')))
          (when
            (implicit_object_call letter?)
            (expressions
              (call PLACEHOLDER_1 pred)))
          (else
            (expressions PLACEHOLDER_1)))))
    (method_def self initials
      (param_list
        (param PLACEHOLDER_2 String)
        (param PLACEHOLDER_3 String)) String
      (expressions
        (string
          (interpolation #{
            (index_call PLACEHOLDER_2 [
              (argument_list 0))) "."
          (interpolation #{
            (index_call PLACEHOLDER_3 [
              (argument_list 0))))))
    (method_def self decrypt_text
      (param_list
        (param PLACEHOLDER_4 String)) String
      (expressions
        (call
          (call
            (call PLACEHOLDER_4 chars) map
            (block
              (param_list
                (param PLACEHOLDER_5))
              (expressions
                (call decrypt_character
                  (argument_list PLACEHOLDER_5))))) join)))
    (method_def self first_letter
      (param_list
        (param PLACEHOLDER_6 String)) Char
      (expressions
        (index_call PLACEHOLDER_6 [
          (argument_list 0))))))
