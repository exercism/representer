(file "src/library_of_luton.cr"
  (class_def Library
    (expressions
      (method_def self decrypt_character
        (param_list
          (param PLACEHOLDER_1 Char)) Char
        (expressions
          (case PLACEHOLDER_1
            (when
              (range
                (char ' a ') ..
                (char ' z '))
              (expressions
                (call
                  (char ' a ') +
                  (argument_list
                    (call
                      (call
                        (call PLACEHOLDER_1 -
                          (argument_list
                            (char ' a '))) +
                        (argument_list 25)) %
                      (argument_list 26))))))
            (when
              (range
                (char ' A ') ..
                (char ' Z '))
              (expressions
                (call
                  (char ' A ') +
                  (argument_list
                    (call
                      (call
                        (call PLACEHOLDER_1 -
                          (argument_list
                            (char ' A '))) +
                        (argument_list 25)) %
                      (argument_list 26))))))
            (else
              (expressions PLACEHOLDER_1)))))
      (method_def self initials
        (param_list
          (param PLACEHOLDER_2 String)
          (param PLACEHOLDER_3 String)) String
        (expressions
          (string
            (interpolation #{
              (index_call PLACEHOLDER_2 [
                (argument_list 0))) "."
            (interpolation #{
              (index_call PLACEHOLDER_3 [
                (argument_list 0))))))
      (method_def self decrypt_text
        (param_list
          (param PLACEHOLDER_4 String)) String
        (expressions
          (call
            (call
              (call PLACEHOLDER_4 chars) map
              (argument_list
                (block_argument &
                  (method_proc -> decrypt_character
                    (param_list Char))))) join)))
      (method_def self first_letter
        (param_list
          (param PLACEHOLDER_5 String)) Char
        (expressions
          (index_call PLACEHOLDER_5 [
            (argument_list 0)))))))
