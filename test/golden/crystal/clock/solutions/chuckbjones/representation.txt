(class_def Clock
  (expressions
    (call getter
      (argument_list hour minute))
    (method_def initialize
      (param_list
        (param (instance_var PLACEHOLDER_1) = 0)
        (param (instance_var PLACEHOLDER_2) = 0))
      (expressions
        (assign PLACEHOLDER_3 PLACEHOLDER_4 =
          (call (instance_var PLACEHOLDER_2) divmod
            (argument_list 60)))
        (assign (instance_var PLACEHOLDER_2) = PLACEHOLDER_4)
        (op_assign (instance_var PLACEHOLDER_1) += PLACEHOLDER_3)
        (op_assign (instance_var PLACEHOLDER_1) %= 24)))
    (method_def +
      (param_list
        (param PLACEHOLDER_5))
      (expressions
        (call Clock new
          (argument_list
            (call hour +
              (argument_list
                (call PLACEHOLDER_5 hour)))
            (call minute +
              (argument_list
                (call PLACEHOLDER_5 minute)))))))
    (method_def -
      (param_list
        (param PLACEHOLDER_5))
      (expressions
        (call Clock new
          (argument_list
            (call hour -
              (argument_list
                (call PLACEHOLDER_5 hour)))
            (call minute -
              (argument_list
                (call PLACEHOLDER_5 minute)))))))
    (method_def ==
      (param_list
        (param PLACEHOLDER_5))
      (expressions
        (and
          (call hour ==
            (argument_list
              (call PLACEHOLDER_5 hour))) &&
          (call minute ==
            (argument_list
              (call PLACEHOLDER_5 minute))))))
    (method_def to_s
      (expressions
        (call sprintf
          (argument_list "%02d:%02d" (instance_var PLACEHOLDER_1) (instance_var PLACEHOLDER_2)))))))
