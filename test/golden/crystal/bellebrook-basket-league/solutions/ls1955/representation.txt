  (expressions
    (module_def TicketingReservation
      (expressions
        (call getter
          (argument_list
            (symbol tickets_available)))
        (method_def PLACEHOLDER_1
          (param_list
            (param PLACEHOLDER_2))
          (expressions
            (assign PLACEHOLDER_3 = tickets_available)
            (if PLACEHOLDER_4
              (then
                (string
                  (interpolation #{ PLACEHOLDER_2) ", your purchase was successful, your ticket number is #"
                  (interpolation #{ PLACEHOLDER_3) ", and the game is played at the "
                  (interpolation #{ stadium) " stadium."))
              (else
                (expressions
                  (string
                    (interpolation #{ PLACEHOLDER_2) ", your purchase was unsuccessful, there are not enough tickets available."))))))
        (method_def PLACEHOLDER_4
          (expressions
            (if
              (call tickets_available <
                (argument_list 100))
              (then
                (return
                  (argument_list false))))
            (op_assign @tickets_available -= 1) true))))
    (class_def TicketSystem
      (expressions
