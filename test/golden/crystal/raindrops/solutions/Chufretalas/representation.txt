(file "src/raindrops.cr"
  (class_def Raindrops
    (expressions
      (method_def self convert
        (param_list
          (param PLACEHOLDER_1 Int32)) String
        (expressions
          (assign PLACEHOLDER_2 = "")
          (if
            (call PLACEHOLDER_1 divisible_by?
              (argument_list 3))
            (then
              (op_assign PLACEHOLDER_2 += "Pling")))
          (if
            (call PLACEHOLDER_1 divisible_by?
              (argument_list 5))
            (then
              (op_assign PLACEHOLDER_2 += "Plang")))
          (if
            (call PLACEHOLDER_1 divisible_by?
              (argument_list 7))
            (then
              (op_assign PLACEHOLDER_2 += "Plong")))
          (conditional
            (call PLACEHOLDER_2 blank?) ?
            (call PLACEHOLDER_1 to_s) PLACEHOLDER_2))))))
