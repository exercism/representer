(class_def Raindrops
  (expressions
    (method_def self convert
      (param_list
        (param PLACEHOLDER_1 Int32)) String
      (expressions
        (assign PLACEHOLDER_2 = "")
        (if
          (call
            (call PLACEHOLDER_1 %
              (argument_list 3)) ==
            (argument_list 0))
          (then
            (assign PLACEHOLDER_2 =
              (call PLACEHOLDER_2 +
                (argument_list "Pling")))))
        (if
          (call
            (call PLACEHOLDER_1 %
              (argument_list 5)) ==
            (argument_list 0))
          (then
            (assign PLACEHOLDER_2 =
              (call PLACEHOLDER_2 +
                (argument_list "Plang")))))
        (if
          (call
            (call PLACEHOLDER_1 %
              (argument_list 7)) ==
            (argument_list 0))
          (then
            (assign PLACEHOLDER_2 =
              (call PLACEHOLDER_2 +
                (argument_list "Plong")))))
        (conditional
          (call PLACEHOLDER_2 empty?) ?
          (call PLACEHOLDER_1 to_s) PLACEHOLDER_2)))))
