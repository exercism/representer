(file "src/bank_account.cr"
  (class_def BankAccount
    (expressions
      (method_def withdraw
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (unless (instance_var PLACEHOLDER_2)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Please open the account beforehand"))))))
          (if
            (call PLACEHOLDER_1 negative?)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Withdraw amount could not be negative"))))))
          (if
            (call balance <
              (argument_list PLACEHOLDER_1))
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Balance is smaller than withdraw amount"))))))
          (op_assign (instance_var PLACEHOLDER_3) -= PLACEHOLDER_1)))
      (method_def deposit
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (unless (instance_var PLACEHOLDER_2)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Please open the account beforehand"))))))
          (if
            (call PLACEHOLDER_1 negative?)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Deposit amount could not be negative"))))))
          (op_assign (instance_var PLACEHOLDER_3) += PLACEHOLDER_1)))
      (method_def open
        (expressions
          (if (instance_var PLACEHOLDER_2)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Account is already open"))))))
          (assign (instance_var PLACEHOLDER_3) = 0)
          (assign (instance_var PLACEHOLDER_2) = true)))
      (method_def close
        (expressions
          (unless (instance_var PLACEHOLDER_2)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "The account is already close"))))))
          (assign (instance_var PLACEHOLDER_2) = false)))
      (method_def balance
        (expressions
          (unless (instance_var PLACEHOLDER_2)
            (then
              (call raise
                (argument_list
                  (call ArgumentError new
                    (argument_list "Please open the account beforehand")))))) (instance_var PLACEHOLDER_3)))
      (method_def initialize
        (expressions
          (assign (instance_var PLACEHOLDER_3) = 0)
          (assign (instance_var PLACEHOLDER_2) = false))))))
