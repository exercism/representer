(class_def WeighingMachine
  (expressions
    (type_declaration @weight Float64)
    (type_declaration @precision Int32)
    (type_declaration @metric Bool)
    (call property
      (argument_list weight))
    (call getter
      (argument_list precision))
    (call setter
      (argument_list metric))
    (method_def weigh String
      (expressions
        (assign weight =
          (conditional @metric ? @weight
            (call @weight *
              (argument_list 2.20462))))
        (assign weight =
          (call weight round
            (argument_list @precision)))
        (call weight to_s)))
    (method_def initialize
      (param_list
        (param precision Int32)
        (param metric Bool))
      (expressions
        (assign @weight = 0)
        (assign @precision = precision)
        (assign @metric = metric)))))
