(file "src/weighing_machine.cr"
  (class_def WeighingMachine
    (expressions
      (method_def weigh String
        (expressions
          (assign weight =
            (conditional @metric ? @weight
              (call @weight *
                (argument_list 2.20462))))
          (assign weight =
            (call weight round
              (argument_list @precision)))
          (call weight to_s)))
      (call getter
        (argument_list precision))
      (call setter
        (argument_list metric))
      (call property
        (argument_list weight))
      (method_def initialize
        (param_list
          (param precision Int32)
          (param metric Bool))
        (expressions
          (assign @precision = precision)
          (assign @metric = metric)
          (assign @weight = 0.0))))))
