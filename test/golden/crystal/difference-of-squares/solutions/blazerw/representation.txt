(module_def Squares
  (expressions
    (extend self)
    (method_def sum_of_squares
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (range 1 .. PLACEHOLDER_1) sum
          (argument_list
            (block_argument &
              (implicit_object_call .**
                (argument_list 2)))))))
    (method_def difference_of_squares
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call square_of_sum
            (argument_list PLACEHOLDER_1)) -
          (argument_list
            (call sum_of_squares
              (argument_list PLACEHOLDER_1))))))
    (method_def square_of_sum
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call
            (range 1 .. PLACEHOLDER_1) sum) **
          (argument_list 2))))))
