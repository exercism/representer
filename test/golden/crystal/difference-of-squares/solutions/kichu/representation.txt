(module_def Squares
  (expressions
    (extend self)
    (method_def sum_of_squares
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call
            (call PLACEHOLDER_1 *
              (argument_list
                (call PLACEHOLDER_1 +
                  (argument_list 1)))) *
            (argument_list
              (call
                (call 2 *
                  (argument_list PLACEHOLDER_1)) +
                (argument_list 1)))) /
          (argument_list 6))))
    (method_def square_of_sum
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call
            (call PLACEHOLDER_1 *
              (argument_list
                (call PLACEHOLDER_1 +
                  (argument_list 1)))) /
            (argument_list 2)) **
          (argument_list 2))))
    (method_def difference_of_squares
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call square_of_sum
            (argument_list PLACEHOLDER_1)) -
          (argument_list
            (call sum_of_squares
              (argument_list PLACEHOLDER_1))))))))
