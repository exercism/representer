(module_def Chess
  (expressions
    (const_assign RANKS =
      (range 1 .. 8))
    (const_assign FILES =
      (range
        (char ' A ') ..
        (char ' H ')))
    (method_def self move_message
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2)
        (param PLACEHOLDER_3))
      (expressions
        (assign nickname =
          (call nickname
            (argument_list PLACEHOLDER_1 PLACEHOLDER_2)))
        (assign PLACEHOLDER_4 =
          (string
            (interpolation #{ nickname) " moved to "
            (interpolation #{ PLACEHOLDER_3)))
        (assign PLACEHOLDER_5 =
          (string
            (interpolation #{ nickname) " attempted to move to "
            (interpolation #{ PLACEHOLDER_3) ", but that is not a valid square"))
        (conditional
          (call valid_square?
            (argument_list
              (call
                (index_call PLACEHOLDER_3 [
                  (argument_list 1)) to_i)
              (index_call PLACEHOLDER_3 [
                (argument_list 0)))) ? PLACEHOLDER_4 PLACEHOLDER_5)))
    (method_def self nickname
      (param_list
        (param PLACEHOLDER_1)
        (param PLACEHOLDER_2))
      (expressions
        (call
          (call
            (index_call PLACEHOLDER_1 [
              (argument_list
                (range 0 .. 1))) +
            (argument_list
              (index_call PLACEHOLDER_2 [
                (argument_list
                  (range
                    (integer -) ..
                    (integer -)))))) upcase)))
    (method_def self valid_square?
      (param_list
        (param PLACEHOLDER_6)
        (param PLACEHOLDER_7))
      (expressions
        (and
          (call RANKS includes?
            (argument_list PLACEHOLDER_6)) &&
          (call FILES includes?
            (argument_list PLACEHOLDER_7)))))))
