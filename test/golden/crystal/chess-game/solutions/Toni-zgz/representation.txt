(file "src/chess_game.cr"
  (module_def Chess
    (expressions
      (const_assign RANKS =
        (range 1 .. 8))
      (const_assign FILES =
        (range
          (char ' A ') ..
          (char ' H ')))
      (method_def self move_message
        (param_list
          (param PLACEHOLDER_1)
          (param PLACEHOLDER_2)
          (param PLACEHOLDER_3))
        (expressions
          (if
            (call valid_square?
              (argument_list
                (call
                  (index_call PLACEHOLDER_3 [
                    (argument_list 1)) to_i)
                (index_call PLACEHOLDER_3 [
                  (argument_list 0))))
            (then
              (string
                (interpolation #{
                  (call nickname
                    (argument_list PLACEHOLDER_1 PLACEHOLDER_2))) " moved to "
                (interpolation #{ PLACEHOLDER_3)))
            (else
              (expressions
                (string
                  (interpolation #{
                    (call nickname
                      (argument_list PLACEHOLDER_1 PLACEHOLDER_2))) " attempted to move to "
                  (interpolation #{ PLACEHOLDER_3) ", but that is not a valid square"))))))
      (method_def self nickname
        (param_list
          (param PLACEHOLDER_1)
          (param PLACEHOLDER_2))
        (expressions
          (call
            (call
              (index_call PLACEHOLDER_1 [
                (argument_list
                  (range 0 .. 1))) +
              (argument_list
                (index_call PLACEHOLDER_2 [
                  (argument_list
                    (range
                      (integer -) ..
                      (integer -)))))) upcase)))
      (method_def self valid_square?
        (param_list
          (param PLACEHOLDER_4)
          (param PLACEHOLDER_5))
        (expressions
          (and
            (call RANKS includes?
              (argument_list PLACEHOLDER_4)) &&
            (call FILES includes?
              (argument_list PLACEHOLDER_5))))))))
