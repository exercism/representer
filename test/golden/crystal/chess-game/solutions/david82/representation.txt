(module_def Chess
  (expressions
    (extend self)
    (const_assign RANKS =
      (range 1 .. 8))
    (const_assign FILES =
      (range
        (char ' A ') ..
        (char ' H ')))
    (method_def move_message
      (param_list
        (param PLACEHOLDER_1 String)
        (param PLACEHOLDER_2 String)
        (param PLACEHOLDER_3 String)) String
      (expressions
        (unless
          (call
            (call PLACEHOLDER_3 size) ==
            (argument_list 2))
          (then
            (return
              (argument_list "Invalid square format"))))
        (assign nickname =
          (call nickname
            (argument_list PLACEHOLDER_1 PLACEHOLDER_2)))
        (assign PLACEHOLDER_4 =
          (call
            (index_call PLACEHOLDER_3 [
              (argument_list
                (integer -))) to_i))
        (assign PLACEHOLDER_5 =
          (index_call PLACEHOLDER_3 [
            (argument_list 0)))
        (if
          (call valid_square?
            (argument_list PLACEHOLDER_4 PLACEHOLDER_5))
          (then
            (string
              (interpolation #{ nickname) " moved to "
              (interpolation #{ PLACEHOLDER_3)))
          (else
            (expressions
              (string
                (interpolation #{ nickname) " attempted to move to "
                (interpolation #{ PLACEHOLDER_3) ", but that is not a valid square"))))))
    (method_def nickname
      (param_list
        (param PLACEHOLDER_1 String)
        (param PLACEHOLDER_2 String)) String
      (expressions
        (call
          (string
            (interpolation #{
              (index_call PLACEHOLDER_1 [
                (argument_list 0 2)))
            (interpolation #{
              (index_call PLACEHOLDER_2 [
                (argument_list
                  (integer -) 2)))) upcase)))
    (method_def valid_square?
      (param_list
        (param PLACEHOLDER_4 Int32)
        (param PLACEHOLDER_5 Char)) Bool
      (expressions
        (and
          (call RANKS includes?
            (argument_list PLACEHOLDER_4)) &&
          (call FILES includes?
            (argument_list PLACEHOLDER_5)))))))
