(class_def CastleDinner
  (expressions
    (method_def self check_drink?
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (conditional
          (index_call
            (call PLACEHOLDER_1 downcase) [
            (argument_list "i") ]?) ? PLACEHOLDER_1 nil)))
    (method_def self check_food?
      (param_list
        (param PLACEHOLDER_2))
      (expressions
        (conditional
          (call PLACEHOLDER_2 ==
            (argument_list "Mushroom pasties")) ? PLACEHOLDER_2 nil)))
    (method_def self replace_drink
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (or
          (call self check_drink?
            (argument_list PLACEHOLDER_1)) || "Apple juice")))))
