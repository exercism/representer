(class_def CastleDinner
  (expressions
    (method_def self check_drink?
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (if
          (call PLACEHOLDER_1 matches?
            (argument_list
              (regex / [iI] /)))
          (then PLACEHOLDER_1))))
    (method_def self check_food?
      (param_list
        (param PLACEHOLDER_2))
      (expressions
        (if
          (call PLACEHOLDER_2 ==
            (argument_list "Mushroom pasties"))
          (then PLACEHOLDER_2))))
    (method_def self replace_drink
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (or
          (call check_drink?
            (argument_list PLACEHOLDER_1)) || "Apple juice")))))
