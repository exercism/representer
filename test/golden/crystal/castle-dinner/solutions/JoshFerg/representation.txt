(class_def CastleDinner
  (expressions
    (method_def self check_drink?
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (if
          (call
            (call PLACEHOLDER_1 downcase) includes?
            (argument_list "i"))
          (then PLACEHOLDER_1)
          (else
            (expressions nil)))))
    (method_def self replace_drink
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (case
          (call check_drink?
            (argument_list PLACEHOLDER_1))
          (when nil
            (expressions "Apple juice"))
          (else
            (expressions PLACEHOLDER_1)))))
    (method_def self check_food?
      (param_list
        (param PLACEHOLDER_2))
      (expressions
        (case PLACEHOLDER_2
          (when "Mushroom pasties"
            (expressions PLACEHOLDER_2))
          (else
            (expressions nil)))))))
