(class_def BinarySearch
  (expressions
    (method_def find
      (param_list
        (param PLACEHOLDER_1 Int32))
      (expressions
        (if
          (or
            (call (instance_var PLACEHOLDER_2) empty?) ||
            (not !
              (call PLACEHOLDER_1 in?
                (argument_list (instance_var PLACEHOLDER_2)))))
          (then
            (call raise
              (argument_list
                (call ArgumentError new)))))
        (call
          (call (instance_var PLACEHOLDER_2) sort) index
          (argument_list PLACEHOLDER_1))))
    (method_def initialize
      (param_list
        (param (instance_var PLACEHOLDER_2)
          (generic_instance_type Array
            (param_list Int32)))))))
