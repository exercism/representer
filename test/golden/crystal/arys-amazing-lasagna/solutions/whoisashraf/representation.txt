(file "src/arys_amazing_lasagna.cr"
  (class_def Lasagna
    (expressions
      (const_assign EXPECTED_MINUTES_IN_OVEN = 40)
      (const_assign (constant PLACEHOLDER_1) = 2)
      (method_def total_time_in_minutes
        (param_list
          (param PLACEHOLDER_2)
          (param PLACEHOLDER_3))
        (expressions
          (assign PLACEHOLDER_4 =
            (call (constant PLACEHOLDER_1) *
              (argument_list PLACEHOLDER_2)))
          (return
            (argument_list
              (call PLACEHOLDER_4 +
                (argument_list PLACEHOLDER_3))))))
      (method_def preparation_time_in_minutes
        (param_list
          (param PLACEHOLDER_5))
        (expressions
          (return
            (argument_list
              (call (constant PLACEHOLDER_1) *
                (argument_list PLACEHOLDER_5))))))
      (method_def remaining_minutes_in_oven
        (param_list
          (param PLACEHOLDER_3))
        (expressions
          (return
            (argument_list
              (call EXPECTED_MINUTES_IN_OVEN -
                (argument_list PLACEHOLDER_3)))))))))
