(file "src/wellingtons_weather_station.cr"
  (class_def Temperature
    (expressions
      (method_def number_missing_sensors
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (assign PLACEHOLDER_2 = 0)
          (assign PLACEHOLDER_3 =
            (call PLACEHOLDER_1 %
              (argument_list 4)))
          (if
            (call PLACEHOLDER_3 >
              (argument_list 0))
            (then
              (assign PLACEHOLDER_2 =
                (call 4 -
                  (argument_list PLACEHOLDER_3))))) PLACEHOLDER_2))
      (method_def to_fahrenheit
        (param_list
          (param PLACEHOLDER_4))
        (expressions
          (call
            (call
              (call PLACEHOLDER_4 *
                (argument_list 1.8)) +
              (argument_list 32)) to_i)))
      (method_def round
        (param_list
          (param PLACEHOLDER_4))
        (expressions
          (call PLACEHOLDER_4 round
            (argument_list 1))))
      (method_def to_kelvin
        (param_list
          (param PLACEHOLDER_4))
        (expressions
          (call PLACEHOLDER_4 +
            (argument_list 273.15)))))))
