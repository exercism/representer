(class_def Temperature
  (expressions
    (method_def to_fahrenheit
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call
          (call
            (call PLACEHOLDER_1 *
              (argument_list 1.8)) to_i) +
          (argument_list 32))))
    (method_def number_missing_sensors
      (param_list
        (param PLACEHOLDER_2))
      (expressions
        (call
          (call 4 -
            (argument_list PLACEHOLDER_2)) %
          (argument_list 4))))
    (method_def round
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call PLACEHOLDER_1 round
          (argument_list 1))))
    (method_def to_kelvin
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (call PLACEHOLDER_1 +
          (argument_list 273.15))))))
