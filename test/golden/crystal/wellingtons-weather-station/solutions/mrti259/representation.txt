(class_def Temperature
  (expressions
    (method_def number_missing_sensors
      (param_list
        (param PLACEHOLDER_1))
      (expressions
        (assign PLACEHOLDER_2 =
          (call PLACEHOLDER_1 %
            (argument_list 4)))
        (if
          (call PLACEHOLDER_2 ==
            (argument_list 0))
          (then
            (return
              (argument_list 0))))
        (call 4 -
          (argument_list PLACEHOLDER_2))))
    (method_def to_fahrenheit
      (param_list
        (param PLACEHOLDER_3))
      (expressions
        (call
          (call 32 +
            (argument_list
              (call 1.8 *
                (argument_list PLACEHOLDER_3)))) to_i)))
    (method_def round
      (param_list
        (param PLACEHOLDER_3))
      (expressions
        (call PLACEHOLDER_3 round
          (argument_list 1))))
    (method_def to_kelvin
      (param_list
        (param PLACEHOLDER_3))
      (expressions
        (call PLACEHOLDER_3 +
          (argument_list 273.15))))))
