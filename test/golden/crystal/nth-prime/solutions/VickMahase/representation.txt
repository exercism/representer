(module_def NthPrime
  (expressions
    (visibility_modifier private
      (method_def self PLACEHOLDER_1
        (param_list
          (param PLACEHOLDER_2 Int32)) Bool
        (expressions
          (if
            (call PLACEHOLDER_2 <=
              (argument_list 1))
            (then
              (return
                (argument_list false))))
          (if
            (call PLACEHOLDER_2 <=
              (argument_list 3))
            (then
              (return
                (argument_list true))))
          (if
            (or
              (call
                (call PLACEHOLDER_2 %
                  (argument_list 2)) ==
                (argument_list 0)) ||
              (call
                (call PLACEHOLDER_2 %
                  (argument_list 3)) ==
                (argument_list 0)))
            (then
              (return
                (argument_list false))))
          (assign PLACEHOLDER_3 = 5)
          (while
            (call
              (call PLACEHOLDER_3 *
                (argument_list PLACEHOLDER_3)) <=
              (argument_list PLACEHOLDER_2))
            (expressions
              (if
                (or
                  (call
                    (call PLACEHOLDER_2 %
                      (argument_list PLACEHOLDER_3)) ==
                    (argument_list 0)) ||
                  (call
                    (call PLACEHOLDER_2 %
                      (argument_list
                        (call PLACEHOLDER_3 +
                          (argument_list 2)))) ==
                    (argument_list 0)))
                (then
                  (return
                    (argument_list false))))
              (op_assign PLACEHOLDER_3 += 6))) true)))
    (method_def self prime
      (param_list
        (param PLACEHOLDER_4 Int32)) Int32
      (expressions
        (if
          (call PLACEHOLDER_4 <=
            (argument_list 0))
          (then
            (call raise
              (argument_list
                (call ArgumentError new
                  (argument_list "There is no zeroth or negative prime"))))))
        (assign PLACEHOLDER_5 = 0)
        (assign PLACEHOLDER_2 = 2)
        (while
          (call PLACEHOLDER_5 <
            (argument_list PLACEHOLDER_4))
          (expressions
            (if
              (call PLACEHOLDER_1
                (argument_list PLACEHOLDER_2))
              (then
                (op_assign PLACEHOLDER_5 += 1)
                (if
                  (call PLACEHOLDER_5 ==
                    (argument_list PLACEHOLDER_4))
                  (then
                    (return
                      (argument_list PLACEHOLDER_2))))))
            (op_assign PLACEHOLDER_2 += 1)))
        (call raise
          (argument_list "Error finding the nth prime"))))))
