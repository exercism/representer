(file "src/crystal_hunter.cr"
  (class_def Rules
    (expressions
      (method_def win?
        (param_list
          (param PLACEHOLDER_1)
          (param PLACEHOLDER_2)
          (param PLACEHOLDER_3))
        (expressions
          (and PLACEHOLDER_1 &&
            (not !
              (call lose?
                (argument_list PLACEHOLDER_2 PLACEHOLDER_3))))))
      (method_def bonus_points?
        (param_list
          (param PLACEHOLDER_2 Bool)
          (param PLACEHOLDER_3 Bool))
        (expressions
          (and PLACEHOLDER_2 && PLACEHOLDER_3)))
      (method_def lose?
        (param_list
          (param PLACEHOLDER_2)
          (param PLACEHOLDER_3))
        (expressions
          (and
            (not ! PLACEHOLDER_2) && PLACEHOLDER_3)))
      (method_def score?
        (param_list
          (param PLACEHOLDER_4)
          (param PLACEHOLDER_5))
        (expressions
          (or PLACEHOLDER_4 || PLACEHOLDER_5))))))
