(file "src/party_robot.cr"
  (class_def PartyRobot
    (expressions
      (method_def self assign_table
        (param_list
          (param PLACEHOLDER_1 String)
          (param PLACEHOLDER_2 String)
          (param PLACEHOLDER_3 Int)
          (param PLACEHOLDER_4 Float64)
          (param PLACEHOLDER_5 String)) String
        (expressions
          (assign PLACEHOLDER_6 =
            (call PLACEHOLDER_4 round
              (argument_list 1)))
          (assign PLACEHOLDER_7 =
            (index_call
              (call PLACEHOLDER_5 split
                (argument_list
                  (char ' " " '))) [
              (argument_list 0)))
          (assign PLACEHOLDER_7 =
            (call
              (call "" +
                (argument_list
                  (index_call PLACEHOLDER_7 [
                    (argument_list 0)))) +
              (argument_list
                (index_call PLACEHOLDER_7 [
                  (argument_list
                    (integer -))))))
          (string "Welcome to my party, "
            (interpolation #{ PLACEHOLDER_1) "!\nYou have been assigned to table "
            (interpolation #{ PLACEHOLDER_3) ". Your table is "
            (interpolation #{ PLACEHOLDER_2) ", exactly "
            (interpolation #{ PLACEHOLDER_6) " meters from here.\nYou will be sitting next to "
            (interpolation #{ PLACEHOLDER_7) ".")))
      (method_def self happy_birthday
        (param_list
          (param PLACEHOLDER_1 String)
          (param PLACEHOLDER_8 Int)) String
        (expressions
          (string "Happy birthday "
            (interpolation #{ PLACEHOLDER_1) "! You are now "
            (interpolation #{ PLACEHOLDER_8) " years old!")))
      (method_def self welcome
        (param_list
          (param PLACEHOLDER_1 String)) String
        (expressions
          (string "Welcome to my party, "
            (interpolation #{ PLACEHOLDER_1) "!"))))))
