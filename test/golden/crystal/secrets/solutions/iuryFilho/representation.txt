(file "src/secrets.cr"
  (module_def Secrets
    (expressions
      (method_def self reverse_xor
        (param_list
          (param PLACEHOLDER_1 UInt8)
          (param PLACEHOLDER_2 UInt8)) UInt8
        (expressions
          (call ~
            (call PLACEHOLDER_1 ^
              (argument_list PLACEHOLDER_2)))))
      (method_def self apply_mask
        (param_list
          (param PLACEHOLDER_1 UInt8)
          (param PLACEHOLDER_3 UInt8)) UInt8
        (expressions
          (call PLACEHOLDER_1 &
            (argument_list PLACEHOLDER_3))))
      (method_def self shift_back
        (param_list
          (param PLACEHOLDER_1 UInt8)
          (param PLACEHOLDER_4 UInt8)) UInt8
        (expressions
          (call PLACEHOLDER_1 <<
            (argument_list PLACEHOLDER_4))))
      (method_def self set_bits
        (param_list
          (param PLACEHOLDER_5 UInt8)
          (param PLACEHOLDER_6 UInt8)) UInt8
        (expressions
          (call PLACEHOLDER_5 |
            (argument_list PLACEHOLDER_6)))))))
