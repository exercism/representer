(file "src/interest_is_interesting.cr"
  (class_def SavingsAccount
    (expressions
      (method_def self interest_rate
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (if
            (call PLACEHOLDER_1 <
              (argument_list 0))
            (then 3.213)
            (elsif
              (call PLACEHOLDER_1 <
                (argument_list 1000))
              (then 0.5)
              (elsif
                (call PLACEHOLDER_1 <
                  (argument_list 5000))
                (then 1.621)
                (else
                  (expressions 2.475)))))))
      (method_def self years_before_desired_balance
        (param_list
          (param PLACEHOLDER_2)
          (param PLACEHOLDER_3))
        (expressions
          (assign PLACEHOLDER_4 = 0)
          (while
            (call PLACEHOLDER_2 <
              (argument_list PLACEHOLDER_3))
            (expressions
              (assign PLACEHOLDER_2 =
                (call self annual_balance_update
                  (argument_list PLACEHOLDER_2)))
              (op_assign PLACEHOLDER_4 += 1))) PLACEHOLDER_4))
      (method_def self interest
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (call
            (call PLACEHOLDER_1 *
              (argument_list
                (call self interest_rate
                  (argument_list PLACEHOLDER_1)))) /
            (argument_list 100))))
      (method_def self annual_balance_update
        (param_list
          (param PLACEHOLDER_1))
        (expressions
          (call PLACEHOLDER_1 +
            (argument_list
              (call self interest
                (argument_list PLACEHOLDER_1)))))))))
