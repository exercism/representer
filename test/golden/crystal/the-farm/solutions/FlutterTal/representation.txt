(expressions
  (class_def InvalidCowsError < Exception
    (expressions
      (method_def initialize
        (param_list
          (param (instance_var PLACEHOLDER_1) String)))))
  (module_def TheFarm
    (expressions
      (method_def self validate_number_of_cows
        (param_list
          (param PLACEHOLDER_2 Int32))
        (expressions
          (if
            (call PLACEHOLDER_2 <
              (argument_list 0))
            (then
              (call raise
                (argument_list
                  (call InvalidCowsError new
                    (argument_list
                      (string
                        (interpolation #{ PLACEHOLDER_2) " cows are invalid: there are no negative cows"))))))
            (elsif
              (call PLACEHOLDER_2 ==
                (argument_list 0))
              (then
                (call raise
                  (argument_list
                    (call InvalidCowsError new
                      (argument_list "0 cows are invalid: no cows don't need food")))))))))
      (method_def self divide_food
        (param_list
          (param food FodderCalculator)
          (param PLACEHOLDER_2 Int32))
        (expressions
          (call
            (call
              (call food fodder_amount!) /
              (argument_list PLACEHOLDER_2)) *
            (argument_list
              (call food fattening_factor!))))
        (rescue rescue
          (expressions 0)))
      (method_def self validate_input_and_divide_food
        (param_list
          (param food FodderCalculator)
          (param PLACEHOLDER_2 Int32))
        (expressions
          (if
            (call PLACEHOLDER_2 <=
              (argument_list 0))
            (then
              (call raise
                (argument_list "Number of cows must be greater than 0"))))
          (call divide_food
            (argument_list food PLACEHOLDER_2)))))))
