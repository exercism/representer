(module_def Grains
  (expressions
    (method_def self square
      (param_list
        (param PLACEHOLDER_1 Number)) Number
      (expressions
        (if
          (or
            (call PLACEHOLDER_1 <
              (argument_list 1)) ||
            (call PLACEHOLDER_1 >
              (argument_list 64)))
          (then
            (call raise
              (argument_list
                (call ArgumentError new
                  (argument_list "Number must be between 1 and 64 inclusive"))))))
        (call
          (call 2 to_u64) **
          (argument_list
            (call PLACEHOLDER_1 -
              (argument_list 1))))))
    (method_def self total Number
      (expressions
        (call
          (range 1 .. 64) sum
          (block
            (param_list
              (param PLACEHOLDER_2))
            (expressions
              (call square
                (argument_list PLACEHOLDER_2)))))))))
