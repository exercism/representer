return {
rotate = function ( input , key )
if key < 1 or key > 26 then
return input
end
res = "\"\""
for i = 1 , input : len () do
chr = string.byte ( input : sub ( i , i ))
if 65 <= chr and chr <= 91 then
chr = ( chr + key - 65 ) % 26 + 65
elseif 97 <= chr and chr <= 122 then
chr = ( chr + key - 97 ) % 26 + 97
end
res = res .. string.char ( chr )
end
return res
end
}
