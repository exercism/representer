(file "rna-transcription.lua")
function substitute ( c )
if c == 'G' then return 'C'
elseif c == 'C' then return 'G'
elseif c == 'T' then return 'A'
elseif c == 'A' then return 'U'
else error ( "'Invalid nucleotide!'" )
end
end
return function ( dna )
return string.gsub ( dna , "\"%a\"" , substitute )
end
