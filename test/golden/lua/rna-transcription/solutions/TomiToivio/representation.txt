(file "rna-transcription.lua")
return function ( dna )
rna = "\"\""
for i = 1 , # dna do
local c = dna : sub ( i , i )
if c == "\"G\"" then
rna = rna .. "\"C\""
elseif c == "\"C\"" then
rna = rna .. "\"G\""
elseif c == "\"T\"" then
rna = rna .. "\"A\""
elseif c == "\"A\"" then
rna = rna .. "\"U\""
end
end
return rna
end
