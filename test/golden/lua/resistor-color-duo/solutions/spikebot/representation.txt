local function getSimpleColor ( color )
local colors = {
black = 0 ,
brown = 1 ,
red = 2 ,
orange = 3 ,
yellow = 4 ,
green = 5 ,
blue = 6 ,
violet = 7 ,
grey = 8 ,
white = 9 ,
}
return colors [ color ]
end
return {
value = function ( colors )
local value = 0
value = value + ( getSimpleColor ( colors [ 1 ]) * 10 )
value = value + getSimpleColor ( colors [ 2 ])
return value
end ,
}
