return function ( n )
if n < 1 then n () end
local primes = { 2 , 3 , 5 }
local q = 3
local q2 = q ^ 2
local p = 7
local step = 4
while # primes < n do
while p > q2 do
q = q + 1
q2 = q ^ 2
end
local i = 3
while primes [ i ] <= q do
if p % primes [ i ] == 0 then
goto continue
end
i = i + 1
end
primes [ # primes + 1 ] = p
:: continue ::
p = p + step
step = step > 2 and 2 or 4
end
return primes [ n ]
end
