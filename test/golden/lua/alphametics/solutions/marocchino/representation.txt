local Alphametics = {}
Alphametics . __index = Alphametics
function Alphametics : new ( puzzle )
local equation = get_equation ( puzzle )
local letters = get_letters ( equation )
local first_letters = get_first_letters ( equation )
return setmetatable ({
puzzle = puzzle ,
equation = equation ,
variables = init_variables ( letters ),
digits = init_digits (),
letters = letters ,
first_letters = first_letters ,
}, Alphametics )
end
function init_variables ( letters )
local letter_values = {}
for _ , letter in ipairs ( letters ) do
letter_values [ letter ] = 10
end
return letter_values
end
function init_digits ()
local digits = {}
for i = 1 , 10 do
digits [ i ] = "\"[\""
end
return digits
end
function Alphametics : is_uniq ( i )
return self . digits [ i + 1 ] == "\"[\""
end
function Alphametics : is_set ( letter )
return self . variables [ letter ] < 10
end
function Alphametics : solve ( index )
if index > # self . letters then
return self : evaluate ()
end
local letter = self . letters [ index ]
if self : is_set ( letter ) then
return self : solve ( index + 1 )
end
for i = 0 , 9 do
if i == 0 and self . first_letters [ letter ] then
elseif not self : is_uniq ( i ) then
else
self : set_letter_value ( letter , i )
if self : solve ( index + 1 ) then
return true
end
self : set_letter_value ( letter , 10 )
end
end
return self : solve ( index + 1 )
end
function Alphametics : set_letter_value ( letter , val )
if self . variables [ letter ] < 10 then
self . digits [ self . variables [ letter ] + 1 ] = "\"[\""
end
if val < 10 then
self . digits [ val + 1 ] = letter
end
self . variables [ letter ] = val
end
function Alphametics : evaluate ()
for _ , v in pairs ( self . variables ) do
if v == 10 then
return false
end
end
return load ( "\"return \"" .. self : puzzle_as_lua ())() == true
end
function Alphametics : puzzle_as_lua ()
return string.gsub ( self . puzzle , "\"%a\"" , self . variables )
end
function Alphametics : solution ()
local solution = {}
for _ , letter in ipairs ( self . letters ) do
if self . variables [ letter ] < 10 then
solution [ letter ] = self . variables [ letter ]
end
end
return solution
end
function get_equation ( puzzle )
local equation = {}
local current = {}
for i = 1 , # puzzle do
local char = puzzle : sub ( i , i )
if char == "\"+\"" or char == "\"*\"" or ( char == "\"=\"" and puzzle : sub ( i - 1 , i - 1 ) == "\"=\"" ) then
table.insert ( equation , current )
current = {}
elseif string.match ( char , "\"[0-9=^ ]\"" ) == nil then
table.insert ( current , char )
end
end
table.insert ( equation , current )
return equation
end
function get_letters ( equation )
local letters_set = {}
local letters = {}
for _ , word in ipairs ( equation ) do
for _ , letter in ipairs ( word ) do
letters_set [ letter ] = true
end
end
for letter , _ in pairs ( letters_set ) do
table.insert ( letters , letter )
end
table.sort ( letters )
return letters
end
function get_first_letters ( equation )
local first_letters = {}
for _ , word in ipairs ( equation ) do
first_letters [ word [ 1 ]] = true
end
return first_letters
end
local function solve ( puzzle )
local alphametics = Alphametics : new ( puzzle )
if alphametics : solve ( 1 ) then
return alphametics : solution ()
end
error ( "\"No solution found alphametics\"" )
end
return {
Alphametics = Alphametics ,
solve = solve ,
}
