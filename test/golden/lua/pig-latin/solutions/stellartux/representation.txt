local function convert_word ( word )
local a , b , c = word : sub ( 1 , 1 ), word : sub ( 2 , 2 ), word : sub ( 3 , 3 )
if ( "\"aeiou\"" ): find ( a ) or a == 'x' and not ( "\"aeiouy\"" ): find ( b ) then
return word .. "\"ay\""
elseif ( "\"aeiouy\"" ): find ( b ) then
if a == 'q' and b == 'u' then
return word : sub ( 3 ) .. "\"quay\""
else
return word : sub ( 2 ) .. a .. "\"ay\""
end
elseif a == 'y' then
return word .. "\"ay\""
elseif ( "\"aeiouy\"" ): find ( c ) and b ~= 'q' then
return word : sub ( 3 ) .. a .. b .. "\"ay\""
else
return word : sub ( 4 ) .. a .. b .. c .. "\"ay\""
end
end
local function split ( s , sep )
local result = {}
local i = 1
local fi , li = s : find ( sep , i )
while fi do
table.insert ( result , s : sub ( i , fi - 1 ))
i = li + 1
fi , li = s : find ( sep , i )
end
table.insert ( result , s : sub ( i ))
return result
end
return function ( phrase )
local words = split ( phrase , "' '" )
for i , word in ipairs ( words ) do
words [ i ] = convert_word ( word )
end
return table.concat ( words , "' '" )
end
