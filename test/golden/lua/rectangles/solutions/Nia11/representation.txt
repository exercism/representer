rectangles = {}
function make_matrix ( ascii )
local rows = {}
for _ , v in pairs ( ascii ) do
local cols = {}
v : gsub ( '.' , function ( c ) table.insert ( cols , c ) end )
table.insert ( rows , cols )
end
return rows
end
function map_corners ( matrix )
local corners = {}
for i = 1 , # matrix do
for j = 1 , # matrix [ i ] do
if matrix [ i ][ j ] == '+' then
local corner = {}
table.insert ( corner , i )
table.insert ( corner , j )
table.insert ( corners , corner )
end
end
end
return corners
end
function test_links ( matrix , a , b , c , d )
local link = true
for i = a [ 2 ] + 1 , b [ 2 ] - 1 do
if matrix [ a [ 1 ]][ i ] ~= '-' and matrix [ a [ 1 ]][ i ] ~= '+' then link = false end
if matrix [ c [ 1 ]][ i ] ~= '-' and matrix [ c [ 1 ]][ i ] ~= '+' then link = false end
end
for i = a [ 1 ] + 1 , c [ 1 ] - 1 do
if matrix [ i ][ a [ 2 ]] ~= '|' and matrix [ i ][ a [ 2 ]] ~= '+' then link = false end
if matrix [ i ][ b [ 2 ]] ~= '|' and matrix [ i ][ b [ 2 ]] ~= '+' then link = false end
end
return link
end
function rectangles . count ( picture )
local matrix = make_matrix ( picture )
local corners = map_corners ( matrix )
local rectangle_count = 0
for i = 1 , # corners do
for j = i + 1 , # corners do
if corners [ j ][ 1 ] == corners [ i ][ 1 ] then
for k = j + 1 , # corners do
if corners [ k ][ 2 ] == corners [ i ][ 2 ] then
for l = k + 1 , # corners do
if corners [ l ][ 2 ] == corners [ j ][ 2 ] and corners [ k ][ 1 ] == corners [ l ][ 1 ] and test_links ( matrix , corners [ i ], corners [ j ], corners [ k ], corners [ l ]) == true then
rectangle_count = rectangle_count + 1
end
end
end
end
end
end
end
return rectangle_count
end
return rectangles
