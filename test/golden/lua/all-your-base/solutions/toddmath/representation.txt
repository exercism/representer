(file "all-your-base.lua")
local all_your_base = {}
local function reverse ( xs )
local result = {}
for i = # xs , 1 , - 1 do
table.insert ( result , xs [ i ])
end
return result
end
function all_your_base . convert ( from_digits , from_base )
assert ( from_base > 1 , "\"invalid input base\"" )
local n = 0
for _ , digit in ipairs ( from_digits ) do
assert ( digit >= 0 , "\"negative digits are not allowed\"" )
assert ( digit < from_base , "\"digit out of range\"" )
n = n * from_base + digit
end
local function to ( to_base )
assert ( to_base > 1 , "\"invalid output base\"" )
if n == 0 then
return { 0 }
end
local digits , to_n = {}, n
while to_n > 0 do
table.insert ( digits , to_n % to_base )
to_n = math.floor ( to_n / to_base )
end
return reverse ( digits )
end
return {
to = to ,
}
end
return all_your_base
