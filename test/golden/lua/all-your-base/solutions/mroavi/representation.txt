(file "all-your-base.lua")
local all_your_base = {}
local function validate_base ( base , str )
assert ( base > 1 , str )
end
local function validate_digit ( digit , base )
assert ( digit >= 0 , "\"negative digits are not allowed\"" )
assert ( digit < base , "\"digit out of range\"" )
end
local function to_base_10 ( from_digits , from_base )
validate_base ( from_base , "\"invalid input base\"" )
local result = 0
for i , from_digit in ipairs ( from_digits ) do
validate_digit ( from_digit , from_base )
result = result + from_digit * math.floor ( from_base ^ ( # from_digits - i ))
end
return result
end
local function from_base_10 ( num , to_base )
local result = {}
while true do
table.insert ( result , 1 , num % to_base )
num = math.floor ( num / to_base )
if num == 0 then break end
end
return result
end
all_your_base . convert = function ( from_digits , from_base )
return {
to = function ( to_base )
validate_base ( to_base , "\"invalid output base\"" )
return from_base_10 ( to_base_10 ( from_digits , from_base ), to_base )
end
}
end
return all_your_base
