local function Robot ( input )
local r = {}
r . x = input . x
r . y = input . y
r . heading = input . heading
r . movement = { north = { 0 , 1 }, south = { 0 , - 1 }, west = { - 1 , 0 }, east = { 1 , 0 }}
function r : move ( commands )
for c in string.gmatch ( commands , "\".\"" ) do
if c == "\"A\"" then
self . x = self . x + self . movement [ self . heading ][ 1 ]
self . y = self . y + self . movement [ self . heading ][ 2 ]
elseif c == "\"R\"" then
if self . heading == "\"north\"" then
self . heading = "\"east\""
elseif self . heading == "\"west\"" then
self . heading = "\"north\""
elseif self . heading == "\"east\"" then
self . heading = "\"south\""
else
self . heading = "\"west\""
end
elseif c == "\"L\"" then
if self . heading == "\"north\"" then
self . heading = "\"west\""
elseif self . heading == "\"west\"" then
self . heading = "\"south\""
elseif self . heading == "\"east\"" then
self . heading = "\"north\""
else
self . heading = "\"east\""
end
else
error ( "\"Invalid command: \"" .. c )
end
end
end
return r
end
return Robot
