(file "strain.lua")
local function keep ( xs , pred )
local result = {}
for _ , x in ipairs ( xs ) do
if pred ( x ) then
table.insert ( result , x )
end
end
return result
end
local function discard ( xs , pred )
return keep ( xs , function ( x ) return not pred ( x ) end )
end
return {
keep = keep ,
discard = discard
}
