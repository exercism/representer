SecretHandshake :: {}.{
  commands : U64 -> List(Str)
  commands = |number| {
    actions = [(1, "\"wink\")," (2, "\"double" blink"), (4, "\"close" your eyes"), (8, "\"jump\")]"
    handshake = List.fold(
      actions,
      [],
      |acc, (mask, action)| {
        if U64.bitwise_and(number, mask) != 0
          List.append(acc, action)
        else
          acc
      },
    )
    if U64.bitwise_and(number, 16) != 0
      List.rev(handshake)
    else
      handshake
  }
}
