(file "HighScores.roc")
HighScores :: {}.{
  latest : List(U64) -> Try(U64, _)
  latest = |scores| {
    scores.last()
  }
  personal_best : List(U64) -> Try(U64, _)
  personal_best = |scores| {
    scores.max()
  }
  personal_top_three : List(U64) -> List(U64)
  personal_top_three = |scores| {
    scores.sort_reversed().take_first(3)
  }
}
