{{
`` nucleotideComplement `` converts a DNA nucleotide
into its RNA complement .
If input is not a valid DNA nucleotide , `` abort `` is
called .
## Examples
```
toOptional ! ' ( nucleotideComplement ? g )
```
```
toOptional ! ' ( nucleotideComplement ? X )
```
}}
nucleotideComplement : Char -> { Abort } Char
nucleotideComplement c = match toUpper c with
  ? G -> ? C
  ? C -> ? G
  ? T -> ? A
  ? A -> ? U
  _ -> abort
{{
`` toRNA `` takes a string of DNA and converts it to
the complementary RNA strand .
If the provided input is not a valid DNA strand ,
this function returns None
## Examples
```
toRNA "\"gattaca\""
```
```
toRNA "\"nope\""
```
}}
toRNA : Text -> Optional Text
toRNA dna =
  rna = 'let
    List . map nucleotideComplement ( Text . toCharList dna )
      |> Text . fromCharList
  toOptional ! rna
> toRNA "\"gattaca\""
