allYourBase . rebase : Int -> [ Int ] -> Int -> Optional [ Int ]
allYourBase . rebase = cases
  inputBase , inputDigits , outputBase
    | any ( flip ( < ) + 2 ) [ inputBase , outputBase ] -> None
    | any ( x -> x < + 0 || ( x >= inputBase )) inputDigits -> None
    | otherwise ->
      unfoldr n =
        go n acc =
          if n == + 0
          then acc
          else go ( n / outputBase ) ( mod n outputBase List .+: acc )
        go n []
      n = List . foldLeft ( acc d -> acc * inputBase + d ) + 0 inputDigits
      match unfoldr n with
        [] -> Some ([ + 0 ])
        ds -> Some ( ds )
