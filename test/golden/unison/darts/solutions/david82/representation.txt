darts . score : Float -> Float -> Nat
darts . score dartX dartY =
  dartDistance = sqrt (( dartX * dartX ) + ( dartY * dartY ))
  distanceToScore dartDistance
distanceToScore : Float -> Nat
distanceToScore = cases
  dartDistance | dartDistance <= innerCircleRadius -> 10
    | dartDistance <= middleCircleRadius -> 5
    | dartDistance <= outerCircleRadius -> 1
    | otherwise -> 0
innerCircleRadius : Float
innerCircleRadius = 1.0
middleCircleRadius : Float
middleCircleRadius = 5.0
outerCircleRadius : Float
outerCircleRadius = 10.0
