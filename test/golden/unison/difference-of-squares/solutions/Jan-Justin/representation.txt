differenceOfSquares . squareOfSum : Nat -> Nat
differenceOfSquares . squareOfSum number = (( number * ( number + 1 )) / 2 ) |> ( flip pow 2 )
differenceOfSquares . sumOfSquares : Nat -> Nat
differenceOfSquares . sumOfSquares number = ( number * ( number + 1 ) * (( 2 * number ) + 1 )) / 6
differenceOfSquares . difference : Nat -> Nat
differenceOfSquares . difference number = squareOfSum number - sumOfSquares number
