allergens = [ "\"eggs\"" , "\"peanuts\"" , "\"shellfish\"" , "\"strawberries\"" , "\"tomatoes\"" , "\"chocolate\"" , "\"pollen\"" , "\"cats\"" ]
lookup allergen = match Map . get allergen
  ( Map . fromList ( List . zip allergens [ 1 , 2 , 4 , 8 , 16 , 32 , 64 , 128 ])) with
    None -> 0
    Some n -> n
allergies . allergicTo : Text -> Nat -> Boolean
allergies . allergicTo allergen score =
  Nat . and score ( lookup allergen ) > 0
allergies . list : Nat -> [ Text ]
allergies . list score =
  List . filter (( flip allergicTo ) score ) allergens
