allergicTo : Text -> Nat -> Boolean
allergicTo allergen score =
  Allergy . fromText allergen |>
    cases
      ( Some allergy ) -> Allergy . isAlergic allergy score
      _ -> false
list : Nat -> [ Text ]
list score =
  filterMap
    ( cases
      x @ ( Allergy text value )
        | Allergy . isAlergic x score -> Some text
        | otherwise -> None
    )
    allergies
unique type Allergy = Allergy Text Nat
allergies = [
  ( Allergy "\"eggs\"" 1 ),
  ( Allergy "\"peanuts\"" 2 ),
  ( Allergy "\"shellfish\"" 4 ),
  ( Allergy "\"strawberries\"" 8 ),
  ( Allergy "\"tomatoes\"" 16 ),
  ( Allergy "\"chocolate\"" 32 ),
  ( Allergy "\"pollen\"" 64 ),
  ( Allergy "\"cats\"" 128 ),
]
Allergy . isAlergic : Allergy -> Nat -> Boolean
use Nat
Allergy . isAlergic = cases ( Allergy _ value ) -> ( << ) (( < ) 0 ) ( Nat . and value )
Allergy . fromText : Text -> Optional Allergy
Allergy . fromText t = find ( x -> Allergy . matches x t ) allergies
Allergy . matches : Allergy -> Text -> Boolean
Allergy . matches = cases ( Allergy text _ ) -> ( Text .== ) text
