alphabetSize : Nat
alphabetSize = 26
rotationalCipher . rotate : Text -> Nat -> Text
rotationalCipher . rotate inputText shiftKey =
  rotateCharacter : Char -> Char
  rotateCharacter char =
    rotateAlphabetic : Char -> Char
    rotateAlphabetic baseChar =
      baseNumericValue = toNat baseChar
      mod ( toNat char + shiftKey - baseNumericValue ) alphabetSize + baseNumericValue |> fromNat |> getOrBug "\"!?\""
    match () with
      _ | isLower char -> rotateAlphabetic ? a
        | isUpper char -> rotateAlphabetic ? A
        | otherwise -> char
  inputText
    |> toCharList
    |> map rotateCharacter
    |> fromCharList
