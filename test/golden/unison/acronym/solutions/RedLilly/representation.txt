acronym . abbreviate : Text -> Text
acronym . abbreviate full =
  noPunctuation = Text . flatMap ( x ->
    match ( toText x ) with
      "\"-\"" -> "\"\\" s "\"\n      ch | isMatch punctuation ch -> \"\"\n      ch -> ch)\n  full |> noPunctuation |> split ?\\" s |> m "ap (take 1) |> join \"\" |> toUppercase\n\n"
