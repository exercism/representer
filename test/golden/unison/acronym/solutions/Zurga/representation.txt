acronym . abbreviate : Text -> Text
acronym . abbreviate test =
  test
    |> Text . toLowercase
    |> Text . flatMap (
      cases
        ? _ -> "\" \""
        ?- -> "\" \""
        c | Char . inRange ? a ? z c -> Char . toText c
        ?\ s -> "\" \""
        _ -> "\"\""
    )
    |> Text . split ?\ s
    |> List . map ( x -> Text . take 1 x )
    |> List . foldLeft ( Text .++ ) "\"\""
    |> Text . toUppercase
