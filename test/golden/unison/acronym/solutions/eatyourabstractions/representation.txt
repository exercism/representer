acronym . abbreviate : Text -> Text
acronym . abbreviate phrase =
  zeroString = "\"\""
  isDelim ch = ( ch == ?- ) || ( ch == ? _ ) || ( ch == ?\ s )
  words = Text . segmentBy ( not << isDelim ) phrase
  letters = words |> map ( str -> ( take 1 str ) |> toUppercase )
  List . foldLeft ( Text .++ ) zeroString letters
