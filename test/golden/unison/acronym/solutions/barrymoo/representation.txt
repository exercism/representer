{{
  @ signature { acronym . abbreviate }
  ` acronym . abbreviate ` will convert a sentence into the appropriate acronym
  ignoring charaters such as ` ' ` .
  ```
  acronym . abbreviate "\"Liquid-crystal display\""
  ```
  ```
  acronym . abbreviate "\"As Soon As Possible\""
  ```
  ```
  acronym . abbreviate "\"Thank George It's Friday!\""
  ```
}}
acronym . abbreviate : Text -> Text
acronym . abbreviate input =
  input
    |> Text . toUppercase
    |> Text . segmentBy ( c -> isAlphaNum c || ( c == ? ' ))
    |> List . map ( Optional . map Tuple . at1 << Text . uncons )
    |> List . somes
    |> Text . fromCharList
test > acronym . abbreviate . tests . u1 =
  check ( acronym . abbreviate "\"Liquid-crystal display\"" == "\"LCD\"" )
test > acronym . abbreviate . tests . u2 =
  check ( acronym . abbreviate "\"As Soon As Possible\"" == "\"ASAP\"" )
test > acronym . abbreviate . tests . u3 =
  check ( acronym . abbreviate "\"Thank George It's Friday!\"" == "\"TGIF\"" )
