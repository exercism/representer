acronym . abbreviate : Text -> Text
acronym . abbreviate test =
  test
    |> toCharList
    |> List . map ( c -> fromCharList [ c ])
    |> List . split ( c -> ( c == "\" \"" ) || ( c == "\"-\"" ) || ( c == "\"_\"" ))
    |> List . map ( c -> c |> List . head |> Optional . getOrElse "\"\"" |> Text . toUppercase )
    |> join "\"\""
