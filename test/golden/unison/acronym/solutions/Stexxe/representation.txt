acronym . abbreviate : Text -> Text
acronym . abbreviate test =
  toCharList test |>
  List . split ( x -> isSpace x || ( x == ?- )) |>
  List . filter ( xs -> not ( List . isEmpty xs )) |>
  List . map ( List . find isLetter ) |>
  List . map ( Optional . getOrBug "\"error\"" ) |>
  List . map Char . toUppercase |>
  fromCharList
