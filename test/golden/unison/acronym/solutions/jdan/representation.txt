acronym . abbreviate : Text -> Text
acronym . abbreviate test =
  use List map
  test
    |> acronym . filter
    |> acronym . split
    |> map Text . toUppercase
    |> map ( x -> Text . take 1 x )
    |> Text . join "\"\""
acronym . filter : Text -> Text
acronym . filter text =
  use Char == inRange
  text
    |> toCharList
    |> List . filter
      ( c ->
        ( inRange ? a ? z c ) || ( inRange ? A ? Z c ) || ( c == ?\ s ) || ( c == ?- ))
    |> fromCharList
acronym . split : Text -> [ Text ]
acronym . split test =
  use Text split
  List . flatMap ( x -> split ?- x ) ( split ?\ s test )
