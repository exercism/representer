abbreviate : Text -> Text
abbreviate test =
  cleaned = Text . dropWhile isPunct test
  match Text . uncons cleaned with
    Some ( head , tail ) ->
      prepend ( Char . ascii . toUpper head ) ( abbreviate ( Text . dropWhile ( char -> not ( isSep char )) tail ))
    None -> cleaned
isSep : Char -> Boolean
isSep char = ( char == ? ) || ( char == ?- )
isPunct : Char -> Boolean
isPunct char = not ( inRange ? a ? z char || inRange ? A ? Z char )
prepend : Char -> Text -> Text
prepend char text = Text . fromCharList ( char +: Text . toCharList text )
