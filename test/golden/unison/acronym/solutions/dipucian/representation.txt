abbreviate : Text -> Text
abbreviate t =
  chars : [ Char ]
  chars = toCharList t
  firstLetter : [ Char ] -> Optional Char
  firstLetter = head >> map toUpper
  letterTable c = if isSpace c || isLetter c || ( c == ? ' ) then c else ?
  chars |> ( map letterTable >> split isSpace >> map firstLetter >> somes >> fromCharList )
