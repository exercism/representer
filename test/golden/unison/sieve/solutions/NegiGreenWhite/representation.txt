sieve . primes : Nat -> [ Nat ]
sieve . primes n =
  go : [( Nat , Boolean )] -> Optional ( Nat , [( Nat , Boolean )])
  go pairs =
    idx = compat . findFirstIndex ( cases ( _ , b ) -> b == true ) pairs
    match idx with
      None -> None
      Some prime ->
        newPairs =
          pairs
            |> chunk prime
            |> map ( modifyAt 0 ( cases ( x , _ ) -> ( x , false )))
            |> map concatOptional
            |> join
        Some ( prime , newPairs )
  baseList = [( 0 , false ), ( 1 , false )] ++ map ( x -> ( x , true )) ( rangeClosed 2 n )
  List . unfold baseList go
compat . findFirstIndex : ( a -> { e } Boolean ) -> [ a ] -> { e } Optional Nat
compat . findFirstIndex f xs =
  go i = cases
    x +: xs
      | f x -> Some i
      | otherwise ->
        use Nat +
        go ( i + 1 ) xs
    _ -> None
  go 0 xs
