use List
allLetters = Text . toCharList "\"ABCDEFGHIJKLMNOPQRSTUVWXYZ\""
diamond . rows : Char -> [ Text ]
diamond . rows letter =
  if letter == ? A then [ "\"A\"" ]
  else
    letters = arr letter allLetters
    row letters 0
diamond . row : [ Char ] -> Nat -> [ Text ]
diamond . row rest depth =
  match rest with
    [] -> []
    letter +: [] -> [ rowText letter [] depth ]
    letter +: rest ->
      [ rowText letter rest depth ] ++ ( row rest ( depth + 1 )) ++ [ rowText letter rest depth ]
diamond . rowText : Char -> [ Char ] -> Nat -> Text
diamond . rowText letter rest depth =
  pad = List . size rest
  if depth == 0 then
  ( spaces pad ) Text .++ "\"A\"" Text .++ ( spaces pad )
  else
  ( spaces pad ) Text .++ ( Char . toText letter ) Text .++ ( spaces ((( depth Nat .- 1 ) Nat .* 2 ) Nat .+ 1 )) Text .++ ( Char . toText letter ) Text .++ ( spaces pad )
diamond . arr : Char -> [ Char ] -> [ Char ]
diamond . arr letter list =
  List . takeWhile ( l -> l <= letter ) list
diamond . spaces : Nat -> Text
diamond . spaces n = Text . leftPad n "\" \"" "\"\""
