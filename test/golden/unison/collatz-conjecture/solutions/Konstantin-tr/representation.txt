(file "collatz.u")
steps : Nat -> Optional Nat
steps =
  steps . recursive 0
steps . recursive : Nat -> Nat -> Optional Nat
steps . recursive counter =
  use Nat
  cases
    x
      | x == 0 -> None
      | x == 1 -> Some counter
      | Nat . isEven x -> steps . recursive ( counter + 1 ) ( x / 2 )
      | Nat . isOdd x -> steps . recursive ( counter + 1 ) ( 3 * x + 1 )
    _ -> None
> steps 1000000
