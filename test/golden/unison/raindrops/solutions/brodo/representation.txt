raindrops . convert : Nat -> Text
raindrops . convert n =
  mapping = Map . fromList [ ( 3 , "\"Pling\"" ), ( 5 , "\"Plang\"" ), ( 7 , "\"Plong\"" ) ]
  textList = List . map ( x -> if ( mod n x ) == 0 then ( Map . get x mapping ) else None ) [ 3 , 5 , 7 ]
  folder = acc x -> match x with
    None -> acc
    Some y -> acc Text .++ y
  resultText = List . foldLeft folder "\"\"" textList
  if resultText == "\"\"" then Nat . toText n else resultText
> convert 28
