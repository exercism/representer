(file "raindrops.u")
raindrops . convert : Nat -> Text
raindrops . convert n =
  raindrop : Nat -> Nat -> Text -> Optional Text
  raindrop n f drop =
    if Nat . mod n f == 0 then
      Some drop
    else
      None
  match List . filterMap ( uncurry <| raindrop n ) [( 3 , "\"Pling\"" ), ( 5 , "\"Plang\"" ), ( 7 , "\"Plong\"" )] with
  drops
    | List . isEmpty drops -> Nat . toText n
    | otherwise -> List . foldRight ( Text .++ ) "\"\"" drops
