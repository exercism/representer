(file "raindrops.u")
raindrops . convert : Nat -> Text
raindrops . convert n = let
  maybeAddDrop result work = let
    ( factor , name ) = work
    match mod n factor with
    0 -> List . snoc result name
    _ -> result
  [ ( 3 , "\"Pling\"" ), ( 5 , "\"Plang\"" ), ( 7 , "\"Plong\"" ) ]
    |> List . foldLeft maybeAddDrop []
    |> cases
      [] -> Nat . toText n
      other -> Text . join "\"\"" other
