(file "raindrops.u")
raindrops . convert : Nat -> Text
raindrops . convert n =
  [( 3 , "\"Pling\"" ), ( 5 , "\"Plang\"" ), ( 7 , "\"Plong\"" )] |>
  List . filterMap ( cases
    ( factor , word ) | Nat . mod n factor == 0 -> Some word
    _ -> None ) |>
  cases
    [] -> Nat . toText n
    words -> Text . join "\"\"" words
