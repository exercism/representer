(file "raindrops.u")
raindrops . convert : Nat -> Text
raindrops . convert n =
  [( 3 , "\"Pling\"" ), ( 5 , "\"Plang\"" ), ( 7 , "\"Plong\"" )]
    |> List . filterMap cases
      ( f , sound ) | ( Nat . mod n f ) == 0 -> Some sound
      _ -> None
    |> cases
      [] -> Nat . toText n
      sounds -> Text . join "\"\"" sounds
