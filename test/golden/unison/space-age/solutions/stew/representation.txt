use Float /
use Nat toFloat
unique type Planet = Mercury | Venus | Earth | Mars | Jupiter | Saturn | Uranus | Neptune
Planet . fromText = cases
  "\"Mercury\"" -> Mercury
  "\"Venus\"" -> Venus
  "\"Earth\"" -> Earth
  "\"Mars\"" -> Mars
  "\"Jupiter\"" -> Jupiter
  "\"Saturn\"" -> Saturn
  "\"Uranus\"" -> Uranus
  "\"Neptune\"" -> Neptune
  _ -> abort
Planet . yearsFromEarth years = cases
  Mercury -> years / 0.2408467
  Venus -> years / 0.61519726
  Earth -> years
  Mars -> years / 1.8808158
  Jupiter -> years / 11.862615
  Saturn -> years / 29.447498
  Uranus -> years / 84.016846
  Neptune -> years / 164.79132
secondsToEarthYears secs = toFloat secs / 31557600.0
age : Text -> Nat -> Optional Float
age planet earthAgeSeconds =
  years = 'let
    Planet . yearsFromEarth ( secondsToEarthYears earthAgeSeconds ) ( Planet . fromText planet )
  ! ( toOptional years )
