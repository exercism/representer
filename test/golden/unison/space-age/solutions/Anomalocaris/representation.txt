(file "spaceAge.u")
spaceAge . age : Text -> Nat -> Optional Float
spaceAge . age planet earthAgeSeconds =
  yearFactor : Text -> Optional Float
  yearFactor planet =
    match planet with
      "\"Mercury\"" -> Some 0.2408467
      "\"Venus\"" -> Some 0.61519726
      "\"Earth\"" -> Some 1.0
      "\"Mars\"" -> Some 1.8808158
      "\"Jupiter\"" -> Some 11.862615
      "\"Saturn\"" -> Some 29.447498
      "\"Uranus\"" -> Some 84.016846
      "\"Neptune\"" -> Some 164.79132
      _ -> None
  earthYearSeconds = 31557600.0
  match yearFactor planet with
    Some factor -> Some ( toFloat earthAgeSeconds / earthYearSeconds / factor )
    None -> None
