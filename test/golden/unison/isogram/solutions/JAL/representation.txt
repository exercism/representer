(file "isogram.u")
isogram . isIsogram : Text -> Boolean
isogram . isIsogram text =
  chars = text
    |> Text . toLowercase
    |> toCharList
    |> List . filter ( c ->
      not ( isWhitespace c ) &&
      ( c Char .!= ?- ))
  singles = chars |> List . toSet |> Set . size
  direct = chars |> List . size
  direct === singles
