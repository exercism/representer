multiples : Nat -> Nat -> Set Nat
multiples upperBound factor =
  List . range 1 (( upperBound + factor - 1 ) / factor )
    |> map ( a -> a * factor )
    |> toSet
sumOfMultiples . sumOfMultiples : [ Nat ] -> Nat -> Nat
sumOfMultiples . sumOfMultiples targets upperBound =
  filter ( a -> a != 0 ) targets
    |> map ( multiples upperBound )
    |> unions
    |> Set . foldLeft ( a -> b -> a Nat .+ b ) 0
