sumOfMultiples . sumOfMultiples : [ Nat ] -> Nat -> Nat
sumOfMultiples . sumOfMultiples targets upperBound =
  fixedTargets = List . filter ( i -> Nat . neq i 0 ) targets
  multipleInList : Nat -> [ Nat ] -> Boolean
  multipleInList i list = List . any ( e -> mod i e == 0 ) list
  r = Nat . range 0 upperBound
  List . foldLeft ( acc i -> if multipleInList i fixedTargets then acc Nat .+ i else acc ) 0 r
