use Nat div
use List head +: size
use Optional getOrElse
change . findFewestCoins : [ Nat ] -> Nat -> Optional [ Nat ]
change . findFewestCoins inputCoins target =
  inputCoinsSorted = sortDescending inputCoins
  maxCoin = getOrElse 1 ( head inputCoinsSorted )
  incrementMaxCoinsUntilSomeUpToLimit maxCoins limit =
    if maxCoins > limit then None else
      res = findFewestCoinsGo [] inputCoinsSorted target maxCoins
      if isSome res then res else incrementMaxCoinsUntilSomeUpToLimit ( increment maxCoins ) limit
  incrementMaxCoinsUntilSomeUpToLimit ( div target maxCoin ) 15
change . findFewestCoinsGo : [ Nat ] -> [ Nat ] -> Nat -> Nat -> Optional [ Nat ]
change . findFewestCoinsGo change inputCoins target maxCoins = match inputCoins with
  [] -> None
  thisCoin +: rest | sum change == target -> Some change
    | sum change > target -> None
    | size change == maxCoins -> None
    | sum change + (( maxCoins - size change ) * thisCoin ) < target
      -> None
    | otherwise ->
    result = findFewestCoinsGo ( thisCoin +: change ) inputCoins target maxCoins
    if isSome result then result
    else findFewestCoinsGo change rest target maxCoins
