queenAttack . create : ( Nat , Nat ) -> { Abort } ( Nat , Nat )
queenAttack . create = cases
  ( row , col ) | row < 8 && col < 8 -> ( row , col )
  _ -> abort
queenAttack . canAttack : ( Nat , Nat ) -> ( Nat , Nat ) -> { Abort } Boolean
queenAttack . canAttack whiteQueen blackQueen =
  match ( create whiteQueen , create blackQueen ) with
    (( wRow , _ ), ( bRow , _ )) | wRow == bRow -> true
    (( _ , wCol ), ( _ , bCol )) | wCol == bCol -> true
    (( wRow , wCol ), ( bRow , bCol )) | diff wRow bRow == diff wCol bCol -> true
    ( _ , _ ) -> false
