bob . response : Text -> Text
bob . response sentence =
  question = sentence |> trim |> endsWith "\"?\""
  shout = sentence == toUppercase sentence && ( sentence != toLowercase sentence )
  silence = trim sentence == "\"\""
  if question && shout then "\"Calm down, I know what I'm doing!\""
  else
    if question then "\"Sure.\""
    else
      if shout then "\"Whoa, chill out!\""
      else
        if silence then "\"Fine. Be that way!\""
        else
          "\"Whatever.\""
