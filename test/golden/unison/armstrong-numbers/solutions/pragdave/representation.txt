zero = Char . toNat ? 0
armstrongNumbers . isArmstrongNumber : Nat -> Boolean
armstrongNumbers . isArmstrongNumber number = let
  nstr = Nat . toText number
  len = Text . size nstr
  nstr
    |> Text . toCharList
    |> map ( ch -> Char . toNat ch - zero )
    |> map ( d -> Nat . pow d len )
    |> sum
    |> ( == ) number
> armstrongNumbers . isArmstrongNumber 153
