use base
armstrongNumbers . isArmstrongNumber : Nat -> Boolean
armstrongNumbers . isArmstrongNumber num =
  lenNum : Nat
  lenNum = Optional . getOrElse 0 ( Float . toNat ( floor ( logBase 10.0 ( toFloat num )))) + 1
  use Nat
  go : Nat -> Nat -> Nat
  go acc num = match ( num / 10 , mod num 10 ) with
    ( rest , n ) | rest == 0 -> ( pow n lenNum + acc )
    ( rest , n ) -> go ( pow n lenNum + acc ) rest
  go 0 num == num
