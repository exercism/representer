structural type zipper . Tree a
  = Branch a ( Optional ( zipper . Tree a )) ( Optional ( zipper . Tree a ))
structural ability zipper . Zipper t a where
  setValue : a -> { zipper . Zipper t a } ()
  setLeft : t a -> { zipper . Zipper t a } ()
  setRight : t a -> { zipper . Zipper t a } ()
  left : { zipper . Zipper t a } ()
  right : { zipper . Zipper t a } ()
  up : { zipper . Zipper t a } ()
  focus : { zipper . Zipper t a } a
zipper . Zipper . handlerImpl : Tree a -> Request { Zipper Tree a } r -> { Store [ Tree a -> Tree a ]} ( r , Tree a )
zipper . Zipper . handlerImpl tree request =
  ( Branch val left right ) = tree
  push = Store . modify << ( List .+: )
  match request with
    { r } -> ( r , tree )
    { setValue newVal -> r } ->
      handle ! r with handlerImpl ( Branch newVal left right )
    { setLeft newBranch -> r } ->
      handle ! r with handlerImpl ( Branch val ( Some newBranch ) right )
    { setRight newBranch -> r } ->
      handle ! r with handlerImpl ( Branch val left ( Some newBranch ))
    { left -> r } ->
      newRoot = match left with
        Some subtree ->
          push ( t -> Branch val ( Some t ) right )
          subtree
        None -> tree
      handle ! r with handlerImpl newRoot
    { right -> r } ->
      newRoot = match right with
        Some subtree ->
          push ( t -> Branch val left ( Some t ))
          subtree
        None -> tree
      handle ! r with handlerImpl newRoot
    { up -> r } ->
      newRoot = match Store . get with
        rewind +: rest ->
          Store . put rest
          rewind tree
        _ -> tree
      handle ! r with handlerImpl newRoot
    { focus -> r } ->
      handle r val with handlerImpl tree
zipper . Zipper . run : Tree a -> ' { Zipper Tree a } r -> r
zipper . Zipper . run tree zipperFunction =
  runWithTree tree zipperFunction |> at1
zipper . Zipper . runWithTree : Tree a -> ' { Zipper Tree a } r -> ( r , Tree a )
zipper . Zipper . runWithTree tree zipperFunction =
  withInitialValue [] ' ( handle ! zipperFunction with handlerImpl tree )
zipper . testTree : Tree Nat
zipper . testTree = Branch 1 ( Some ( Branch 2 None None )) ( Some ( Branch 3 None None ))
zipper . newRoot : ' { Zipper Tree Nat } Nat
zipper . newRoot _ =
  Zipper . right
  setValue 400
  Zipper . up
  setValue 500
  setLeft testTree
  focus
zipper . exampleProgram : ( Nat , Tree Nat )
zipper . exampleProgram = runWithTree testTree newRoot
