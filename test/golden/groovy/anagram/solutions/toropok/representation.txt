class Anagram {
def subject
def subject_map
def public Anagram ( String subject ){
this . subject = subject . toLowerCase ()
this . subject_map = fq_map ( this . subject )
}
Map fq_map ( String s ){
def res = [:]
s . each { letter -> res [ letter ] = res . get ( letter , 0 ) + 1 }
res . sort ()*. key
return res
}
def find ( candidates ) {
List anagrams = []
candidates . each { word ->
if ( word . toLowerCase () == subject ) return
if ( subject_map . equals ( fq_map ( word . toLowerCase ()))) anagrams << word
}
return anagrams
}
}
