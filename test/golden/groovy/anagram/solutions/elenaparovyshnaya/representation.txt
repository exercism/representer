(file "src/main/groovy/Anagram.groovy")
class Anagram {
private final Decomposed subject
Anagram ( String subject ) {
this . subject = new Decomposed ( subject )
}
List < String > find ( List < String > candidates ) {
candidates . findAll { new Decomposed ( it ). anagramOf ( subject ) }
}
private static class Decomposed {
private final Map < String , Integer > content
private final String source
Decomposed ( String input ) {
source = input . toLowerCase ()
content = source . toList ()
. groupBy { it }
. collectEntries { k , v -> [( k ): v . size ()] }
}
boolean anagramOf ( Decomposed origin ) {
( source != origin . source ) && ( content == origin . content )
}
}
}
