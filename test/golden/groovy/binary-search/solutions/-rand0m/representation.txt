import groovy.transform.TailRecursive
class BinarySearch {
List data
BinarySearch ( List data ) {
this . data = data
}
@TailRecursive
private int index ( d , item , offset ) {
if ( d . size () == 1 )
return item in d ? offset : - 1
int n = d . size () / 2
if ( d [ n ] > item )
return index ( d [ 0 ..( n - 1 )], item , offset )
else
return index ( d [ n ..- 1 ], item , offset + n )
}
int indexOf ( item ) {
data ? index ( data , item , 0 ) : - 1
}
}
