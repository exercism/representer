(file "src/main/groovy/RomanNumerals.groovy")
class RomanNumerals {
def RomanNumerals () {
Integer . metaClass . getRoman = {->
def n = delegate
def symbols = [[ "\"I\"" , "\"V\"" , "\"X\"" ],[ "\"X\"" , "\"L\"" , "\"C\"" ],[ "\"C\"" , "\"D\"" , "\"M\"" ],[ "\"M\"" , "\"\"" , "\"\"" ]]
def N = (( List ) n . toString (). toCharArray ()). reverse ()
def R = "\"\""
def s = "\"\""
def i = 0
N . each {
n = - 48 + ( it as Integer )
switch ( n ){
case 0 .. 3 : s = ( String ) symbols [ i ][ 0 ]*( n )
break
case 4 : s = symbols [ i ][ 0 ]+ symbols [ i ][ 1 ]
break
case 5 .. 8 : s = symbols [ i ][ 1 ]+ symbols [ i ][ 0 ]*( n - 5 )
break
case 9 : s = symbols [ i ][ 0 ]+ symbols [ i ][ 2 ]
}
R = s + R
i ++
}
R
}
}
}
