(file "src/main/groovy/RomanNumerals.groovy")
class RomanNumerals {
def romanNumbersMap = [
1000 : "\"M\"" ,
900 : "\"CM\"" ,
500 : "\"D\"" ,
400 : "\"CD\"" ,
100 : "\"C\"" ,
90 : "\"XC\"" ,
50 : "\"L\"" ,
40 : "\"XL\"" ,
10 : "\"X\"" ,
9 : "\"IX\"" ,
5 : "\"V\"" ,
4 : "\"IV\"" ,
1 : "\"I\""
]
RomanNumerals () {
Integer . metaClass . getRoman = {
def roman = new StringBuilder ()
def number = delegate
romanNumbersMap . each { decimalValue , romanValue ->
while ( number >= decimalValue ) {
number -= decimalValue
roman << romanValue
}
}
roman . toString ()
}
}
}
