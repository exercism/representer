class Proverb {
static recite ( strings ) {
String tmpMsg = '' ;
if ( strings . size () == 0 ) {
return tmpMsg ;
}
if ( strings . size () > 1 ) {
for ( int i = 1 ; i < strings . size (); i ++) {
tmpMsg = tmpMsg . concat ( "\"For want of a ${strings[i-1]} the ${strings[i]} was lost.\\n\"" );
}
}
tmpMsg = tmpMsg . concat ( "\"And all for the want of a ${strings[0]}.\"" );
return tmpMsg ;
}
}
