class Sieve {
static primes ( Integer limit ) {
if ( limit < 2 ) return []
def candidates = ( 2 .. limit ). collectEntries { n -> [ n , true ] }
( 2 .. limit ). inject ([]) { primes , n ->
def multiplicand = 2
while ( n * multiplicand <= limit ) {
candidates . put ( n * multiplicand , false )
multiplicand ++
}
if ( candidates . get ( n )) primes << n
primes
}
}
}
