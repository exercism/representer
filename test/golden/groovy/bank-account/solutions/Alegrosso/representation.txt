class BankAccount {
private int balance
boolean isOpen
@Synchronized
void open () {
if ( isOpen ) throw new UnsupportedOperationException ()
balance = 0
isOpen = true
}
@Synchronized
void close () {
if (! isOpen ) throw new UnsupportedOperationException ()
balance = 0
isOpen = false
}
@Synchronized
void deposit ( int amount ) {
if (! isOpen ) throw new UnsupportedOperationException ()
if ( amount < 0 ) throw new IllegalArgumentException ()
balance += amount
}
@Synchronized
void withdraw ( int amount ) {
if (! isOpen ) throw new UnsupportedOperationException ()
if ( amount < 0 || amount > balance ) throw new IllegalArgumentException ()
balance -= amount
}
@Synchronized
int getBalance () {
if (! isOpen ) throw new UnsupportedOperationException ()
return balance
}
}
