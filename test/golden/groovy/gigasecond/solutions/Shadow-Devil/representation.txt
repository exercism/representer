private static final Duration gigasecond = Duration . ofSeconds ( 1 _000_000_000 )
static LocalDateTime add ( moment ){
if ( moment instanceof LocalDate ){
moment = moment . atStartOfDay ()
}
moment + gigasecond
}
}
