class Queen {
def row
def column
Queen ( int row , int column ) {
this . row = row
this . column = column
if ( this . row < 0 || this . row > 7 || this . column < 0 || this . column > 7 )
throw new Exception ()
}
}
class QueenAttack {
static canAttack ( Queen q1 , Queen q2 ) {
if ( q1 . row == q2 . row || q1 . column == q2 . column )
return true
if ( Math . abs ( q1 . row - q2 . row ) == Math . abs ( q1 . column - q2 . column ))
return true
else
return false
}
}
