class Luhn {
static valid ( String value ) {
if (!( value ==~ /[\d\s]*/ )) return false
if (!( value ==~ /.*\d.*\d.*/ )) return false
value . toList ()
. findAll { it ==~ /\d/ }
. collect { it . toInteger ()}
. reverse ()
. withIndex ()
. collect { item , index -> index % 2 == 1 ? item * 2 : item }
. collect { it > 9 ? it - 9 : it }
. sum () % 10 == 0
}
}
