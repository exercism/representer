(file "src/main/groovy/Bob.groovy")
class Bob {
static yelling ( it ) {
it == it ?. toUpperCase ()&& it ?. toUpperCase () != it ?. toLowerCase ()
}
static question ( it ) {
it ?. trim ()?. endsWith ( "\"?\"" )
}
static nothing ( it ) {
! it ?. trim ()
}
static response ( input ) {
switch ( input ) {
case { yelling ( it ) && question ( it )}:
return "\"Calm down, I know what I'm doing!\""
case { yelling ( it )}:
return "\"Whoa, chill out!\""
case { question ( it )}:
return "\"Sure.\""
case { nothing ( it )}:
return "\"Fine. Be that way!\""
default:
"\"Whatever.\""
}
}
}
