(file "src/main/groovy/Bob.groovy")
class Bob {
static response ( input ) {
String inputModify = input . replaceAll (~ "\"[^a-zA-Z\\\\r\\\\n\\\\t\\\\f\\\\v]\"" , '' )
Boolean isQuestion = input . trim (). endsWith ( '?' )
Boolean isUpper = inputModify ? inputModify . toCharArray (). every { it . isUpperCase ()} : false
Boolean isSpace = input . toCharArray (). every { it . isWhitespace ()}
if ( isQuestion && isUpper )
return "'Calm down, I know what I\\'m doing!'"
if ( isQuestion )
return 'Sure.'
if ( isUpper )
return "'Whoa, chill out!'"
if ( isSpace )
return "'Fine. Be that way!'"
return 'Whatever.'
}
}
