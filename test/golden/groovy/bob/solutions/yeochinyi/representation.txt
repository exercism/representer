class Bob {
static response ( input ) {
if ( input == null || input . trim ()== '' ){
return "'Fine. Be that way!'"
}
switch ( input . trim ()){
case ~ /^[0-9\W]+\?$/ :
return "\"Sure.\""
case ~ /^[A-Z0-9\W]+\?$/ :
return "\"Calm down, I know what I'm doing!\""
case ~ /^[A-Za-z0-9\W]+\?$/ :
return "\"Sure.\""
case ~ /^[0-9\W]+$/ :
return "\"Whatever.\""
case ~ /^[A-Z0-9\W]+$/ :
return "\"Whoa, chill out!\""
default:
return "\"Whatever.\""
}
}
static void main ( args ){
println Bob . response ( args [ 0 ])
}
}
