class Triangle {
private float a
private float b
private float c
private boolean isValidTriangle
Triangle ( a , b , c ) {
this . a = a
this . b = b
this . c = c
isValidTriangle = doesNotViolateTriangleInequality ()
&& allGreaterThanZero ()
}
boolean isEquilateral () {
isValidTriangle && ( a == b && a == c )
}
boolean isIsosceles () {
isValidTriangle && ( a == b || a == c || b == c )
}
boolean isScalene () {
isValidTriangle && ( a != b && a != c && b != c )
}
private boolean allGreaterThanZero () {
a > 0 && b > 0 && c > 0
}
private boolean doesNotViolateTriangleInequality () {
a + b >= c && b + c >= a && c + a >= b
}
}
