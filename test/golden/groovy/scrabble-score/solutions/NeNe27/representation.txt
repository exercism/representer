class ScrabbleScore {
static scoreWord ( String word ) {
def one = [ 'A' , 'E' , 'I' , 'O' , 'U' , 'L' , 'N' , 'R' , 'S' , 'T' ]
def two = [ 'D' , 'G' ]
def three = [ 'B' , 'C' , 'M' , 'P' ]
def four = [ 'F' , 'H' , 'V' , 'W' , 'Y' ]
def five = [ 'K' ]
def eight = [ 'J' , 'X' ]
def ten = [ 'Q' , 'Z' ]
int score = 0
for ( i in word . toUpperCase ()){
if ( i in one ){
score += 1
}
else if ( i in two ){
score += 2
}
else if ( i in three ){
score += 3
}
else if ( i in four ){
score += 4
}
else if ( i in five ){
score += 5
}
else if ( i in eight ){
score += 8
}
else if ( i in ten ){
score += 10
}
}
return score
}
}
