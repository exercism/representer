(file "src/main/groovy/ScrabbleScore.groovy")
class ScrabbleScore {
static scoreWord ( String word ) {
word . toUpperCase (). findAll (). collect {
switch ( it ) {
case ~ /[AEIOULNRST]/ : return 1
case ~ /[DG]/ : return 2
case ~ /[BCMP]/ : return 3
case ~ /[FHVWY]/ : return 4
case 'K' : return 5
case ~ /[JX]/ : return 8
case ~ /[QZ]/ : return 10
default: 0
}
}. sum ( 0 )
}
}
