class ScrabbleScore {
static scoreWord ( String word ) {
def value1 = [ 'A' , 'E' , 'I' , 'O' , 'U' , 'L' , 'N' , 'R' , 'S' , 'T' ]
def value2 = [ 'D' , 'G' ]
def value3 = [ 'B' , 'C' , 'M' , 'P' ]
def value4 = [ 'F' , 'H' , 'V' , 'W' , 'Y' ]
def value5 = [ 'K' ]
def value8 = [ 'J' , 'X' ]
def value10 = [ 'Q' , 'Z' ]
int totalPoints
for ( letter in word ){
switch ( letter ){
case value1 + value1 *. toLowerCase ():
totalPoints += 1
break
case value2 + value2 *. toLowerCase ():
totalPoints += 2
break
case value3 + value3 *. toLowerCase ():
totalPoints += 3
break
case value4 + value4 *. toLowerCase ():
totalPoints += 4
break
case value5 + value5 *. toLowerCase ():
totalPoints += 5
break
case value8 + value8 *. toLowerCase ():
totalPoints += 8
break
case value10 + value10 *. toLowerCase ():
totalPoints += 10
break
}
}
totalPoints
}
}
