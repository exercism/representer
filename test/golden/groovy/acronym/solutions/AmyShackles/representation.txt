(file "src/main/groovy/Acronym.groovy")
class Acronym {
static String abbreviate ( String phrase ) {
def words = phrase . replace ( '-' , "\" \"" ). replace ( "\"_\"" , "\"\"" ). split ( "' '" );
def acronym = "\"\"" ;
for ( word in words ) {
if ( word . length () > 0 ) {
acronym += word [ 0 ]. toUpperCase ();
}
}
return acronym ;
}
}
