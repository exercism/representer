class Raindrops {
def convert ( num ) {
String result = '' ;
if ( num % 3 == 0 ){
result = result + 'Pling' ;
}
if ( num % 5 == 0 ){
result = result + 'Plang' ;
}
if ( num % 7 == 0 ){
result = result + 'Plong' ;
}
if ( result == '' ){
return num as String ;
} else {
return result ;
}
}
static void main ( String [] args ){
println ( new Raindrops (). convert ( 3 ));
}
}
