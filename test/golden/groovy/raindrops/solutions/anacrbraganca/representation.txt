class Raindrops {
def convert ( num ) {
def String result = ''
if ( num % 3 != 0 && num % 5 != 0 && num % 7 != 0 ) {
return Integer . toString ( num )
}
if ( num % 3 == 0 ) {
result = result + 'Pling'
}
if ( num % 5 == 0 ) {
result = result + 'Plang'
}
if ( num % 7 == 0 ) {
result = result + 'Plong'
}
return result
}
}
