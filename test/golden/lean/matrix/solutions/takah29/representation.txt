namespace Matrix
def getRowList ( xs : String ) : List ( List String ) :=
  xs . splitToList ( · == '\ n ' ) |>. map ( · . splitOn )
def row ( xs : String ) ( n : Nat ) : List Nat :=
  ( getRowList xs )[ n - 1 ] ! |> . map String . toNat !
def column ( xs : String ) ( n : Nat ) : List Nat :=
  ( getRowList xs ) . map ( · [ n - 1 ] !. toNat ! )
end Matrix
