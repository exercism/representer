(file "Series.lean")
namespace Series
def slices ( series : String ) ( sliceLength : Nat ) : Option ( Array String ) :=
  let len := series . length
  if sliceLength == 0 || len == 0 || sliceLength > len then
    none
  else
    let count := len - sliceLength + 1
    some ( Array . ofFn ( n := count ) fun i =>
      series . extract ( series . pos ! ⟨ i . val ⟩ ) ( series . pos ! ⟨ i . val + sliceLength ⟩ ))
end Series
