namespace RotationalCipher
private
def baseLower : UInt32 := ' a ' . val
private
def baseUpper : UInt32 := ' A ' . val
def rotate ( shiftKey : UInt32 ) ( text : String ) : String :=
  text . map ( fun ch =>
    if ch . isAlpha then
      let ord := if ch . isLower then
        ( ch . val - baseLower + shiftKey ) % 26 + baseLower
      else
        ( ch . val - baseUpper + shiftKey ) % 26 + baseUpper
      Char . ofNat ord . toNat
    else
      ch )
end RotationalCipher
