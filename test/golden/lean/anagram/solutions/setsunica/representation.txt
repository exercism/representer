import Std . Data
namespace Anagram
def findAnagrams ( subject : String ) ( candidates : List String ) : List String :=
  let subject ' := subject . toLower
  let subjectHm := toHashMap subject '
  candidates . filter λ cand =>
    let cand ' := cand . toLower
    cand ' != subject ' && toHashMap cand ' == subjectHm
  where
    toHashMap s := Id . run do
      let mut hm : Std . HashMap Char Nat := {}
      for c in s do
        hm := hm . alter c λ
          | some n => some $ n + 1
          | none => some 1
      return hm
end Anagram
