namespace SpaceAge
inductive Planet where
  | Mercury
  | Venus
  | Earth
  | Mars
  | Jupiter
  | Saturn
  | Uranus
  | Neptune
private def earthYearInSeconds : Float := 31557600 .0
@ [ inline ]
private def inv ( orbitalYears : Float ) : Float :=
  1 .0 / ( earthYearInSeconds * orbitalYears )
@ [ inline ]
private def invSecondsPerPlanetYear ( planet : Planet ) : Float :=
  match planet with
  | . Mercury => inv 0 .2408467
  | . Venus => inv 0 .61519726
  | . Earth => inv 1 .0
  | . Mars => inv 1 .8808158
  | . Jupiter => inv 11 .862615
  | . Saturn => inv 29 .447498
  | . Uranus => inv 84 .016846
  | . Neptune => inv 164 .79132
@ [ inline ]
def age ( planet : Planet ) ( seconds : Nat ) : Float :=
  seconds . toFloat * invSecondsPerPlanetYear planet
end SpaceAge
