namespace Strain
class Partition (α : Type ) ( C : Type ) where
  keep : (α → Bool ) → C → C
  discard : (α → Bool ) → C → C
def keep ( p : α → Bool ) : List α → List α
  | [] => []
  | x :: xs => if p x then x :: keep p xs else keep p xs
def discard ( p : α → Bool ) : List α → List α
  | [] => []
  | x :: xs => if p x then discard p xs else x :: discard p xs
instance : Partition α ( List α) where
  keep := Strain . keep
  discard := Strain . discard
end Strain
