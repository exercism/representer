namespace Diamond
def rows ( letter : Char ) : List String :=
  let getChar ( i : Nat ) := Char . ofNat ( ' A ' . toNat + i )
  let rec loop ( n : Nat ) : List String :=
    match n with
    | 0 => [ "\"A\"" ]
    | n + 1 =>
      let subRows := loop n
      let c := getChar ( n + 1 )
      let maxRow := s ! "\"{c}{\"\".pushn ' ' (2 * n + 1)}{c}\""
      subRows . map ( fun s => s ! "\" {s} \"" ) |>. concat maxRow
  loop ( letter . toUpper . toNat - ' A ' . toNat ) |> fun l => l ++ l . reverse . drop 1
end Diamond
