namespace ArmstrongNumbers
private def numDigits ( number : Nat ) : Nat := Id . run do
  if number == 0 then return 1
  let mut num := number
  let mut res := 0
  while num > 0 do
    num := num . div 10
    res := res + 1
  return res
#guard numDigits 5 == 1
#guard numDigits 50 == 2
#guard numDigits 100 == 3
def isArmstrongNumber ( number : Nat ) : Bool := Id . run do
  let len := numDigits number
  let mut num := number
  let mut sum := 0
  while num > 0 do
    let n := num . mod 10
    sum := sum + n ^ len
    num := num . div 10
  return number == sum
#guard isArmstrongNumber 4
end ArmstrongNumbers
