namespace IsbnVerifier
def isValidList ( isbn : String ) : Bool :=
  let digits := isbn . toList . filter ( · != ' - ' )
  if digits . length != 10 then false
  else
    let result := digits . foldl ( fun ( acc : Nat × Nat × Bool ) c =>
      let ( sum , idx , ok ) := acc
      let weight := 10 - idx
      if ok then
        if idx < 9 then
          if c . isDigit then ( sum + weight * ( c . toNat - ' 0 ' . toNat ), idx + 1 , true )
          else ( sum , idx + 1 , false )
        else
          if c . isDigit then ( sum + weight * ( c . toNat - ' 0 ' . toNat ), idx + 1 , true )
          else if c = ' X ' then ( sum + weight * 10 , idx + 1 , true )
          else ( sum , idx + 1 , false )
      else ( sum , idx + 1 , false )
    ) ( 0 , 0 , true )
    let ( sum , _ , ok ) := result
    ok && sum % 11 == 0
def isValid ( isbn : String ) : Bool :=
  let result := isbn . foldl ( fun ( acc : UInt32 × UInt32 × Bool ) c =>
    let ( sum , count , ok ) := acc
    if c == ' - ' then acc
    else
      let count ' := count + 1
      if ! ok || count >= 10 then ( sum , count ' , false )
      else
        let weight : UInt32 := 10 - count
        if count < 9 then
          if c . isDigit then ( sum + weight * ( c . val - ' 0 ' . val ), count ' , true )
          else ( sum , count ' , false )
        else
          if c . isDigit then ( sum + ( c . val - ' 0 ' . val ), count ' , true )
          else if c == ' X ' then ( sum + 10 , count ' , true )
          else ( sum , count ' , false )
  ) ( 0 , 0 , true )
  let ( sum , count , ok ) := result
  ok && count == 10 && sum % 11 == 0
end IsbnVerifier
