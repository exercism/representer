namespace FlattenArray
inductive Box (α : Type ) : Type where
  | zero
  | one ( value : α)
  | many ( boxes : Array ( Box α))
def flatten ( box : Box Int ) : Array Int :=
  let rec recur array box := match box with
    | . zero => array
    | . one val => array . push val
    | . many boxes => boxes . foldl recur array
  recur Array . empty box
end FlattenArray
