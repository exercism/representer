namespace PhoneNumber
def clean ( phrase : String ) : Option String :=
  let digits := phrase . toList . filter Char . isDigit
  let base := match digits with
    | ' 1 ' :: rest => rest
    | _ => digits
  match base with
  | [ n1 , _ , _ , n4 , _ , _ , _ , _ , _ , _ ] =>
    if n1 >= ' 2 ' && n4 >= ' 2 ' then some ( String . mk base ) else none
  | _ => none
end PhoneNumber
