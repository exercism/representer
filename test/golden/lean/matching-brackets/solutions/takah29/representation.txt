namespace MatchingBrackets
private def validate ( xs : List Char ) ( stack : List Char := []) : Bool :=
  match xs , stack with
  | ' ( ' :: xs ' , _ => validate xs ' ( ' ) ' :: stack )
  | ' [ ' :: xs ' , _ => validate xs ' ( ' ] ' :: stack )
  | '{' :: xs ' , _ => validate xs ' ( '}' :: stack )
  | x :: xs ' , y :: stack ' => if x == y then validate xs ' stack ' else false
  | [], [] => true
  | _ , _ => false
def isPaired ( value : String ) : Bool :=
  let brackets := value . toList . filter ( "\"()[]{}\"" . contains · )
  validate brackets
end MatchingBrackets
