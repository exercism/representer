namespace Isogram
def isIsogram ( s : String ) : Bool :=
  let rec loop ( cs : List Char ) ( seen : List Char ) : Bool :=
    match cs with
    | [] => true
    | c :: rest =>
      if c == ' ' || c == ' - ' then
        loop rest seen
      else
        let lc := c . toLower
        if seen . contains lc then
          false
        else
          loop rest ( lc :: seen )
  loop s . toList []
end Isogram
