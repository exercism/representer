import Std . Data . HashMap
namespace WordCount
def countWords ( sentence : String ) : Std . HashMap String Nat :=
  let words := sentence . toLower |>. split ( fun c => ! c . isAlphanum && c != '\'' )
  words . fold ( init := {} ) fun acc w =>
    let w := w . dropPrefix '\'' |>. dropSuffix '\''
    if w . isEmpty then
      acc
    else
      let word := w . toString
      acc . insert word ( acc . getD word 0 + 1 )
end WordCount
