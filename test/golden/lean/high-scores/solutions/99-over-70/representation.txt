namespace HighScores
def latestList ( scores : List Nat ) : Nat :=
  scores . getLast ? . getD 0
def latestArray ( scores : Array Nat ) : Nat :=
  scores . back ? . getD 0
def personalBestList ( scores : List Nat ) : Nat :=
  scores . foldl max 0
def personalBestList ' ( scores : List Nat ) : Nat :=
  scores . max ? . getD 0
def personalBestList '' ( scores : List Nat ) : Nat :=
  scores
  |>. mergeSort ( · > · )
  |>. getD 0 0
def personalBestList ''' ( scores : List Nat ) : Nat := Id . run do
  let mut max_score := 0
  for score in scores do
    if score >= max_score then
      max_score := score
  return max_score
def personalBestArray ( scores : Array Nat ) : Nat :=
  scores . foldl max 0
def personalBestArray ' ( scores : Array Nat ) : Nat :=
  scores . max ? . getD 0
def personalBestArray '' ( scores : Array Nat ) : Nat :=
  scores
  |>. qsort ( · > · )
  |>. getD 0 0
def personalTopThreeList ( scores : List Nat ) : List Nat :=
  ( scores . mergeSort ( · > · )) . take 3
def personalTopThreeArray ( scores : Array Nat ) : Array Nat :=
  ( scores . qsort ( · > · )) . take 3
end HighScores
