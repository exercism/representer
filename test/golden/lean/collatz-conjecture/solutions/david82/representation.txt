namespace CollatzConjecture
def Positive := { x : Nat // 0 < x }
private partial def stepsU64 ( n : UInt64 ) ( acc : Nat ) : Nat :=
  if n ≤ 1 then acc
  else if n &&& 1 == 0 then stepsU64 ( n >>> 1 ) ( acc + 1 )
  else stepsU64 ( n * 3 + 1 ) ( acc + 1 )
def steps ( n : Positive ) : Nat :=
  stepsU64 n . val . toUInt64 0
end CollatzConjecture
