namespace QueenAttack
structure Queen where
  row : Int
  col : Int
  h : row ≥ 0 ∧ row < 8 ∧ col ≥ 0 ∧ col < 8
  deriving Repr , BEq
def create ? ( row col : Int ) : Option Queen :=
  if h : 0 ≤ row ∧ row < 8 ∧ 0 ≤ col ∧ col < 8 then
    some ⟨ row , col , h ⟩
  else
    none
def canAttack ( white black : Queen ) : Bool :=
  white . row == black . row ||
  white . col == black . col ||
  ( white . row - black . row ) . natAbs == ( white . col - black . col ) . natAbs
end QueenAttack
