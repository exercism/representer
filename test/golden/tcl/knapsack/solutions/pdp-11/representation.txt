proc maximumValue { W is } {
set l [ llength $is ]
set m [ lrepeat [ expr { $l +1 }] [ lrepeat [ expr { $W+1 }] 0 ]]
for { set i 0 } { $i <$l } { incr i } {
set item [ lindex $is $i ]
set iw [ dict get $item weight ]
set iv [ dict get $item value ]
for { set w 0 } { $w <= $W } { incr w } {
set v1 [ lindex $m $i $w ]
if { $w < $iw } {
lset m $i+1 $w $v1
} else {
set v2 [ lindex $m $i $w-$iw ]
lset m $i+1 $w [ expr { max ( $v1 , $iv+$v2 )}]
}
}
}
lindex $m $l $W
}
