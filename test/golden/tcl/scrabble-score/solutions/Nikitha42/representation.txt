variable scores {
{ A E I O U L N R S T } 1
{ D G } 2
{ B C M P } 3
{ F H V W Y } 4
{ K } 5
{ J X } 8
{ Q Z } 10
}
variable scoresDict [ dict create ]
foreach { chars score } $scores {
foreach char $chars {
dict set scoresDict $char $score
}
}
proc scrabbleScore { word } {
set sum 0
foreach c [ split [ string toupper $word ] "\"\"" ] {
incr sum [ dict get $::scoresDict $c ]
}
return $sum
}
