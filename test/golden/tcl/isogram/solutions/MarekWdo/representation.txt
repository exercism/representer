proc isIsogram { input } {
set lowInput [ string tolower $input ]
for { set i 0 } { $i < [ string length $input ]} { incr i } {
set letter1 [ string index $lowInput $i ]
if {[ string is wordchar $letter1 ]} {
for { set j 0 } { $j < [ string length $input ]} { incr j } {
set letter2 [ string index $lowInput $j ]
if { $i != $j } {
if { $letter1 == $letter2 } {
return false
}
}
}
}
}
return true
}
