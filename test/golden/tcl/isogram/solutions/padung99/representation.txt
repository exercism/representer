proc isIsogram { input } {
set char [ dict create ]
foreach ch [ split $input "\"\"" ] {
if { $ch != "\"-\"" } { dict incr char [ string tolower $ch ] }
}
puts "\"length " [ llength $char ] ", " $char "\""
set len [ expr [ llength $char ] /2 ]
set flag 0
set cnt 0
foreach i $char {
if { $i == "\" \"" } { break } else {
if {[ expr $cnt%2 ] ==1 } {
if { $i != 1 } { set flag 1 }
}
}
incr cnt
}
if { $flag == 0 } { return "\"true\"" } else { return "\"false\"" }
}
