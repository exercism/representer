proc isIsogram { input } {
set input [ string tolower $input ]
array set mapping {
a 0
b 0
c 0
d 0
e 0
f 0
g 0
h 0
i 0
j 0
k 0
l 0
m 0
n 0
o 0
p 0
q 0
r 0
s 0
t 0
u 0
v 0
w 0
x 0
y 0
z 0
}
set ans 1
for { set pos 0 } {[ expr $pos < [ string length $input ]]} { incr pos } {
set val [ string index $input $pos ]
if {[ string is alpha $val ]} {
if {[ expr $mapping ( $val ) == 1 ]} {
set ans 0
break
}
set mapping ( $val ) 1
}
}
return $ans
}
