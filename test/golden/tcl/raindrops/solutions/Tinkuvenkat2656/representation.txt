(file "raindrops.tcl")
proc raindrops { number } {
set result {}
if ([ expr $number % 3 == 0 ]) {
lappend result "\"Pling\""
}
if ([ expr $number % 5 == 0 ]) {
lappend result "\"Plang\""
}
if ([ expr $number % 7 == 0 ]) {
lappend result "\"Plong\""
}
if ([ expr [ llength $result ] == 0 ]) {
return $number
} else {
return [ join $result "\"\"" ]
}
}
