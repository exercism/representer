(file "raindrops.tcl")
proc raindrops { number } {
set result "\"\""
if { [ expr $number % 3 ] eq 0 } {
append result "\"Pling\""
}
if { [ expr $number % 5 ] eq 0 } {
append result "\"Plang\""
}
if { [ expr $number % 7 ] eq 0 } {
append result "\"Plong\""
}
if { $result eq "\"\"" } {
append result $number
}
return $result
}
