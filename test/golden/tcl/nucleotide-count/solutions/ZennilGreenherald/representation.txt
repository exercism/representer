proc nucleotideCounts { strand } {
set counts [ dict create "\"A\"" 0 "\"C\"" 0 "\"G\"" 0 "\"T\"" 0 ]
foreach c [ split $strand "\"\"" ] {
if {[ string first $c "\"ACGT\"" ] == -1 } {
throw FOO "\"Invalid nucleotide in strand\""
}
dict set counts $c [ expr { 1 + [ dict get $counts $c ]}]
}
return $counts
}
