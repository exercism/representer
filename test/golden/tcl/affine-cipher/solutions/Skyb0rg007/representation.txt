proc affine_encode { phrase a b } {
modular_inverse $a 26
binary scan $phrase c* bytes
set result {}
foreach byte $bytes {
if { 65 <= $byte && $byte <= 90 } {
incr byte 32 ; # Convert uppercase to lowercase
}
if { 97 <= $byte && $byte <= 122 } {
if {[ llength $result ] % 6 == 5 } {
lappend result 32
}
lappend result [ expr {( $a * ( $byte - 97 ) + $b ) % 26 + 97 }]
} elseif { 48 <= $byte && $byte <= 57 } {
if {[ llength $result ] % 6 == 5 } {
lappend result 32
}
lappend result $byte
}
}
binary format c* $result
}
proc affine_decode { phrase a b } {
set a_inv [ modular_inverse $a 26 ]
binary scan $phrase c* bytes
set result {}
foreach byte $bytes {
if { 97 <= $byte && $byte <= 122 } {
lappend result [ expr { $a _inv * ( $byte - 97 - $b ) % 26 + 97 }]
} elseif { 48 <= $byte && $byte <= 57 } {
lappend result $byte
}
}
binary format c* $result
}
proc modular_inverse { a m } {
set t 0
set newt 1
set r $m
set newr $a
while { $newr != 0 } {
set quotient [ expr { $r / $newr }]
set tmp $newt
set newt [ expr { $t - $quotient * $newt }]
set t $tmp
set tmp $newr
set newr [ expr { $r - $quotient * $newr }]
set r $tmp
}
if { $r > 1 } {
return -code error "\"a and m must be coprime.\""
}
set t
}
