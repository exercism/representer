proc abbreviate { phrase } {
# clean special caracters
regsub -all { ' } $phrase "\"\"" phrase
regsub -all { \W } $phrase "\" \"" phrase
regsub -all { _ } $phrase "\" \"" phrase
set phrase [ string toupper $phrase ]
#puts $phrase
set words [ split $phrase "\" \"" ]
set words [ lsearch -all -inline -not -exact $phrase {}]
#puts $words
set abb "\"\""
#set firstLetters [ lmap x $words { string index $x 0 } ]
foreach word $words {
set first [ string index $word 0 ]
set abb "\"" $abb$first "\""
}
#puts $abb
return $abb
}
