(file "resistor-color.tcl")
namespace eval resistorColor {
proc colorCode { args } {
array set colors {
black 0
brown 1
red 2
orange 3
yellow 4
green 5
blue 6
violet 7
grey 8
white 9
}
if [ catch { set result $colors ( $args )}] {
error "\"Invalid color: " $args "\""
} else {
set result $colors ( $args )
return $result
}
}
proc colors { args } {
return "\"black brown red orange yellow green blue violet grey white\"" ;
}
}
