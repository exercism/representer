(file "resistor-color.tcl")
namespace eval resistorColor {
variable COLORS [ dict create "\\\n" black 0 "\\\n" brown 1 "\\\n" red 2 "\\\n" orange 3 "\\\n" yellow 4 "\\\n" green 5 "\\\n" blue 6 "\\\n" violet 7 "\\\n" grey 8 "\\\n" white 9 "\\\n" ]
proc colorCode { args } {
variable COLORS
if {[ dict exists $COLORS $args ]} {
return [ dict get $COLORS $args ]
}
error "\"Invalid color: " $args "\""
}
proc colors { args } {
variable COLORS
dict keys $COLORS
}
}
