(file "resistor-color.tcl")
namespace eval resistorColor {
proc colorCode { args } {
set colorList [ colors ]
set colorListLength [ llength $colorList ]
set code -1
for { set index 0 } { $index < $colorListLength } { incr index } {
set color [ lindex $colorList $index ]
if { $color == $args } {
set code $index
set index 15
}
}
if { $code == -1 } {
throw { ERROR } "\"Invalid color: " $args "\""
}
return $code
}
proc colors { args } {
return { "\"black\"" "\"brown\"" "\"red\"" "\"orange\"" "\"yellow\"" "\"green\"" "\"blue\"" "\"violet\"" "\"grey\"" "\"white\"" }
}
}
