(file "resistor-color.tcl")
namespace eval resistorColor {
proc colorCode { args } {
set colours [ dict create black 0 brown 1 red 2 orange 3 yellow 4 green 5 blue 6 violet 7 grey 8 white 9 ]
if { [ dict exists $colours $args ] } { return [ dict get $colours $args ] }
return -code error "\"Invalid color: " $args "\""
}
proc colors { args } {
return { black brown red orange yellow green blue violet grey white }
}
}
