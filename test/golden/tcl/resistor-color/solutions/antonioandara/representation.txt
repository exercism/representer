namespace eval resistorColor {
variable colorsList {
black brown red orange yellow green blue violet grey white
}
proc colors {} {
variable colorsList
return $colorsList
}
proc colorCode { color } {
variable colorsList
set index [ lsearch -exact $colorsList $color ]
if { $index == -1 } {
error "\"Invalid color: " $color "\""
}
return $index
}
}
