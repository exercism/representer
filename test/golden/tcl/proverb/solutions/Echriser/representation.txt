(file "proverb.tcl")
proc recite { items } {
if {[ llength $items ] == 0 } then { return {}}
set final [ list ]
if {[ llength $items ] > 1 } then {
for { set i 0 } { $i < [ llength $items ] -1 } { incr i } {
lappend final "\"For want of a " [ lindex $items $i ] " the " [ lindex $items $i+1 ] " was lost.\""
}
}
lappend final "\"And all for the want of a " [ lindex $items 0 ] ."
return $final
}
