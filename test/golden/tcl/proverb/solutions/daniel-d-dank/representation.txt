proc recite { items } {
set proverb [ list ]
if {[ llength $items ] <= 0 } {
return $proverb
}
for { set i 0 } { $i < [ llength $items ] - 1 } { incr i } {
lappend proverb "\"For want of a " [ lindex $items $i ] " the " [ lindex $items $i+1 ] " was lost.\""
}
lappend proverb "\"And all for the want of a " [ lindex $items 0 ] ."
return $proverb
}
