proc heyBob { input } {
if {[ regexp { ^ \s *$ } $input ]} {
return "\"Fine. Be that way!\""
}
set q [ regexp { \?\s *$ } $input ]
set nolower [ regexp { ^ [ ^a-z ] *$ } $input ]
set hasupper [ regexp {[ A-Z ]} $input ]
if { $q && $nolower && $hasupper } {
return "\"Calm down, I know what I'm doing!\""
} elseif { $q } {
return "\"Sure.\""
} elseif { $nolower && $hasupper } {
return "\"Whoa, chill out!\""
} else {
return "\"Whatever.\""
}
}
