(file "resistor-color-duo.tcl")
namespace eval ::resistorColor {
proc value { first second args } {
set resCol [ dict create black 0 brown 1 red 2 orange 3 yellow 4 green 5 blue 6 violet 7 grey 8 white 9 ]
foreach color [ list $first $second ] {
if { ! [ dict exists $resCol $color ] } {
return -code error -errorinfo 1 "\"Invalid color*\""
}
}
return -code ok -errorinfo 0 [ expr { [ dict get $resCol $first ] *10 + [ dict get $resCol $second ] }]
}
}
