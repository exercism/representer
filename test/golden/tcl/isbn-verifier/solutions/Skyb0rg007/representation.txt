(file "isbn-verifier.tcl")
proc isValid { isbn } {
set sum 0
set i 0
foreach c [ split [ regsub -all {[ - ] + } $isbn "\"\"" ] "\"\"" ] {
if {[ string is digit $c ]} {
incr sum [ expr { $c * ( 10 - $i )}]
} elseif { $i == 9 && $c eq "\"X\"" } {
incr sum 10
} else {
return false
}
incr i
}
if { $i != 10 } {
return false
}
expr { $sum % 11 == 0 }
}
