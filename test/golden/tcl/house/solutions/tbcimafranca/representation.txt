proc recite { from to } {
set subjects [ list "\\\n" "\"house that Jack built\"" "\\\n" "\"malt\"" "\\\n" "\"rat\"" "\\\n" "\"cat\"" "\\\n" "\"dog\"" "\\\n" "\"cow with the crumpled horn\"" "\\\n" "\"maiden all forlorn\"" "\\\n" "\"man all tattered and torn\"" "\\\n" "\"priest all shaven and shorn\"" "\\\n" "\"rooster that crowed in the morn\"" "\\\n" "\"farmer sowing his corn\"" "\\\n" "\"horse and the hound and the horn\"" ]
set actions [ list "\\\n" "\"\"" "\\\n" "\"lay in\"" "\\\n" "\"ate\"" "\\\n" "\"killed\"" "\\\n" "\"worried\"" "\\\n" "\"tossed\"" "\\\n" "\"milked\"" "\\\n" "\"kissed\"" "\\\n" "\"married\"" "\\\n" "\"woke\"" "\\\n" "\"kept\"" "\\\n" "\"belonged to\"" ]
set verses [ list ]
for { set verse $from } { $verse <= $to } { incr verse } {
set index [ expr { $verse - 1 }]
set sentence "\"This is the " [ lindex $subjects $index ] "\""
for { set i $index } { $i > 0 } { incr i -1 } {
append sentence "\" that " [ lindex $actions $i ] " the " [ lindex $subjects [ expr { $i - 1 }]] "\""
}
append sentence "\".\""
lappend verses $sentence
}
return [ join $verses "\"\\n\"" ]
}
