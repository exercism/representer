proc hammingDistance { left right } {
if {[ string length $left ] == [ string length $right ]} {
set sum 0
for { set ind 0 } { $ind < [ string length $left ]} { incr ind } {
if {[ string index $left $ind ] != [ string index $right $ind ] } {
incr sum
}
}
} else {
error "\"strands must be of equal length\""
}
return $sum
}
