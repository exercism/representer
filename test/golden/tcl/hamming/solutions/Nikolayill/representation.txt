(file "hamming.tcl")
proc hammingDistance { left right } {
if {[ string length $left ] != [ string length $right ]} {
error "\"strands must be of equal length\""
}
set diffCount 0
set leftChars [ split $left {}]
set rightChars [ split $right {}]
set maxLen [ llength $rightChars ]
for { set i 0 } { $i < $maxLen } { incr i } {
puts "\"" [ lindex $leftChars $i ] " to " [ lindex $rightChars $i ] "\""
if {[ lindex $leftChars $i ] != [ lindex $rightChars $i ]} {
incr diffCount
}
}
return $diffCount
}
