proc bracketsMatch { input } {
if { $input == "\"\"" } { return true }
set brackets [ list ]
set openB [ list \{ \[ \( ]
set closeB [ list \} \] \) ]
foreach char [ split $input "\"\"" ] {
if { $char in $openB } {
lappend brackets $char
} elseif { $char in $closeB } {
set clsB [ lindex $brackets end ]
if {( $char == "\"\\}\"" && $clsB == "\"\\{\"" ) || ( $char == "\"\\]\"" && $clsB == "\"\\[\"" ) || ( $char == "\"\\)\"" && $clsB == "\"\\(\"" )} {
set brackets [ lreplace $brackets end end ]
} else {
return false
}
}
}
return [ expr [ llength $brackets ] ==0 ]
}
