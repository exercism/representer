oo::class create HighScores {
variable scores
constructor {} {
set scores {}
}
method addScores { args } {
set scores [ concat $scores $args ]
}
method scores {} {
return $scores
}
method latest {} {
return [ lindex $scores end ]
}
method personalBest {} {
return [ lindex [ lsort -integer -decreasing $scores ] 0 ]
}
method topThree {} {
lrange [ lsort -integer -decreasing $scores ] 0 2
}
}
