oo::class create Node {
variable datum
variable next
constructor { value } {
set datum $value
set next "\"\""
}
method datum {} {
return $datum
}
method next { args } {
if {[ llength $args ] > 0 } {
set next $args
}
return $next
}
}
oo::class create SimpleLinkedList {
variable head
constructor {{ values {}}} {
set head [ Node new "\"\"" ]
foreach value $values {
my push [ Node new $value ]
}
}
method head {} {
return $head
}
method push { args } {
set temphead $args
$temphead next $head
set head $temphead
return [ self ]
}
method pop {} {
if {[ $head datum ] eq "\"\"" } { return "\"\"" }
set oldhead $head
set head [ $head next ]
return $oldhead
}
method listnode {} {
set nodelist [ list ]
set temphead $head
while { 1 == 1 } {
#if current head datum is "\"\"" , we have reach last node. So break out
#before adding the empty head to nodelist
if {[ $temphead datum ] eq "\"\"" } {
break
}
lappend nodelist $temphead ; #add the current node to nodelist
set temphead [ $temphead next ]; #change current node to next node
}
return $nodelist
}
method foreach { var scr } {
foreach node [ my listnode ] {
# go to caller level. Tell the caller to set $var = ( current node in listnode ) then run the expression to multiply node datum by 2
uplevel 1 "\"set " $var " " $node "; " $scr "\""
}
}
method toList {} {
set datumlist [ list ]
foreach node [ my listnode ] {
lappend datumlist [ $node datum ]
}
return $datumlist
}
