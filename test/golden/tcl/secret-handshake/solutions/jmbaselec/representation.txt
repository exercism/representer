(file "secret-handshake.tcl")
proc secretHandshake { code } {
set output [ list ]
if { $code & 1 } { lappend output "\"wink\"" }
if { $code & 2 } { lappend output "\"double blink\"" }
if { $code & 4 } { lappend output "\"close your eyes\"" }
if { $code & 8 } { lappend output "\"jump\"" }
if { $code & 16 } { return [ lreverse $output ]}
return $output
}
