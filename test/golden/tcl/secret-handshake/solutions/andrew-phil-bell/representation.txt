(file "secret-handshake.tcl")
proc secretHandshake { code } {
set field_codes [ list 0 "\"wink\"" 1 "\"double blink\"" 2 "\"close your eyes\"" "\\\n" 3 "\"jump\"" 4 "\"reverse\"" ]
set hshake [ list ]
foreach { bfield behav } $field_codes {
set mask [ expr ( 1 << $bfield )]
if { ( $mask & $code ) > 0 } {
lappend hshake $behav
}
}
if { [ lindex $hshake end ] eq "\"reverse\"" } {
return [ lreverse [ lrange $hshake 0 end-1 ]]
}
return $hshake
}
