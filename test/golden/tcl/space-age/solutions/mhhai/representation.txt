proc onEarth { age } {
return [ expr { double ( $age ) / 31557600 }]
}
proc onMercury { age } {
return [ expr { double ( $age ) / ( 0.2408467 * 31557600 )}]
}
proc onVenus { age } {
return [ expr { double ( $age ) / ( 0.61519726 * 31557600 )}]
}
proc onMars { age } {
return [ expr { double ( $age ) / ( 1.8808158 * 31557600 )}]
}
proc onJupiter { age } {
return [ expr { double ( $age ) / ( 11.862615 * 31557600 )}]
}
proc onSaturn { age } {
return [ expr { double ( $age ) / ( 29.447498 * 31557600 )}]
}
proc onUranus { age } {
return [ expr { double ( $age ) / ( 84.016846 * 31557600 )}]
}
proc onNeptune { age } {
return [ expr { double ( $age ) / ( 164.79132 * 31557600 )}]
}
proc onSun { age } {
error "\"not a planet\""
}
