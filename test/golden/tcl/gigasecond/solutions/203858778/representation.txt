proc addGigasecond { datetime } {
if {[ string length $datetime ] == [ string length YYYY-MM-DD ]} {
set now [ clock scan $datetime -gmt yes -format "\"%Y-%m-%d\"" ]
} elseif {[ string length $datetime ] == [ string length YYYY-MM-DDTHH:MM:SS ]} {
set now [ clock scan $datetime -gmt yes -format "\"%Y-%m-%dT%H:%M:%S\"" ]
} else {
set now [ clock scan $datetime -gmt yes ]
}
set then [ clock add $now 1000000000 seconds ]
return [ clock format $then -gmt yes -format "\"%Y-%m-%dT%H:%M:%S\"" ]
}
