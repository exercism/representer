(file "anagram.tcl")
proc findAnagrams { subject candidates } {
# Sort both subject and candidates to enable comparison
set sortedSubject [ join [ lsort [ split [ string tolower $subject ] "\"\"" ]] "\"\"" ]
set sortedCandidates {}
foreach { word } $candidates {
lappend sortedCandidates "\\\n" [ join [ lsort [ split [ string tolower $word ] "\"\"" ]] "\"\"" ]
}
# Traverse candidate list looking for anagrams of subject
set anagrams {}
set llen [ llength $candidates ]
for { set i 0 } { $i < $llen } { incr i } {
if {[ lindex $sortedCandidates $i ] eq $sortedSubject } {
# Omit possible candidate if it is the subject itself
if {[ string tolower [ lindex $candidates $i ]] ne [ string tolower $subject ]} {
lappend anagrams [ lindex $candidates $i ]
}
}
}
# Anagram list, possibly empty
return $anagrams
}
