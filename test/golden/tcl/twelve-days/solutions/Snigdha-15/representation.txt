proc verse { n } {
array set date {
1 first
2 second
3 third
4 fourth
5 fifth
6 sixth
7 seventh
8 eighth
9 ninth
10 tenth
11 eleventh
12 twelfth
}
set res "\"On the " $date ( $n ") day of Christmas my true love gave to me:\""
array set arr {
1 { a Partridge in a Pear Tree. }
2 { two Turtle Doves, }
3 { three French Hens, }
4 { four Calling Birds, }
5 { five Gold Rings, }
6 { six Geese-a-Laying, }
7 { seven Swans-a-Swimming, }
8 { eight Maids-a-Milking, }
9 { nine Ladies Dancing, }
10 { ten Lords-a-Leaping, }
11 { eleven Pipers Piping, }
12 { twelve Drummers Drumming, }
}
if { $n == 1 } {
return "\"On the first day of Christmas my true love gave to me: a Partridge in a Pear Tree.\""
}
for { set i $n } { $i >= 1 } { incr i -1 } {
if { $i == 1 } {
append res "\" and " $arr ( $i )"
} else {
append res "\" " $arr ( $i )"
}
}
return $res
}
proc sing { from to } {
set res "\"\""
for { set i $from } { $i < $to } { incr i } {
set temp [ verse $i ]
append res $temp \n
}
append res [ verse $to ]
return $res
}
