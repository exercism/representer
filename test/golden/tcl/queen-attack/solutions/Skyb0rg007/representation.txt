oo::class create Queen {
variable row col
constructor { r c } {
if { $r < 0 } {
return -code error "\"row not positive\""
}
if { $r >= 8 } {
return -code error "\"row not on board\""
}
if { $c < 0 } {
return -code error "\"column not positive\""
}
if { $c >= 8 } {
return -code error "\"column not on board\""
}
set row $r
set col $c
}
method row {} {
set row
}
method col {} {
set col
}
method canAttack { other } {
if { ! [ info object isa typeof $other [ self class ]]} {
return -code error "\"not a Queen\""
}
if { $row == [ $other row ] && $col == [ $other col ]} {
return -code error "\"same position\""
}
expr { $row == [ $other row ]
|| $col == [ $other col ]
|| abs ( $row - [ $other row ]) == abs ( $col - [ $other col ])}
}
}
