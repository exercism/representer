unit ProteinTranslation ;
interface
uses SysUtils ;
type
TStrArray = Array Of String ;
function proteins ( const strand : string ) : TStrArray ;
implementation
function protein ( const codon : string ) : string ;
begin
case codon of
'AUG' : result := 'Methionine' ;
'UUU' : result := 'Phenylalanine' ;
'UUC' : result := 'Phenylalanine' ;
'UUA' : result := 'Leucine' ;
'UUG' : result := 'Leucine' ;
'UCU' : result := 'Serine' ;
'UCC' : result := 'Serine' ;
'UCA' : result := 'Serine' ;
'UCG' : result := 'Serine' ;
'UAU' : result := 'Tyrosine' ;
'UAC' : result := 'Tyrosine' ;
'UGU' : result := 'Cysteine' ;
'UGC' : result := 'Cysteine' ;
'UGG' : result := 'Tryptophan' ;
else
raise Exception . Create ( "'Invalid codon'" );
end ;
end ;
function proteins ( const strand : string ) : TStrArray ;
var
index : integer ;
codon : string ;
begin
result := nil ;
index := 0 ;
while index < strand . length do
begin
if strand . length < index + 3 then
raise Exception . Create ( "'Invalid codon'" );
codon := copy ( strand , index + 1 , 3 );
case codon of
'UAA' : break ;
'UAG' : break ;
'UGA' : break ;
else
insert ( protein ( codon ), result , length ( result ));
end ;
inc ( index , 3 );
end ;
end ;
end .
