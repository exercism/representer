(file "Pangram.pas")
unit Pangram ;
interface
function IsPangram ( const Sentence : string ) : boolean ;
implementation
uses SysUtils ;
function IsPangram ( const Sentence : string ) : boolean ;
var
LowerSentence : String ;
Letters : Array [ 'a' .. 'z' ] of integer ;
i : char ;
j : integer ;
begin
LowerSentence := LowerCase ( Sentence );
for i := 'a' to 'z' do
Letters [ i ] := 0 ;
for j := 1 to Length ( LowerSentence ) do
if ( LowerSentence [ j ] >= 'a' ) and ( LowerSentence [ j ] <= 'z' ) then
Letters [ LowerSentence [ j ]] += 1 ;
result := true ;
for i := 'a' to 'z' do
if Letters [ i ] = 0 then
result := false
end ;
end .
