(file "Luhn.pas")
unit Luhn ;
interface
function valid ( const value : string ) : boolean ;
implementation
function valid ( const value : string ) : boolean ;
var
total : integer ;
count : integer ;
index : integer ;
current : integer ;
begin
result := true ;
total := 0 ;
count := 0 ;
for index := length ( value ) downto 1 do
begin
current := ord ( value [ index ]) - 48 ;
if ( current >= 0 ) and ( current <= 9 ) then
begin
count += 1 ;
if count mod 2 = 0 then
begin
current *= 2 ;
if current > 9 then
current -= 9 ;
end ;
total += current ;
end
else if current <> - 16 then
result := false ;
end ;
result := result and ( count >= 2 ) and ( total mod 10 = 0 );
end ;
end .
