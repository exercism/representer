(file "SaddlePoints.pas")
unit SaddlePoints ;
interface
type
TMatrix = array of array of integer ;
TPoint = record
row : integer ;
column : integer ;
end ;
TPoints = array of TPoint ;
function saddlePoints ( const matrix : TMatrix ): TPoints ;
function RowMax ( const matrix : TMatrix ; const r : integer ): integer ;
function ColMin ( const matrix : TMatrix ; const c : integer ): integer ;
implementation
uses SysUtils , Math ;
function RowMax ( const matrix : TMatrix ; const r : integer ): integer ;
var
c : integer ;
begin
Result := - 1 ;
for c := 0 to high ( matrix [ r ]) do
Result := max ( Result , matrix [ r ][ c ]);
end ;
function ColMin ( const matrix : TMatrix ; const c : integer ): integer ;
var
r : integer ;
begin
Result := maxint ;
for r := 0 to high ( matrix ) do
Result := min ( Result , matrix [ r ][ c ]);
end ;
function saddlePoints ( const matrix : TMatrix ): TPoints ;
var
point : TPoint ;
r , c : integer ;
begin
Result := nil ;
for r := 0 to high ( matrix ) do
for c := 0 to high ( matrix [ r ]) do
if ( matrix [ r ][ c ] = RowMax ( matrix , r )) and
( matrix [ r ][ c ] = ColMin ( matrix , c )) then
begin
point . row := r + 1 ;
point . column := c + 1 ;
SetLength ( Result , length ( Result ) + 1 );
Result [ high ( Result )] := point ;
end ;
end ;
end .
