unit ResistorColor ;
interface
type
TStrArray = Array Of String ;
function ColorCode ( const color : string ) : int8 ;
function colors : TStrArray ;
implementation
uses SysUtils ;
var
arr : array [ 0..9 ] of string = ( 'black' , 'brown' , 'red' , 'orange' , 'yellow' , 'green' , 'blue' , 'violet' , 'grey' , 'white' );
function ColorCode ( const color : string ) : int8 ;
var
i : int8 ;
begin
for i := 0 to 9 do
begin
if arr [ i ] = color then
begin
result := i ;
exit ;
end ;
end ;
end ;
function colors : TStrArray ;
begin
result := arr ;
end ;
end .
