unit KillerSudokuHelper ;
interface
type
TIntArray = array of integer ;
TIntArray2D = array of array of integer ;
TDigitSet = set of 1..9 ;
function combinations ( const sum , size : word ; const exclude : TIntArray ): TIntArray2D ;
implementation
uses SysUtils ;
procedure FindCombinations ( const sum , size : integer ; const exclude : TDigitSet ;
var combinations : TIntArray2D ; comb : TIntArray = nil ; prev : integer = 0 );
var
n : integer ;
begin
if size = 0 then
begin
if sum = 0 then
insert ( comb , combinations , length ( combinations ));
exit ;
end ;
for n := prev + 1 to 10 - size do
if not ( n in exclude ) then
begin
insert ( n , comb , length ( comb ));
FindCombinations ( sum - n , size - 1 , exclude , combinations , comb , n );
SetLength ( comb , length ( comb ) - 1 );
end ;
end ;
function combinations ( const sum , size : word ; const exclude : TIntArray ): TIntArray2D ;
var
n : integer ;
ExcludeSet : TDigitSet = [];
begin
Result := nil ;
for n in exclude do include ( ExcludeSet , n );
FindCombinations ( sum , size , ExcludeSet , Result );
end ;
end .
