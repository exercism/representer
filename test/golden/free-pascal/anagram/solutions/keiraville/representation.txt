unit Anagram ;
interface
type
TStrArray = Array Of String ;
function FindAnagrams (
const subject : string ;
const candidates : TStrArray
) : TStrArray ;
implementation
uses SysUtils ;
function IsAnagram (
const subject : string ;
const candidate : string
) : Boolean ;
var
tallies : array [ 0..25 ] of integer ;
index : integer ;
tally : integer ;
ch : char ;
begin
if candidate = subject then
exit ( false );
for index := 25 downto 0 do
tallies [ index ] := 0 ;
for ch in subject do
inc ( tallies [ ord ( ch ) - ord ( 'A' )]);
for ch in candidate do
dec ( tallies [ ord ( ch ) - ord ( 'A' )]);
for tally in tallies do
if tally <> 0 then
exit ( false );
result := true ;
end ;
function FindAnagrams (
const subject : string ;
const candidates : TStrArray
) : TStrArray ;
var
candidate : string ;
begin
result := nil ;
for candidate in candidates do
if IsAnagram ( UpperCase ( subject ), UpperCase ( candidate )) then
insert ( candidate , result , length ( result ));
end ;
end .
