unit CollatzConjecture ;
interface
function steps ( const number : integer ) : integer ;
implementation
uses SysUtils ;
function steps ( const number : integer ) : integer ;
var
i , k : integer ;
begin
i := 0 ;
K := number ;
if k <= 0 then
raise Exception . Create ( "'Only positive integers are allowed'" );
repeat
i := i + 1 ;
if k = 1 then
begin
k := 1 ;
i := 0 ;
end
else if k mod 2 = 0 then
begin
k := k div 2 ;
end
else begin
k := ( k * 3 ) + 1 ;
end ;
until k = 1 ;
result := i ;
end ;
end .
