(file "CollatzConjecture.pas")
unit CollatzConjecture ;
interface
function steps ( const number : integer ) : integer ;
implementation
uses SysUtils ;
function steps ( const number : integer ) : integer ;
var
n : Integer ;
begin
n := number ;
result := 0 ;
if n < 1 then
raise Exception . Create ( "'Only positive integers are allowed'" );
while n > 1 do
begin
if n mod 2 = 1 then
begin
n := n * 3 + 1 ;
result += 1 ;
end ;
n := n div 2 ;
result += 1 ;
end ;
end ;
end .
