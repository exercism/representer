unit AffineCipher ;
interface
const
MODBASE : integer = 26 ;
type
TCharArr = array of char ;
function encode ( const phrase : string ; const a , b : longword ): string ;
function decode ( const phrase : string ; const a , b : longword ): string ;
function GCD ( a , b : integer ): integer ;
function IsCoprime ( a , b : integer ): boolean ;
procedure ExtendedGCD ( a , b : integer ; out gcd , x , y : integer );
function ModularMultiplicativeInverse ( a , m : integer ): integer ;
function FloorDiv ( const a , m : integer ): integer ;
function PositiveMod ( const a , m : integer ): integer ;
function EncryptChar ( a , b : longword ; c : char ): char ;
function DecryptChar ( a , b : longword ; c : char ): char ;
function NormalizeInput ( phrase : string ): TCharArr ;
implementation
uses SysUtils ;
function NormalizeInput ( phrase : string ): TCharArr ;
var
c : char ;
begin
Result := nil ;
for c in LowerCase ( phrase ) do
if c in [ 'a' .. 'z' , '0' .. '9' ] then
begin
SetLength ( Result , length ( Result ) + 1 );
Result [ high ( Result )] := c ;
end ;
end ;
function FloorDiv ( const a , m : integer ): integer ;
begin
Result := a div m ;
if (( a mod m ) <> 0 ) and (( a < 0 ) <> ( m < 0 )) then
Dec ( Result );
end ;
function PositiveMod ( const a , m : integer ): integer ;
begin
Result := (( a mod m ) + m ) mod m ;
end ;
function GCD ( a , b : integer ): integer ;
var
temp : integer ;
begin
while b <> 0 do
begin
temp := b ;
b := a mod b ;
a := temp ;
end ;
Result := a ;
end ;
function IsCoprime ( a , b : integer ): boolean ;
begin
Result := GCD ( a , b ) = 1 ;
end ;
procedure ExtendedGCD ( a , b : integer ; out gcd , x , y : integer );
var
x1 , y1 : integer ;
begin
if b = 0 then
begin
gcd := a ;
x := 1 ;
y := 0 ;
end
else
begin
ExtendedGCD ( b , PositiveMod ( a , b ), gcd , x1 , y1 );
x := y1 ;
y := x1 - FloorDiv ( a , b ) * y1 ;
end ;
end ;
function ModularMultiplicativeInverse ( a , m : integer ): integer ;
var
gcd , xv , yv : integer ;
begin
gcd := 0 ;
xv := 0 ;
yv := 0 ;
ExtendedGCD ( a , m , gcd , xv , yv );
Result := PositiveMod ( xv , m );
end ;
function EncryptChar ( a , b : longword ; c : char ): char ;
begin
if not ( c in [ 'a' .. 'z' ]) then exit ( c );
Result := chr ( 97 + ((( a * ( Ord ( c ) - 97 )) + b ) mod MODBASE ));
end ;
function DecryptChar ( a , b : longword ; c : char ): char ;
var
mmi : integer ;
begin
if not ( c in [ 'a' .. 'z' ]) then exit ( c );
mmi := ModularMultiplicativeInverse ( a , MODBASE );
Result := chr ( 97 + PositiveMod (( mmi * ( Ord ( c ) - 97 - b )), MODBASE ));
end ;
function encode ( const phrase : string ; const a , b : longword ): string ;
var
c : char ;
i : integer ;
begin
if not IsCoprime ( a , MODBASE ) then
raise Exception . Create ( "'a and m must be coprime.'" );
Result := '' ;
i := 0 ;
for c in NormalizeInput ( phrase ) do
begin
Result += EncryptChar ( a , b , c );
Inc ( i );
if ( i mod 5 = 0 ) then Result += "' '" ;
end ;
Result := Trim ( Result );
end ;
function decode ( const phrase : string ; const a , b : longword ): string ;
var
c : char ;
begin
if not IsCoprime ( a , MODBASE ) then
raise Exception . Create ( "'a and m must be coprime.'" );
Result := '' ;
for c in NormalizeInput ( phrase ) do
Result += DecryptChar ( a , b , c );
end ;
end .
