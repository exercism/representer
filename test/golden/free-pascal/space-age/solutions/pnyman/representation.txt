unit SpaceAge ;
interface
function orbital ( const planet : string ): double ;
function age ( const planet : string ; const seconds : longint ): double ;
implementation
uses SysUtils ;
function orbital ( const planet : string ): double ;
begin
case ( planet ) of
'Mercury' : Result := 0.2408467 ;
'Venus' : Result := 0.61519726 ;
'Earth' : Result := 1.0 ;
'Mars' : Result := 1.8808158 ;
'Jupiter' : Result := 11.862615 ;
'Saturn' : Result := 29.447498 ;
'Uranus' : Result := 84.016846 ;
'Neptune' : Result := 164.79132 ;
else
raise Exception . Create ( "'not a planet'" );
end ;
end ;
function age ( const planet : string ; const seconds : longint ): double ;
var
earthyear : longint ;
begin
earthyear := 31557600 ;
Result := seconds / ( earthyear * orbital ( planet ));
end ;
end .
