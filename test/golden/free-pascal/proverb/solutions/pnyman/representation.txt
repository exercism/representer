(file "Proverb.pas")
unit Proverb ;
interface
function recite ( const strings : array of string ): string ;
implementation
uses SysUtils , Classes ;
function recite ( const strings : array of string ): string ;
var
i : integer ;
begin
Result := '' ;
if length ( strings ) = 0 then
exit ( Result );
for i := 0 to length ( strings ) - 2 do
Result += format ( "'For want of a %s the %s was lost.'" + # 10 , [ strings [ i ], strings [ i + 1 ]]);
Result += format ( "'And all for the want of a %s.'" , [ strings [ 0 ]]);
end ;
end .
