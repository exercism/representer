let factor = 10 ** Array . findIndex ( colors , String . equal ( _ , resistors [ 2 ] -> Option . getOr ( "\"black\"" )))
let sum = ( first -> BigInt . fromInt + second -> BigInt . fromInt ) * factor -> BigInt . fromInt
