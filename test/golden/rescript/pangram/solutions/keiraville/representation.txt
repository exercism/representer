let isPangram = ( sentence ) => {
let sentence = sentence -> String . toLowerCase
let seen = ref ( 0 )
for i in 0 to String . length ( sentence ) - 1 {
switch ( sentence -> String . charCodeAt ( i )) {
| Some ( code ) if code >= 97 && code <= 122 => seen : = seen . contents ||| ( 1 << ( code - 97 ))
| _ => ()
}
}
seen . contents == 0x3ffffff
}
