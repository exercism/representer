type planet =

let age = ( planet , ageInSeconds ) => {
let earthAge = Int . toFloat ( ageInSeconds ) / 60 . 0 / 60 . 0 / 24 . 0 / 365 . 0
let conversion = switch planet {
| Mercury => 0 . 2408467
| Venus => 0 . 61519726
| Earth => 1 . 0
| Mars => 1 . 8808158
| Jupiter => 11 . 862615
| Saturn => 29 . 447498
| Neptune => 164 . 79132
| Uranus => 84 . 016846
}
earthAge / conversion
}
