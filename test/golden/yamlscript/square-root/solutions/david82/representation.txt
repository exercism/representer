defn square-root(radicand):
  cond:
    radicand == 0: 0
    radicand == 1: 1
    else:
      loop [low 0 high radicand]:
        mid =: (low + high) / 2
        square =: mid * mid
        cond:
          square == radicand: mid
          abs(square - radicand) < 1: round(mid)
          square < radicand: recur(mid high)
          else: recur(low mid)
