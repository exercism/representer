defn is-silence(text):
  # First trim the string to remove leading/trailing whitespace
  trimmed =: text.trim()
  # Check if the trimmed string is empty or only contains whitespace
  =>: trimmed == "\"\""
defn is-shouting(text):
  # Check if there are any letters and if they're all uppercase
  has-letters =: text =~ /[A-Za-z]/
  is-uppercase =: text == uc(text)
  =>: has-letters && is-uppercase
defn is-question(text):
  # Check if trimmed text ends with "\"?\""
  =>: text.trim() =~ /\?$/
defn response(hey-bob):
  # Order matters here - check silence first
  cond:
    is-silence(hey-bob):
      "\"Fine." Be that way!"
    is-shouting(hey-bob) && is-question(hey-bob):
      "\"Calm" down, I know what I'm doing!"
    is-shouting(hey-bob):
      "\"Whoa," chill out!"
    is-question(hey-bob):
      "\"Sure.\""
    else:
      "\"Whatever.\""
