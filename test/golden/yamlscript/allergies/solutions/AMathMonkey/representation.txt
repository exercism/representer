!YS-v0
allergies =: qw(eggs peanuts shellfish strawberries tomatoes chocolate pollen cats)
defn allergic-to(item score): item.in?(score:list-allergens)
defn list-allergens(score):
  loop n score rest-v allergies acc []:
    cond:
      n.! || rest-v.!: acc
      n.bit-and(1).?: recur(n.bit-shift-right(1) rest-v:rest acc.conj(rest-v:first))
      else: recur(n.bit-shift-right(1) rest-v:rest acc)
