defn largest-product(digits span):
  span < 0 &&: die('span must not be negative')
  span > digits.# &&: die('span must be smaller than string length')
  digits =~ /\D/ &&: die('digits input must only contain digits')
  if span == 0:
    1
    digits.map(comp(I S))
      .partition(span 1 _)
      .map(partial(apply mul))
      .max(*)
