defn create(queen):
  row col =: queen.position.slice(q(row column))
  when row < 0: die('row not positive')
  when row > 7: die('row not on board')
  when col < 0: die('column not positive')
  when col > 7: die('column not on board')
  =>: 0
defn can-attack(white-queen black-queen):
  w-row w-col =: white-queen.position.slice(q(row column))
  b-row b-col =: black-queen.position.slice(q(row column))
  or:
    w-row ==: b-row
    w-col ==: b-col
    w-row.sub(b-row):abs ==: w-col.sub(b-col):abs
