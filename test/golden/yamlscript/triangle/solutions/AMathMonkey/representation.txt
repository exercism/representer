defn valid(sides):
  +[a b c] =: sides
  and: every?(pos? sides) (a + b).ge(c) (a + c).ge(b) (b + c).ge(a)
defn equilateral(sides):
  +[a b c] =: sides
  and: valid(sides) (a == b) (b == c)
defn isosceles(sides):
  +[a b c] =: sides
  and: valid(sides) or((a == b) (b == c) (a == c))
defn scalene(sides):
  +[a b c] =: sides
  and: valid(sides) (a != b) (b != c) (a != c)
