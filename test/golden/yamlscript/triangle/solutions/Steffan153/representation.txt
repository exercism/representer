defn is-valid(sides):
  -[a b c] =: sort(sides)
  =>: sides.every?(gt(0)) && ((a + b) >= c)
defn equilateral(sides):
  is-valid(sides) && (sides.distinct().# == 1)
defn isosceles(sides):
  is-valid(sides) && (sides.distinct().# <= 2)
defn scalene(sides):
  is-valid(sides) && (sides.distinct().# == 3)
