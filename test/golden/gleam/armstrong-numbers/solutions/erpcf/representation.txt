(file "src/armstrong_numbers.gleam"
  (source_file
    (import gleam/int)
    (import gleam/list)
    (function pub is_armstrong_number
      (function_parameters
        (function_parameter PLACEHOLDER_1 :
          (type Int))) ->
      (type Bool)
      (block
        (let_assert
          (record_pattern Ok
            (record_pattern_arguments
              (record_pattern_argument (variable PLACEHOLDER_2)))) =
          (function_call
            (field_access int . digits)
            (arguments
              (argument PLACEHOLDER_1)
              (argument 10))))
        (let (variable PLACEHOLDER_3) =
          (function_call
            (field_access list . length)
            (arguments
              (argument PLACEHOLDER_2))))
        (let (variable PLACEHOLDER_4) =
          (anonymous_function
            (function_parameters
              (function_parameter PLACEHOLDER_5))
            (block
              (function_call
                (field_access int . product)
                (arguments
                  (argument
                    (function_call
                      (field_access list . repeat)
                      (arguments
                        (argument PLACEHOLDER_5)
                        (argument PLACEHOLDER_3)))))))))
        (let (variable PLACEHOLDER_6) =
          (binary_expression
            (binary_expression PLACEHOLDER_2 |>
              (function_call
                (field_access list . map)
                (arguments
                  (argument PLACEHOLDER_4)))) |>
            (field_access int . sum)))
        (binary_expression PLACEHOLDER_1 == PLACEHOLDER_6)))))
