(file "src/bettys_bike_shop.gleam"
  (source_file
    (import gleam/float)
    (import gleam/int)
    (import gleam/string)
    (function pub pence_to_pounds
      (function_parameters
        (function_parameter PLACEHOLDER_1 :
          (type Int))) ->
      (type Float)
      (block
        (binary_expression
          (function_call
            (field_access int . to_float)
            (arguments
              (argument PLACEHOLDER_1))) /. 100.0)))
    (function pub pounds_to_string
      (function_parameters
        (function_parameter PLACEHOLDER_2))
      (block
        (binary_expression
          (binary_expression PLACEHOLDER_2 |>
            (function_call
              (field_access float . to_string)
              (arguments))) |>
          (function_call
            (field_access string . append)
            (arguments
              (argument to : "\u00A3"))))))))
