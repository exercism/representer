(file "triangle.8th")
: valid? \ n n n -- n n n T
3dup 0 > -rot 0 > -rot 0 > and and
!if false ;then
3dup n:+ >
if false ;then
3dup rot n:+ >
if false ;then
3dup -rot n:+ > not
;
: equilateral? \ n n n -- T
valid? !if 3drop false ;then
dup -3 roll n:= -rot n:= and
;
: isosceles? \ n n n -- T
valid? !if 3drop false ;then
over 3 pick 2 pick n:= -rot n:= or -rot n:= or
;
: scalene? \ n n n -- T
valid? !if 3drop false ;then
isosceles? not
;
