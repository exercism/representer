\ convert to a more useful representation. All score words will use a map of counts.
: empty-dice-map
m:new ( I 0 m:! ) 6 times
;
: count
empty-dice-map swap
( dup -rot m:@ n:1+ rot swap m:! ) a:each! drop
;
: score.n \ m n
tuck m:_@ n:*
;
: ones 1 score.n ;
: twos 2 score.n ;
: threes 3 score.n ;
: fours 4 score.n ;
: fives 5 score.n ;
: sixes 6 score.n ;
: foak
0 swap
( 3 n:> if >n 4 n:* nip else drop then ) m:each drop
;
: little.straight

;
: big.straight

;
: full.house
0 swap
( 3 n:= if >n 3 n:* nip else drop then ) m:each
over 0 n:= if drop ;then
0 swap

n:+
;
: choice
0 swap
( swap >n n:* n:+ ) m:each drop
;

m:vals 5 ' n:= a:indexof nip null? if 0 else 50 then nip nip
