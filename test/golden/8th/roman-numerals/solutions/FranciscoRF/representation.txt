900 r@ n:> !if "\"CM\"" s:+ -900 n:r+ ;then
500 r@ n:> !if "\"D\"" s:+ -500 n:r+ ;then
400 r@ n:> !if "\"CD\"" s:+ -400 n:r+ ;then
100 r@ n:> !if "\"C\"" s:+ -100 n:r+ ;then
90 r@ n:> !if "\"XC\"" s:+ -90 n:r+ ;then
50 r@ n:> !if "\"L\"" s:+ -50 n:r+ ;then
40 r@ n:> !if "\"XL\"" s:+ -40 n:r+ ;then
10 r@ n:> !if "\"X\"" s:+ -10 n:r+ ;then
9 r@ n:> !if "\"IX\"" s:+ -9 n:r+ ;then
5 r@ n:> !if "\"V\"" s:+ -5 n:r+ ;then
4 r@ n:> !if "\"IV\"" s:+ -4 n:r+ ;then
1 r@ n:> !if "\"I\"" s:+ -1 n:r+ ;then
) w:exec r@ while! rdrop
