\ Extract first letter of each word and convert to uppercase
: first-letter \ s -- s
0 1 s:slice s:uc ;
\ Check if string is non-empty: ( str -- bool )
: non-empty?
s:len nip 0 n:= not ;
\ Main acronym function: ( str -- str )
: acronym \ s -- s
s:lc \ Convert to lowercase first
/[^a-z0-9']+/ s:/ \ Split on non-alphanumeric chars (preserving apostrophes)
( non-empty? ) a:filter \ Remove empty strings
( first-letter ) a:map \ Get first letter of each word and uppercase
"\"\"" a:join \ Join all letters into final acronym
;
