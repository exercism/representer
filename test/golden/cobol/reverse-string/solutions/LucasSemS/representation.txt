IDENTIFICATION DIVISION .
  PROGRAM-ID . reverse-string .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-STRING PIC X ( 64 ).
  01 WS-RESULT PIC X ( 64 ).
  01 IND PIC 9 ( 2 ).
  PROCEDURE DIVISION .
  REVERSE-STRING .
    MOVE SPACES TO WS-RESULT .
    MOVE 64 TO IND .
    PERFORM UNTIL IND = 0
      STRING WS-RESULT ( 2 : 63 ) WS-STRING ( IND : 1 ) INTO WS-RESULT
      SUBTRACT 1 FROM IND
    END-PERFORM .
    MOVE FUNCTION TRIM ( WS-RESULT ) TO WS-STRING .
