(file "src/armstrong-numbers.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . armstrong-numbers .
  ENVIRONMENT DIVISION .
  CONFIGURATION SECTION .
  REPOSITORY . FUNCTION ALL INTRINSIC .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NUMBER PIC X ( 8 ).
  01 WS-NUMBER-ORIG PIC 9 ( 8 ).
  01 WS-NUMERIC PIC 9 ( 8 ).
  01 WS-RESULT PIC 9 .
  01 WS-SUM PIC 9 ( 18 ).
  01 WS-CURR-DIGIT PIC 9 ( 1 ).
  01 WS-COUNT PIC 9 ( 1 ).
  PROCEDURE DIVISION .
  ARMSTRONG-NUMBERS .
    MOVE 0 to WS-SUM .
    MOVE 0 to WS-COUNT .
    MOVE WS-NUMBER TO WS-NUMERIC .
    MOVE WS-NUMBER TO WS-NUMBER-ORIG .
    PERFORM ARMSTRONG-COUNT-DIGIT UNTIL WS-NUMERIC = 0 .
    MOVE WS-NUMBER TO WS-NUMERIC .
    PERFORM ARMSTRONG-ADD-DIGIT UNTIL WS-NUMERIC = 0 .
    IF WS-SUM = WS-NUMBER-ORIG THEN
      MOVE 1 to WS-RESULT
    ELSE
      MOVE 0 to WS-RESULT
    END-IF .
  ARMSTRONG-COUNT-DIGIT .
    DIVIDE WS-NUMERIC BY 10 GIVING WS-NUMERIC REMAINDER WS-CURR-DIGIT .
    ADD 1 TO WS-COUNT .
  ARMSTRONG-ADD-DIGIT .
    DIVIDE WS-NUMERIC BY 10 GIVING WS-NUMERIC REMAINDER WS-CURR-DIGIT .
    COMPUTE WS-SUM = WS-SUM + ( WS-CURR-DIGIT ** WS-COUNT ).
  ARMSTRONG-EXIT .
    EXIT .
