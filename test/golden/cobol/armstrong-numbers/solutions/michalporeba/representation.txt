IDENTIFICATION DIVISION .
  PROGRAM-ID . armstrong-numbers .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NUMBER PIC X ( 8 ).
  01 WS-NUMERIC PIC 9 ( 8 ).
  01 WS-INDEX PIC 9 .
  01 WS-DIGIT PIC 9 .
  01 WS-DIGITS PIC 9 .
  01 WS-SUM PIC 9 ( 8 ).
  01 WS-RESULT PIC 9 .
  PROCEDURE DIVISION .
  ARMSTRONG-NUMBERS .
  MOVE WS-NUMBER TO WS-NUMERIC .
  SET WS-SUM TO ZERO .
  MOVE FUNCTION LENGTH ( FUNCTION TRIM ( WS-NUMBER )) TO WS-DIGITS .
  PERFORM LOOP-PARA
    WITH TEST AFTER
    VARYING WS-INDEX
    FROM 1 BY 1
    UNTIL WS-INDEX = WS-DIGITS .
  IF WS-NUMERIC = WS-SUM THEN
    SET WS-RESULT TO 1
  ELSE
    SET WS-RESULT TO 0
  END-IF .
  END-RUN .
  LOOP-PARA .
  MOVE WS-NUMBER ( WS-INDEX : 1 ) TO WS-DIGIT .
  COMPUTE WS-SUM = WS-SUM + WS-DIGIT ** WS-DIGITS .
