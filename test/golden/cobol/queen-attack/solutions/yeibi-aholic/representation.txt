IDENTIFICATION DIVISION .
  PROGRAM-ID . QUEEN-ATTACK .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-QUEEN PIC X ( 9 ).
  01 WS-COORD .
    02 WS-ROW PIC S9 .
    02 WS-COL PIC S9 .
  01 WS-WHITE_QUEEN PIC X(9).
  01 WS-COORD-W .
    02 WS-ROW-W PIC S9 .
    02 WS-COL-W PIC S9 .
  01 WS-BLACK_QUEEN PIC X(9).
  01 WS-COORD-B .
    02 WS-ROW-B PIC S9 .
    02 WS-COL-B PIC S9 .
  01 WS-PROPERTY PIC X ( 11 ).
  01 WS-SLOPE PIC 99V99.
  01 WS-RESULT PIC 9 .
  PROCEDURE DIVISION .
  QUEEN-ATTACK .
    MOVE ZEROS TO WS-RESULT
    EVALUATE WS-PROPERTY
      WHEN 'create'
        UNSTRING WS-QUEEN DELIMITED BY ','
          INTO WS-ROW , WS-COL
        IF ( WS-ROW >= 0 AND <= 7 ) AND
          ( WS-COL >= 0 AND <= 7 )
          MOVE 1 TO WS-RESULT
        END-IF
      WHEN 'canAttack'
        UNSTRING WS-WHITE_QUEEN DELIMITED BY ','
          INTO WS-ROW-W , WS-COL-W
        UNSTRING WS-BLACK_QUEEN DELIMITED BY ','
          INTO WS-ROW-B , WS-COL-B
        IF ( WS-ROW-W = WS-ROW-B ) OR
          ( WS-COL-W = WS-COL-B )
          MOVE 1 TO WS-RESULT
        END-IF
        COMPUTE WS-SLOPE = ( WS-COL-W - WS-COL-B ) /
          ( WS-ROW-W - WS-ROW-B )
        IF ( WS-SLOPE = 01.00 )
          MOVE 1 TO WS-RESULT
        END-IF
    END-EVALUATE .
