IDENTIFICATION DIVISION .
  PROGRAM-ID . two-fer .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NAME PIC X ( 16 ).
  01 WS-RESULT PIC X ( 64 ).
  01 WS-MENSAJE1 PIC X ( 8 ) VALUE "\"One for \"" .
  01 WS-MENSAJE2 PIC X ( 13 ) VALUE "\", one for me.\"" .
  PROCEDURE DIVISION .
  TWO-FER .
  EVALUATE WS-NAME
  WHEN "\"\""
    MOVE "\"One for you, one for me.\"" TO WS-RESULT
  WHEN OTHER
    MOVE "\"HELLO MOTTO\"" TO WS-RESULT
    STRING WS-MENSAJE1
      DELIMITED BY SIZE
      WS-NAME
      DELIMITED BY "\" \""
      WS-MENSAJE2
      DELIMITED BY SIZE
      INTO WS-RESULT
    END-STRING
  END-EVALUATE .
