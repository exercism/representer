IDENTIFICATION DIVISION .
  PROGRAM-ID . PANGRAM .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-SENTENCE PIC X ( 60 ).
  01 WS-RESULT PIC 9 .
  01 AUX PIC 9 .
  01 ABC .
    05 LETTER PIC X OCCURS 26 TIMES INDEXED BY L .
  PROCEDURE DIVISION .
  PANGRAM .
    MOVE 1 TO WS-RESULT .
    MOVE 'ABCDEFGHIJKLMNOPQRSTUVWXYZ' TO ABC .
    MOVE FUNCTION UPPER-CASE ( WS-SENTENCE ) TO WS-SENTENCE
    MOVE 1 TO L
    PERFORM UNTIL L = 27 OR WS-RESULT = 0
      MOVE 0 TO AUX
      INSPECT WS-SENTENCE TALLYING AUX FOR ALL LETTER ( L )
      IF AUX = 0 THEN
        MOVE 0 TO WS-RESULT
      END-IF
      ADD 1 TO L
    END-PERFORM
    CONTINUE .
  PANGRAM-EXIT .
    EXIT .
