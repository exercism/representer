IDENTIFICATION DIVISION .
  PROGRAM-ID . PANGRAM .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-SENTENCE PIC X ( 60 ).
  01 WS-RESULT PIC 9 .
  01 CONTADOR PIC 9 ( 03 ).
  01 CONT-LETRAS PIC 9 ( 03 ).
  01 CONT-LETRAS2 PIC 9 ( 03 ).
  01 POSICION PIC 9 ( 02 ).
  01 ABCDARIO PIC X ( 26 ) VALUE 'abcdefghijklmnopqrstuvwxyz' .
  01 TABLA-ABC REDEFINES ABCDARIO .
    05 TABLA-ABC-OC OCCURS 26 TIMES .
      10 LETRAS PIC X ( 01 ).
  01 ABCDARIO2 PIC X ( 26 ) VALUE 'ABCDEFGHIJKLMNOPQRSTUVWXYZ' .
  01 TABLA-ABC2 REDEFINES ABCDARIO2 .
    05 TABLA-ABC-OC2 OCCURS 26 TIMES .
      10 LETRAS2 PIC X ( 01 ).
  PROCEDURE DIVISION .
  PANGRAM .
    INITIALIZE CONTADOR .
    DISPLAY WS-SENTENCE
    IF WS-SENTENCE NOT = space
      PERFORM PROCESO VARYING POSICION FROM 1 BY 1
        UNTIL POSICION > 26
    END-IF
    IF CONTADOR = 26
      DISPLAY "'ES UN PANGRAMA'"
      MOVE 1 TO WS-RESULT
    ELSE
      DISPLAY "'NO ES UN PANGRAMA'"
      MOVE 0 TO WS-RESULT
    END-IF
    CONTINUE .
  PANGRA-EXIT .
    EXIT .
  PROCESO .
    INITIALIZE CONT-LETRAS
      CONT-LETRAS2
    INSPECT WS-SENTENCE TALLYING CONT-LETRAS
      FOR ALL LETRAS ( POSICION ).
    INSPECT WS-SENTENCE TALLYING CONT-LETRAS2
      FOR ALL LETRAS2 ( POSICION ).
    IF (( CONT-LETRAS NOT = ZEROS ) OR
      ( CONT-LETRAS2 NOT = ZEROS ))
      ADD 1 TO CONTADOR
    END-IF .
