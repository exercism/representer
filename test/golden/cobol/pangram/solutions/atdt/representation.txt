(file "src/pangram.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . PANGRAM .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-SENTENCE PIC X ( 60 ).
  01 WS-RESULT PIC 9 VALUE 1 .
  01 WS-ALPHABET PIC X ( 26 ) VALUE 'ABCDEFGHIJKLMNOPQRSTUVWXYZ' .
  LOCAL-STORAGE SECTION .
  01 SENTENCE-UPPER PIC X ( 60 ).
  01 CNT BINARY-SHORT VALUE 0 .
  01 I BINARY-SHORT .
  PROCEDURE DIVISION .
  PANGRAM .
    MOVE FUNCTION UPPER-CASE ( WS-SENTENCE ) TO SENTENCE-UPPER
    MOVE 1 TO WS-RESULT
    PERFORM VARYING I FROM 1 BY 1 UNTIL I > LENGTH OF WS-ALPHABET
      OR WS-RESULT = 0
      INSPECT SENTENCE-UPPER TALLYING CNT FOR ALL WS-ALPHABET ( I : 1 )
      IF CNT = 0 THEN
        MOVE 0 TO WS-RESULT
      END-IF
      MOVE 0 TO CNT
    END-PERFORM .
  PANGRAM-EXIT .
    EXIT .
