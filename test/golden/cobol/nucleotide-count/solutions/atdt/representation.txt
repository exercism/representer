(file "src/nucleotide-count.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . nucleotide-count .
  ENVIRONMENT DIVISION .
  CONFIGURATION SECTION .
  REPOSITORY . FUNCTION ALL INTRINSIC .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-DNA PIC X ( 128 ).
  01 WS-A PIC 9 ( 4 ).
  01 WS-C PIC 9 ( 4 ).
  01 WS-G PIC 9 ( 4 ).
  01 WS-T PIC 9 ( 4 ).
  01 WS-ERROR PIC X ( 36 ).
  PROCEDURE DIVISION .
  NUCLEOTIDE-COUNT .
    INSPECT FUNCTION UPPER-CASE ( WS-DNA )
      TALLYING WS-A FOR ALL 'A'
        WS-C FOR ALL 'C'
        WS-G FOR ALL 'G'
        WS-T FOR ALL 'T'
    IF WS-A + WS-C + WS-G + WS-T < FUNCTION LENGTH ( WS-DNA ) THEN
      MOVE "'ERROR: Invalid nucleotide in strand'" TO WS-ERROR
    END-IF .
  NUCLEOTIDE-COUNT-EXIT .
    EXIT .
