IDENTIFICATION DIVISION .
  PROGRAM-ID . high-scores .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-PROPERTY PIC A ( 20 ).
  01 WS-SCORES .
    02 SCORES PIC X ( 3 ) OCCURS 20 TIMES
      INDEXED BY IDX .
  01 WS-RESULT-STRING PIC X ( 60 ).
  01 WS-RESULT-VALUE PIC 999 .
  LOCAL-STORAGE SECTION .
  01 LS-LEN PIC 99 .
  01 LS-INDEX PIC 99 .
  01 LS-BEST-SCORES .
    03 LS-BEST-SCORE OCCURS 3 PIC XXX .
  PROCEDURE DIVISION .
  HIGH-SCORES .
    MOVE SPACES TO WS-RESULT-STRING .
    IF WS-PROPERTY = "\"scores\""
      MOVE WS-SCORES TO WS-RESULT-STRING
      MOVE LENGTH OF FUNCTION TRIM ( WS-SCORES , TRAILING )
        TO WS-RESULT-VALUE
    END-IF .
    IF WS-PROPERTY = "\"latest\""
      MOVE LENGTH OF FUNCTION TRIM ( WS-SCORES , TRAILING )
        TO LS-LEN
      COMPUTE LS-LEN = LS-LEN - 2
      MOVE WS-SCORES ( LS-LEN : 3 ) TO WS-RESULT-VALUE
    END-IF .
    IF WS-PROPERTY = "\"personalBest\""
      MOVE LENGTH OF FUNCTION TRIM ( WS-SCORES , TRAILING )
        TO LS-LEN
      MOVE 0 TO LS-BEST-SCORE ( 1 )
      PERFORM VARYING LS-INDEX
      FROM 1 BY 3 UNTIL LS-INDEX >= LS-LEN
        IF WS-SCORES ( LS-INDEX : 3 ) > LS-BEST-SCORE ( 1 )
          MOVE WS-SCORES ( LS-INDEX : 3 ) TO LS-BEST-SCORE ( 1 )
        END-IF
      END-PERFORM
      MOVE LS-BEST-SCORE ( 1 ) TO WS-RESULT-VALUE
    END-IF .
    IF WS-PROPERTY = "\"personalTopThree\""
      MOVE LENGTH OF FUNCTION TRIM ( WS-SCORES , TRAILING )
        TO LS-LEN
      MOVE "\"   \"" TO LS-BEST-SCORE ( 1 )
      MOVE "\"   \"" TO LS-BEST-SCORE ( 2 )
      MOVE "\"   \"" TO LS-BEST-SCORE ( 3 )
      PERFORM VARYING LS-INDEX
      FROM 1 BY 3 UNTIL LS-INDEX >= LS-LEN
        IF WS-SCORES ( LS-INDEX : 3 ) > LS-BEST-SCORE ( 1 )
          MOVE LS-BEST-SCORE ( 2 ) TO LS-BEST-SCORE ( 3 )
          MOVE LS-BEST-SCORE ( 1 ) TO LS-BEST-SCORE ( 2 )
          MOVE WS-SCORES ( LS-INDEX : 3 ) TO LS-BEST-SCORE ( 1 )
        ELSE IF WS-SCORES ( LS-INDEX : 3 ) > LS-BEST-SCORE ( 2 )
          MOVE LS-BEST-SCORE ( 2 ) TO LS-BEST-SCORE ( 3 )
          MOVE WS-SCORES ( LS-INDEX : 3 ) TO LS-BEST-SCORE ( 2 )
        ELSE
          MOVE WS-SCORES ( LS-INDEX : 3 ) TO LS-BEST-SCORE ( 3 )
        END-IF
      END-PERFORM
      STRING
        WS-RESULT-STRING DELIMITED BY SPACE
        LS-BEST-SCORE ( 1 )
        LS-BEST-SCORE ( 2 )
        LS-BEST-SCORE ( 3 )
        INTO WS-RESULT-STRING
    END-IF .
