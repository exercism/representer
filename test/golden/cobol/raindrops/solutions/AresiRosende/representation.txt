(file "src/raindrops.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . raindrops .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NUMBER PIC 9 ( 4 ).
  01 WS-RESULT PIC X ( 20 ) VALUE "\"\"" .
  01 PLING PIC 9 VALUE 0 .
  01 PLANG PIC 9 VALUE 0 .
  01 PLONG PIC 9 VALUE 0 .
  PROCEDURE DIVISION .
  RAINDROPS .
  IF FUNCTION MOD ( WS-NUMBER , 3 ) = 0 THEN
    STRING WS-RESULT DELIMITED BY SPACES "\"Pling\"" INTO WS-RESULT
  END-IF
  IF FUNCTION MOD ( WS-NUMBER , 5 ) = 0 THEN
    STRING WS-RESULT DELIMITED BY SPACES "\"Plang\"" INTO WS-RESULT
  END-IF
  IF FUNCTION MOD ( WS-NUMBER , 7 ) = 0 THEN
    STRING WS-RESULT DELIMITED BY SPACES "\"Plong\"" INTO WS-RESULT
  END-IF
  IF WS-RESULT = "\"\"" THEN
    MOVE WS-NUMBER TO WS-RESULT
  END-IF .
