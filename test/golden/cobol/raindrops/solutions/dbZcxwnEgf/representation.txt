  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NUMBER PIC 9 ( 4 ).
  01 WS-RESULT PIC X ( 20 ).
  PROCEDURE DIVISION .
  RAINDROPS .
  MOVE "\"\"" TO WS-RESULT
  IF FUNCTION MOD ( WS-NUMBER , 3 ) = 0
    STRING WS-RESULT DELIMITED BY SPACE "\"Pling\"" INTO WS-RESULT
  END-IF
  IF FUNCTION MOD ( WS-NUMBER , 5 ) = 0
    STRING WS-RESULT DELIMITED BY SPACE "\"Plang\"" INTO WS-RESULT
  END-IF
  IF FUNCTION MOD ( WS-NUMBER , 7 ) = 0
    STRING WS-RESULT DELIMITED BY SPACE "\"Plong\"" INTO WS-RESULT
  END-IF
  IF WS-RESULT = SPACES
    MOVE WS-NUMBER TO WS-RESULT
  END-IF
  .
