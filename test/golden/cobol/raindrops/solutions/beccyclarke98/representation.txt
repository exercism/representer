(file "src/raindrops.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . raindrops .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-NUMBER PIC 9 ( 4 ).
  01 WS-RESULT PIC X ( 20 ).
  01 WS-FAC-OF-THREE PIC X ( 100 ) VALUE "\"Pling\"" .
  01 WS-FAC-OF-FIVE PIC X ( 100 ) VALUE "\"Plang\"" .
  01 WS-FAC-OF-SEVEN PIC X ( 100 ) VALUE "\"Plong\"" .
  PROCEDURE DIVISION .
  RAINDROPS .
  IF FUNCTION MOD ( WS-NUMBER , 3 ) = 0
    STRING WS-FAC-OF-THREE INTO WS-RESULT
  END-IF
  IF FUNCTION MOD ( WS-NUMBER , 5 ) = 0
    STRING WS-RESULT DELIMITED BY SPACE WS-FAC-OF-FIVE
      INTO WS-RESULT
  END-IF
    IF FUNCTION MOD ( WS-NUMBER , 7 ) = 0
    STRING WS-RESULT DELIMITED BY SPACE WS-FAC-OF-SEVEN
      INTO WS-RESULT
  END-IF
  IF WS-RESULT = SPACE MOVE WS-NUMBER TO WS-RESULT END-IF .
