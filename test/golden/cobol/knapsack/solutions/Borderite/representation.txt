IDENTIFICATION DIVISION .
  PROGRAM-ID . KNAPSACK .
    ENVIRONMENT DIVISION .
    DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-INPUTS .
    05 MAXIMUM-WEIGHT PIC 999 BINARY .
    05 NO-OF-ITEMS PIC 99 BINARY .
  01 KNAPSACKTABLE .
    05 ITEMS OCCURS 1 TO 20
      DEPENDING ON NO-OF-ITEMS .
      10 WS-WEIGHT PIC 999 BINARY .
      10 WS-VALUE PIC 999 BINARY .
  01 WS-OUTPUTS .
    05 WS-RESULT PIC 99999 BINARY .
  01 VALUE-TABLE .
  05 VALUE-TABLE-ELT PIC 99999 BINARY OCCURS 1 TO 1000
    DEPENDING ON MAX-WEIGHT-PLS-1 .
  01 MAX-WEIGHT-PLS-1 PIC 9999 BINARY .
  01 CUR-WEIGHT PIC 9999 BINARY .
  01 CUR-VALUE PIC 999 BINARY .
  01 I PIC 9999 BINARY .
  01 W pic 9999 BINARY .
  PROCEDURE DIVISION .
  TEST-PARAGRAPH .
    MOVE 5 TO NO-OF-ITEMS
    INITIALIZE KNAPSACKTABLE
    MOVE ZERO TO WS-RESULT
    MOVE 10 TO MAXIMUM-WEIGHT
    MOVE 2 TO WS-WEIGHT ( 1 )
    MOVE 5 TO WS-VALUE ( 1 )
    MOVE 2 TO WS-WEIGHT ( 2 )
    MOVE 5 TO WS-VALUE ( 2 )
    MOVE 2 TO WS-WEIGHT ( 3 )
    MOVE 5 TO WS-VALUE ( 3 )
    MOVE 2 TO WS-WEIGHT ( 4 )
    MOVE 5 TO WS-VALUE ( 4 )
    MOVE 10 TO WS-WEIGHT ( 5 )
    MOVE 21 TO WS-VALUE ( 5 )
    PERFORM VARYING I FROM 1 BY 1 UNTIL I > NO-OF-ITEMS
      DISPLAY I "': WEIGHT = '" WS-WEIGHT ( I )
        "', VALUE = '" WS-VALUE ( I )
    END-PERFORM
    PERFORM MAXIMUM-VALUE
    DISPLAY "'WS-RESULT = '" WS-RESULT .
  TEST-PARAGRAPH-END .
    STOP RUN .
  MAXIMUM-VALUE .
    ADD 1 TO MAXIMUM-WEIGHT GIVING MAX-WEIGHT-PLS-1
    INITIALIZE VALUE-TABLE
    PERFORM VARYING I FROM 1 BY 1 UNTIL I > NO-OF-ITEMS
      MOVE WS-WEIGHT ( I ) TO CUR-WEIGHT
      MOVE WS-VALUE ( I ) TO CUR-VALUE
      PERFORM VARYING W FROM MAX-WEIGHT-PLS-1
        BY -1 UNTIL W <= CUR-WEIGHT
        MOVE FUNCTION MAX ( VALUE-TABLE-ELT ( W ),
          VALUE-TABLE-ELT ( W - CUR-WEIGHT ) + CUR-VALUE )
          TO VALUE-TABLE-ELT ( W )
      END-PERFORM
    END-PERFORM
    MOVE VALUE-TABLE-ELT ( MAX-WEIGHT-PLS-1 ) TO WS-RESULT .
  END PROGRAM KNAPSACK .
