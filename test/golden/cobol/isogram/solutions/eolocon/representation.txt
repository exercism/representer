IDENTIFICATION DIVISION .
  PROGRAM-ID . ISOGRAM .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-PHRASE PIC X ( 60 ).
  01 WS-SENTENCE PIC X ( 60 ).
  01 WS-RESULT PIC 9 .
  01 WS-COUNT-ARRAY .
    02 WS-COUNT-ELEMENT PIC 99 OCCURS 26 TIMES .
  01 WS-I PIC 99 .
  01 WS-LOOKUP-RESULT PIC 99 .
  01 WS-TARGET PIC X .
  PROCEDURE DIVISION .
  ISOGRAM .
    MOVE WS-PHRASE TO WS-SENTENCE
    MOVE 1 TO WS-RESULT .
    MOVE ZEROS TO WS-COUNT-ARRAY .
    PERFORM VARYING WS-I
    FROM 1 BY 1
    UNTIL WS-I > 60
      MOVE WS-SENTENCE ( WS-I : 1 ) TO WS-TARGET
      PERFORM LOOKUP
      IF WS-LOOKUP-RESULT IS GREATER THAN 0
        ADD 1 TO WS-COUNT-ELEMENT ( WS-LOOKUP-RESULT )
    END-PERFORM .
    PERFORM VARYING WS-I
    FROM 1 BY 1
    UNTIL WS-I > 26
      IF WS-COUNT-ELEMENT ( WS-I ) IS GREATER THAN 1
        MOVE 0 TO WS-RESULT
    END-PERFORM .
  LOOKUP .
    MOVE 0 TO WS-LOOKUP-RESULT .
    EVALUATE WS-TARGET
      WHEN "\"A\"" WHEN "\"a\""
        MOVE 1 TO WS-LOOKUP-RESULT
      WHEN "\"B\"" WHEN "\"b\""
        MOVE 2 TO WS-LOOKUP-RESULT
      WHEN "\"C\"" WHEN "\"c\""
        MOVE 3 TO WS-LOOKUP-RESULT
      WHEN "\"D\"" WHEN "\"d\""
        MOVE 4 TO WS-LOOKUP-RESULT
      WHEN "\"E\"" WHEN "\"e\""
        MOVE 5 TO WS-LOOKUP-RESULT
      WHEN "\"F\"" WHEN "\"f\""
        MOVE 6 TO WS-LOOKUP-RESULT
      WHEN "\"G\"" WHEN "\"g\""
        MOVE 7 TO WS-LOOKUP-RESULT
      WHEN "\"H\"" WHEN "\"h\""
        MOVE 8 TO WS-LOOKUP-RESULT
      WHEN "\"I\"" WHEN "\"i\""
        MOVE 9 TO WS-LOOKUP-RESULT
      WHEN "\"J\"" WHEN "\"j\""
        MOVE 10 TO WS-LOOKUP-RESULT
      WHEN "\"K\"" WHEN "\"k\""
        MOVE 11 TO WS-LOOKUP-RESULT
      WHEN "\"L\"" WHEN "\"l\""
        MOVE 12 TO WS-LOOKUP-RESULT
      WHEN "\"M\"" WHEN "\"m\""
        MOVE 13 TO WS-LOOKUP-RESULT
      WHEN "\"N\"" WHEN "\"n\""
        MOVE 14 TO WS-LOOKUP-RESULT
      WHEN "\"O\"" WHEN "\"o\""
        MOVE 15 TO WS-LOOKUP-RESULT
      WHEN "\"P\"" WHEN "\"p\""
        MOVE 16 TO WS-LOOKUP-RESULT
      WHEN "\"Q\"" WHEN "\"q\""
        MOVE 17 TO WS-LOOKUP-RESULT
      WHEN "\"R\"" WHEN "\"r\""
        MOVE 18 TO WS-LOOKUP-RESULT
      WHEN "\"S\"" WHEN "\"s\""
        MOVE 19 TO WS-LOOKUP-RESULT
      WHEN "\"T\"" WHEN "\"t\""
        MOVE 20 TO WS-LOOKUP-RESULT
      WHEN "\"U\"" WHEN "\"u\""
        MOVE 21 TO WS-LOOKUP-RESULT
      WHEN "\"V\"" WHEN "\"v\""
        MOVE 22 TO WS-LOOKUP-RESULT
      WHEN "\"W\"" WHEN "\"w\""
        MOVE 23 TO WS-LOOKUP-RESULT
      WHEN "\"X\"" WHEN "\"x\""
        MOVE 24 TO WS-LOOKUP-RESULT
      WHEN "\"Y\"" WHEN "\"y\""
        MOVE 25 TO WS-LOOKUP-RESULT
      WHEN "\"Z\"" WHEN "\"z\""
        MOVE 26 TO WS-LOOKUP-RESULT
      END-EVALUATE .
