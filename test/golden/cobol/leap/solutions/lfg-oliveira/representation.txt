IDENTIFICATION DIVISION .
  PROGRAM-ID . LEAP .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
    WORKING-STORAGE SECTION .
      01 WS-YEAR BINARY-LONG VALUE 0 .
      01 WS-RESULT PIC 1 VALUE 0 .
      01 GIVE BINARY-LONG VALUE 0 .
      01 REM BINARY-LONG VALUE 0 .
  PROCEDURE DIVISION .
  LEAP .
    MOVE 0 TO WS-RESULT .
    DIVIDE WS-YEAR BY 4 GIVING GIVE REMAINDER REM .
    IF REM = 0 THEN
      MOVE 1 TO WS-RESULT
    END-IF .
    DIVIDE WS-YEAR BY 100 GIVING GIVE REMAINDER REM .
    IF REM = 0 THEN
      MOVE 0 TO WS-RESULT
    END-IF .
    DIVIDE WS-YEAR BY 400 GIVING GIVE REMAINDER REM .
    IF REM = 0 THEN
      MOVE 1 TO WS-RESULT
    END-IF .
  LEAP-EXIT .
    EXIT .
