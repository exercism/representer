IDENTIFICATION DIVISION .
  PROGRAM-ID . LEAP .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-YEAR PIC 9 ( 4 ).
  01 DIV4 PIC 9 ( 4 ).
  01 DIV100 PIC 9 ( 4 ).
  01 DIV400 PIC 9 ( 4 ).
  01 WS-RESULT PIC 9 .
  PROCEDURE DIVISION .
  LEAP .
    COMPUTE DIV4 = FUNCTION MOD ( WS-YEAR , 4 )
    COMPUTE DIV100 = FUNCTION MOD ( WS-YEAR , 100 )
    COMPUTE DIV400 = FUNCTION MOD ( WS-YEAR , 400 )
    IF DIV4 = 0
      IF DIV100 = 0
        IF DIV400 = 0
          THEN MOVE 1 TO WS-RESULT
          GO TO END-IF-CHAIN
        ELSE
          MOVE 0 TO WS-RESULT
          GO TO END-IF-CHAIN
        END-IF
      END-IF
    MOVE 1 TO WS-RESULT
    ELSE
      MOVE 0 TO WS-RESULT
    END-IF .
  END-IF-CHAIN .
    EXIT PROGRAM .
