IDENTIFICATION DIVISION .
  PROGRAM-ID . LEAP .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-YEAR PIC 9 ( 4 ).
  01 WS-RESULT PIC 9 ( 1 ) VALUE 0 .
  01 WS-YYYY PIC 9 ( 4 ) value 000 .
  PROCEDURE DIVISION .
  LEAP .
    MOVE 0 TO WS-YYYY .
    MOVE 0 TO WS-RESULT .
    COMPUTE WS-YYYY = FUNCTION REM ( WS-YEAR , 400 ).
    IF WS-YYYY = 0 THEN
      DISPLAY WS-YEAR "' IS A LEAP YEAR'"
      MOVE 1 TO WS-RESULT
    ELSE COMPUTE WS-YYYY = FUNCTION REM ( WS-YEAR , 4 )
      IF WS-YYYY = 0 THEN
        COMPUTE WS-YYYY = FUNCTION REM ( WS-YEAR , 100 )
        IF WS-YYYY = 0 THEN
          DISPLAY WS-YEAR "' IS NOT A LEAP YEAR'"
        ELSE DISPLAY WS-YEAR "' IS A LEAP YEAR'"
          MOVE 1 TO WS-RESULT
        END-IF
      END-IF
      END-IF .
  LEAP-EXIT .
    EXIT .
