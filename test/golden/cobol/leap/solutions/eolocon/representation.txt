IDENTIFICATION DIVISION .
  PROGRAM-ID . LEAP .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-YEAR PIC 9999 .
  77 WS-RESULT PIC 9 VALUE 0 .
  77 QUOTIENT PIC 9999 .
  77 REM PIC 9999 .
  77 DIVISIBLE-BY-4 PIC 9999 VALUE ZEROS .
  77 DIVISIBLE-BY-100 PIC 9999 VALUE ZEROS .
  77 DIVISIBLE-BY-400 PIC 9999 VALUE ZEROS .
  PROCEDURE DIVISION .
  LEAP .
    MOVE ZERO TO WS-RESULT .
    DIVIDE WS-YEAR BY 4
      GIVING QUOTIENT REMAINDER DIVISIBLE-BY-4 .
    DIVIDE WS-YEAR BY 100
      GIVING QUOTIENT REMAINDER DIVISIBLE-BY-100 .
    DIVIDE WS-YEAR BY 400
      GIVING QUOTIENT REMAINDER DIVISIBLE-BY-400 .
    DISPLAY DIVISIBLE-BY-4
    DISPLAY DIVISIBLE-BY-100
    DISPLAY DIVISIBLE-BY-400 .
    IF ( DIVISIBLE-BY-4 IS EQUAL TO ZEROS
      AND DIVISIBLE-BY-100 IS NOT EQUAL TO ZEROS )
    OR ( DIVISIBLE-BY-4 IS EQUAL TO ZEROS
      AND DIVISIBLE-BY-100 IS EQUAL TO ZEROS
      AND DIVISIBLE-BY-400 IS EQUAL TO ZEROS ) THEN
      MOVE 1 TO WS-RESULT .
    EXIT .
