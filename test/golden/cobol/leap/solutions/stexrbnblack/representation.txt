  IDENTIFICATION DIVISION .
  PROGRAM-ID . YEAR-BISIESTO .
  DATA DIVISION .
  FILE SECTION .
  WORKING-STORAGE SECTION .
  01 WS-YEAR PIC 9 ( 4 ).
  01 WS-RESULT PIC 9 ( 4 ).
  01 YEAR-LEAP-4 PIC 999 .
  01 YEAR-LEAP-100 PIC 999 .
  01 YEAR-LEAP-400 PIC 999 .
  01 YEAR-LEAP-SI-NO PIC 999 .
  01 VALIDAR-RESULT PIC 9 .
  01 WS-RESIDUO-4 PIC 999 .
  01 WS-RESIDUO-400 PIC 999 .
  01 WS-RESIDUO-100 PIC 999 .
  01 CONTINUAR PIC X .
  PROCEDURE DIVISION .
  MAIN-PROCEDURE .
    DISPLAY "\"INTRODUSCA EL A\u00D1O A VERIFICAR COMO BISIESTO\"" .
    ACCEPT WS-YEAR .
    PERFORM VALIDAR-BISIESTO .
    PERFORM LEAP .
    VALIDAR-BISIESTO .
      DIVIDE WS-YEAR BY 4 GIVING YEAR-LEAP-4
      REMAINDER WS-RESIDUO-4
      DISPLAY WS-YEAR "\"/4 = \"" YEAR-LEAP-4
      DISPLAY "\"RESTO = \"" WS-RESIDUO-4
      MOVE 0 TO YEAR-LEAP-SI-NO
      DIVIDE WS-YEAR BY 400 GIVING YEAR-LEAP-400
      REMAINDER WS-RESIDUO-400
      DISPLAY WS-YEAR "\"/400 = \"" YEAR-LEAP-400
      DISPLAY "\"RESTO = \"" WS-RESIDUO-400
      MOVE 0 TO YEAR-LEAP-SI-NO
      DIVIDE WS-YEAR BY 100 GIVING YEAR-LEAP-100
      REMAINDER WS-RESIDUO-100
      DISPLAY WS-YEAR "\"/100 = \"" YEAR-LEAP-100
      DISPLAY "\"RESTO = \"" WS-RESIDUO-100
      MOVE 1 TO YEAR-LEAP-SI-NO .
    LEAP .
      IF WS-RESIDUO-4 = 0 AND
        ( WS-RESIDUO-400 = 0 OR WS-RESIDUO-400 > 0 ) AND
        ( WS-RESIDUO-100 = 0 OR WS-RESIDUO-100 > 0 )
        MOVE 0 TO WS-RESULT
        DISPLAY "\"SI ES BISIETO\""
      ELSE
        MOVE 1 TO WS-RESULT
        DISPLAY "\"NO ES BISIETO\""
      END-IF
      STOP RUN .
  END PROGRAM YEAR-BISIESTO .
