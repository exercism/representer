IDENTIFICATION DIVISION .
  PROGRAM-ID . BOB .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-HEYBOB PIC X ( 60 ).
  01 WS-RESULT PIC X ( 40 ).
  01 WS-STATEMENT PIC X ( 60 ).
  01 WS-STATEMENT-LENGTH PIC 99 VALUE ZERO .
  01 WS-CANIDATE PIC X
    OCCURS 60 TIMES
    INDEXED BY WS-CANIDATE-INDEX .
    88 WS-CAPITAL VALUE 'A' THRU 'I'
      'J' THRU 'R'
      'S' THRU 'Z' .
    88 WS-COMMON VALUE 'a' THRU 'i'
      'j' THRU 'r'
      's' THRU 'z' .
  01 WS-LETTER-FOUND PIC 99 .
    88 WS-HAS-LETTER VALUE 1 .
    88 WS-NO-LETTER-FOUND VALUE 0 .
  01 WS-LAST-CHAR PIC X VALUE SPACE .
    88 WS-IS-QUESTION VALUE "\"?\"" .
  01 WS-RESPONSES .
    05 WS-WHEN-QUESTIONED PIC X ( 5 ) VALUE "\"Sure.\"" .
    05 WS-WHEN-YELLED-AT PIC X ( 16 ) VALUE
      "\"Whoa, chill out!\"" .
    05 WS-WHEN-YELLED-QUESTION PIC X ( 33 ) VALUE
      "\"Calm down, I know what I'm doing!\"" .
    05 WS-WHEN-ADDRESSED-NOTHING-SAID
      PIC X ( 18 ) VALUE
      "\"Fine. Be that way!\"" .
    05 WS-CATCHALL PIC X ( 9 ) VALUE "\"Whatever.\"" .
  PROCEDURE DIVISION .
  BOB .
    MOVE FUNCTION TRIM ( WS-HEYBOB ) TO WS-STATEMENT
    MOVE FUNCTION LENGTH ( FUNCTION TRIM ( WS-STATEMENT )) TO
      WS-STATEMENT-LENGTH
    MOVE WS-STATEMENT ( WS-STATEMENT-LENGTH : 1 ) TO WS-LAST-CHAR
    SET WS-NO-LETTER-FOUND TO TRUE
    PERFORM FIND-LETTERS-IN-INPUT
    EVALUATE TRUE
    WHEN WS-IS-QUESTION
      IF WS-HAS-LETTER AND WS-STATEMENT =
        FUNCTION UPPER-CASE ( WS-STATEMENT ) THEN
        MOVE WS-WHEN-YELLED-QUESTION TO WS-RESULT
      ELSE
        MOVE WS-WHEN-QUESTIONED TO WS-RESULT
      END-IF
    WHEN WS-HAS-LETTER AND WS-STATEMENT =
      FUNCTION UPPER-CASE ( WS-STATEMENT )
        MOVE WS-WHEN-YELLED-AT TO WS-RESULT
    WHEN WS-STATEMENT-LENGTH = 0
      MOVE WS-WHEN-ADDRESSED-NOTHING-SAID TO WS-RESULT
    WHEN OTHER
      MOVE WS-CATCHALL TO WS-RESULT
    END-EVALUATE .
  FIND-LETTERS-IN-INPUT .
    PERFORM
      VARYING WS-CANIDATE-INDEX
      FROM 1 BY 1
      UNTIL WS-CANIDATE-INDEX > WS-STATEMENT-LENGTH
        MOVE WS-STATEMENT ( WS-CANIDATE-INDEX : 1 ) TO
          WS-CANIDATE ( WS-CANIDATE-INDEX )
        IF WS-CAPITAL ( WS-CANIDATE-INDEX ) OR
          WS-COMMON ( WS-CANIDATE-INDEX )
          THEN
          SET WS-HAS-LETTER TO TRUE
          EXIT PERFORM
        END-IF
    END-PERFORM .
