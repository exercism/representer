IDENTIFICATION DIVISION .
  PROGRAM-ID . BOB .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-HEYBOB PIC X ( 60 ).
  01 WS-RESULT PIC X ( 40 ).
    88 WS-RSP-Q VALUE "\"Sure.\"" .
    88 WS-RSP-YELL VALUE "\"Whoa, chill out!\"" .
    88 WS-RSP-YELL-Q VALUE "\"Calm down, I know what I'm doing!\"" .
    88 WS-RSP-SILENCE VALUE "\"Fine. Be that way!\"" .
    88 WS-RSP-DEFAULT VALUE "\"Whatever.\"" .
  01 WS-IS-Q PIC 9 VALUE 0 .
  01 WS-IS-YELL PIC 9 VALUE 0 .
  01 WS-IS-ALPHABETIC PIC 9 VALUE 0 .
  01 WS-IS-CHAR PIC X .
    88 WS-CHAR VALUE "\"A\"" THRU "\"Z\"" .
  01 WS-LENGTH PIC 99 .
  01 I PIC 99 .
  PROCEDURE DIVISION .
  BOB .
    SET WS-RSP-DEFAULT TO TRUE
    MOVE 0 TO WS-IS-Q WS-IS-YELL WS-IS-ALPHABETIC
    COMPUTE WS-LENGTH = FUNCTION LENGTH ( FUNCTION TRIM ( WS-HEYBOB ))
    PERFORM VARYING I FROM 1 BY 1 UNTIL ( I > WS-LENGTH )
      MOVE WS-HEYBOB ( I : 1 ) TO WS-IS-CHAR
      IF ( WS-CHAR )
        MOVE 1 TO WS-IS-ALPHABETIC
        EXIT PERFORM
      END-IF
    END-PERFORM
    IF ( WS-HEYBOB ( WS-LENGTH : 1 ) = "\"?\"" )
      MOVE 1 TO WS-IS-Q
    END-IF
    IF ( WS-HEYBOB = FUNCTION UPPER-CASE ( WS-HEYBOB )) AND
      ( WS-IS-ALPHABETIC = 1 )
      MOVE 1 TO WS-IS-YELL
    END-IF
    EVALUATE TRUE
      WHEN ( WS-HEYBOB = SPACES )
        SET WS-RSP-SILENCE TO TRUE
      WHEN ( WS-IS-Q = 0 ) AND ( WS-IS-YELL = 1 )
        SET WS-RSP-YELL TO TRUE
      WHEN ( WS-IS-Q = 1 )
        IF ( WS-IS-YELL = 1 )
          SET WS-RSP-YELL-Q TO TRUE
        ELSE
          SET WS-RSP-Q TO TRUE
        END-IF
    END-EVALUATE .
