IDENTIFICATION DIVISION .
  PROGRAM-ID . hamming .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-DNA-1 PIC X ( 32 ).
  01 REDEFINES WS-DNA-1 .
    05 C-DNA-1 PIC X OCCURS 32 TIMES .
  01 WS-LEN-1 PIC 99 .
  01 WS-DNA-2 PIC X ( 32 ).
  01 REDEFINES WS-DNA-2 .
    05 C-DNA-2 PIC X OCCURS 32 TIMES .
  01 WS-LEN-2 PIC 99 .
  01 WS-HAMMING PIC 9 ( 2 ).
  01 WS-ERROR PIC X ( 31 ).
  01 ITER PIC 99 .
  PROCEDURE DIVISION .
  HAMMING .
  COMPUTE WS-LEN-1 = FUNCTION LENGTH ( FUNCTION TRIM ( WS-DNA-1 )).
  COMPUTE WS-LEN-2 = FUNCTION LENGTH ( FUNCTION TRIM ( WS-DNA-2 )).
  IF WS-LEN-1 <> WS-LEN-2 THEN
    MOVE "\"Strands must be of equal length\"" TO WS-ERROR
    EXIT PARAGRAPH
  END-IF .
  PERFORM VARYING ITER FROM 1 BY 1 UNTIL ITER > WS-LEN-1
    IF C-DNA-1 ( ITER ) <> C-DNA-2 ( ITER ) THEN
      ADD 1 TO WS-HAMMING
    END-IF
  END-PERFORM .
  DISPLAY "\"Hamming Distance is: \"" WS-HAMMING .
  DISPLAY WS-ERROR .
