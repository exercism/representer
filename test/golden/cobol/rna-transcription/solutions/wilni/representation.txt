IDENTIFICATION DIVISION .
  PROGRAM-ID . rna-transcription .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-COMPLEMENT PIC X ( 64 ).
  01 WS-DNA PIC X ( 4 ) VALUE "\"ACGT\"" .
  01 WS-NUC .
    05 WS-NUCS PIC X ( 1 ) OCCURS 12 times indexed by I .
  PROCEDURE DIVISION .
  RNA-TRANSCRIPTION .
  SET I TO 1 .
  perform function length ( WS-COMPLEMENT ) times
    EVALUATE TRUE
      WHEN WS-COMPLEMENT ( I : 1 ) = "\"G\""
        Move "\"C\"" to WS-NUCS ( I )
      WHEN WS-COMPLEMENT ( I : 1 ) = "\"C\""
        Move "\"G\"" to WS-NUCS ( I )
      WHEN WS-COMPLEMENT ( I : 1 ) = "\"T\""
        Move "\"A\"" to WS-NUCS ( I )
      WHEN WS-COMPLEMENT ( I : 1 ) = "\"A\""
        Move "\"U\"" to WS-NUCS ( I )
    END-EVALUATE
    SET I UP BY 1
  END-PERFORM
  MOVE WS-NUC TO WS-COMPLEMENT .
