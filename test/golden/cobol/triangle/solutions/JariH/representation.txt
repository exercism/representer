(file "src/triangle.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . TRIANGLE .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 WS-SIDES PIC X ( 20 ).
  01 WS-SIDE1 PIC 9 ( 4 ) V99 .
  01 WS-SIDE2 PIC 9 ( 4 ) V99 .
  01 WS-SIDE3 PIC 9 ( 4 ) V99 .
  01 WS-PROPERTY PIC X ( 11 ).
  01 WS-RESULT PIC 9 .
  PROCEDURE DIVISION .
  TRIANGLE .
    move 0 to WS-RESULT
    UNSTRING WS-SIDES DELIMITED BY "\",\""
      INTO WS-SIDE1 , WS-SIDE2 , WS-SIDE3
    END-UNSTRING
    if WS-SIDE1 <= 0 OR WS-SIDE2 <= 0 OR WS-SIDE3 <= 0
      goback
    end-if
    if ( ws-side1 + ws-side2 < ws-side3 )
      or ( ws-side1 + ws-side3 < ws-side2 )
      or ( ws-side2 + ws-side3 < ws-side1 )
      goback
    end-if
    evaluate WS-PROPERTY
      when "\"equilateral\""
        if ( ws-side1 = ws-side2 and ws-side3 )
          move 1 to WS-RESULT
        end-if
      when "\"isosceles\""
        if ( ws-side1 = ws-side2 or ws-side3 )
          or ws-side2 = ws-side3
          move 1 to WS-RESULT
        end-if
      when "\"scalene\""
        if ( ws-side1 = ws-side2 or ws-side3 )
          or ws-side2 = ws-side3
          move 0 to WS-RESULT
        else
          move 1 to WS-RESULT
        end-if
      when other
        move 0 to WS-RESULT
    end-evaluate
    .
