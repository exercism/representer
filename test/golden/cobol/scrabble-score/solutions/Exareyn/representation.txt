IDENTIFICATION DIVISION .
  PROGRAM-ID . SCRABBLE-SCORE .
  ENVIRONMENT DIVISION .
  DATA DIVISION .
  WORKING-STORAGE SECTION .
  01 I PIC 9 ( 2 ).
  01 WS-LEN PIC 9 ( 2 ).
  01 WS-WORD PIC X ( 60 ).
  01 WS-UPPER PIC X ( 60 ).
  01 WS-RESULT PIC 99 .
  PROCEDURE DIVISION .
  SCRABBLE-SCORE .
  move 0 to WS-RESULT
  move function length ( function trim ( WS-WORD )) to WS-LEN .
  move function upper-case ( WS-WORD ) to WS-UPPER .
  perform varying I from 1 by 1 until I > WS-LEN
    evaluate WS-UPPER ( I : 1 )
      WHEN "\"A\"" WHEN "\"E\"" WHEN "\"I\"" WHEN "\"O\"" WHEN "\"U\""
      WHEN "\"L\"" WHEN "\"N\"" WHEN "\"R\"" WHEN "\"S\"" WHEN "\"T\""
        ADD 1 TO WS-RESULT
      WHEN "\"D\"" WHEN "\"G\""
        ADD 2 TO WS-RESULT
      WHEN "\"B\"" WHEN "\"C\"" WHEN "\"M\"" WHEN "\"P\""
        ADD 3 TO WS-RESULT
      WHEN "\"F\"" WHEN "\"H\"" WHEN "\"V\"" WHEN "\"W\"" WHEN "\"Y\""
        ADD 4 TO WS-RESULT
      WHEN "\"K\""
        ADD 5 TO WS-RESULT
      WHEN "\"J\"" WHEN "\"X\""
        ADD 8 TO WS-RESULT
      WHEN "\"Q\"" WHEN "\"Z\""
        ADD 10 TO WS-RESULT
    END-EVALUATE
  END-PERFORM .
