  01 WS-VALUE PIC 99 BINARY .
  PROCEDURE DIVISION .
  TEST-PARAGRAPH .
    MOVE "\"a\"" TO WS-WORD
    DISPLAY "'WS-WORD = \"'" WS-WORD '"'
    PERFORM SCRABBLE-SCORE
    DISPLAY "'WS-RESULT = '" WS-RESULT .
  TEST-PARAGRAPH-END .
    STOP RUN .
  SCRABBLE-SCORE .
    MOVE WS-WORD TO WS-UC-WORD
    INSPECT WS-UC-WORD CONVERTING
      "\"abcdefghijklmnopqrstuvwxyz\"" TO
      "\"ABCDEFGHIJKLMNOPQRSTUVWXYZ\""
    COMPUTE WS-UC-BASE = FUNCTION ORD ( 'A' ) - 1
    COMPUTE WS-LEN = FUNCTION STORED-CHAR-LENGTH ( WS-UC-WORD )
    MOVE 0 TO WS-RESULT
    PERFORM VARYING WS-I FROM 1 BY 1 UNTIL WS-I > WS-LEN
      PERFORM GET-VALUE
      ADD WS-VALUE TO WS-RESULT
    END-PERFORM .
  GET-VALUE .
    COMPUTE WS-ORDER =
      FUNCTION ORD ( WS-UC-WORD ( WS-I : 1 )) - WS-UC-BASE
    MOVE WS-VALUES ( WS-ORDER ) TO WS-VALUE .
  END PROGRAM SCRABBLE-SCORE .
