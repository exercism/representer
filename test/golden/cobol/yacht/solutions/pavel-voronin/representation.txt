(file "src/yacht.cob")
IDENTIFICATION DIVISION .
  PROGRAM-ID . YACHT .
  DATA DIVISION .
    WORKING-STORAGE SECTION .
      01 WS-RESULT PIC 99 .
      01 WS-CATEGORY PIC X ( 15 ).
      01 WS-DICE PIC 9 ( 5 ).
      01 DICE-ARRAY PIC 9 OCCURS 5 .
      01 DICE-COUNTS PIC 9 OCCURS 6 .
      01 NOMINAL PIC 9 .
      01 I PIC 9 .
      01 J PIC 9 .
  PROCEDURE DIVISION .
    FIND_YACHT.
      IF DICE-COUNTS ( 1 ) = 5
        OR DICE-COUNTS ( 2 ) = 5
        OR DICE-COUNTS ( 3 ) = 5
        OR DICE-COUNTS ( 4 ) = 5
        OR DICE-COUNTS ( 5 ) = 5
      THEN
        MOVE 50 TO WS-RESULT
      END-IF .
    FIND_NOMINAL.
      COMPUTE WS-RESULT =
        DICE-COUNTS ( NOMINAL ) * NOMINAL .
    FIND_FULL_HOUSE.
      PERFORM VARYING I FROM 1 BY 1 UNTIL I > 6
        IF DICE-COUNTS ( I ) = 3 THEN
          PERFORM VARYING J FROM 1 BY 1 UNTIL J > 6
            IF I <> J AND DICE-COUNTS ( J ) = 2 THEN
              PERFORM FIND_SUM
            END-IF
          END-PERFORM
        END-IF
      END-PERFORM .
    FIND_SUM.
      PERFORM VARYING I FROM 1 BY 1 UNTIL I > 5
        ADD DICE-ARRAY ( I ) TO WS-RESULT
      END-PERFORM .
    FIND_BIG_STRAIGHT.
      IF DICE-COUNTS ( 1 ) = 0
        AND DICE-COUNTS ( 2 ) = 1
        AND DICE-COUNTS ( 3 ) = 1
        AND DICE-COUNTS ( 4 ) = 1
        AND DICE-COUNTS ( 5 ) = 1
        AND DICE-COUNTS ( 6 ) = 1
      THEN
        MOVE 30 TO WS-RESULT
      END-IF .
    FIND_SMALL_STRAIGHT.
      IF DICE-COUNTS ( 1 ) = 1
        AND DICE-COUNTS ( 2 ) = 1
        AND DICE-COUNTS ( 3 ) = 1
        AND DICE-COUNTS ( 4 ) = 1
        AND DICE-COUNTS ( 5 ) = 1
        AND DICE-COUNTS ( 6 ) = 0
      THEN
        MOVE 30 TO WS-RESULT
      END-IF .
    FIND_FOUR_OF_A_KIND.
      PERFORM VARYING I FROM 1 BY 1 UNTIL I > 6
        IF DICE-COUNTS ( I ) >= 4 THEN
          COMPUTE WS-RESULT = I * 4
        END-IF
      END-PERFORM .
    YACHT .
      MOVE 0 TO WS-RESULT .
      PERFORM VARYING I FROM 1 BY 1 UNTIL I > 6
        MOVE 0 TO DICE-COUNTS ( I )
      END-PERFORM
      PERFORM VARYING I FROM 1 BY 1 UNTIL I > 5
        MOVE WS-DICE ( I : 1 ) TO DICE-ARRAY ( I )
        ADD 1 TO DICE-COUNTS ( DICE-ARRAY ( I ))
      END-PERFORM
      IF WS-CATEGORY = "\"yacht\"" THEN
        PERFORM FIND_YACHT
      ELSE IF WS-CATEGORY = "\"ones\"" THEN
        MOVE 1 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"twos\"" THEN
        MOVE 2 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"threes\"" THEN
        MOVE 3 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"fours\"" THEN
        MOVE 4 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"fives\"" THEN
        MOVE 5 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"sixes\"" THEN
        MOVE 6 TO NOMINAL
        PERFORM FIND_NOMINAL
      ELSE IF WS-CATEGORY = "\"full house\"" THEN
        PERFORM FIND_FULL_HOUSE
      ELSE IF WS-CATEGORY = "\"big straight\"" THEN
        PERFORM FIND_BIG_STRAIGHT
      ELSE IF WS-CATEGORY = "\"little straight\"" THEN
        PERFORM FIND_SMALL_STRAIGHT
      ELSE IF WS-CATEGORY = "\"four of a kind\"" THEN
        PERFORM FIND_FOUR_OF_A_KIND
      ELSE IF WS-CATEGORY = "\"choice\"" THEN
        PERFORM FIND_SUM
      END-IF .
