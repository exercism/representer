enum Classification {
perfect ,
abundant ,
deficient ,
}
class PerfectNumbers {
Classification classify ( int n ) {
if ( n < = 0 ) {
throw ArgumentError ( "\"Number must be positive\"" );
}
final aliquotSum = _aliquotSum ( n );
if ( aliquotSum == n ) {
return Classification . perfect ;
} else if ( aliquotSum > n ) {
return Classification . abundant ;
} else {
return Classification . deficient ;
}
}
int _aliquotSum ( int n ) {
var sum = 0 ;
for ( var i = 1 ; i < = n ~/ 2 ; i ++ ) {
if ( n % i == 0 ) {
sum += i ;
}
}
return sum ;
}
}
void main () {
final perfectNumbers = PerfectNumbers ();
print ( "\"6 \u2192 ${perfectNumbers.classify(6)}\"" );
print ( "\"28 \u2192 ${perfectNumbers.classify(28)}\"" );
print ( "\"12 \u2192 ${perfectNumbers.classify(12)}\"" );
print ( "\"24 \u2192 ${perfectNumbers.classify(24)}\"" );
print ( "\"8 \u2192 ${perfectNumbers.classify(8)}\"" );
print ( "\"15 \u2192 ${perfectNumbers.classify(15)}\"" );
print ( "\"13 \u2192 ${perfectNumbers.classify(13)}\"" );
}
