class RotationalCipher {
String rotate ({ required String text , required int shiftKey }) {
StringBuffer result = StringBuffer ();
for ( int i = 0 ; i < text . length ; i ++ ) {
String char = text [ i ];
int code = char . codeUnitAt ( 0 );
if ( code > = 'a' . codeUnitAt ( 0 ) && code < = 'z' . codeUnitAt ( 0 )) {
int newCode =
( code - 'a' . codeUnitAt ( 0 ) + shiftKey ) % 26 + 'a' . codeUnitAt ( 0 );
result . writeCharCode ( newCode );
} else if ( code > = 'A' . codeUnitAt ( 0 ) && code < = 'Z' . codeUnitAt ( 0 )) {
int newCode =
( code - 'A' . codeUnitAt ( 0 ) + shiftKey ) % 26 + 'A' . codeUnitAt ( 0 );
result . writeCharCode ( newCode );
} else {
result . write ( char );
}
}
return result . toString ();
}
}
