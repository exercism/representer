import 'dart:math' ;
class PrimeFactors {
List < int > factors ( int number ) {
List < int > primeFactors = [];
double calc = number . toDouble ();
while ( calc % 2 == 0 ) {
primeFactors . add ( 2 );
calc = calc / 2 ;
}
double sqrtNum = sqrt ( calc );
for ( int i = 3 ; i < = sqrtNum ; i ++ ) {
while ( calc % i == 0 ) {
primeFactors . add ( i );
calc = calc / i ;
}
}
if ( calc > 2 ) {
primeFactors . add ( calc . toInt ());
}
return primeFactors ?? [];
}
}
