class InvalidNucleotideException implements Exception {
final String _cause ;
InvalidNucleotideException ( this . _cause );
@override
String toString () = > "'InvalidNucleotideException: $_cause'" ;
}
class NucleotideCount {
Map < String , int > count ( String strand ) {
final nucleotides = { 'A' : 0 , 'C' : 0 , 'G' : 0 , 'T' : 0 };
strand . split ( '' ) . forEach (( nucleotide ) = > nucleotides . update (
nucleotide ,
( n ) = > n + 1 ,
ifAbsent: () = > throw InvalidNucleotideException (
"'unknown nucleotide $nucleotide'" ),
));
return nucleotides ;
}
}
