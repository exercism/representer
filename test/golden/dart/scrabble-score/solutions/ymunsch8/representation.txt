int score ( String name ){
int res = 0 ;
name . toUpperCase () . split ( '' ) . forEach (( letter ) {
switch ( letter ){
case 'A' :
case 'E' :
case 'I' :
case 'O' :
case 'U' :
case 'L' :
case 'N' :
case 'R' :
case 'S' :
case 'T' : res += 1 ; break ;
case 'D' :
case 'G' : res += 2 ; break ;
case 'B' :
case 'C' :
case 'M' :
case 'P' : res += 3 ; break ;
case 'F' :
case 'H' :
case 'V' :
case 'W' :
case 'Y' : res += 4 ; break ;
case 'K' : res += 5 ; break ;
case 'J' :
case 'X' : res += 8 ; break ;
case 'Q' :
case 'Z' : res += 10 ; break ;
}
});
return res ;
}
