class NthPrime {
int prime ( int n ) {
if ( n < 1 ) throw ArgumentError ( "'There is no zeroth prime'" );
int counter = 1 , latestPrime = 2 , nextVal = 3 ;
while ( counter < n ) {
for ( int i = 2 ; i < = nextVal ; i ++ ) {
if ( i == nextVal ) {
counter ++ ;
latestPrime = nextVal ;
}
if ( nextVal % i == 0 ) break ;
}
nextVal ++ ;
}
return latestPrime ;
}
}
