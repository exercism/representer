class Diamond {
static const _alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ' ;
List < String > rows ( String letter ) {
final letterIndex = _alphabet . indexOf ( letter );
if ( letter . length != 1 || letterIndex == - 1 ) {
throw ArgumentError ();
}
final diamond = < String >[];
final rowLastIndex = 2 * letterIndex ;
var left = letterIndex ;
var right = left ;
for ( var i = 0 ; i < = letterIndex ; i ++ ) {
final row = StringBuffer ();
for ( var j = 0 ; j < = rowLastIndex ; j ++ ) {
if ( j == left || j == right ) {
row . write ( _alphabet [ i ]);
} else {
row . write ( "' '" );
}
}
diamond . add ( row . toString ());
left -= 1 ;
right += 1 ;
}
return _mirrorDiamond ( diamond );
}
List < String > _mirrorDiamond ( List < String > diamond ) {
final reversedDiamond = diamond . reversed . skip ( 1 );
return [.. . diamond , .. . reversedDiamond ];
}
}
