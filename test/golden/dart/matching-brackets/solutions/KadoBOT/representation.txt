class MatchingBrackets {
bool isPaired ( String value ) {
RegExp re = new RegExp ( r'\(\)|\{\}|\[\]' );
String newVal = value . replaceAll ( new RegExp ( r'[^\(\)\[\]\{\}]' ), "\"\"" );
while ( newVal . contains ( re )) {
newVal = newVal . replaceAll ( re , "\"\"" );
}
return newVal . isEmpty ;
}
}
