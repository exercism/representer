class MatchingBrackets {
bool isPaired ( String s ) {
RegExp validChars = RegExp ( r'[^\(\)\[\]\{\}]' );
List < String > cleanedInput = s . replaceAll ( validChars , '' ) . split ( '' );
List < String > stack = [];
for ( int i = 0 ; i < cleanedInput . length ; i ++ ) {
switch ( cleanedInput [ i ]) {
case ']' :
if ( stack . isEmpty ) return false ;
if ( stack . removeLast () != '[' ) return false ;
break ;
case '}' :
if ( stack . isEmpty ) return false ;
if ( stack . removeLast () != '{' ) return false ;
break ;
case ')' :
if ( stack . isEmpty ) return false ;
if ( stack . removeLast () != '(' ) return false ;
break ;
default :
stack . add ( cleanedInput [ i ]);
break ;
}
}
return stack . isEmpty ;
}
}
