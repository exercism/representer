class Acronym {
String abbreviate ( String phrase ) {
final cleanPhrase = phrase
. replaceAll ( '-' , "' '" )
. replaceAll ( '_' , "' '" )
. replaceAll ( RegExp ( r"[^\w\s']" ), '' );
final words = cleanPhrase . split ( RegExp ( r'\s+' ));
final acronym = words
. where (( word ) = > word . isNotEmpty )
. map (( word ) = > word [ 0 ] . toUpperCase ())
. join ( '' );
return acronym ;
}
}
