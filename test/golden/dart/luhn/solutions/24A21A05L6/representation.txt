class Luhn {
bool valid ( String input ) {
String number = input . replaceAll ( "' '" , '' );
if ( number . length < = 1 ) {
return false ;
}
if ( ! RegExp ( r'^\d+$' ) . hasMatch ( number )) {
return false ;
}
int sum = 0 ;
bool doubleDigit = false ;
for ( int i = number . length - 1 ; i > = 0 ; i -- ) {
int digit = int . parse ( number [ i ]);
if ( doubleDigit ) {
digit *= 2 ;
if ( digit > 9 ) {
digit -= 9 ;
}
}
sum += digit ;
doubleDigit = ! doubleDigit ;
}
return sum % 10 == 0 ;
}
}
