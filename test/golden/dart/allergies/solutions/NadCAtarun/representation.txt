class Allergies {
static const Map < String , int > kItems = {
'eggs' : 0 ,
'peanuts' : 1 ,
'shellfish' : 2 ,
'strawberries' : 3 ,
'tomatoes' : 4 ,
'chocolate' : 5 ,
'pollen' : 6 ,
'cats' : 7 ,
};
String _toBinary ( int score ) = > score . toRadixString ( 2 );
bool _featuresBit ( String binaryScore , int ? bitPower )
= > bitPower != null
&& bitPower < binaryScore . length
&& binaryScore [ binaryScore . length - bitPower - 1 ] == '1' ;
bool allergicTo ( String allergen , int score ) {
return _featuresBit ( _toBinary ( score ), kItems [ allergen ]);
}
List < String > list ( int score ) = > kItems . keys . where (( allergen ) = > _featuresBit ( _toBinary ( score ), kItems [ allergen ])) . toList ();
}
