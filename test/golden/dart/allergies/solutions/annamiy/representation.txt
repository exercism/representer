(file "lib/allergies.dart")
class Allergies {
final List < String > allergies = [
'eggs' ,
'peanuts' ,
'shellfish' ,
'strawberries' ,
'tomatoes' ,
'chocolate' ,
'pollen' ,
'cats'
];
bool allergicTo ( String item , int score ) = >
toBitList ( score )[ allergies . indexOf ( item )] == '1' ;
List < String > list ( int score ) {
List < String > bitArray = toBitList ( score );
List < String > result = [];
for ( int i = 0 ; i < allergies . length ; i ++ ) {
if ( bitArray [ i ] == '1' ) result . add ( allergies [ i ]);
}
return result ;
}
List < String > toBitList ( int score ) {
return score
. toRadixString ( 2 )
. padLeft ( allergies . length , '0' )
. split ( '' )
. reversed
. toList ();
}
}
