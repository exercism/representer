class Allergies {
bool allergicTo ( String allergy , int allergyScore ) {
final List < String > allergyList = list ( allergyScore );
return allergyList . contains ( allergy );
}
List < String > list ( int allergyScore ) {
final List < bool > allergyMarkers = allergyMarkerListByScore ( allergyScore );
final List < String > allergyList = [];
for ( var i = 0 ; i < allergyMarkers . length ; i ++ ) {
if ( allergyMarkers [ i ]) {
allergyList . add ( allergyNameByIndex ( i ));
}
}
return allergyList ;
}
}
List < bool > allergyMarkerListByScore ( int score ) {
final String binaryScore = score . toRadixString ( 2 );
final Iterable < String > allergyMarkers = binaryScore . split ( '' ) . reversed ;
final Iterable < String > relevantAllergyMarkers = allergyMarkers . take ( 8 );
return relevantAllergyMarkers . map (( marker ) = > marker == '1' ) . toList ();
}
String allergyNameByIndex ( int index ) {
final allergyEnumValue = AllergyTypes . values [ index ];
return allergyEnumValue . name . split ( '.' ) . last ;
}
enum AllergyTypes {
eggs ,
peanuts ,
shellfish ,
strawberries ,
tomatoes ,
chocolate ,
pollen ,
cats ,
}
