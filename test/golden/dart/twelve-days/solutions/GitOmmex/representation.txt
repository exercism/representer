class TwelveDays {
final List < String > _gifts = [
"'a Partridge in a Pear Tree.'" ,
"'two Turtle Doves,'" ,
"'three French Hens,'" ,
"'four Calling Birds,'" ,
"'five Gold Rings,'" ,
"'six Geese-a-Laying,'" ,
"'seven Swans-a-Swimming,'" ,
"'eight Maids-a-Milking,'" ,
"'nine Ladies Dancing,'" ,
"'ten Lords-a-Leaping,'" ,
"'eleven Pipers Piping,'" ,
"'twelve Drummers Drumming,'" ,
];
final List < String > _ordinals = [
'first' , 'second' , 'third' , 'fourth' , 'fifth' , 'sixth' ,
'seventh' , 'eighth' , 'ninth' , 'tenth' , 'eleventh' , 'twelfth'
];
String recite ( int startVerse , int endVerse ) {
List < String > verses = [];
for ( int i = startVerse ; i < = endVerse ; i ++ ) {
final day = _ordinals [ i - 1 ];
final header = "'On the $day day of Christmas my true love gave to me:'" ;
final List < String > lines = [];
for ( int j = i - 1 ; j > = 0 ; j -- ) {
if ( j == 0 && i != 1 ) {
lines . add ( "'and ${_gifts[j]}'" );
} else {
lines . add ( _gifts [ j ]);
}
}
final verse = "'$header ${lines.join(' ')}'" ;
verses . add ( verse );
}
return verses . join ( '\n' );
}
}
