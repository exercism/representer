class Anagram {
List < String > findAnagrams ( String word , List < String > listOfWords ) {
return listOfWords
. where (( e ) = >
sortWord ( word ) == sortWord ( e ) &&
word . toLowerCase () != e . toLowerCase ())
. toList ();
}
String sortWord ( String unsortedWord ) {
return ( unsortedWord . toLowerCase () . split ( '' ). . sort ()) . join ();
}
}
