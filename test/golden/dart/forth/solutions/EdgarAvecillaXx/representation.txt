(file "lib/forth.dart")
enum Errors { DivisionByZero , InvalidDefinition , UnknownCommand }
class Forth {
List < int > stack = [];
late Map < String , dynamic > _operators ;
void _upsertStack ( int res ) {
this . stack
. . removeLast ()
. . removeLast ()
. . add ( res );
}
bool _isNumber ( String value ) = > int . tryParse ( value ) . runtimeType == int ;
Forth () {
this . _operators = < String , dynamic >{
'+' : ( List < int > stack ) = >
_upsertStack ( stack [ stack . length - 2 ] + stack . last ),
'-' : ( List < int > stack ) = >
_upsertStack ( stack [ stack . length - 2 ] - stack . last ),
'*' : ( List < int > stack ) = >
_upsertStack ( stack [ stack . length - 2 ] * stack . last ),
'/' : ( List < int > stack ) {
if ( stack . last == 0 ) throw Errors . DivisionByZero ;
_upsertStack ( stack [ stack . length - 2 ] ~/ stack . last );
},
'dup' : ( List stack ) = > stack . add ( stack . last ),
'drop' : ( List stack ) = > stack . removeLast (),
'swap' : ( List < int > stack ) {
int idx = stack . length - 2 ;
int a = stack [ idx ];
stack
. . removeAt ( idx )
. . add ( a );
},
'over' : ( List stack ) = > stack . add ( stack [ stack . length - 2 ]),
};
}
void evaluate ( String values ) {
List words = < dynamic >[];
List < String > temp = ( values . toLowerCase ()) . split ( "' '" );
try {
if ( temp [ 0 ] == ':' && temp [ temp . length - 1 ] == ';' ) {
if ( _isNumber ( temp [ 1 ])) throw Errors . InvalidDefinition ;
for ( int idx = 2 ; idx < temp . length - 1 ; idx ++ ) {
this . _operators . containsKey ( temp [ idx ])
? words . add ( this . _operators [ temp [ idx ]])
: words . add ( temp [ idx ]);
}
this . _operators [ temp [ 1 ]] = ( List < int > stack ) {
List defs = words ;
defs . forEach (( dynamic el ) {
_isNumber ( el . toString ())
? stack . add ( int . parse ( el . toString ()))
: ( el as Function )( stack );
});
};
} else {
for ( String value in temp ) {
if ( _isNumber ( value )) {
this . stack . add ( int . parse ( value ));
} else {
if ( ! this . _operators . containsKey ( value ))
throw Errors . UnknownCommand ;
dynamic operator = this . _operators [ value ];
operator ( this . stack );
}
}
}
} catch ( e ) {
throw e . runtimeType == Errors
? Exception ( handleError ( e as Errors ))
: Exception ( "'Stack empty'" );
}
}
}
String handleError ( Errors error ) {
if ( error == Errors . DivisionByZero ) {
return "'Division by zero'" ;
} else if ( error == Errors . InvalidDefinition ) {
return "\"Invalid definition\"" ;
} else {
return "\"Unknown command\"" ;
}
}
