class Forth {
List < int > stack = [];
Map < String , List < String >> defs = {};
void evaluate ( String code ) {
final parts = code . toLowerCase () . split ( "\" \"" );
while ( parts . length > 0 ) {
final current_part = parts . removeAt ( 0 );
final def = defs [ current_part ];
if ( def != null ) {
parts . addAll ( def );
continue ;
}
final action = _ACTIONS [ current_part ];
if ( action != null ) {
action ();
continue ;
}
if ( current_part == "\":\"" ) {
final name = parts . removeAt ( 0 );
if ( int . tryParse ( name ) != null ) throw Exception ( "\"Invalid definition\"" );
final subParts = < String >[];
while ( true ) {
final subPart = parts . removeAt ( 0 );
if ( subPart == "\";\"" ) break ;
final subDef = defs [ subPart ];
if ( subDef == null ) {
subParts . add ( subPart );
} else {
subParts . addAll ( subDef );
}
}
defs [ name ] = subParts ;
continue ;
}
try {
stack . add ( int . parse ( current_part ));
} on FormatException {
throw Exception ( "\"Unknown command\"" );
}
}
}
int _peak () {
_ensureStackNotEmpty ();
return stack . last ;
}
int _pop () {
_ensureStackNotEmpty ();
return stack . removeLast ();
}
void _ensureStackNotEmpty () {
if ( stack . length == 0 ) throw Exception ( "\"Stack empty\"" );
}
late final _ACTIONS = {
"\"+\"" : () = > stack . add ( _pop () + _pop ()),
"\"-\"" : () = > stack . add ( - _pop () + _pop ()),
"\"*\"" : () = > stack . add ( _pop () * _pop ()),
"\"/\"" : () {
final v1 = _pop ();
final v2 = _pop ();
try {
stack . add ( v2 ~/ v1 );
} on IntegerDivisionByZeroException {
throw Exception ( "\"Division by zero\"" );
}
},
"\"dup\"" : () = > stack . add ( _peak ()),
"\"drop\"" : _pop ,
"\"swap\"" : () {
final v1 = _pop ();
final v2 = _pop ();
stack . add ( v1 );
stack . add ( v2 );
},
"\"over\"" : () {
final v1 = _pop ();
final v2 = _peak ();
stack . add ( v1 );
stack . add ( v2 );
},
};
}
