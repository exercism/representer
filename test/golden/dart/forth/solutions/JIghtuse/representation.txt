import 'dart:convert' ;
import 'dart:io' ;
import 'forth.dart' ;
void main () {
final forth = Forth ();
stdout . writeln ( "'<Simple forth interpreter>'" );
forth . evaluate ( '?' );
stdin
. transform ( utf8 . decoder )
. transform ( const LineSplitter ())
. listen (( final line ) {
try {
forth . evaluate ( line );
} on Exception catch ( e ) {
stderr . writeln ( e );
} finally {
stdout . writeln ( forth . stack );
}
});
}
import 'dart:io' ;
enum _InterpreterState {
waitingForToken ,
readingUserCommandName ,
readingUserCommandTokens ,
}
enum _TokenKind {
number ,
command ,
define ,
endDefinition ,
unknown ,
}
const _predefinedCommands = {
'+' ,
'-' ,
'*' ,
'/' ,
'?' ,
'dup' ,
'drop' ,
'swap' ,
'over' ,
};
final _isNumber = RegExp ( r'^-?\d+$' );
final _isCommand = RegExp ( r'^[a-z-]+$' );
extension _StringTokenKind on String {
_TokenKind kind () {
assert ( this != '' , "'Should not treat empty string as token'" );
if ( _predefinedCommands . contains ( this ) || _isCommand . hasMatch ( this )) {
return _TokenKind . command ;
}
if ( _isNumber . hasMatch ( this )) {
return _TokenKind . number ;
}
if ( this == ':' ) {
return _TokenKind . define ;
}
if ( this == ';' ) {
return _TokenKind . endDefinition ;
}
return _TokenKind . unknown ;
}
}
class Forth {
Iterable < String > _getTokens ( final String code ) sync * {
for ( final s in code . split ( "' '" )) {
if ( s != '' ) {
yield s . toLowerCase ();
}
}
}
void _ensureHasAtLeast ({ required final int items }) {
if ( stack . length < items ) {
throw Exception ( "'Stack empty'" );
}
}
void _processPredefinedCommand ( final String token ) {
assert ( token != '' , "'Should not treat empty string as token'" );
switch ( token ) {
case '+' :
_ensureHasAtLeast ( items: 2 );
stack . add ( stack . removeLast () + stack . removeLast ());
break ;
case '-' :
_ensureHasAtLeast ( items: 2 );
stack . add ( - stack . removeLast () + stack . removeLast ());
break ;
case '*' :
_ensureHasAtLeast ( items: 2 );
stack . add ( stack . removeLast () * stack . removeLast ());
break ;
case '/' :
_ensureHasAtLeast ( items: 2 );
if ( stack . last == 0 ) {
throw Exception ( "'Division by zero'" );
}
final a = stack . removeLast ();
final b = stack . removeLast ();
stack . add ( b ~/ a );
break ;
case 'dup' :
_ensureHasAtLeast ( items: 1 );
stack . add ( stack . last );
break ;
case 'drop' :
_ensureHasAtLeast ( items: 1 );
stack . removeLast ();
break ;
case 'swap' :
_ensureHasAtLeast ( items: 2 );
final a = stack . removeLast ();
final b = stack . removeLast ();
stack . add ( a );
stack . add ( b );
break ;
case 'over' :
_ensureHasAtLeast ( items: 2 );
stack . add ( stack [ stack . length - 2 ]);
break ;
case '?' :
stdout . writeln ( "'Predefined commands: $_predefinedCommands'" );
stdout . writeln ( "'Known user commands: ${_userCommands.keys}'" );
break ;
default :
stderr . writeln ( "\"Unkown command '$token'\"" );
throw Exception ( "'Unknown command'" );
}
}
void _processUserCommand ( final String command ) {
final commandTokens = _userCommands [ command ];
if ( commandTokens == null ) {
return ;
}
_evaluateTokens ( commandTokens );
}
List < String > _expandTokens ( final List < String > tokens ) {
final expanded = < String >[];
for ( final token in tokens ) {
_expandUserCommand ( token ) . forEach ( expanded . add );
}
return expanded ;
}
List < String > _expandUserCommand ( final String command ) {
final commandTokens = _userCommands [ command ];
return commandTokens == null ? [ command ] : _expandTokens ( commandTokens );
}
void _addUserCommand () {
_userCommands [ _pendingUserCommandName ] =
_expandTokens ( _pendingUserCommandTokens );
_pendingUserCommandName = '' ;
_pendingUserCommandTokens . clear ();
}
void _evaluateTokens ( final Iterable < String > tokens ) {
for ( final token in tokens ) {
final tokenKind = token . kind ();
switch ( _state ) {
case _InterpreterState . waitingForToken :
switch ( tokenKind ) {
case _TokenKind . number :
stack . add ( int . parse ( token ));
break ;
case _TokenKind . define :
_state = _InterpreterState . readingUserCommandName ;
break ;
case _TokenKind . command :
if ( _userCommands . containsKey ( token )) {
_processUserCommand ( token );
} else {
_processPredefinedCommand ( token );
}
break ;
case _TokenKind . unknown :
stderr . writeln ( "\"Unknown word '$token'\"" );
break ;
case _TokenKind . endDefinition :
assert ( false , "'Should not get here'" );
break ;
}
break ;
case _InterpreterState . readingUserCommandName :
if ( tokenKind == _TokenKind . number ) {
throw Exception ( "'Invalid definition'" );
}
_pendingUserCommandName = token ;
_state = _InterpreterState . readingUserCommandTokens ;
break ;
case _InterpreterState . readingUserCommandTokens :
if ( tokenKind == _TokenKind . endDefinition ) {
_state = _InterpreterState . waitingForToken ;
_addUserCommand ();
} else {
_pendingUserCommandTokens . add ( token );
}
break ;
}
}
}
void evaluate ( final String code ) {
_evaluateTokens ( _getTokens ( code ));
}
var _state = _InterpreterState . waitingForToken ;
var _pendingUserCommandName = '' ;
final _pendingUserCommandTokens = < String >[];
final _userCommands = < String , List < String >>{};
final stack = < int >[];
}
