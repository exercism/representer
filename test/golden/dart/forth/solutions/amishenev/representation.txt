import 'dart:collection' ;
class Forth {
Queue < int > stack ;
late Map < String , Function > functions ;
Map < String , String > userFunctions ;
Forth ()
: stack = Queue (),
userFunctions = Map () {
functions = {
'+' : _plus ,
'-' : _minus ,
'*' : _mul ,
'/' : _div ,
'dup' : _dup ,
'drop' : _drop ,
'swap' : _swap ,
'over' : _over ,
};
}
void evaluate ( String s ) {
var fields = s . split ( "' '" ) . map (( e ) = > e . toLowerCase ()) . toList ();
if ( fields . first == ':' && fields . last == ';' ) {
var name = fields [ 1 ];
if ( _isNum ( name )) throw Exception ( "\"Invalid definition\"" );
userFunctions [ name ] =
_parseUserOperation ( fields . getRange ( 2 , fields . length - 1 ));
} else {
for ( var item in fields ) {
if ( userFunctions . containsKey ( item ))
evaluate ( userFunctions [ item ] ! );
else if ( functions . containsKey ( item ))
functions [ item ] ! ();
else if ( _isNum ( item ))
stack . addLast ( int . parse ( item ));
else
throw Exception ( "\"Unknown command\"" );
}
}
}
bool _isNum ( String value ) = > int . tryParse ( value ) != null ;
String _parseUserOperation ( Iterable < String > operations ) {
List < String > result = [];
for ( var item in operations ) {
if ( userFunctions . containsKey ( item )) {
result . add ( _parseUserOperation ([ userFunctions [ item ] ! ]));
} else if ( functions . containsKey ( item ) || _isNum ( item )) {
result . add ( item );
}
}
return result . join ( "\" \"" );
}
int _tryRemoveLast () {
if ( stack . isEmpty ) throw Exception ( "\"Stack empty\"" );
return stack . removeLast ();
}
( int , int ) _tryRemoveLastTwo () {
var second = _tryRemoveLast ();
var first = _tryRemoveLast ();
return ( first , second );
}
void _plus () {
var ( first , second ) = _tryRemoveLastTwo ();
stack . addLast ( first + second );
}
void _minus () {
var ( first , second ) = _tryRemoveLastTwo ();
stack . addLast ( first - second );
}
void _mul () {
var ( first , second ) = _tryRemoveLastTwo ();
stack . addLast ( first * second );
}
void _div () {
var ( first , second ) = _tryRemoveLastTwo ();
if ( second == 0 ) throw Exception ( "\"Division by zero\"" );
stack . addLast ( first ~/ second );
}
void _dup () {
var value = _tryRemoveLast ();
stack . addLast ( value );
stack . addLast ( value );
}
void _drop () {
_tryRemoveLast ();
}
void _swap () {
var ( first , second ) = _tryRemoveLastTwo ();
stack . addLast ( second );
stack . addLast ( first );
}
void _over () {
var ( first , second ) = _tryRemoveLastTwo ();
stack . addLast ( first );
stack . addLast ( second );
stack . addLast ( first );
}
}
