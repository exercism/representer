class UnknownCommandException implements Exception {
final String message ;
const UnknownCommandException ([ this . message = "'Exception: Unknown command'" ]);
@override
String toString () = > message ;
}
class InvalidDefinitionException implements Exception {
final String message ;
const InvalidDefinitionException (
[ this . message = "'Exception: Invalid definition'" ]);
@override
String toString () = > message ;
}
class EmpyStackException implements Exception {
final String message ;
const EmpyStackException ([ this . message = "'Exception: Stack empty'" ]);
@override
String toString () = > message ;
}
class DivisionByZeroException implements Exception {
final String message ;
const DivisionByZeroException ([ this . message = "'Exception: Division by zero'" ]);
@override
String toString () = > message ;
}
class Forth {
static const UserDefinitionPrefix = ':' ;
static const UserDefinitionSuffix = ';' ;
final wordPattern = RegExp ( r'^\w+$' );
final wordSplitPattern = RegExp ( r'\s+' );
List < int > stack = [];
Map < String , List < String >> userDefinitions = {};
Map < String , Function > operations = {
'+' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
self . stack . add ( self . stack . removeLast () + self . stack . removeLast ());
},
'-' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
var b = self . stack . removeLast ();
var a = self . stack . removeLast ();
self . stack . add ( a - b );
},
'*' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
self . stack . add ( self . stack . removeLast () * self . stack . removeLast ());
},
'/' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
if ( self . stack . last == 0 ) throw DivisionByZeroException ();
var b = self . stack . removeLast ();
var a = self . stack . removeLast ();
self . stack . add ( a ~/ b );
},
'dup' : ( Forth self ) {
if ( self . stack . isEmpty ) throw EmpyStackException ();
self . stack . add ( self . stack . last );
},
'drop' : ( Forth self ) {
if ( self . stack . isEmpty ) throw EmpyStackException ();
self . stack . removeLast ();
},
'swap' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
var b = self . stack . removeLast ();
var a = self . stack . removeLast ();
self . stack . addAll ([ b , a ]);
},
'over' : ( Forth self ) {
if ( self . stack . length < 2 ) throw EmpyStackException ();
self . stack . add ( self . stack [ self . stack . length - 2 ]);
},
};
void evaluate ( String sentence ) {
var words = sentence . toLowerCase () . split ( wordSplitPattern );
if ( isUserDefinition ( words )) {
addUserDefinition ( words . sublist ( 1 , words . length - 1 ));
} else {
parse ( words );
}
}
void parse ( List < String > words ) {
var wordsStack = words . reversed . toList ();
while ( wordsStack . isNotEmpty ) {
var word = wordsStack . removeLast ();
var maybeOperand = int . tryParse ( word );
if ( maybeOperand != null ) {
stack . add ( maybeOperand );
} else if ( userDefinitions . containsKey ( word )) {
wordsStack . addAll ( userDefinitions [ word ] !. reversed );
} else if ( operations . containsKey ( word )) {
operations [ word ] ! ( this );
} else {
throw UnknownCommandException ();
}
}
}
void addUserDefinition ( List < String > words ) {
if ( words . length < 2 || isNumber ( words [ 0 ])) {
throw InvalidDefinitionException ();
}
var newWord = words [ 0 ];
var definition = words . sublist ( 1 );
definition =
definition . expand (( word ) = > userDefinitions [ word ] ?? [ word ]) . toList ();
userDefinitions [ newWord ] = definition ;
}
bool isUserDefinition ( List < String > words ) {
return words . first == UserDefinitionPrefix &&
words . last == UserDefinitionSuffix ;
}
bool isNumber ( String word ) {
return int . tryParse ( word ) != null ;
}
}
