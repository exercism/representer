validTriangle?: function [x y z][
sides: sort @[x y z]
a: sides\0 ; shortest side
b: sides\1 ; middle side
c: sides\2 ; longest side
and? -> a > 0
-> c =< a + b
]
equilateral?: $[x y z][
sides: sort @[x y z]
a: sides\0 ; shortest side
c: sides\2 ; longest side
and? -> validTriangle? x y z
-> a = c
]
isosceles?: $[x y z][
sides: sort @[x y z]
a: sides\0 ; shortest side
b: sides\1 ; middle side
c: sides\2 ; longest side
and? -> validTriangle? x y z
-> any? @[a = b b = c]
]
scalene?: function [x y z][
sides: sort @[x y z]
a: sides\0 ; shortest side
b: sides\1 ; middle side
c: sides\2 ; longest side
and? -> validTriangle? x y z
-> all? @[a <> b b <> c]
]
