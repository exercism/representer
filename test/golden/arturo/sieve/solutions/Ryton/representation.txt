(file "src/sieve.art")
;sieve
VERBOSE: false
primes: $[limit][
if one? limit -> return []
knownprimes: []
founddivisor: false
loop 2..(limit) 'number [
'founddivisor: false
if (not? zero? size knownprimes) [
loop (knownprimes) 'p [ ; go over primes small to big
if VERBOSE -> print ~"lets check div |p|"
if (zero? (number % p)) [
'founddivisor: true
break ; do nothing, do not add it.
]
] ; outside of loop
]
if not? founddivisor ['knownprimes ++ @[number]]
]
return knownprimes
]
