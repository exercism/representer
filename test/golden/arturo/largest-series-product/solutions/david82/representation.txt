(file "src/largest-series-product.art")
largestProduct: function [str span][
; Handle special cases first
if span = 0 -> return 1
if span < 0 -> return null
if empty? str [
if span = 0 -> return 1
return null
]
if span > size str -> return null
; Validate digits using pattern matching
if not? match? str {^[0-9]+$} -> return null
; Convert string to digits
digits: map split str => [to :integer &]
; Calculate products
maxProduct: 0
endIndex: (size str) - span
loop 0..endIndex 'i [
product: 1
loop 0..span-1 'j [
product: product * digits\[i+j]
]
if product > maxProduct [
maxProduct: product
]
]
maxProduct
]
