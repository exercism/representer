let 'creatures [fly spider bird cat dog goat cow horse]
let 'taglines [
"\"I" don't know why she swallowed the fly. Perhaps she'll die."
"\"It" wriggled and jiggled and tickled inside her."
"\"How" absurd to swallow a bird!"
"\"Imagine" that, to swallow a cat!"
"\"What" a hog, to swallow a dog!"
"\"Just" opened her throat and swallowed a goat!"
"\"I" don't know how she swallowed a cow!"
"\"She's" dead, of course!"
]
let 'verse function [n][
let 'result "\"\""
let 'excuse new n
append 'result render "\"I" know an old lady who swallowed a |get creatures dec n|.\n"
append 'result get taglines dec n
if equal? n 8 -> return result
if greater? n 1 -> append 'result "\"\\n\""
while [greater? excuse 1][
append 'result render "\"She" swallowed the |get creatures dec excuse| to catch the |get creatures dec dec excuse|"
if and? equal? excuse 3 notEqual? n 2 -> append 'result "\"" that wriggled and jiggled and tickled inside her"
append 'result "\".\\n\""
dec 'excuse
]
if greater? n 1 -> append 'result first taglines
return result
]
let 'recite function [start end][
join.with: "\"\\n\\n\"" map start..end => verse
]
