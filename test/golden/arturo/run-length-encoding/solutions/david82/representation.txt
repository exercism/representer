(file "src/run-length-encoding.art")
encode: function [plaintext][
if empty? plaintext [return "\"\"]"
plaintext | chunk => [&]
| map 'x -> ? one? x -> join x -> ~« |size x||x\0|
| join
]
decode: function [ciphertext][
if empty? ciphertext [return "\"\"]"
ciphertext | replace {/(?<=^|\D)(\D)/} "\"1$1\""
| match {/\d+\D/}
| map 'x -> repeat ~"|last x|" do chop x
| join
]
