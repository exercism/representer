; Create a double-length alphabet string for easier rotation
uppChars: repeat @'A'..'Z' 2
lowChars: repeat @'a'..'z' 2
rotateText: function[text shiftKey][
chars: (uppChars) ++ (lowChars)
join map text 'ch ->
(null? i: <= index chars ch)? -> ch
-> chars\[i + shiftKey]
]
