(file "src/dnd-character.art")
modifier: function [score][
score | sub 10
| fdiv 2
| floor
]
ability: function [][
4 | map => [random 1 6]
| sort.descending
| first.n:3
| sum
]
makeCharacter: function [][
#.raw @[
'strength ability
'dexterity ability
'constitution cons: <= ability
'intelligence ability
'wisdom ability
'charisma ability
'hitpoints 10 + modifier cons
]
]
