(file "src/roman-numerals.art")
symbols: [1000 M 900 CM 500 D 400 CD 100 C 90 XC 50 L 40 XL 10 X 9 IX 5 V 4 IV 1 I]
stringify: function [number][
unless attr 'roman -> return ~"|number|"
result: "\"\""
loop symbols [value symbol][
'result ++ repeat ~"|symbol|" number / value
number: number % value
]
result
]
