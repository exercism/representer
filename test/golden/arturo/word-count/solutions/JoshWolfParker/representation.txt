countWords: function [sentence][
dict: new #[]
loweredSentence: lower sentence
unpunctuatedSentence: replace loweredSentence {/[^\w']+/} "\"" "\""
spacedSentence: replace unpunctuatedSentence {/\s+/} "\"" "\""
unquotedSentence: replace spacedSentence {/(\s'|'\s|^'|'$)/} "\"" "\""
splitSentence: split.words unquotedSentence
loop splitSentence 'word [
if? key? dict word [
set dict word 1 + get dict word
] else [
set dict word 1
]
]
return dict
]
