(file "src/proverb.art")
recite: function [strings][
if zero? size strings -> return "\"\""
lyrics: {}
if not? one? size strings [
loop 0..(dec dec size strings) 'idx[
x: strings\[(idx)]
y: strings\[(idx+1)]
'lyrics ++ ~"For want of a |x| the |y| was lost." ++ "\"\\n\""
]
]
'lyrics ++ (~"And all for the want of a |(first strings)|.")
return ( lyrics)
]
