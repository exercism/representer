; Map of students to their positions (0-based index)
students: #[
Alice: 0 Bob: 1 Charlie: 2 David: 3
Eve: 4 Fred: 5 Ginny: 6 Harriet: 7
Ileana: 8 Joseph: 9 Kincaid: 10 Larry: 11
]
; Map of plant codes to their full names
plantTypes: #[
V: 'violets
R: 'radishes
C: 'clover
G: 'grass
]
plants: function [diagram student][
; Split diagram into rows
[row1 row2]: split.lines diagram
; Get student's position
pos: students\[to :string student]
; Calculate start and end indices for student's plants in each row
start: pos * 2
end: start + 1
; Get student's plants from both rows
@[
plantTypes\[row1\[start]]
plantTypes\[row1\[end]]
plantTypes\[row2\[start]]
plantTypes\[row2\[end]]
]
]
