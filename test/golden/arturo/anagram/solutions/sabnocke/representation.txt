(file "src/anagram.art")
count_map: function [str :string] [
dct: #[]
loop str 'x [
s: to :string lower x
if contains? keys dct s [
tmp: get dct s
tmp: inc tmp
set dct s tmp
continue
]
set dct s 1
]
return dct
]
findAnagrams: function[subject candidates][
counted: count_map subject
lower_subject: lower subject
return select candidates [item][
a: notEqual? lower_subject lower item
b: equal? counted count_map item
and? a b
]
]
