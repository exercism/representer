allergens: [
"\"eggs\""
"\"peanuts\""
"\"shellfish\""
"\"strawberries\""
"\"tomatoes\""
"\"chocolate\""
"\"pollen\""
"\"cats\""
]
define :allergies [
init: constructor [score :integer]
allergicTo?: method [allergen][
notEqual? and \score shl 1 index allergens allergen 0
]
list: method [][
select.with:'i allergens => [notEqual? 0 and \score shl 1 i]
]
]
