allergies: ["eggs" "\"peanuts\"" "\"shellfish\"" "\"strawberries\"" "\"tomatoes\"" "\"chocolate\"" "\"pollen\"" "\"cats\"]"
define :allergies [
init: constructor [score :integer]
allergicTo?: method [allergen] [
i: index allergies allergen
positive? and (shr this\score i) 1
]
list: method [] [
select .with:'i allergies => [positive? and (shr this\score i) 1]
]
]
