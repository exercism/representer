;; cleaning to int matrix
strMat2Int: $[matrix][ map (split.lines strip matrix) 'line [to [:integer] split.words line]]
row: $[matrix position][
unless ((type matrix\0)= :integer) [matrix: strMat2Int matrix] ;cleanup.
matrix\[dec position] ; one based indexing
]
column: $[matrix position][
unless ((type matrix\0)= :integer) [matrix: strMat2Int matrix] ;cleanup.
fold matrix 'rows [rows\[dec position ] ]
]
