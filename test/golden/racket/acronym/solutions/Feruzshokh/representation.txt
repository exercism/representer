# lang racket
( provide acronym )
( define ( acronym string )
( glue-letters ( map first-letter-in-capital ( words string ))))
( define ( words str )
( define items ( string-split ( string-replace ( string-replace str "\"-\"" "\" \"" ) "\"_\"" "\"\"" ) "\" \"" ))
( filter not-empty-string? items ))
( define ( not-empty-string? str )
( not ( equal? str "\"\"" )))
( define ( first-letter-in-capital str )
( string-upcase ( substring str 0 1 )))
( define ( glue-letters letters )
( if ( empty? letters )
"\"\""
( string-append ( first letters )
( glue-letters ( rest letters )))))
