# lang racket
( require racket/set )
( provide isogram? )
( define ( isogram? s )
( define chars ( mutable-set ))
( andmap
( lambda ( char ) ( cond
[( or ( equal? char #\- ) ( char-whitespace? char )) #t ]
[( set-member? chars char ) #f ]
[ else ( begin ( set-add! chars char ) #t )]))
( string->list ( string-downcase s ))
)
)
