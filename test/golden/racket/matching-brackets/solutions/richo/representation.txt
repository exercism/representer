# lang racket
( provide balanced? )
( define ( check-balance str chr )
( cond
[( null? str ) ( null? chr )]
[( eq? #\( ( car str ))
( check-balance ( cdr str ) ( cons #\) chr ))]
[( eq? # \ [ ( car str ))
( check-balance ( cdr str ) ( cons # \ ] chr ))]
[( eq? # \ { ( car str ))
( check-balance ( cdr str ) ( cons # \ } chr ))]
[( member ( car str ) ' ( #\ ) # \ } # \ ]))
( and ( not ( null? chr ))
( eq? ( car str ) ( car chr ))
( check-balance ( cdr str ) ( cdr chr )))]
[ else ( check-balance ( cdr str ) chr )]))
( define ( balanced? str )
( let (( test ( string->list str )))
( check-balance test ' ())))
