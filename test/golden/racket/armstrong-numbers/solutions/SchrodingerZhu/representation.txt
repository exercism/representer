# lang racket
( provide armstrong-number? )
( define ( split-digits n [ acc ' ()] [ counter 0 ])
( if ( = n 0 )
( values acc counter )
( split-digits ( quotient n 10 ) ( cons ( remainder n 10 ) acc ) ( + counter 1 ))))
( define ( armstrong-number? n )
( let*-values
([( digits count ) ( split-digits n )]
[( sum ) ( apply + ( map ( lambda ( x ) ( expt x count )) digits ))])
( = sum n )))
