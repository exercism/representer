(file "armstrong-numbers.rkt")
# lang racket
( provide armstrong-number? )
( define ( integer->digits n [ acc ' ()])
( define-values ( q r ) ( quotient/remainder n 10 ))
( if ( zero? q )
( append acc ( list r ))
( integer->digits q ( append acc ( list r )))))
( define ( sum-power-digits n )
( let* ([ digits ( integer->digits n )]
[ exponent ( length digits )])
( foldl ( lambda ( digit sum ) ( + sum ( expt digit exponent ))) 0 digits )))
( define ( armstrong-number? n )
( = n ( sum-power-digits n )))
