(file "armstrong-numbers.rkt")
# lang racket
( provide armstrong-number? )
( define ( number-length n )
( cond
[( < n 10 ) 1 ]
[ else ( + 1 ( number-length ( / n 10 )))]))
( define ( armstrong-sum n length )
( cond
[( < n 10 ) ( expt n length )]
[ else ( + ( expt ( modulo n 10 ) length ) ( armstrong-sum ( quotient n 10 ) length ))]))
( define ( armstrong-number? n )
( let ([ num-length ( number-length n )])
( = n ( armstrong-sum n num-length ))))
