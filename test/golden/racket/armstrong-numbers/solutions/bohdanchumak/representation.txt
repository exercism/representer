(file "armstrong-numbers.rkt")
# lang racket
( provide armstrong-number? )
( define ( armstrong-number? n )
( define lon ( number->digits n ))
( = n ( foldr + 0 ( lon-expt lon ( length lon )))))
( define ( number->digits n )
( map ( lambda ( c ) ( - ( char->integer c ) ( char->integer #\0 )))
( string->list ( number->string n ))))
( define ( lon-expt lon p )
( map ( lambda ( n ) ( expt n p )) lon ))
