( define ( recurse-digit n )
( if ( = ( quotient n 10 ) 0 )
n
( list ( remainder n 10 ) ( recurse-digit ( quotient n 10 )))))
( define ( digits n )
( flatten ( recurse-digit n )))
( define ( armstrong-number? n )
( = n
( apply +
( map ( lambda ( x )
( expt x ( string-length ( format "\"~v\"" n )))) ( digits n ) ))))
