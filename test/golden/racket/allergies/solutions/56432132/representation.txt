# lang racket
( provide list-allergies allergic-to? )
( define allergies ' ( "\"eggs\"" "\"peanuts\"" "\"shellfish\"" "\"strawberries\""
"\"tomatoes\"" "\"chocolate\"" "\"pollen\"" "\"cats\"" ))
( define ( list-allergies score )
( for/list ([( a i ) ( in-indexed allergies )]
#:when ( bitwise-bit-set? score i ))
( list-ref allergies i )))
( define ( allergic-to? str score )
( bitwise-bit-set? score ( index-of allergies str )))
