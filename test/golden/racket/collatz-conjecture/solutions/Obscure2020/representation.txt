(program
  (extension racket)
  (list provide PLACEHOLDER_1)
  (list define
    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_2))
    (list define
      (list
        (definition PLACEHOLDER_3)
        (definition PLACEHOLDER_2)
        (definition PLACEHOLDER_4))
      (list cond
        (list
          (list = PLACEHOLDER_2 1) PLACEHOLDER_4)
        (list
          (list even? PLACEHOLDER_2)
          (list PLACEHOLDER_3
            (list / PLACEHOLDER_2 2)
            (list + PLACEHOLDER_4 1)))
        (list else
          (list PLACEHOLDER_3
            (list +
              (list * PLACEHOLDER_2 3) 1)
            (list + PLACEHOLDER_4 1)))))
    (list cond
      (list
        (list inexact? PLACEHOLDER_2)
        (list error "Inexact input."))
      (list
        (list < PLACEHOLDER_2 1)
        (list error "Input too small."))
      (list else
        (list PLACEHOLDER_3 PLACEHOLDER_2 0)))))
