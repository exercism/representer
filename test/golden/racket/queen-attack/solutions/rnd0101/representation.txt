  (extension racket)
  (list provide PLACEHOLDER_1 PLACEHOLDER_2)

    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_3)
      (definition PLACEHOLDER_4))

      (list
        (list and

          (list < -1 PLACEHOLDER_4 8))
        (list cons PLACEHOLDER_3 PLACEHOLDER_4))
      (list else
        (list error
          (list quote PLACEHOLDER_1) "row or column out of bounds: ~a ~a" PLACEHOLDER_3 PLACEHOLDER_4))))
  (list define
    (list
      (definition PLACEHOLDER_2)
      (definition PLACEHOLDER_5)
      (definition PLACEHOLDER_6))
    (list let
      (list
        (list
          (definition PLACEHOLDER_7)
          (list car PLACEHOLDER_5))

          (definition PLACEHOLDER_8)
          (list cdr PLACEHOLDER_5))
        (list
          (definition PLACEHOLDER_9)
          (list car PLACEHOLDER_6))
        (list
          (definition PLACEHOLDER_10)
          (list cdr PLACEHOLDER_6)))
      (list or
        (list = PLACEHOLDER_7 PLACEHOLDER_9)
        (list = PLACEHOLDER_8 PLACEHOLDER_10)
        (list =
          (list - PLACEHOLDER_7 PLACEHOLDER_9)
          (list - PLACEHOLDER_8 PLACEHOLDER_10))
        (list =
          (list - PLACEHOLDER_9 PLACEHOLDER_7)
          (list - PLACEHOLDER_8 PLACEHOLDER_10))))))
