(program
  (extension racket)
  (list provide PLACEHOLDER_1)
  (list define
    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_2)
      (definition PLACEHOLDER_3))
    (list if
      (list not
        (list equal?
          (list string-length PLACEHOLDER_2)
          (list string-length PLACEHOLDER_3)))
      (list error "strings should be of equal length")
      (list length
        (list filter
          (list lambda
            (list
              (definition PLACEHOLDER_4))
            (list not
              (list char=?
                (list car PLACEHOLDER_4)
                (list cadr PLACEHOLDER_4))))
          (list map list
            (list string->list PLACEHOLDER_2)
            (list string->list PLACEHOLDER_3)))))))
