# lang racket
( provide triangle? )
( define ( valid-triangle? sides )
( < ( apply max sides )
( / ( apply + sides ) 2 )))
( define ( equal-sides a b c )
( + ( if ( = a b ) 1 0 )
( if ( = a c ) 1 0 )
( if ( = b c ) 1 0 )))
( define ( triangle? sides kind )
( and ( valid-triangle? sides )
( let ([ equals ( apply equal-sides sides )])
( case kind
[ 'equilateral ( = equals 3 )]
[ 'isosceles ( > equals 0 )]
[ 'scalene ( = equals 0 )]))))
