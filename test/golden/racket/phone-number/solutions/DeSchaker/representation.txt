(program
  (extension racket)
  (list provide PLACEHOLDER_1)
  (list define
    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_2))
    (list let*
      (list
        (list
          (definition PLACEHOLDER_3)
          (list filter char-numeric?
            (list string->list PLACEHOLDER_2)))
        (list
          (definition PLACEHOLDER_4)
          (list lambda
            (list
              (definition PLACEHOLDER_5))
            (list not
              (list or
                (list equal? PLACEHOLDER_5 "#\\0")
                (list equal? PLACEHOLDER_5 "#\\1"))))))
      (list cond
        (list
          (list and
            (list equal?
              (list length PLACEHOLDER_3) 10)
            (list PLACEHOLDER_4
              (list car PLACEHOLDER_3))
            (list PLACEHOLDER_4
              (list fourth PLACEHOLDER_3)))
          (list list->string PLACEHOLDER_3))
        (list
          (list and
            (list equal?
              (list length PLACEHOLDER_3) 11)
            (list equal?
              (list car PLACEHOLDER_3) "#\\1")
            (list PLACEHOLDER_4
              (list second PLACEHOLDER_3))
            (list PLACEHOLDER_4
              (list fifth PLACEHOLDER_3)))
          (list list->string
            (list rest PLACEHOLDER_3)))
        (list else
          (list error "number not right"))))))
