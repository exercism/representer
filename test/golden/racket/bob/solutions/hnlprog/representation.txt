# lang racket
( provide response-for )
( define ( response-for [ input "\"\"" ])
( define str ( string-trim input ))
( cond
[( silence? str ) "\"Fine. Be that way!\"" ]
[( and [ question? str ]
[ yelling? str ]) "\"Calm down, I know what I'm doing!\"" ]
[( question? str ) "\"Sure.\"" ]
[( yelling? str ) "\"Whoa, chill out!\"" ]
[ else "\"Whatever.\"" ]))
( define ( question? str )
( string-suffix? str "\"?\"" ))
( define ( silence? str )
( equal? str "\"\"" ))
( define ( yelling? str )
( and
[ not ( equal? ( string-downcase str ) str )]
[ equal? ( string-upcase str ) str ]))
