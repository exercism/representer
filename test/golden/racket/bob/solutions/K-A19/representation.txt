(program
  (extension racket)
  (list provide PLACEHOLDER_1)
  (list define
    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_2))
    (list define
      (definition PLACEHOLDER_3)
      (list string-trim PLACEHOLDER_2))
    (list define
      (definition PLACEHOLDER_4)
      (list if
        (list regexp-match
          (regex #px "\\?$") PLACEHOLDER_3) #t #f))
    (list define
      (definition PLACEHOLDER_5)
      (list regexp-match*
        (regex #px "[a-zA-Z]+") PLACEHOLDER_3))
    (list define
      (definition PLACEHOLDER_6)
      (list and
        (list not
          (list null? PLACEHOLDER_5))
        (list for/and
          (list
            (list
              (definition PLACEHOLDER_7) PLACEHOLDER_5))
          (list regexp-match?
            (regex #px "^[A-Z]+$") PLACEHOLDER_7))))
    (list define
      (definition PLACEHOLDER_8)
      (list and PLACEHOLDER_4 PLACEHOLDER_6))
    (list define
      (definition PLACEHOLDER_9)
      (list if
        (list regexp-match
          (regex #px "^\\s*$") PLACEHOLDER_3) #t #f))
    (list cond
      (list PLACEHOLDER_9 "Fine. Be that way!")
      (list PLACEHOLDER_8 "Calm down, I know what I'm doing!")
      (list PLACEHOLDER_6 "Whoa, chill out!")
      (list PLACEHOLDER_4 "Sure.")
      (list else "Whatever."))))
