( provide grep )
( define ( is-match? pattern line [ inverted? #f ] [ case-insensitive? #f ] [ entire-line? #f ])
( define pattern_
( if case-insensitive? ( string-upcase pattern ) pattern ))
( define line_
( if case-insensitive? ( string-upcase line ) line ))
( define is-pattern-match?
( if entire-line?
( string=? pattern_ line_ )
( string-contains? line_ pattern_ )))
( if inverted?
( not is-pattern-match? )
is-pattern-match? ))
( define ( grep-in-file pattern filename [ inverted? #f ] [ case-insensitive? #f ] [ entire-line? #f ])
( define input
( open-input-file filename #:mode 'text ))
( define ( search found line-number )
( define line ( read-line input ))
( if ( eof-object? line )
( reverse found )
( if ( is-match? pattern line inverted? case-insensitive? entire-line? )
( search
( cons ( list line-number line ) found )
( + 1 line-number ))
( search found ( + 1 line-number )))))
( search ' () 1 ))
( define ( contains? item lst )
( define ( search remaining )
( if ( null? remaining )
#f
( let ([ head ( car remaining )]
[ tail ( cdr remaining )])
( if ( string=? item head )
#t
( search tail )))))
( search lst ))
( define ( normalize-output matches filename single-file? line-number? )
( define ( prepend-filename line ) ( format "\"~A:~A\"" filename ( cadr line )))
( define ( prepend-linenumber line ) ( format "\"~A:~A\"" ( car line ) ( cadr line )))
( define ( prepend-filename-linenumber line )
( format "\"~A:~A:~A\"" filename ( car line ) ( cadr line )))
( if ( and single-file?
( not line-number? ))
( map cadr matches )
( if ( and single-file? line-number? )
( map prepend-linenumber matches )
( if ( not line-number? )
( map prepend-filename matches )
( map prepend-filename-linenumber matches )))))
( define ( grep flags pattern files )
( define line-number? ( contains? "\"-n\"" flags ))
( define filename-only? ( contains? "\"-l\"" flags ))
( define case-insensitive? ( contains? "\"-i\"" flags ))
( define inverted? ( contains? "\"-v\"" flags ))
( define entire-line? ( contains? "\"-x\"" flags ))
( define single-file? ( = 1 ( length files )))
( define ( search results files_ )
( if ( null? files_ )
results
( let* ([ filename ( car files_ )]
[ tail ( cdr files_ )]
[ matches ( grep-in-file pattern filename inverted? case-insensitive? entire-line? )])
( if ( null? matches )
( search results tail )
( if filename-only?
( search ( append results ( list filename )) tail )
( let ([ output ( normalize-output matches filename single-file? line-number? )])
( search ( append results output ) tail )))))))
( define result
( if ( = 0 ( length files ))
"\"\""
