(file "grep.rkt")
# lang racket
( provide grep )
( define ( grep flags pattern files )
( let* ([ -n ( member "\"-n\"" flags )]
[ -l ( member "\"-l\"" flags )]
[ -x ( member "\"-x\"" flags )]
[ -v ( member "\"-v\"" flags )]
[ pattern ( regexp
( if ( member "\"-i\"" flags )
( format "\"(?i:~a)\"" pattern )
pattern ))]
[ matches
( apply
append
( for/list ([ f files ])
( with-input-from-file f
( lambda ()
( for/list ([ linum ( in-naturals 1 )]
[ line ( in-lines )]
#:when ( xor -v
(( if -x regexp-match-exact? regexp-match? )
pattern line )))
` (( filename . , f )
( linum . , linum )
( line . , line )))))))])
( if -l
( remove-duplicates
( for/list ([ m matches ])
( cdr ( assq 'filename m ))))
( if -n
( if ( > ( length files ) 1 )
( for/list ([ m matches ])
( format "\"~a:~a:~a\""
( cdr ( assq 'filename m ))
( cdr ( assq 'linum m ))
( cdr ( assq 'line m ))))
( for/list ([ m matches ])
( format "\"~a:~a\""
( cdr ( assq 'linum m ))
( cdr ( assq 'line m )))))
( if ( > ( length files ) 1 )
( for/list ([ m matches ])
( format "\"~a:~a\""
( cdr ( assq 'filename m ))
( cdr ( assq 'line m ))))
( for/list ([ m matches ])
( cdr ( assq 'line m ))))))))
