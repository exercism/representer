(file "rna-transcription.rkt")
# lang racket
( define ( transcribe-nucleotide n ) ( cond
[( equal? n #\G ) #\C ]
[( equal? n #\C ) #\G ]
[( equal? n #\T ) #\A ]
[( equal? n #\A ) #\U ]))
( define ( to-rna dna )
( define dna-list ( string->list dna ))
( define rna-list ( map transcribe-nucleotide dna-list ))
( apply string rna-list ))
( provide to-rna )
