(file "rna-transcription.rkt")
# lang racket
( define dna-to-rna # hash (( "\"G\"" . "\"C\"" )
( "\"C\"" . "\"G\"" )
( "\"T\"" . "\"A\"" )
( "\"A\"" . "\"U\"" )))
( define ( to-rna dna )
( define ( lookup k ) ( hash-ref dna-to-rna k ))
( regexp-replace* # px" . "\"" dna lookup ))
( provide to-rna )
