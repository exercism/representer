# lang racket
( provide to-rna )
( define ht ( hash #\G "\"C\""
#\C "\"G\""
#\T "\"A\""
#\A "\"U\"" ))
( define ( to-rna dna )
( if ( dna? dna )
( string-join ( for/list ([ i ( string->list dna )])
( hash-ref ht i )) "\"\"" )
( error "\"no dna sequence\"" )))
( define ( dna? sequence )
( andmap ( λ ( x ) ( for/or ([ type ( hash-keys ht )])
( equal? type x ))) ( string->list sequence )))
