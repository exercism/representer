# lang racket
( provide color-code )
( define color-values
' (( "\"black\"" . 0 )
( "\"brown\"" . 1 )
( "\"red\"" . 2 )
( "\"orange\"" . 3 )
( "\"yellow\"" . 4 )
( "\"green\"" . 5 )
( "\"blue\"" . 6 )
( "\"violet\"" . 7 )
( "\"grey\"" . 8 )
( "\"white\"" . 9 )))
( define ( value color )
( cdr ( assoc color color-values )))
( define ( magnitude colors )
( define digits ( map value colors ))
( define tens ( * 10 ( car digits )))
( define units ( cadr digits ))
( define scale ( expt 10 ( caddr digits )))
( * scale ( + tens units )))
( define ( scale-down resistance )
( do ([ t 0 ( add1 t )]
[ r resistance ( quotient r 1000 )])
(( or ( zero? r ) ( not ( = ( remainder r 1000 ) 0 ))) ( cons r t ))))
( define ( prefix-name si )
( list-ref ' ( "\"\"" "\"kilo\"" "\"mega\"" "\"giga\"" ) si ))
( define ( color-code colors )
( define r ( magnitude colors ))
( define sf ( scale-down r ))
( define msd ( car sf ))
( define prefix ( prefix-name ( cdr sf )))
( ~a msd "\" \"" prefix "\"ohms\"" ))
