-- Grains on a given square (1..64).
def square (n: i32): u64 =
-- Guard the valid range and return the computed value.
assert (1 <= n && n <= 64)
(1 << u64.i32 (n - 1))
-- Total grains on the board: sum_{i=0}^{63} (1 << i) = 2^64 - 1.
def total (): u64 =
reduce (+) 0 (map (\(i: i64) -> 1 << u64.i64 i) (iota 64))
