def to_upper(c:u8) = if c >= 'a' && c <= 'z' then c-32 else c
def is_upper(c:u8) = c >= 'A' && c <= 'Z'
-- "\"rising" fronts" of a given boolean array (including first element)
def rising [n] (bs:[n]bool) :[n]bool =
let inner = map2 (\a b -> a && (not b))
(bs[1:n] :> [n-1]bool) (bs[0:n-1] :> [n-1]bool)
in (concat [bs[0]] inner :> [n]bool)
entry abbreviate (phrase:[]u8) =
let up = map to_upper phrase
in
map (\a -> is_upper a || a =='\'') up
|> rising
|> zip up
|> filter (.1)
|> unzip
|> (.0)
