def cost: []i32 = [800, 2 * 760, 3 * 720, 4 * 640, 5 * 600]
def count (basket: []i32): []i32 =
let m = i32.maximum basket
in loop xs = (replicate (i64.i32 m) 0)
for b in basket do xs with [b - 1] = 1 + xs[b - 1]
def insert [n] (xs: *[n]i32) (i: i64): [n]i32 =
if i == 0 || xs[i - 1] <= xs[i] then xs
else
loop ys = xs for j < i do
if ys[i - j - 1] <= ys[i - j] then ys
else
let u = copy ys[i - j - 1]
let v = copy ys[i - j]
in ys with [i - j - 1] = v with [i - j] = u
def sort [n] (xs: *[n]i32): [n]i32 =
loop ys = xs for i < length xs do insert ys i
def add_title (c: i32) (groups: *[]i32): []i32 =
let (g, b) =
loop (g, b) = (copy groups, c)
for i in [2, 3, 1, 0] do
let t = i32.min b groups[i]
in (g with [i + 1] = g[i + 1] + t with [i] = g[i] - t, b - t)
in g with [0] = g[0] + b
def distribute (copies: []i32): []i32 =
loop groups = (replicate 5 0)
for c in copies do
add_title c groups
def total (basket: []i32): i32 =
let cs = count (copy basket)
let groups = sort cs |> reverse |> distribute
in
map2 (*) groups cost |> i32.sum
