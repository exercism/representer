def clean (phrase: []u8): []u8 =
map (\c -> if c >= 'A' && c <= 'Z' then c | 0x20 else c) phrase
|> filter (\c -> c >= 'a' && c <= 'z' || c >= '0' && c <= '9')
def reflect (letters: []u8): []u8 =
map (\c -> if c >= 'a' && c <= 'z' then 'a' + 'z' - c else c) letters
def group (letters: []u8): []u8 =
loop g = []
for i < length letters do
if i != 0 && i % 5 == 0
then g ++ [' ', letters[i]]
else g ++ [letters[i]]
def encode (phrase: []u8): []u8 =
clean phrase |> reflect |> group
def decode (phrase: []u8): []u8 =
clean phrase |> reflect
