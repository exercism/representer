-- Lower-case an ASCII byte.
let to_lower (c: u8): u8 =
if c >= 65 && c <= 90 then c + 32 else c -- 'A'..'Z' -> 'a'..'z'
-- Keep only alphabetic ASCII letters.
let is_alpha (c: u8): bool =
c >= 97 && c <= 122 -- 'a'..'z'
let is_isogram (phrase: []u8): bool =
-- Normalize: lower-case and drop non-letters (spaces, hyphens, etc.).
let letters = filter is_alpha (map to_lower phrase)
-- Scan letters, setting a bit for each seen letter.
-- If a bit is already set, we found a duplicate.
let (_, ok) =
loop (mask: u32, ok: bool) = (0, true) for c in letters do
let pos: u32 = u32.u8 (c - 97) -- 0 for 'a', 25 for 'z'
let bit: u32 = 1 << pos
let dup = (mask & bit) != 0
in (mask | bit, ok && not dup)
in ok
