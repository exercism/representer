def lsum (digits: []u8): i32 =
let (sum, _) =
loop (s, dbl) =
(0, 2 - length digits % 2)
for c in digits do
let d = i64.u8 (c - '0')
let x = dbl * d + if dbl == 2 && d > 4 then 1 else 0
in (s + x, 3 - dbl)
in i32.i64 sum
def valid (value: []u8): bool =
let digits = filter (!= ' ') value
in
length digits > 1 &&
all (\d -> d >= '0' && d <= '9') digits &&
lsum digits % 10 == 0
