def initial_guess (n: i32): i32 =
let (_, t) = loop (x, a) = (n, 0) while x != 0 do (x / 2, a + 1)
in 2**((1 + t)/2)
def step (n: i32, x: i32): i32 =
(x + n / x) / 2
def square_root (n: i32): i32 =
let g = initial_guess n
let (r, _) = loop (x, y) = (g, step (n, g)) while x > y do (y, step (n, y))
in r
