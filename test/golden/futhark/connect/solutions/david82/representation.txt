-- Board access with bounds checking
def contents [m] [n] (board: [m][n]u8) (r: i64) (c: i64): u8 =
if r >= 0 && r < m && c >= 0 && c < n then board[r][r + 2 * c] else '.'
-- Calculate columns in hexagonal board
def cols (m: i64) (n: i64): i64 = 1 + (n - m) / 2
-- Position type
type pos = (i64, i64)
-- Get edge positions for any player and edge type
def edge_pos [m] [n] (board: [m][n]u8) (player: u8) (edge: i32): []pos =
let c = cols m n
let positions = if edge == 0 then map (\col -> (0, col)) (iota c) -- top
else if edge == 1 then map (\row -> (row, 0)) (iota m) -- left
else if edge == 2 then map (\col -> (m-1, col)) (iota c) -- bottom
else map (\row -> (row, c-1)) (iota m) -- right
in filter (\(r, c) -> contents board r c == player) positions
-- Starting positions
def startO [m] [n] (board: [m][n]u8): []pos = edge_pos board 'O' 0
def startX [m] [n] (board: [m][n]u8): []pos = edge_pos board 'X' 1
-- Final position checks
def endO [m] [n] (board: [m][n]u8) (r: i64) (c: i64): bool = r == m-1 && contents board r c == 'O'
def endX [m] [n] (board: [m][n]u8) (r: i64) (c: i64): bool = c == cols m n-1 && contents board r c == 'X'
-- Hexagonal neighbors
def neighbors [m] [n] [p] (board: [m][n]u8) (player: u8) (r: i64) (c: i64) (visited: [m][p]bool): []pos =
let cols_count = cols m n
let candidates = [(r-1,c), (r-1,c+1), (r,c-1), (r,c+1), (r+1,c-1), (r+1,c)]
in filter (\(i,j) -> i >= 0 && j >= 0 && i < m && j < cols_count && !visited[i][j] && contents board i j == player) candidates
-- BFS path finding
def find_path [m] [n] (board: [m][n]u8) (player: u8) (start: [m][n]u8 -> []pos) (end_check: [m][n]u8 -> i64 -> i64 -> bool): bool =
let cols_count = cols m n
let starts = start board
in if length starts == 0 then false
else
let (found, _, _) = loop (_, q, v) = (false, starts, replicate m (replicate cols_count false))
while length q != 0 do
let (r,c) = head q
let rest = tail q
let v = v with [r,c] = true
in if end_check board r c then (true, [], v)
else (false, rest ++ neighbors board player r c v, v)
in found
-- Connection checks
def connectsO [m] [n] (board: [m][n]u8): bool = find_path board 'O' startO endO
def connectsX [m] [n] (board: [m][n]u8): bool = find_path board 'X' startX endX
-- Winner determination
def winner [m] [n] (board: [m][n]u8): u8 =
if connectsO board then 'O' else if connectsX board then 'X' else '.'
-- Utility functions
def count [m] [n] (board: [m][n]u8) (player: u8): i64 =
let cols_count = cols m n
in length (filter (\(r,c) -> contents board r c == player)
(map (\r -> map (\c -> (r,c)) (iota cols_count)) (iota m) |> flatten))
def valid [m] [n] (board: [m][n]u8): bool = count board 'O' > 0 && count board 'X' > 0
-- Quick checks
def has_edge [m] [n] (board: [m][n]u8) (player: u8) (edge: i32): bool = length (edge_pos board player edge) > 0
def quick_win [m] [n] (board: [m][n]u8): u8 =
if has_edge board 'O' 0 && has_edge board 'O' 2 then 'O'
else if has_edge board 'X' 1 && has_edge board 'X' 3 then 'X'
else '.'
