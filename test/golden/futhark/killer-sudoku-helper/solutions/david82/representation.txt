-- Universal set of digits
def digits: []i32 = [1,2,3,4,5,6,7,8,9]
-- Set membership predicate: x in xs
def contains (xs: []i32) (x: i32): bool =
length (filter (== x) xs) != 0
-- Set difference: xs - ys = {x in xs : x not in ys}
def subtract (xs: []i32) (ys: []i32): []i32 =
filter (\x -> !(contains ys x)) xs
-- Queue data structure for BFS combination generation
type queue [n] = { nextd: i64, remaining: i32, total: i32, parts: [n]i32 }
-- Create initial queue state
def create_initial_queue (size: i64): queue [] =
{ nextd = 0, remaining = i32.i64 size, total = 0, parts = replicate size 0 }
-- Check if queue element represents a valid solution
def is_solution (q: queue []) (target_sum: i32): bool =
q.total == target_sum && q.remaining == 0
-- Check if queue element should be pruned (invalid state)
def should_prune (q: queue []) (target_sum: i32) (max_digits: i64): bool =
q.remaining == 0 || q.total > target_sum || q.nextd == max_digits
-- Generate next queue states from current state
def generate_next_states (q: queue []) (digits: []i32) (size: i64): []queue [] =
let current_digit = digits[q.nextd]
let include_state = { nextd = q.nextd + 1,
remaining = q.remaining - 1,
total = q.total + current_digit,
parts = copy q.parts with [size - i64.i32 q.remaining] = current_digit }
let exclude_state = q with nextd = q.nextd + 1
in [include_state, exclude_state]
-- Generic Queue Processing Engine
-- Processes queue using BFS to find all valid combinations
def process_queue (digits: []i32) (target_sum: i32) (size: i64): [][]i32 =
let max_digits = length digits
in if max_digits < size then []
else
let (combos, _) =
loop (combos, q) = ([], [create_initial_queue size])
while length q != 0 do
let h = head q
let rest = tail q
in
if is_solution h target_sum
then ([h.parts] ++ combos, rest)
else if should_prune h target_sum max_digits
then (combos, rest)
else
let next_states = generate_next_states h digits size
in (combos, rest ++ next_states)
in combos
-- Main combination generator using queue-based approach
def combinations (sum: i32) (size: i32) (exclude: []i32): [][]i32 =
let available_digits = subtract (copy digits) exclude
let size_i64 = i64.i32 size
in process_queue available_digits sum size_i64
