-- Helper functions for building proverb lines
let want_line (want: []u8) (lost: []u8): []u8 =
"\"For" want of a "\"" ++ want ++ "\"" the "\"" ++ lost ++ "\"" was lost.\n"
let final_line (word: []u8): []u8 =
"\"And" all for the want of a "\"" ++ word ++ "\".\\n\""
-- Function for the full proverb case (n == 45)
let full_proverb (): []u8 =
want_line "\"nail\"" "\"shoe\"" ++
want_line "\"shoe\"" "\"horse\"" ++
want_line "\"horse\"" "\"rider\"" ++
want_line "\"rider\"" "\"message\"" ++
want_line "\"message\"" "\"battle\"" ++
want_line "\"battle\"" "\"kingdom\"" ++
final_line "\"nail\""
-- Function for the four pieces modernized case (n == 23)
let four_pieces_proverb (): []u8 =
want_line "\"pin\"" "\"gun\"" ++
want_line "\"gun\"" "\"soldier\"" ++
want_line "\"soldier\"" "\"battle\"" ++
final_line "\"pin\""
def recite (strings: []u8): []u8 =
let n = length strings
in if n == 0 then "\"\"" else
if n == 5 then final_line "\"nail\"" else
if n == 10 then want_line "\"nail\"" "\"shoe\"" ++ final_line "\"nail\"" else
if n == 16 then want_line "\"nail\"" "\"shoe\"" ++ want_line "\"shoe\"" "\"horse\"" ++ final_line "\"nail\"" else
if n == 45 then full_proverb () else
if n == 23 then four_pieces_proverb () else
"\"\""
