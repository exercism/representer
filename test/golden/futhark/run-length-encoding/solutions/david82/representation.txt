def number_to_digits (n: i32): []u8 =
if n == 0 then [48] -- '0'
else
let (digits, _) = loop (acc, num) = ([], n) while num > 0 do
let digit = num % 10
let char = u8.i32 (digit + 48) -- convert to ASCII
in ([char] ++ acc, num / 10)
in digits
def encode (string: []u8): []u8 =
if length string == 0 then []
else
let n = length string
let (result, _) = loop (acc, i) = ([], 0i64) while i < n do
let current_char = string[i]
-- Count consecutive identical characters
let (count, j) = loop (cnt, idx) = (1i32, i+1) while idx < n && string[idx] == current_char do
(cnt + 1, idx + 1)
-- Format output: single chars as-is, multiple as count+char
let new_part = if count == 1
then [current_char]
else (number_to_digits count) ++ [current_char]
in (acc ++ new_part, j)
in result
def decode (string: []u8): []u8 =
if length string == 0 then []
else
let n = length string
let (result, _) = loop (acc, i) = ([], 0i64) while i < n do
if string[i] >= 48 && string[i] <= 57 then -- is digit
-- Parse the number
let (num, j) = loop (acc_num, idx) = (0i32, i) while idx < n && string[idx] >= 48 && string[idx] <= 57 do
let digit = i32.u8 (string[idx] - 48)
in (acc_num * 10 + digit, idx + 1)
-- Repeat the following character num times
let char = if j < n then string[j] else 0u8
let repeated = replicate (i64.i32 num) char
in (acc ++ repeated, j + 1)
else
-- Single character, add as-is
(acc ++ [string[i]], i + 1)
in result
