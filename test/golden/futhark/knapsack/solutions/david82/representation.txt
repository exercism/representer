def maximum_value [n] (weights: [n]i32) (values: [n]i32) (maximum_weight: i32): i32 =
let init_row = replicate (i64.i32 (maximum_weight + 1)) 0
let final_row = loop row = init_row for i < n do
let w = weights[i]
let v = values[i]
in tabulate (i64.i32 (maximum_weight + 1)) (\j ->
if w > i32.i64 j then row[j]
else i32.max row[j] (row[j - i64.i32 w] + v)
)
in final_row[i64.i32 maximum_weight]
