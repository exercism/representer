# $a0 - input
strlen:
li $t0,0 # iterador / tamaño
strlen_bucle:
# condición
add $t1,$a0,$t0
lbu $t1,0($t1)
beq $t1,$zero,strlen_bucle_fin
# repetir
addi $t0,$t0,1
b strlen_bucle
strlen_bucle_fin:
move $v0,$t0
jr $ra
# $a0 - letra
toupper:
blt $a0,'a',toupper_mayus
addi $a0,$a0,-32

move $v0,$a0
jr $ra
# $a0 - str
# $a1 - char
find:
move $fp,$sp
#volcar
addi $sp,$sp,-16

sw $s0,4($sp)
sw $s1,8($sp)
sw $s2,12($sp)
#cuerpo
li $s2,0 # resultado
jal strlen # $v0 - size
li $s0,0 #iterador
find_for:
#condición
beq $s0,$v0,find_for_
#cuerpo
add $s1,$a0,$s0
lbu $s1,0($s1)
bne $s1,$a1,find_false
li $s2,1 # true
b find_for_
find_false:
#repetrir
addi $s0,$s0,1
b find_for
find_for_:
move $v0,$s2
#desvolcar
lw $ra,0($sp)
lw $s0,4($sp)

lw $s2,12($sp)

addi $sp,$sp,-16

sw $s0,4($sp)

#cuerpo

jal strlen
move $s1,$v0

bucle:
#condición
beq $s0,$s1,bucle_
#cuerpo
#load char

#encontrar el char en las listas
