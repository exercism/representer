(file "impl.mips")
.globl hex_convert
hex_convert:
move $t7, $a0 #$a0 has string
lbu $t3, 0($t7) #ascii string, 32-bit word stored in $t1
ori $t9, $0, 48 #ascii '0'
ori $t5, $0, 58 #ascii ':', immediately after ascii '9'
ori $t4, $0, 97 #ascii 'a'
#ascii 'A' is 65; 7 = 65 - 58
#ascii 'a' is 97; 39 = 97 - 58
ori $v0, $0, 0 #outreg = 0
while: beq $t3, $0, endwhile #while char not null
sll $v0, $v0, 4 #four bits for new hex value
sub $t6, $t3, $t9 #sub 48 from ascii char for numeric value
blt $t3, $t5, endif #if ascii char <= '9', no extra parsing
blt $t3, $t4, uppercase
addi $t6, $t6, -39 #if ascii lowercase a-f
beq $0, $0, endif
uppercase: #if ascii uppercase A-F
addi $t6, $t6, -7 #sub 7 to get value starting from 'A'
endif:
or $v0, $v0, $t6 #hex digit added to output register
### printing for debugging purposes ###
#move $a1, $v0
#move $a0, $t6
#ori $v0, $0, 1
#syscall
#move $v0, $a0
#######
add $t7, $t7, 1 #char pointer incremented to next ascii char
lbu $t3, 0($t7) #load next ascii char
j while #continue while loop
endwhile:
jr $ra #return to caller
