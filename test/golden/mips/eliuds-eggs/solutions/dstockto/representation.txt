(file "impl.mips")
.globl egg_count
.text
egg_count:
move $v0, $zero # v0 = 0
beq $a0, $zero, report # if a0 == 0 gosub report
move $t0, $a0 # t0 = a0
advance:
addi $v0, $v0, 1 # v0++
sub $t1, $zero, $t0 # t1 = -$t0
and $t1, $t0, $t1 # t1 = t0 & t1
sub $t0, $t0, $t1 # clear least significant 1 bit
bne $t0, $zero, advance # if t0 != 0 goto advance
report:
jr $ra
