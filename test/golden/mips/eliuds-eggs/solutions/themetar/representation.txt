(file "impl.mips")
.globl egg_count
egg_count:
li $t0, 1 # moveable bit
li $v0, 0 # counter; return result
loop:
and $t1, $a0, $t0 # bitwise-and input (a0) and bit (t0)
beq $t1, 0, cont # if t1 is 0, skip increment
add $v0, $v0, 1 # if it wasn't 0, increment counter
cont: sll $t0, $t0, 1 # shift bit to the left
bne $t0, 0, loop # goto loop, until t0 is 0 (the bit has left the register)
jr $ra
