# | Register | Usage | Type | Description |
# | -------- | ------------ | ------- | ----------------------------- |
# | `$a0` | input | address | null-terminated input string |
# | `$a1` | input/output | address | null-terminated output string |
# | `$t0-9` | temporary | any | used for temporary storage |
# $t0 current char
# $t1 is valid char
# $t2 is space char
# $t3 is acronym
#.include "\"impl.mips\""
.globl abbreviate
abbreviate:
move $t9, $ra
beq $a0, $zero, invalid_input
li $t3, 1
j check_char
check_char_return:
sb $zero 0($a1)
move $ra, $t9
jr $ra
check_char:
li $t1, 0
li $t2, 0
lbu $t0, 0($a0)
beq $t0, $zero, check_char_return
li $t8, 64
sgtu $t4, $t0, $t8
slti $t5, $t0, 91
and $t4, $t4, $t5
li $t8, 96
sgtu $t6, $t0, $t8
slti $t7, $t0, 123
and $t6, $t6, $t7
or $t1, $t4, $t6 #in valid char range
and $t1, $t1, $t3 #valid char and is acronym
beq $t1, $zero, check_char_non_valid_char
jal write_char
addi $a0, $a0, 1
j check_char
check_char_non_valid_char:
li $t8, 32
seq $t4, $t0, $t8
li $t8, 45
seq $t5, $t0, $t8
or $t4, $t4, $t5
beq $t4, $zero, check_char_ignore
li $t3, 1
addi $a0, $a0, 1
j check_char
check_char_ignore:
addi $a0, $a0, 1
j check_char
write_char:
slti $t4, $t0, 91
bne $t4, $zero, uppered
addi $t0, $t0, -32 #toUpper()
uppered:
sb $t0 0($a1)
addi $a1, $a1, 1
li $t3, 0
jr $ra
invalid_input:
jr $ra
