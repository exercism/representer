.globl raindrops
.data
tmp: .space 10 # This space will be used to convert a number
# to a string of decimal digits. The biggest
# 32-bit number is 2,147,483,647 so 10 digits
# is enough.
choices: .byte 3
.asciiz "\"Pling\""
.byte 5
.asciiz "\"Plang\""
.byte 7
.asciiz "\"Plong\""
.byte 0
.text
raindrops:
move $t1, $a1 # the place to write an answer to
la $t0, choices
d: lb $t2, 0($t0) # read divisor
beq $t2, $zero, nd # if it is zero, we are done
addi $t0, $t0, 1 # now t0 points to the corresponding string
div $a0, $t2
mfhi $t3 # if t3 is zero, the number was divisible
l1: lb $t2, 0($t0) # read next character
addi $t0, $t0, 1 # move t0 to the next character
beq $t2, $zero, d # if we read ending zero, we are done with
# this divisor
bne $t3, $zero, l1 # if the number was not divisible, do
# nothing
sb $t2, 0($t1) # otherwise store in an output string
addi $t1, $t1, 1
j l1
nd: bne $t1, $a1, out # if we wrote anything, we are done
# We did not write anything. The master plan: store in tmp the digits
# of our number (in reverse order) and copy them to output.
move $t4, $a0
li $t5, 10
la $t6, tmp
move $t7, $t6 # now both t6 (future output) and t7
# point to tmp
l2: beq $t4, $zero, l3 # no more digits
div $t4, $t5
mflo $t4
mfhi $t8
addi $t8, $t8, '0' # digit is in t8
sb $t8, 0($t6)
addi $t6, $t6, 1
j l2 # not 'bne' here because the input
# number can be zero, so we need to
# bail out before writing any digits
l3: beq $t6, $t7, out # all digits are written
addi $t6, $t6, -1
lb $t4, 0($t6)
sb $t4, 0($t1)
addi $t1, $t1, 1
j l3 # see comment about 'bne' above
out: sb $zero, 0($t1) # store terminating zero byte
jr $ra
