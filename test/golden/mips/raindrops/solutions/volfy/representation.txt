(file "impl.mips")
.globl raindrops
.data
pling: .asciiz "\"Pling\""
plang: .asciiz "\"Plang\""
plong: .asciiz "\"Plong\""
.text
raindrops:
sub $sp, $sp, 16
sw $a0, 12($sp)
sw $a1, 8($sp)
sw $a2, 4($sp)
sw $ra, 0($sp)
# a0 = inp, a1 = output
move $t1, $a1
li $t3, 3
li $t4, 5
li $t5, 7
div $a0, $t3
mfhi $t0
bnez $t0, endpling
la $a2, pling
jal writestr
endpling:
div $a0, $t4
mfhi $t0
bnez $t0, endplang
la $a2, plang
jal writestr
endplang:
div $a0, $t5
mfhi $t0
bnez $t0, endplong
la $a2, plong
jal writestr
endplong:
bne $a1, $t1, endops # if output addr has moved, something was written
li $t3, 10
jal writedigits
endops:
lw $ra, 0($sp)
lw $a2, 4($sp)
lw $a1, 8($sp)
lw $a0, 12($sp)
add $sp, $sp, 16
jr $ra
writestr:
# a2 = inp addr
# a1 = output addr
loop:
lb $t0, ($a2)
beqz $t0, endloop
sb $t0, ($a1)
add $a2, $a2, 1
add $a1, $a1, 1
b loop
endloop:
jr $ra
writedigits:
div $a0, $t3
mflo $t1 # quotient
mfhi $t2 # remainder
bge $t1, 10, rec
beqz $t1, writeremainder # don't write quotient if it is zero
add $t1, $t1, '0'
sb $t1, ($a1) # write quotient (52 => 5)
addi $a1, $a1, 1
b writeremainder
rec:
sub $sp, $sp, 8
sw $t2, 4($sp) # store remainder
sw $ra, 0($sp) # store ra
move $a0, $t1 # call writedigits with quotient as arg
jal writedigits
lw $t2, 4($sp)
lw $ra, 0($sp)
add $sp, $sp, 8
writeremainder:
add $t2, $t2, '0'
sb $t2, ($a1) # write remainder (52 => 2)
add $a1, $a1, 1
jr $ra
