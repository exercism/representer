binary_convert:
# la $a0, test_str # 本地调试用
add $s5, $a0, $zero
addi $t0, $zero, 0 # 偏移量计数器
addi $t6, $zero, 0 # answer
addi $t5, $zero, 2 # constant 2
process_char:
# 计算字符的位置偏移量
add $t2, $s5, $t0
lbu $t1, 0($t2)
# 字符串是否结束
beq $t1, $zero, process_end
# 字符串未结束，*2
mul $t6, $t6, $t5
# 比较字符是不是1
# 1的ascii码是49
addi $t3, $zero, 49
beq $t1, $t3, is_one
j is_zero
is_one:
addi $t6, $t6, 1
j comp_end
is_zero:
j comp_end
comp_end:
# add $a0, $zero, $t6
# li $v0, 1
# syscall

# li $v0, 4
# syscall
addi $t0, $t0, 1 # 计数器偏移至下一个地址
j process_char
process_end:
add $v0, $zero, $t6 # 存储答案
