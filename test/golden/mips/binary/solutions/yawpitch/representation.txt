# Return the decimal equivalent of a given binary string (ie "\"11010\"" -> 26).
.globl binary_convert
# $a0 - argument for input string, as starting address
# $v0 - result, as integer value of the string
# $t0 - holds the ascii / integer value of the current char
binary_convert:
li $v0, 0 # initialize the accumulator
loop:
lb $t0, 0($a0) # load the next char
beq $t0, $zero, end # if it's NULL, go to end
sll $v0, $v0, 1 # shift the accumulator left one place
subi $t0, $t0, '0' # calculate the int value of the char (ie "\"1\"" (49) - "\"0\"" (48) = 1)
or $v0, $v0, $t0 # set the 0 position bit of the accumulator
addi $a0, $a0, 1 # move the address forward
j loop # continue the loop
end:
jr $ra
