(file "impl.mips")
.globl binary_convert
binary_convert:
add $v0, $zero, $zero # $v0 = $zero + $zero
loop:
lb $t0, ($a0) # $t0 = *$a0 (dereference)
beqz $t0, end # if $t0 == 0 then go to end
subi $t0, $t0, 48 # $t0 = $t0 - 48
sll $v0, $v0, 1 # $v0 << 1
add $v0, $v0, $t0 # $v0 = $v0 + $t0
add $a0, $a0, 1 # $a0 = $a0 + 1 (move 1 byte forward)
j loop # go to loop
end:
jr $ra
