.globl binary_convert
.text
binary_convert:
addi $t0, $zero, 0x30 # $t0 = 30 ASCII code for '0'
addi $t1, $zero, 0x31 # $t0 = 31 ASCII code for '1'
add $v0, $zero, $zero # $v0 = 0 initialise result
add $t2, $a0, $zero # $t2 = 0 initialise char address
PARSE_CHAR:
lb $t3, 0($t2) # $t3 = current char value
beq $t3, $zero, EXIT # if $t3 == NULL then EXIT
addi $t2, $t2, 1 # $t2++ increment char address
sll $v0, $v0, 1 # $v0 = $v0 << 1 double result
beq $t3, $t1, ADD_ONE # if $t3 == '1' then ADD_ONE
bne $t3, $t0, ERROR # if $t3 != '0' then ERROR
j PARSE_CHAR
ERROR:
j EXIT
ADD_ONE:
addi $v0, $v0, 1 # $v0++
j PARSE_CHAR
EXIT:
jr $ra # jump to $ra
