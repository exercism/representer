.globl is_leap_year
.macro CanBeDivided(%return, %a, %b)
div %a, %b
mfhi %return # if return != 0: a % b != 0
# else: a % b = 0
.end_macro
.text
is_leap_year:
if_4_mult:
addi $t1, $0, 4
CanBeDivided($t0, $a0, $t1)
bnez $t0, not_leap
if_100_mult:
addi $t1, $0, 100
CanBeDivided($t0, $a0, $t1)
bnez $t0, is_leap
if_400_mult:
addi $t1, $0, 400
CanBeDivided($t0, $a0, $t1)
bnez $t0, not_leap
is_leap:
li $v0, 1
jr $ra
not_leap:
li $v0, 0
jr $ra
