(file "impl.mips")
.globl is_leap_year
.text
is_leap_year:
move $s4 $ra #Guardando o reponse para o main
li $t0, 4 #Divisor
div $a0, $t0 #Divisao por 4
mfhi $t1 #Pegando o resto
bne $t1, $zero,isElifBissexto
li $t0, 100 #Divisor
div $a0, $t0 #Divisao por 100
mfhi $t1 #Pegando o resto
beq $t1, $zero,isElifBissexto
j responseBissexto
isElifBissexto:
li $t0, 400 #Divisor
div $a0, $t0 #Divisao por 400
mfhi $t1 #Pegando o resto
bne $t1, $zero,responseNBissexto
responseBissexto:
li $v0, 1
jr $s4
responseNBissexto:
li $v0, 0
jr $s4
