# Return if a year is a leap year
.globl is_leap_year
# $a0 - argument for year, as int
# $v0 - result, as bool
is_leap_year:
rem $t0, $a0, 4 # a year must divide by 4
bne $t0, $zero, is_not_leap # or it is not a leap year
rem $t0, $a0, 100 # AND it must NOT divide by 100
bne $t0, $zero, is_leap # to be a leap year
rem $t0, $a0, 400 # OR, if it divides by 400
beq $t0, $zero, is_leap # then it IS a leap year
is_not_leap:
move $v0, $zero # year is not not a leap
jr $ra
is_leap:
li $v0, 1 # year is a leap
jr $ra
