.globl is_leap_year
is_leap_year:
li $t0, 0
li $t1, 4
li $t2, 100
li $t3, 400
rem $t4, $a0, $t1 # year % 4
bne $t4, $zero, not_divisible_by_4
rem $t5, $a0, $t2 # year % 100
beqz $t5, check_divisible_by_400
rem $t6, $a0, $t3 # year % 400
addi $t0, $zero, 1
j exit
check_divisible_by_400:
rem $t6, $a0, $t3 # year % 400
bnez $t6, set_false
addi $t0, $zero, 1
j exit
set_false:
addi $t0, $zero, 0
jr $ra
not_divisible_by_4:
add $v0, $zero, $t0 # Set $v0 to 0 (false) to indicate it's not a leap year
j exit
exit:
add $v0,$zero, $t0
jr $ra
