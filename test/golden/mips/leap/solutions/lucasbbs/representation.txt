.globl is_leap_year
.data
fourYears: .word 4
hundredYears: .word 100
fourHundredYears: .word 400
.text
is_leap_year:
li $v0, 0
lw $t0, fourYears
divu $a0,$t0
mfhi $t1
bne $t1, $zero, exit
lw $t0, hundredYears
divu $a0,$t0
mfhi $t1
bne $t1, $zero, leap_year
lw $t0, fourHundredYears
divu $a0,$t0
mfhi $t1
bne $t1, $zero, exit
leap_year:
li $v0,1
exit:
jr $ra
