.globl difference_of_squares
difference_of_squares:
li $v0, 0
add $t0, $a0, $zero # copy of $a0
li $t1, 0 # square of the sum : (1 + 2 + ... + 10)²
li $t2, 0 # sum of the squares : 1² + 2² + ... + 10²
loop_1:
beq $a0, 0, loop_1_exit
add $t1, $t1, $a0
addi $a0, $a0, -1
j loop_1
loop_1_exit:
mult $t1, $t1
mflo $t1
add $a0, $t0, $zero
loop_2:
beq $a0, 0, loop_2_exit
mult $a0, $a0
mflo $t0
add $t2, $t2, $t0
addi $a0, $a0, -1
j loop_2
loop_2_exit:
