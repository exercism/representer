# | Register | Usage | Type | Purpose |
# | -------- | --------- | ------- | -------------------------- |
# | `$a0` | input | integer | given value |
# | `$v0` | output | integer | difference of squares |
# | `$t0-9` | temporary | any | used for temporary storage |
# $t0 - counter
# $t1 - square of sum
# $t2 - sum of squares
# $t3 - temporary for calculations
.globl difference_of_squares
difference_of_squares:
# to comment
# li $a0, 3
move $t0, $a0
li $t1, 0
li $t2, 0
j main
main:
bgt $t0, 0, loop
mult $t1, $t1
mflo $t3
sub $v0, $t3, $t2
jr $ra
loop:
add $t1, $t1, $t0
mult $t0, $t0
mflo $t3
add $t2, $t2, $t3
subi $t0, $t0, 1
j main
