# | Register | Usage | Type | Description |
# | -------- | --------- | ------- | -------------------------- |
# | `$a0` | input | integer | x |
# | `$a1` | input | integer | y |
# | `$v0` | output | integer | points earned |
# | `$t0-9` | temporary | any | used for temporary storage |
.globl score
score:
mulo $t0, $a0, $a0
mulo $t1, $a1, $a1
add $t2, $t0, $t1
ble $t2, 1, score_ten
ble $t2, 25, score_five
ble $t2, 100, score_one
li $v0, 0
jr $ra
score_ten:
li $v0, 10
jr $ra
score_five:
li $v0, 5
jr $ra
score_one:
li $v0, 1
jr $ra
