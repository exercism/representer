(file "impl.mips")
.globl rotate
.text
rotate:
# a0 - input string, for callee
# a1 - shift key, for callee
# a2 - pointer to output string, for callee
loop: # loop over all characters of input string
lbu $t0, ($a0)
beq $t0, $zero, return
# skip non-alphabetic characters
blt $t0, 65, continue # ascii(A) == 65
ble $t0, 90, uppercase # ascii(Z) == 90
blt $t0, 97, continue # ascii(a) == 97
bgt $t0, 122, continue # ascii(z) == 122
# lowercase
add $t0, $t0, $a1
ble $t0, 122, continue # ascii(z) == 122
j subtract_26
uppercase:
add $t0, $t0, $a1
ble $t0, 90, continue # ascii(Z) == 90
subtract_26:
sub $t0, $t0, 26
continue:
sb $t0, ($a2)
add $a0, $a0, 1 # i++
add $a2, $a2, 1 # j++
j loop
return:
sb $zero, ($a2) # zero-terminate output string
jr $ra
