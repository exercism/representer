# is_isogram must:
# - be named is_isogram and declared as global
# - read input address of string from a0
# - follow the convention of using the t0-9 registers for temporary storage
# - (if it uses s0-7 then it is responsible for pushing existing values to the stack then popping them back off before returning)
# - write boolean result to v0
# haven't programmed in assembly since ~1999, it was fun to do it again!
.globl is_isogram
.text
is_isogram:
# $t1 - binary map of chars
li $t1, 0
loop:
# $t2 - current char
lb $t2, 0($a0)
# end of string
beq $t2, $zero, isogram
# we have no interest in non-alphabetical chars
# 0x41 - ascii value of 'A'
li $t3, 0x41
blt $t2, $t3, next_char
# lowercase already, jump to normalization
# 0x61, ascii value of 'a'
li $t3, 0x61
bge $t2, $t3, normalize
# assume uppercase, lowercase it
addi $t2, $t2, 0x20
normalize:
sub $t2, $t2, $t3
# $t0 - position of char in map
li $t0, 1
sllv $t0, $t0, $t2
# have we seen this char?
and $t4, $t1, $t0
bne $t4, $zero, not_isogram
# no, lets mark it in map
or $t1, $t1, $t0
next_char:
addi $a0, $a0, 1
j loop
isogram:
li $v0, 1
jr $ra
not_isogram:
li $v0, 0
jr $ra
