# Return the number of grains on the given square of a 64 square chessboard.
.globl square
# $a0 - input square number valid in the range 1..64, as integer
# $v0 - lo 32 bits of the 64 bit output, as integer
# $v1 - hi 32 bits of the 64 bit output, as integer
square:
li $v0, 0 # lo 32 bits of output to 0
li $v1, 0 # hi 32 bits of output to 0
li $t0, 1 # set t0 to constant 1
li $t1, 32 # set t1 to constant 32
blt $a0, $t0, err # error if input < 1
sub $a0, $a0, $t0 # subtract 1 from input
blt $a0, $t1, low # if input < 32, set output low bits
sub $a0, $a0, $t1 # otherwise subtract 32 from input
blt $a0, $t1, high # if input < 64, set output high bits
err:
jr $ra # error if input is not in 1..64
low:
sllv $v0, $t0, $a0 # set output low bits to 2^(input-1)
jr $ra
high:
sllv $v1, $t0, $a0 # set output high bits to 2^(input-1)
jr $ra
