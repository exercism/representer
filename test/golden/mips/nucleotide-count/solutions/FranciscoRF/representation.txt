.globl nucleotide_counts
.text
nucleotide_counts:
#set up
move $t0, $a0 # load string address
li $t2, 0 # reset A count
li $t3, 0 # reset C count
li $t4, 0 # reset G count
li $t5, 0 # reset T count
li $t6, 'A' # load 'A' byte
li $t7, 'C' # load 'C' byte
li $t8, 'G' # load 'G' byte
li $t9, 'T' # load 'T' byte
loop:
lb $t1, 0($t0) # load char from current address
addi $t0, $t0, 1 # prepare next char's address
beq $t1, $zero, exit_ok # end program if null (end of string)
beq $t1, $t6, case_a # branch if char is 'A'
beq $t1, $t7, case_c # branch if char is 'C'
beq $t1, $t8, case_g # branch if char is 'G'
beq $t1, $t9, case_t # branch if char is 'T'
j exit_fail # end program if char is unexpected value
exit_ok:
sw $t2, 0($a1) # move 'A' count to array in memory
sw $t3, 4($a1) # move 'C' count to array in memory
sw $t4, 8($a1) # move 'G' count to array in memory
sw $t5, 12($a1) # move 'T' count to array in memory
jr $ra # return
exit_fail:
li $t1, -1 # load error return value
sw $t1, 0($a1) # set error value in array
sw $t1, 4($a1) # "\"" "\""
sw $t1, 8($a1) # "\"" "\""
sw $t1, 12($a1) # "\"" "\""
jr $ra # return
case_a:
addi $t2, $t2, 1 # increase 'A' count
j loop # restart loop
case_c:
addi $t3, $t3, 1 # increase 'C' count
j loop # restart loop
case_g:
addi $t4, $t4, 1 # increase 'G' count
j loop # restart loop
case_t:
addi $t5, $t5, 1 # increase 'T' count
j loop # restart loop
