(file "impl.mips")
.globl trinary_convert
.text
trinary_convert:
move $t0, $a0 # move input string address
move $v0, $zero # reset return register
li $t3, 3 # set base
__loop:
# read next char
lb $t1, ($t0) # load current char
beqz $t1, __return # return if string is terminated
addi $t0, $t0, 1 # prepare next char address
# accumulate result
mul $v0, $v0, $t3 # multiply current result by base
sub $t1, $t1, '0' # convert ascii digit to integer
add $v0, $v0, $t1 # add integer to return value
j __loop
__return:
jr $ra
