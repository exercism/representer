(file "impl.mips")
# $t0 is the address of the test string
# $v0 is an accumulator for the result
# $t1 contains the current character
.text
.globl trinary_convert
trinary_convert:
move $t0, $a0
move $v0, $0
loop:
lb $t1, 0($t0) # load next char from string
beq $t1, $0, end # if it's a NULL byte, exit
mul $v0, $v0, 3 # multiply current result by 3
addiu $t1, $t1, -48 # convert char from ASCII to int
add $v0, $v0, $t1 # add new value to result
addiu $t0, $t0, 1 # increment iterator
j loop # do it all again!!!!
end:
jr $ra
