package Deque ;
use strict ;
use warnings ;
sub new {
my ( $class ) = shift ;
bless { first => undef ,
last => undef }, $class
}
sub push {
my ( $self , $val ) = @_ ;
my $node = Node -> new (
prev => $self -> { last },
val => $val ,
next => undef
);
$self -> { last }{ next } = $node if ( $self -> { last });
$self -> { last } = $node ;
$self -> { first } = $node unless ( $self -> { first });
}
sub pop {
my ( $self ) = @_ ;
my $node = $self -> { last };
$self -> { last } = $node -> { prev };
$self -> { last }{ next } = undef ;
$node -> { val };
}
sub shift {
my ( $self ) = @_ ;
my $node = $self -> { first };
$self -> { first } = $node -> { next };
$self -> { first }{ prev } = undef ;
$node -> { val };
}
sub unshift {
my ( $self , $val ) = @_ ;
my $node = Node -> new (
prev => undef ,
val => $val ,
next => $self -> { first }
);
$self -> { first }{ prev } = $node if ( $self -> { first });
$self -> { first } = $node ;
$self -> { last } = $node unless ( $self -> { last });
}
package Node ;
use strict ;
use warnings ;
sub new {
my ( $class , %attr ) = @_ ;
bless \ %attr , $class ;
}
1 ;
