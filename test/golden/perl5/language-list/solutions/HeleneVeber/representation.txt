(file "lib/LanguageList.pm")
package LanguageList ;
use v5 . 38 ;
our @Languages ;
sub add_language ($language) {
push ( @Languages , $language );
return @Languages ;
}
sub remove_language () {
pop ( @Languages );
return @Languages ;
}
sub first_language () {
return $Languages [ 0 ];
}
sub last_language () {
return @Languages [ - 1 ];
}
sub get_languages (@elements) {
my @elem = map { $_ - 1 } @elements ;
my @result = @Languages [ @elem ];
return @result ;
}
sub has_language ($language) {
for my $lang ( @Languages ) {
if ( $lang eq $language ) {
return 1 ;
}
}
}
