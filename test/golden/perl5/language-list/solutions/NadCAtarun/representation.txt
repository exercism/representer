package LanguageList ;
use v5 . 38 ;
our @Languages ;
sub add_language ($l){ push ( @Languages , $l )}
sub remove_language (){ pop ( @Languages )}
sub first_language (){ @Languages [ 0 ]}
sub last_language (){ @Languages [ - 1 ]}
sub get_languages (@e){ @Languages [ map { $_ - 1 } @e ]}
sub has_language ($l){ grep ( /^$l$/ , @Languages )}
