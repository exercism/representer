(file "lib/LanguageList.pm")
package LanguageList ;
use v5 . 38 ;
our @Languages ;
sub add_language ($language) {
push ( @Languages , $language );
}
sub remove_language () {
pop ( @Languages );
}
sub first_language () {
return $Languages [ 0 ];
}
sub last_language () {
return $Languages [ - 1 ];
}
sub get_languages (@elements) {
my @slice ;
foreach my $n ( @elements ) {
push ( @slice , $Languages [ $n - 1 ]);
}
return @slice ;
}
sub has_language ($language) {
return grep { $_ eq $language } @Languages ;
}
