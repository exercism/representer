package Matrix ;
use v5 . 42 ;
use Exporter qw<import> ;
our @EXPORT_OK = "qw<extract_row extract_column>" ;
sub _parse_string ($str) {
return [ map { [ split ' " " ', $_ ] } split "\"" \n "\"," $str ];
}
sub extract_row ( $str, $index ) {
return _parse_string ( $str ) -> [ $index - 1 ];
}
sub extract_column ( $str, $index ) {
return [ map { $_ -> [ $index - 1 ] } _parse_string ( $str ) -> @ * ];
}
