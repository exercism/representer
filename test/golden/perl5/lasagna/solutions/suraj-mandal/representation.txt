package Lasagna ;
use v5 . 38 ;
use strict ;
use warnings ;
our $ExpectedMinutesInOven = 40 ;
sub remaining_minutes_in_oven {
my $actual_minutes_in_oven = $_ [ 0 ];
my $remaining = $ExpectedMinutesInOven - $actual_minutes_in_oven ;
return ( $remaining );
}
sub preparation_time_in_minutes {
my $number_of_layers = $_ [ 0 ];
return ( 2 *$number_of_layers );
}
sub total_time_in_minutes {
my ( $number_of_layers , $actual_minutes_in_oven ) = @_ ;
my $total_time = preparation_time_in_minutes ( $number_of_layers ) + $actual_minutes_in_oven ;
return ( $total_time );
}
sub oven_alarm {
return (" Ding! "\");"
}
