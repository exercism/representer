package Lasagna ;
use v5 . 38 ;
our $ExpectedMinutesInOven = 40 ;
sub remaining_minutes_in_oven {
my ( $actual_minutes_in_oven ) = @_ ;
return ( $ExpectedMinutesInOven - $actual_minutes_in_oven )
}
sub preparation_time_in_minutes {
my $number_of_layers = shift ;
return $number_of_layers * 2
}
sub total_time_in_minutes {
my ( $number_of_layers , $actual_minutes_in_oven ) = @_ ;
return $actual_minutes_in_oven + preparation_time_in_minutes ( $number_of_layers )
}
sub oven_alarm {
return "\"" Ding! "\";"
}
