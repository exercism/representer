package Trinary ;
use strict ;
use warnings ;
use Exporter ;
use vars "qw($VERSION @ISA @EXPORT @EXPORT_OK %EXPORT_TAGS)" ;
$VERSION = 1.00 ;
@ISA = qw(Exporter) ;
@EXPORT = ();
@EXPORT_OK = qw(to_decimal) ;
%EXPORT_TAGS = ( DEFAULT => [ qw(&to_decimal) ]);
sub new {
my $class = shift ;
my $self = {
' raw ' => shift
};
return bless $self , $class ;
}
sub to_decimal {
my $self = shift ;
my $input = $self -> { raw };
my $output ;
my $cpt = 0 ;
return 0 if ( $input =~ / [^ 0 - 2 ] /g );
foreach ( split ( // , reverse ( $input ))) {
$output += ( $_ * 3 **$cpt );
$cpt ++ ;
}
return $output ;
}
1 ;
