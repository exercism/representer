package Gigasecond ;
use v5 . 38 ;
use Time:: Piece ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<add_gigasecond> ;
my $datetime_fmt = ' %Y-%m-%dT%H:%M:%S ';
sub add_gigasecond ($time) {
my $timep = Time:: Piece -> strptime ( $time , $datetime_fmt );
return ( $timep + 1000000000 ) -> strftime ( $datetime_fmt );
}
