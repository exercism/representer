package Triangle ;
use v5 . 40 ;
use Exporter qw<import> ;
our @EXPORT_OK = "qw<is_equilateral is_isosceles is_scalene>" ;
sub is_triangle ($triangle) {
my @leg = sort { $a <=> $b } @$triangle ;
return 0 if ( $leg [ 0 ] + $leg [ 1 ] <= $leg [ 2 ]);
return 1 ;
}
sub is_equilateral ($triangle) {
return 0 unless ( is_triangle ( $triangle ));
my ( $a , $b , $c ) = @$triangle ;
return ( $a == $b && $b == $c );
}
sub is_isosceles ($triangle) {
return 0 unless ( is_triangle ( $triangle ));
my ( $a , $b , $c ) = @$triangle ;
if ( $a == $b && $b != $c ) { return 1 ;}
elsif ( $a == $c && $b != $a ) { return 1 ;}
elsif ( $b == $c && $a != $c ) { return 1 ;}
elsif ( is_equilateral ( $triangle )) { return 1 ;}
else { return 0 ;}
}
sub is_scalene ($triangle) {
return 0 unless ( is_triangle ( $triangle ));
return ( ! is_isosceles ( $triangle ) && ! is_equilateral ( $triangle ));
}
1 ;
