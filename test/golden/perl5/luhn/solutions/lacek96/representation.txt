package Luhn ;
use v5 . 38 ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<is_luhn_valid> ;
sub is_luhn_valid ($ number )
{
$number =~ s"\s""g ;
my @chars = reverse split ("", $number );
my $doubling ;
my $sum = 0 ;
my $num_of_digits = 0 ;
for my $char ( @chars )
{
if ( $char =~ /\d/ )
{
$num_of_digits ++ ;
if ( $doubling )
{
$char = $char * 2 ;
if ( $char > 9 )
{
$char -= 9 ;
}
}
$sum += $char ;
$doubling = not $doubling ;
}
else
{
$sum = 1 ;
last ;
}
}
return ( $sum % 10 == 0 ) && ( $num_of_digits >= 2 );
}
