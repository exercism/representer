package Luhn ;
use v5 . 38 ;
use warnings ;
use strict ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<is_luhn_valid> ;
sub is_luhn_valid ($number) {
my $d ;
my $s = 0 ;
my $even = 0 ;
$number =~ "s/ //g" ;
if ( length ( $number ) < 2 ) {
return 0 ;
}
if ( $number =~ / [^ \d ] / ) {
return 0 ;
}
foreach $d ( reverse split / "/, $number" ) " " {
"        $even = 1 - $even;"
"        if " ( $even ) " " {
"            $s " + "= $d;"
"            next;"
"        " }
"        if " ( "$d > 4" ) " " {
"            $s " + "= 2 " * " $d - 9"
"        " } " else " {
"            $s " + "= 2 " * " $d;"
"        " }
"    " }
"    return $s % 10 == 0;"
}
