(file "lib/Luhn.pm")
package Luhn ;
use v5 . 38 ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<is_luhn_valid> ;
sub has_letter ($number) {
return $number =~ m/ [ a - zA - Z ] / ;
}
sub valid_length ($number) {
return length ( $number ) > 1 ;
}
sub is_luhn_valid ($number) {
$number =~ s/\s//g ;
my $len = length ( $number );
return 0 if ! valid_length ( $number );
return 0 if has_letter ( $number );
my $even_number_list = $len % 2 == 0 ;
my @number = split ( // , $number );
my $sum = 0 ;
for ( my $i = 0 ; $i < $len ; $i ++ ) {
my $n = substr ( $number , $i , 1 );
if (
$even_number_list && $i % 2 == 0
|| ! $even_number_list && $i % 2 != 0
) {
$n *= 2 ;
}
$n -= 9 if $n > 9 ;
$sum += $n ;
}
return $sum % 10 == 0 ;
}
