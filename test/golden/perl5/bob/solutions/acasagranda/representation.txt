package Bob ;
use v5 . 38 ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<hey> ;
sub hey ($msg) {
return "\"" "Fine. Be that way!" "\"" if $msg =~ /^\s * $/a ;
if ( $msg eq uc $msg and $msg =~ / [ A - Z ] / ){
return ( $msg =~ /\?\s * $/ ) ? "\"" "Calm down, I know what I'm doing!" "\":\"" "Whoa, chill out!" "\";"
}
return ( $msg =~ /\?\s * $/ ) ? "\"" Sure. "\":\"" Whatever. "\"" ;
}
