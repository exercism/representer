use v5 . 38 ;
use List:: Util "qw < " pairmap > ;
sub hamming_distance ( $strand1, $strand2 ) {
die ' "strands must be of equal length" '
unless length $strand1 == length $strand2 ;
length ( $strand1 )
-
scalar
grep { / 0 / }
pairmap { $a cmp $b }
map { substr ( $strand1 , $_ , 1 ), substr ( $strand2 , $_ , 1 ) }
0 .. length ( $strand1 ) - 1
}
