(file "BeerSong.pm")
package BeerSong ;
use strict ;
use warnings ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<sing> ;
sub poem {
my $this = shift ;
my $next = $this - 1 ;
my $result = "\"\";"
if ( $this > 2 ) {
$result = "\"" "$this bottles of beer on the wall, $this bottles of beer.\\nTake one down and pass it around, $next bottles of beer on the wall." "\";"
}
elsif ( $this == 2 ) {
$result = "\"" "$this bottles of beer on the wall, $this bottles of beer.\\nTake one down and pass it around, 1 bottle of beer on the wall." "\";"
}
elsif ( $this == 1 ) {
$result = "\"" "$this bottle of beer on the wall, $this bottle of beer.\\nTake it down and pass it around, no more bottles of beer on the wall." "\";"
}
elsif ( $this == 0 ) {
$result = "\"" "No more bottles of beer on the wall, no more bottles of beer.\\nGo to the store and buy some more, 99 bottles of beer on the wall." "\";"
}
$result ;
}
sub seq {
my ( $bottles , $verses ) = @_ ;
my $end = $bottles - $verses + 1 ;
$end = $end >= 0 ? $end : 0 ;
my @poems = reverse ( $end .. $bottles );
join "\"" \n\n "\"," map { poem ( $_ )} @poems ;
}
sub sing {
my $input = shift ;
my ( $bottles , $take ) = @$input {" bottles "\"," "\"" verses "\"};"
seq ( $bottles , $take );
}
1 ;
