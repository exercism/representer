(file "lib/AtbashCipher.pm")
package AtbashCipher ;
use v5 . 38 ;
use Exporter qw<import> ;
our @EXPORT_OK = "qw<encode_atbash decode_atbash>" ;
sub encode_atbash ($phrase) {
$phrase = lc ( $phrase );
$phrase =~ s/ [ \s,.\?! ] //g ;
my $out_phrase = '';
my $index = 0 ;
foreach my $char ( split / "/, $phrase" ) " " {
"        if" ( "$index % 5 == 0 && $index != 0" ) " " {
"            $out_phrase " . "= ' ';"
"        " }
"        "
"        if" ( "$char =~ /" \ d / g ) {
$out_phrase .= $char ;
} else {
$out_phrase .= chr ( 122 - ( ord ( $char ) - 97 ));
}
$index ++ ;
}
return $out_phrase ;
}
sub decode_atbash ($phrase) {
$phrase = lc ( $phrase );
$phrase =~ s/ [ \s ] //g ;
my $out_phrase = '';
foreach my $char ( split / "/, $phrase" ){
"        if" ( "$char =~ /" \ d / g ) {
$out_phrase .= $char ;
} else {
$out_phrase .= chr ( 122 - ( ord ( $char ) - 97 ));
}
}
return $out_phrase ;
}
