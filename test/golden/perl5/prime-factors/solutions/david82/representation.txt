package PrimeFactors ;
use v5 . 42 ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<factors> ;
sub factors ($number) {
my @prime_factors ;
my $val = $number ;
while ( $val % 2 == 0 && $val > 1 ) {
push @prime_factors , 2 ;
$val /= 2 ;
}
my $divisor = 3 ;
while ( $divisor * $divisor <= $val ) {
if ( $val % $divisor == 0 ) {
push @prime_factors , $divisor ;
$val /= $divisor ;
} else {
$divisor += 2 ;
}
}
if ( $val > 1 ) {
push @prime_factors , $val ;
}
return \ @prime_factors ;
}
1 ;
