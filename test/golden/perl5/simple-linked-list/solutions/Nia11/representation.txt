package LinkedList ;
use strict ;
use warnings ;
use Exporter ' import ';
our @EXPORT_OK = "qw( new from_array to_array next reverse data )" ;
sub new {
my ( $class , $data , $next ) = @_ ;
my $self = { data => $data , prev => undef , next => $next };
$next -> { prev } = $self if ( defined $next );
return bless $self , $class ;
}
sub data {
my $self = shift ;
return $self -> { data };
}
sub next {
my $self = shift ;
return $self -> { next };
}
sub from_array {
my ( $self , $array ) = @_ ;
return undef if ( scalar @ { $array } == 0 );
my $list ;
for ( reverse @ { $array }){
$list = defined $list ? $self -> new ( $_ , $list ) : $self -> new ( $_ );
}
return $list ;
}
sub to_array {
my $self = shift ;
my @array ;
do {
push @array , $self -> { data };
$self = $self -> { next };
} while ( defined $self );
return [ @array ];
}
sub reverse {
my $self = shift ;
while ( defined $self -> { next }){
( $self -> { prev }, $self -> { next }) = ( $self -> { next }, $self -> { prev });
$self = $self -> { prev };
};
( $self -> { prev }, $self -> { next }) = ( $self -> { next }, $self -> { prev });
return $self ;
}
1 ;
