package Raindrops ;
use v5 . 40 ;
use Exporter qw<import> ;
our @EXPORT_OK = qw<raindrop> ;
sub raindrop ($number) {
my @rules = ([ 3 , ' Pling '], [ 5 , ' Plang '], [ 7 , ' Plong ']);
my $result = join '', map { $_ -> [ 1 ] } grep { $number % $_ -> [ 0 ] == 0 } @rules ;
return $result || $number ;
}
1 ;
