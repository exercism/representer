- module ( isogram ).
- export ([ is_isogram / 1 , test_version / 0 ]).
is_isogram ( String ) ->
is_isogram ( lists : sort ( string : to_lower ( String )), $. ).
is_isogram ([], _ Prev ) ->
true ;
is_isogram ([ H |_ T ], H ) when H >= $a andalso H =< $z ->
false ;
is_isogram ([ H | T ], _ Prev ) ->
is_isogram ( T , H ).
test_version () ->
1 .
