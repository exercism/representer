- module ( isogram ).
- export ([ is_isogram / 1 ]).
- compile ([ debug_info ]).
is_isogram (_ Phrase ) ->
_ P = lists : filtermap (
fun ( X ) ->
I = string : to_lower ( X ),
case I >= $a andalso I =< $z of
true -> { true , I };
_ -> false
end
end ,
_ Phrase ),
length (_ P ) =:= length ( lists : usort (_ P )).
