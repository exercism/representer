- module ( resistor_color_duo ).
- export ([ value / 1 ]).
mapper ( Atom ) ->
Mapper = # {
black => 0 ,
brown => 1 ,
red => 2 ,
orange => 3 ,
yellow => 4 ,
green => 5 ,
blue => 6 ,
violet => 7 ,
grey => 8 ,
white => 9
},
maps : get ( Atom , Mapper ).
value ( Colours ) ->
{ Interesting , _} = lists : split ( 2 , Colours ),
list_to_integer ( lists : concat ( lists : map ( fun mapper / 1 , Interesting ))).
