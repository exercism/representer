(file "src/rational_numbers.erl")
- module ( rational_numbers ).
- export ([ absolute / 1 , add / 2 , divide / 2 , exp / 2 , mul / 2 , reduce / 1 , sub / 2 ]).
absolute ({ N , D }) -> reduce ({ sign ( N ) * N , sign ( D ) * D }).
add ({ N1 , D1 }, { N2 , D2 }) -> reduce ({ N1 * D2 + N2 * D1 , D1 * D2 }).
divide ( R1 , { N2 , D2 }) -> mul ( R1 , { D2 , N2 }).
exp ( Base , { N , D }) -> math : pow ( Base , N / D );
exp ({ N , D }, Exp ) when Exp < 0 -> exp ({ D , N }, - Exp );
exp ({_, _} = R , Exp ) -> lists : foldl ( fun (_, Acc ) -> mul ( R , Acc ) end , { 1 , 1 }, lists : seq ( 1 , Exp )).
mul ({ N1 , D1 }, { N2 , D2 }) -> reduce ({ N1 * N2 , D1 * D2 }).
reduce ({ N , D }) ->
SgnN = sign ( N ),
SgnD = sign ( D ),
AbsN = SgnN * N ,
AbsD = SgnD * D ,
Gcd = gcd ( AbsN , AbsD ),
{ SgnN * SgnD * AbsN div Gcd , AbsD div Gcd }.
sub ( R1 , { N2 , D2 }) -> add ( R1 , { - N2 , D2 }).
sign ( N ) when N < 0 -> - 1 ;
sign (_) -> 1 .
gcd ( A , 0 ) -> A ;
gcd ( 0 , B ) -> B ;
gcd ( A , A ) -> A ;
gcd ( A , B ) when A > B -> gcd ( A rem B , B );
gcd ( A , B ) -> gcd ( A , B rem A ).
