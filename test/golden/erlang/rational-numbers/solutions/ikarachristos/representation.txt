(file "src/rational_numbers.erl")
- module ( rational_numbers ).
- export ([ absolute / 1 , add / 2 , divide / 2 , exp / 2 , mul / 2 , reduce / 1 , sub / 2 ]).
absolute ({ A , B }) -> reduce ({ abs ( A ), abs ( B )}).
add ({ A1 , B1 }, { A2 , B2 }) -> reduce ({ A1 * B2 + A2 * B1 , B1 * B2 }).
sub ({ A1 , B1 }, { A2 , B2 }) -> reduce ({ A1 * B2 - A2 * B1 , B1 * B2 }).
mul ({ A1 , B1 }, { A2 , B2 }) -> reduce ({ A1 * A2 , B1 * B2 }).
divide ({ A1 , B1 }, { A2 , B2 }) -> reduce ({ A1 * B2 , A2 * B1 }).
exp ({ A , B }, Exponent ) when is_float ( Exponent ) ->
reduce ({ math : pow ( A , Exponent ), math : pow ( B , Exponent )});
exp ({ A , B }, Exponent ) when Exponent >= 0 ->
reduce ({ trunc ( math : pow ( A , Exponent )), trunc ( math : pow ( B , Exponent ))});
exp ({ A , B }, Exponent ) ->
reduce ({ trunc ( math : pow ( B , abs ( Exponent ))), trunc ( math : pow ( A , abs ( Exponent )))});
exp ( X , { A , B }) -> math : pow ( math : pow ( X , A ), 1 / B ).
reduce ({ A , B }) ->
G = gcd ( A , B ),
format_negative ({ A div G , B div G }).
format_negative ({ A , B }) when B < 0 -> { - A , - B };
format_negative ( R ) -> R .
gcd ( A , 0 ) -> A ;
gcd ( A , B ) -> gcd ( B , A rem B ).
