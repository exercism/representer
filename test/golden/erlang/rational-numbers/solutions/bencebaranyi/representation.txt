- module ( rational_numbers ).
- export ([ absolute / 1 , add / 2 , divide / 2 , exp / 2 , mul / 2 , reduce / 1 , sub / 2 ]).
absolute ( R ) ->
{ N , D } = reduce ( R ),
{ abs ( N ), abs ( D )}.
add ({ N1 , D1 } = _ R1 , { N2 , D2 } = _ R2 ) ->
R = {( N1 * D2 + N2 * D1 ), ( D1 * D2 )},
reduce ( R ).
divide ({ N1 , D1 } = _ R1 , { N2 , D2 } = _ R2 ) ->
R = { N1 * D2 , N2 * D1 },
reduce ( R ).
exp ({ N , D }, X ) when X >= 0 , is_integer ( X ) ->
R = { trunc ( math : pow ( N , X )), trunc ( math : pow ( D , X ))},
reduce ( R );
exp ({ N , D }, X ) when X < 0 , is_integer ( X ) ->
M = abs ( X ),
R = { trunc ( math : pow ( D , M )), trunc ( math : pow ( N , M ))},
reduce ( R );
exp ({ N , D }, X ) when is_float ( X ) ->
math : pow ( N , X ) / math : pow ( D , X );
exp ( X , { N , D } = _ Rational ) ->
math : pow ( X , N / D ).
mul ({ N1 , D1 } = _ R1 , { N2 , D2 } = _ R2 ) ->
R = { N1 * N2 , D1 * D2 },
reduce ( R ).
reduce ({ N , D } = _ R ) ->
GCD = abs ( gcd ( N , D )),
case D < 0 of
true ->
{ - N div GCD , - D div GCD };
false ->
{ N div GCD , D div GCD }
end .
sub ({ N1 , D1 } = _ R1 , { N2 , D2 } = _ R2 ) ->
R = {( N1 * D2 - N2 * D1 ), ( D1 * D2 )},
reduce ( R ).
gcd ( A , 0 ) -> A ;
gcd ( A , B ) -> gcd ( B , A rem B ).
