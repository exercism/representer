- module ( word_count ).
- export ([ count_words / 1 ]).
count_words ( Sentence ) ->
{ match , Words } =
re : run ( Sentence , "\"\\\\b(?:\\\\w+(?:'\\\\w+)*|\\\\d+)\\\\b\"" , [ global , { capture , first , list }]),
io : fwrite ( "\"Szavak: ~w\"" , [ Words ]),
lists : foldl ( fun ( Word , Acc ) ->
maps : update_with ( string : lowercase ( Word ), fun ( N ) -> N + 1 end , 1 , Acc ) end , # {}, Words ).
