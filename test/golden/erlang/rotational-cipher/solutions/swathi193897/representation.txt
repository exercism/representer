- module ( rotational_cipher ).
decrypt ( String , Key ) -> encrypt ( String , 26 - Key ).
encrypt ( String , Key ) -> lists : map ( fun ( Character ) -> rotate ( Character , Key ) end , String ).
rotate ( Character , Key ) when $A =< Character andalso Character =< $Z ->
Rotated = Character + Key ,
if Rotated =< $Z -> Rotated ;
Rotated > $Z -> Rotated - 26
end ;
rotate ( Character , Key ) when $a =< Character andalso Character =< $z ->
Delta = $a - $A ,
rotate ( Character - Delta , Key ) + Delta ;
rotate ( Character , _) -> Character .
