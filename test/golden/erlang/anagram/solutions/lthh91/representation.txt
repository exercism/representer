(file "src/anagram.erl")
- module ( anagram ).
- export ([ find_anagrams / 2 , are_anagrams / 2 ]).
find_anagrams (_ Subject , _ Candidates ) ->
lists : filter ( fun (_ Candidate ) -> string : casefold (_ Candidate ) /= string : casefold (_ Subject )
andalso
are_anagrams ( string : casefold (_ Subject ),
string : casefold (_ Candidate )) end ,
_ Candidates ).
are_anagrams ([], []) -> true ;
are_anagrams (_, []) -> false ;
are_anagrams ([], _) -> false ;
are_anagrams ([ H | T ], _ Compare ) ->
case lists : member ( H , _ Compare ) of
true -> are_anagrams ( T , lists : delete ( H , _ Compare ));
false -> false
end .
