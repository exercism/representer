- module ( anagram ).
- export ([ find_anagrams / 2 ]).
- spec find_anagrams ( string (), [ string ()]) -> [ string ()].
find_anagrams ( Subject , Candidates ) ->
lists : filter ( fun ( Candidate ) ->
is_anagram_of ( Subject , Candidate ) end ,
Candidates ).
is_anagram_of ( Subject , Candidate ) ->
Word = string : lowercase ( Subject ),
Other = string : lowercase ( Candidate ),
Word /= Other andalso lists : sort ( Word ) == lists : sort ( Other ).
