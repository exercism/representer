- module ( anagram ).
- export ([ find_anagrams / 2 ]).
find_anagrams (_ Subject , _ Candidates ) ->
case [ Word || Word <- _ Candidates , ( lists : sort ( string : lowercase (_ Subject )) == lists : sort ( string : lowercase ( Word ))) and ( string : lowercase (_ Subject ) /= string : lowercase ( Word ))] of
Word -> Word
end .
