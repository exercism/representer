(file "src/anagram.erl")
- module ( anagram ).
- export ([ find_anagrams / 2 ]).
find_anagrams ( Subject , Candidates ) ->
lists : filter ( pred ( Subject ), Candidates ).
pred ( Subject ) ->
fun
( Candidate ) ->
( lists : sort ( string : to_lower ( Subject )) == lists : sort ( string : to_lower ( Candidate )))
and
( string : to_lower ( Subject ) /= string : to_lower ( Candidate ))
end .
