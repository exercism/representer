- module ( anagram ).
- export ([ find_anagrams / 2 ]).
form (_ Map , [ H0 | T ]) ->
H = string : to_lower ( H0 ),
_ Is_key = maps : is_key ( H ,_ Map ),
form (_ Map # { H => if _ Is_key -> maps : get ( H ,_ Map ) + 1 ;
true -> 1 end }, T );
form (_ Map ,[]) -> _ Map .
form_map (_ List ) -> form ( # {},_ List ).
find_anagrams (_ Subject , _ Candidates ) ->
Subj_Map = form_map (_ Subject ),
_ Lower_Subj = [ string : to_lower ( Ch ) || Ch <- _ Subject ],
Trav_List = fun Trav (_ Result , [ H | T ]) ->
_ Is_equal = ( form_map ( H ) =:= Subj_Map ),
_ Lower_str = [ string : to_lower ( Ch ) || Ch <- H ],
Trav ( if (_ Is_equal ) andalso (_ Lower_str /= _ Lower_Subj ) -> [ H |_ Result ];
true -> _ Result end , T );
Trav (_ Result , []) -> _ Result end ,
Trav_List ([], _ Candidates ).
