- module ( darts ).
- export ([ score / 2 ]).
score ( X , Y ) ->
score ( distance_from_middle ( X , Y )).
score ( Dist ) when Dist =< 1 -> 10 ;
score ( Dist ) when Dist =< 5 -> 5 ;
score ( Dist ) when Dist =< 10 -> 1 ;
score (_) -> 0 .
distance_from_middle ( X , Y ) ->
math : sqrt ( math : pow ( X , 2 ) + math : pow ( Y , 2 )).
