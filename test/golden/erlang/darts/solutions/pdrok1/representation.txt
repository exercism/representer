(file "src/darts.erl")
- module ( darts ).
- export ([ score / 2 ]).
dist ( X1 , Y1 , X2 , Y2 ) ->
math : sqrt ( math : pow ( X2 - X1 , 2 ) + math : pow ( Y2 - Y1 , 2 )).
score (_ X , _ Y ) ->
Dist = dist ( 0 , 0 , _ X , _ Y ),
A = if
Dist =< 1 -> 10 ;
Dist =< 5 -> 5 ;
Dist =< 10 -> 1 ;
true -> 0
end ,
A .
