(file "src/grains.erl")
- module ( grains ).
- export ([ square / 1 , total / 0 ]).
square (_ Square ) when _ Square > 0 , _ Square < 65 -> 1 bsl (_ Square - 1 );
square (_) -> { error , "\"square must be between 1 and 64\"" }.
total () ->
lists : sum ([ square ( X ) || X <- lists : seq ( 1 , 64 )]).
