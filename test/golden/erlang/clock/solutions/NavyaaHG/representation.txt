- module ( clock ).
create ( Hour , Minutes ) ->
minutes_add ( 0 , Hour * 60 + Minutes ).
is_equal (_ Clock1 , _ Clock2 ) ->
io : format ( "\"~w ~w\"" ,[_ Clock1 , _ Clock2 ]),
_ Clock1 == _ Clock2 .
minutes_add ( M , A ) ->
N = ( M + A ) rem 1440 ,
case N < 0 of
false -> N ;
true -> N + 1440
end .
to_string ( Minutes ) ->
{ H , M } = { Minutes div 60 , Minutes rem 60 },
[ H div 10 + $0 , H rem 10 + $0 , $: , M div 10 + $0 , M rem 10 + $0 ].
