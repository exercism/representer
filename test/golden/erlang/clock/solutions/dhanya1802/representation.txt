(file "src/clock.erl")
- module ( clock ).
- export ([ create / 2 , is_equal / 2 , minutes_add / 2 , to_string / 1 ]).
create ( Hour , Minute ) when Minute < 0 -> create ( Hour - 1 , 60 + Minute );
create ( Hour , Minute ) when Minute >= 60 -> create ( Hour + 1 , Minute - 60 );
create ( Hour , Minute ) when Hour >= 24 -> create ( Hour - 24 , Minute );
create ( Hour , Minute ) when Hour < 0 -> create ( 24 + Hour , Minute );
create ( Hour , Minute ) -> { Hour , Minute }.
is_equal ( Clock1 , Clock2 ) -> Clock1 =:= Clock2 .
minutes_add (_ Clock = { Hour , Minute }, Minutes ) -> create ( Hour , Minute + Minutes ).
to_string ({ Hour , Minute }) -> io_lib : format ( "\"~2.10.0B:~2.10.0B\"" ,[ Hour , Minute ]).
