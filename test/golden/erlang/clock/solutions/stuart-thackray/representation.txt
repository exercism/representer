- module ( clock ).
- export ([ create / 2 , is_equal / 2 , minutes_add / 2 , to_string / 1 ]).
- define ( MIN_IN_DAY , ( 60 * 24 )).
create ( H , M ) ->
Time = ( H * 60 + M ) rem ? MIN_IN_DAY ,
TempT = if ( Time < 0 ) ->
? MIN_IN_DAY + Time ;
true -> Time
end ,
{ TempT div 60 , TempT rem 60 }.
is_equal ( Clock1 , Clock1 ) -> true ;
is_equal (_,_) -> false .
minutes_add ({ H , M }, Mins ) ->
Time = ( H * 60 + M + Mins ) rem ? MIN_IN_DAY ,
TempT = if ( Time < 0 ) ->
? MIN_IN_DAY + Time ;
true -> Time
end ,
{ TempT div 60 , TempT rem 60 }.
to_string ({ H , M }) -> io_lib : format ( "\"~2..0B:~2..0B\"" , [ H , M ]).
