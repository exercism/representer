- module ( hamming ).
distance ( Strand1 , Strand2 ) ->
if length ( Strand1 ) == length ( Strand2 ) ->
check_hd ( Strand1 , Strand2 , 0 );
true ->
{ error , badarg }
end .
check_hd ([],[], Dis ) -> Dis ;
check_hd ([ H | Strand1 ],[ H | Strand2 ], Dis ) -> check_hd ( Strand1 , Strand2 , Dis );
check_hd ([_| Strand1 ],[_| Strand2 ], Dis ) -> check_hd ( Strand1 , Strand2 , Dis + 1 ).
