(file "src/grade_school.erl")
- module ( grade_school ).
- export ([ add / 3 , get / 2 , get / 1 , new / 0 ]).
add ( Name , Grade , School ) ->
Names = lists : sort ([ S || {_, S } <- School ]),
case lists : member ( Name , Names ) of
true -> School ;
_ -> [{ Grade , Name } | School ]
end .
get ( Grade , School ) ->
[ Student || {_, Student } <- lists : sort ([{ G , _ S } || { G , _ S } <- School , G =:= Grade ])].
get ( School ) ->
lists : sort ([ Student || {_, Student } <- School ]).
new () -> [].
