- module ( sieve ).
primes (_ Limit ) -> get_prime ( 2 ,_ Limit ,[]).
get_prime (_ Start ,_ End , Res ) when _ Start =< _ End ->
case prime_number (_ Start , 2 ) of
true -> get_prime (_ Start + 1 ,_ End ,[_ Start | Res ]);
false -> get_prime (_ Start + 1 ,_ End , Res ) end ;
get_prime (_ Start ,_ End ,_ Res ) -> lists : reverse (_ Res ).
prime_number (_ Start ,_ Num ) when _ Start rem _ Num =:= 0 andalso _ Num =:= _ Start ->
true ;
prime_number (_ Start ,_ Num ) when _ Start rem _ Num =/= 0 andalso _ Num < _ Start ->
prime_number (_ Start ,_ Num + 1 );
prime_number (_ Start ,_ Num ) ->
false .
