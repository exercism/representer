- module ( dominoes ).
- export ([ can_chain / 1 , build_graph / 1 , build_visited / 1 ]).
can_chain ([]) -> true ;
can_chain ([{ A , A }]) -> true ;
can_chain ([{_, _}]) -> false ;
can_chain ( Dominoes ) ->
Graph = build_graph ( Dominoes ),
Visited = build_visited ( Dominoes ),
can_chain ( 1 , Visited , Graph ).
can_chain ( Current , Visited , Graph ) ->
Connected = get_unvisited_neighbors ( Current , Visited , Graph ),
NewVisited = maps : put ( Current , true , Visited ),
case Connected of
[] -> lists : all ( fun ( X ) -> X end , maps : values ( NewVisited ));
_ -> lists : any ( fun ( X ) -> can_chain ( X , NewVisited , Graph ) end , Connected )
end .
get_unvisited_neighbors ( Current , Visited , Graph ) ->
[ I || I <- maps : get ( Current , Graph , []), not maps : get ( I , Visited )].
build_graph ( Dominoes ) ->
build_graph ( enumerate ( Dominoes ), enumerate ( Dominoes ), maps : new ()).
build_graph ([], _, Graph ) -> Graph ;
build_graph ([{ I , { A , B }}| T ], Dominoes , Graph ) ->
IsConnected = fun ({ J , { C , D }}) -> ( J /= I ) andalso ( C == A orelse C == B orelse D == A orelse D == B ) end ,
Connected = lists : filter ( IsConnected , Dominoes ),
Indices = lists : map ( fun ({ J , _}) -> J end , Connected ),
build_graph ( T , Dominoes , maps : put ( I , Indices , Graph )).
build_visited ( Dominoes ) ->
build_visited ( enumerate ( Dominoes ), maps : new ()).
build_visited ([], Visited ) -> Visited ;
build_visited ([{ I , _}| T ], Visited ) ->
build_visited ( T , maps : put ( I , false , Visited )).
enumerate ( List ) ->
{ List1 , _ } = lists : mapfoldl ( fun ( T , Acc ) -> {{ Acc , T }, Acc + 1 } end , 1 , List ),
List1 .
