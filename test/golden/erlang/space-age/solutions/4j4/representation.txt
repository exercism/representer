- module ( space_age ).
age ( earth , _ Seconds ) -> calc (_ Seconds , 1 . 0 );
age ( mercury , _ Seconds ) -> calc (_ Seconds , 0 . 2408467 );
age ( venus , _ Seconds ) -> calc (_ Seconds , 0 . 61519726 );
age ( mars , _ Seconds ) -> calc (_ Seconds , 1 . 8808158 );
age ( jupiter , _ Seconds ) -> calc (_ Seconds , 11 . 862615 );
age ( saturn , _ Seconds ) -> calc (_ Seconds , 29 . 447498 );
age ( uranus , _ Seconds ) -> calc (_ Seconds , 84 . 016846 );
age ( neptune , _ Seconds ) -> calc (_ Seconds , 164 . 79132 ).
calc ( Sec , Fac ) ->
Sec / 31557600 / Fac .
