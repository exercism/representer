- module ( difference_of_squares ).
square_of_sum (_ Number ) -> trunc ( math : pow ( sum (_ Number ), 2 )).
sum_of_squares ( 1 ) -> 1 ;
sum_of_squares (_ Number ) -> trunc ( math : pow (_ Number , 2 )) + sum_of_squares (_ Number - 1 ).
difference_of_squares (_ Number ) -> square_of_sum (_ Number ) - sum_of_squares (_ Number ).
sum ( 1 ) -> 1 ;
sum (_ Number ) -> _ Number + sum (_ Number - 1 ).
