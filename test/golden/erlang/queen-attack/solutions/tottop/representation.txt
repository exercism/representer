(file "src/queen_attack.erl")
- module ( queen_attack ).
- export ([ can_attack / 2 ]).
can_attack ({ W1 , W2 }, { B1 , B2 }) ->
Sum = { W1 - B1 , W2 - B2 },
can_attack ( Sum ).
can_attack ({ A , B }) when ( A == B ); ( A == - B ) -> true ;
can_attack ({ A , B }) when ( A == 0 ); ( B == 0 ) -> true ;
can_attack (_) -> false .
