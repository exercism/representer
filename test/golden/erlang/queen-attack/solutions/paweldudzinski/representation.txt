- module ( queen_attack ).
- export ([ can_attack / 2 ]).
can_attack ( WhiteQueen , BlackQueen ) ->
Diagonal = are_on_diagonal ( WhiteQueen , BlackQueen ),
SameRow = are_on_the_same_row ( WhiteQueen , BlackQueen ),
Diagonal orelse SameRow .
are_on_diagonal ({ WhiteX , WhiteY }, { BlackX , BlackY }) when abs ( WhiteX - BlackX ) =:= abs ( WhiteY - BlackY ) ->
true ;
are_on_diagonal (_, _) ->
false .
are_on_the_same_row ({ WhiteX , _}, { BlackX , _}) when WhiteX =:= BlackX ->
true ;
are_on_the_same_row ({_, WhiteY }, {_, BlackY }) when WhiteY =:= BlackY ->
true ;
are_on_the_same_row (_, _) ->
false .
