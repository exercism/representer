(file "src/queen_attack.erl")
- module ( queen_attack ).
- export ([ can_attack / 2 ]).
can_attack ( WhiteQueen , BlackQueen ) ->
{ Wx , Wy } = WhiteQueen ,
{ Bx , By } = BlackQueen ,
case { abs ( Wx - Bx ), abs ( Wy - By )} of
{_ X , _ X } -> true ;
{ 0 , _ } -> true ;
{ _, 0 } -> true ;
_ -> false
end .
