- module ( queen_attack ).
can_attack ( WhiteQueen , BlackQueen ) ->
{ Wx , Wy } = WhiteQueen ,
{ Bx , By } = BlackQueen ,
{ Xdiff , Ydiff } = { Wx - Bx , Wy - By },
Can_diagonally = abs ( Xdiff ) == abs ( Ydiff ),
Can_vertically = Wy == By ,
Can_horizontally = Wx == Bx ,
Can_diagonally orelse Can_vertically orelse Can_horizontally .
