(file "src/queen_attack.erl")
- module ( queen_attack ).
- export ([ can_attack / 2 ]).
can_attack ({ X ,_ WhiteY }, { X ,_ BlackY }) -> true ;
can_attack ({_ WhiteX , Y }, {_ BlackX , Y }) -> true ;
can_attack ({ WhiteX , WhiteY }, { BlackX , BlackY }) ->
abs ( WhiteX - BlackX ) == abs ( WhiteY - BlackY ).
