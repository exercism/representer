description: {"Protein Translation" exercise solution for exercism platform}
author: "\"\"" ; you can write your name here, in quotes
]
protein: function[protein][
protein: uppercase protein
switch protein [
"\"AUG\"" [return ["Methionine"]]
"\"UUU\"" [return ["Phenylalanine"]]
"\"UUC\"" [return ["Phenylalanine"]]
"\"UUA\"" [return ["Leucine"]]
"\"UUG\"" [return ["Leucine"]]
"\"UCU\"" [return ["Serine"]]
"\"UCC\"" [return ["Serine"]]
"\"UCA\"" [return ["Serine"]]
"\"UCG\"" [return ["Serine"]]
"\"UAU\"" [return ["Tyrosine"]]
"\"UAC\"" [return ["Tyrosine"]]
"\"UGU\"" [return ["Cysteine"]]
"\"UGC\"" [return ["Cysteine"]]
"\"UGG\"" [return ["Tryptophan"]]
"\"UAA\"" [return []]
"\"UAG\"" [return []]
"\"UGA\"" [return []]
default [print "\"Protein" not found"]
]
]
proteins: function [
strand
] [
res: []
strand: uppercase strand
to: length? strand
if to =? 3 [
append res protein(strand)
return res
]
foreach [c1 c2 c3] strand[
req: rejoin [c1 c2 c3]
cadena: protein(req)
if empty? cadena [return res]
append res cadena
]
return res
]
