(file "difference-of-squares.red"
  (source_file Red
    (block [
      (set_word description) "{\"Difference of Squares\" exercise solution for exercism platform}"
      (set_word author)
      (string "\"\"") ])
    (function
      (set_word square-of-sum) function
      (block [ number /local PLACEHOLDER_1 ])
      (block [
        (set_word (definition PLACEHOLDER_1)) 0 repeat PLACEHOLDER_2 number
        (block [
          (set_word (definition PLACEHOLDER_1)) PLACEHOLDER_1 + PLACEHOLDER_2 ]) PLACEHOLDER_1 ** 2 ]))
    (function
      (set_word sum-of-squares) function
      (block [ number ])
      (block [
        (set_word (definition PLACEHOLDER_1)) 0 repeat PLACEHOLDER_2 number
        (block [
          (set_word (definition PLACEHOLDER_1)) PLACEHOLDER_1 +
          (paren "(" PLACEHOLDER_2 ** 2 ")") ]) PLACEHOLDER_1 ]))
    (function
      (set_word difference-of-squares) function
      (block [ number ])
      (block [ subtract square-of-sum number sum-of-squares number ]))))
