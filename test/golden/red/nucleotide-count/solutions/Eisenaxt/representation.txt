]
nucleotide-counts: function [
strand [string!]
return: [map!]
] [
nucleo-map: make map! [A: 0 C: 0 G: 0 T: 0]
forall strand [
if not find "\"ACGT\"" first strand [
do make error! "\"Invalid" nucleotide in strand"
]
put nucleo-map to-word first strand 1 + select nucleo-map to-word first strand
]
return nucleo-map
]
