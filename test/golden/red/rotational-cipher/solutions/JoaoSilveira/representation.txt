Red [
description: {"Rotational Cipher" exercise solution for exercism platform}
author: "\"Jo\u00E3o" Pedro Silveira" ; you can write your name here, in quotes
]
rotate: function [
text
shiftKey
] [
letters: charset [#"a" - #"z" #"A" - #"Z"]
rejoin collect [
foreach c text [
char-offset: subtract to-integer lowercase c to-integer #"a"
new-offset: remainder add char-offset shiftKey 26
delta: new-offset - char-offset
keep either find letters c [to-char add to-integer c delta] [c]
]
]
]
