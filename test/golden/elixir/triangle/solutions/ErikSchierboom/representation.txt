    (call defmodule
      (arguments Triangle)
      (do_block
        (call def
          (arguments
            (binary_operator
              (call kind
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2) (variable PLACEHOLDER_3))) when
              (binary_operator
                (binary_operator
                  (binary_operator PLACEHOLDER_1 <= 0) or
                  (binary_operator PLACEHOLDER_2 <= 0)) or
                (binary_operator PLACEHOLDER_3 <= 0))))
          (do_block
            (tuple :error "all side lengths must be positive")))
        (call def
          (arguments
            (binary_operator
              (call kind
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2) (variable PLACEHOLDER_3))) when
              (binary_operator
                (binary_operator
                  (binary_operator
                    (binary_operator PLACEHOLDER_1 + PLACEHOLDER_2) < PLACEHOLDER_3) or
                  (binary_operator
                    (binary_operator PLACEHOLDER_2 + PLACEHOLDER_3) < PLACEHOLDER_1)) or
                (binary_operator
                  (binary_operator PLACEHOLDER_1 + PLACEHOLDER_3) < PLACEHOLDER_2))))
          (do_block
            (tuple :error "side lengths violate triangle inequality")))
        (call def
          (arguments
            (call kind

          (do_block
            (call case
              (arguments
                (binary_operator
                  (binary_operator
                    (list PLACEHOLDER_1 PLACEHOLDER_2 PLACEHOLDER_3) |>
                    (call
                      (dot Enum . uniq)
                      (arguments))) |>
                  (call
                    (dot Enum . count)
                    (arguments))))

                (stab_clause
                  (arguments 1) ->
                  (body

                  (body
                    (tuple :ok :scalene)))))))))))
