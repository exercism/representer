(file "lib/guessing_game.ex"
  (source
    (call defmodule
      (arguments GuessingGame)
      (do_block
        (call def
          (arguments
            (binary_operator
              (call compare
                (arguments (variable PLACEHOLDER_1)
                  (binary_operator (variable PLACEHOLDER_2) "\\\\" :no_guess))) when
              (binary_operator PLACEHOLDER_2 == :no_guess)))
          (do_block "Make a guess"))
        (call def
          (arguments
            (binary_operator
              (call compare
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2))) when
              (binary_operator PLACEHOLDER_1 == PLACEHOLDER_2)))
          (do_block "Correct"))
        (call def
          (arguments
            (binary_operator
              (call compare
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2))) when
              (binary_operator
                (binary_operator PLACEHOLDER_2 - PLACEHOLDER_1) > 1)))
          (do_block "Too high"))
        (call def
          (arguments
            (binary_operator
              (call compare
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2))) when
              (binary_operator
                (binary_operator PLACEHOLDER_2 - PLACEHOLDER_1) <
                (unary_operator - 1))))
          (do_block "Too low"))
        (call def
          (arguments
            (binary_operator
              (call compare
                (arguments (variable PLACEHOLDER_1) (variable PLACEHOLDER_2))) when
              (binary_operator
                (call abs
                  (arguments
                    (binary_operator PLACEHOLDER_2 - PLACEHOLDER_1))) == 1)))
          (do_block "So close"))))))
