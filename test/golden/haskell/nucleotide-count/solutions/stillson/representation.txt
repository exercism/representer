module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map )
import qualified Data.Map as Map
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts [] = Right Map . empty
nucleotideCounts ( nucleotide : dna ) = case ( nucleotideCounts dna , c2n nucleotide ) of ( Left l , _ ) -> Left l
  ( Right _ , Nothing ) -> Left "\"Bad Nucleotide\""
  ( Right r , Just n ) -> Right $ Map . insertWith ( + ) n 1 r
c2n :: Char -> Maybe Nucleotide
c2n 'A' = Just A
c2n 'C' = Just C
c2n 'G' = Just G
c2n 'T' = Just T
c2n _ = Nothing
