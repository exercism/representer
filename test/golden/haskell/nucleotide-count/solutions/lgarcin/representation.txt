(file "src/DNA.hs")
module DNA
  ( nucleotideCounts
  , Nucleotide ( .. )
  )
where
import Data.Map ( Map
  , adjust
  , fromList
  )
data Nucleotide
  = A
  | C
  | G
  | T
  deriving ( Eq , Ord , Show )
convert :: Char -> Either String Nucleotide
convert 'A' = Right A
convert 'C' = Right C
convert 'G' = Right G
convert 'T' = Right T
convert c = Left [ c ]
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs =
  foldr ( adjust ( + 1 )) ( fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )])
    <$> traverse convert xs
