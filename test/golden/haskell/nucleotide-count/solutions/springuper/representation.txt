module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList , adjust )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts "\"\"" = Right $ fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )]
nucleotideCounts ( x : xs ) = case x of
  'A' -> adjust ( + 1 ) A <$> nucleotideCounts xs
  'C' -> adjust ( + 1 ) C <$> nucleotideCounts xs
  'G' -> adjust ( + 1 ) G <$> nucleotideCounts xs
  'T' -> adjust ( + 1 ) T <$> nucleotideCounts xs
  _ -> Left $ "\"wrong nucleotide\"" ++ [ x ]
