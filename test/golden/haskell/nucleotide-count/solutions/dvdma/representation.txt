module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map.Strict ( Map )
import Data.Map.Strict as Map
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideCounts' :: Map Nucleotide Int -> String -> Either String ( Map Nucleotide Int )
nucleotideCounts' m [] = Right m
nucleotideCounts' m dna @ ( x : xs )
  | x == 'A' = incAndRepeat A
  | x == 'C' = incAndRepeat C
  | x == 'G' = incAndRepeat G
  | x == 'T' = incAndRepeat T
  | otherwise = Left dna
  where
    incrementNuc nuc = Map . insertWith ( + ) nuc 1 m
    incAndRepeat nuc = Right ( incrementNuc nuc ) >>= flip nucleotideCounts' xs
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts = nucleotideCounts' Map . empty
