module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map as Map
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
mapChar :: Char -> Either String Nucleotide
mapChar c =
  case c of
    'A' -> Right A
    'C' -> Right C
    'G' -> Right G
    'T' -> Right T
    _ -> Left [ c ]
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs =
  Map . fromListWith ( + ) . flip zip ( repeat 1 ) <$> mapM mapChar xs
