module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.List ( group , sort )
import Data.Map ( Map , fromList , union )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
toNucleotide :: Char -> Either String Nucleotide
toNucleotide c = case c of
  'A' -> Right A
  'C' -> Right C
  'G' -> Right G
  'T' -> Right T
  _ -> Left "\"Invalid strand\""
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs = merge . count <$> traverse toNucleotide xs
  where
    zeroMap = fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )]
    merge = flip union zeroMap . fromList
    count = map ( \ x -> ( head x , length x )) . group . sort
