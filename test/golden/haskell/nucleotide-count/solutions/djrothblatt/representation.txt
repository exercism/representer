module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromListWith )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
count :: Char -> Either String ( Nucleotide , Int )
count 'A' = Right ( A , 1 )
count 'C' = Right ( C , 1 )
count 'G' = Right ( G , 1 )
count 'T' = Right ( T , 1 )
count _ = Left "\"Bad nucleotides!\""
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts = fmap ( fromListWith ( + )) . traverse count
