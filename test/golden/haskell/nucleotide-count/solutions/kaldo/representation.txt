module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map )
import qualified Data.Map as Map
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
countInc :: Nucleotide -> ( Map Nucleotide Int ) -> ( Map Nucleotide Int )
countInc nucleotide = Map . insertWith ( + ) nucleotide 1
parseNucleotide :: Char -> Either Char Nucleotide
parseNucleotide 'A' = Right A
parseNucleotide 'C' = Right C
parseNucleotide 'G' = Right G
parseNucleotide 'T' = Right T
parseNucleotide c = Left c
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts str = case traverse parseNucleotide str of
  Left c -> Left [ c ]
  Right xs -> Right ( foldr countInc Map . empty xs )
