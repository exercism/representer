module DNA
  ( nucleotideCounts
  , Nucleotide ( .. )
  ) where
import qualified Data.Map as M
data Nucleotide
  = A
  | C
  | G
  | T
  deriving ( Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( M . Map Nucleotide Int )
nucleotideCounts =
  let aux acc [] = Right $ M . fromAscList $ zip [ A , C , G , T ] acc
    aux [ a , c , g , t ] ( 'A' : as ) = aux [ a + 1 , c , g , t ] as
    aux [ a , c , g , t ] ( 'C' : as ) = aux [ a , c + 1 , g , t ] as
    aux [ a , c , g , t ] ( 'G' : as ) = aux [ a , c , g + 1 , t ] as
    aux [ a , c , g , t ] ( 'T' : as ) = aux [ a , c , g , t + 1 ] as
    aux _ _ = Left "\"Invalid\""
    in aux [ 0 , 0 , 0 , 0 ]
