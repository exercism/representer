module DNA ( nucleotideCounts , Nucleotide ( .. )) where
  import Data.Map.Strict as Map
  data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
  nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
  nucleotideCounts xs = countAllNucleotides xs ( fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )])
  countAllNucleotides :: String -> Map Nucleotide Int -> Either String ( Map Nucleotide Int )
  countAllNucleotides [] m = Right m
  countAllNucleotides ( x : xs ) nucleotideMap
    | x == 'A' = countAllNucleotides xs ( insertWith ( + ) A 1 nucleotideMap )
    | x == 'C' = countAllNucleotides xs ( insertWith ( + ) C 1 nucleotideMap )
    | x == 'G' = countAllNucleotides xs ( insertWith ( + ) G 1 nucleotideMap )
    | x == 'T' = countAllNucleotides xs ( insertWith ( + ) T 1 nucleotideMap )
    | otherwise = Left ( "\"Not a valid DNA\"" )
