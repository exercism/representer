module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList , insertWith )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideFromString :: Char -> Either String ( Nucleotide , Int )
nucleotideFromString c =
  case c of
    'A' -> Right ( A , 0 )
    'C' -> Right ( C , 0 )
    'G' -> Right ( G , 0 )
    'T' -> Right ( T , 0 )
    _ -> Left "\"\""
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs =
  let
    empty = fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )]
    nList = traverse nucleotideFromString xs
  in
    fmap ( foldr ( \ ( k , _ ) m -> insertWith ( + ) k 1 m ) empty ) nList
