module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , adjust , fromList )
data Nucleotide = A | C | G | T deriving ( Bounded , Enum , Eq , Ord , Read , Show )
allNucleotides :: [ Nucleotide ]
allNucleotides = [ minBound :: Nucleotide .. ]
allStrNucleotides :: [ String ]
allStrNucleotides = map show allNucleotides
fromChar :: Char -> Either String Nucleotide
fromChar c | [ c ] ` elem ` allStrNucleotides = Right ( read [ c ])
  | otherwise = Left ( "\"invalid nucleotide: \"" ++ show c )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts input = ( foldr sumFunc emptyCounter ) <$> ( traverse fromChar input )
  where emptyCounter = fromList ( zip allNucleotides ( cycle [ 0 ]))
    sumFunc = ( \ nt acc -> adjust ( + 1 ) nt acc )
