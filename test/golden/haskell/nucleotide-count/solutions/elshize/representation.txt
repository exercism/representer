module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList , adjust )
import Text.Read ( readEither )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show , Enum , Read )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts = foldl update ( Right ( fromList [( A , 0 ), ( C , 0 ), ( G , 0 ), ( T , 0 )]))
update :: Either String ( Map Nucleotide Int ) -> Char -> Either String ( Map Nucleotide Int )
update mm c = readEither [ c ] >>= \ nuc -> mm >>= Right . adjust ( + 1 ) nuc
