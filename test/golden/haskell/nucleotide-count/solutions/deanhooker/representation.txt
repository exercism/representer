module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList , insertWith )
import Control.Monad
data Nucleotide = A | C | G | T deriving ( Enum , Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs = foldM resultFn resultMap xs
  where resultMap = fromList $ zip [ A .. ] ( repeat 0 )
    resultFn r x = let nuc = charToNucleotide x
      in ( insertWith ( + )) <$> nuc <*> Right 1 <*> Right r
charToNucleotide :: Char -> Either String Nucleotide
charToNucleotide c = case c of
  'A' -> Right A
  'C' -> Right C
  'G' -> Right G
  'T' -> Right T
  _ -> Left ( "\"Unknown char: \"" ++ [ c ])
