module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs
  | ( length $ filter (` elem ` "\"ACGT\"" ) xs ) == length xs = Right (
    fromList [
      ( A , ( countOcur 'A' xs ))
    , ( C , ( countOcur 'C' xs ))
    , ( G , ( countOcur 'G' xs ))
    , ( T , ( countOcur 'T' xs )) ])
  | otherwise = Left ( xs )
  where countOcur x = length . filter ( == x )
