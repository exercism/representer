module DNA ( nucleotideCounts , Nucleotide ( .. )) where
import Data.Map ( Map , fromList )
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
countLetterCount :: Char -> String -> Int
countLetterCount c str = length $ filter ( == c ) str
isNucleotide :: Char -> Bool
isNucleotide c = c ` elem ` [ 'A' , 'C' , 'G' , 'T' ]
isDNA :: String -> Bool
isDNA str = all isNucleotide str
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs
  | isDNA xs = Right $ fromList [( A , countLetterCount 'A' xs ), ( C , countLetterCount 'C' xs ), ( G , countLetterCount 'G' xs ), ( T , countLetterCount 'T' xs )]
  | otherwise = Left "\"Not DNA\""
