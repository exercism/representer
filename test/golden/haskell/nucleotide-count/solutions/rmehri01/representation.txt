{-# LANGUAGE LambdaCase #-}
module DNA
  ( nucleotideCounts
  , Nucleotide ( .. )
  )
where
import Data.Map ( Map )
import qualified Data.Map as M
data Nucleotide = A | C | G | T deriving ( Eq , Ord , Show )
nucleotideCounts :: String -> Either String ( Map Nucleotide Int )
nucleotideCounts xs =
  foldr ( \ n m -> M . insertWith ( + ) n 1 m ) M . empty
    <$> traverse charToNucleotide xs
  where
    charToNucleotide = \ case
      'G' -> Right G
      'C' -> Right C
      'T' -> Right T
      'A' -> Right A
      c -> Left $ "\"failed on character \"" ++ [ c ] ++ "\" in string \"" ++ xs
