module Matrix ( saddlePoints ) where
import Data.Array ( Array , indices , assocs , listArray )
saddlePoints :: Array ( Int , Int ) Int -> [( Int , Int )]
saddlePoints matrix =
  [( x , y ) | (( x , y ), v ) <- assocs matrix , all ( v >= ) ( vxs x ) && all ( v <= ) ( vys y )]
  where
    vxs c = [ v | (( x , y ), v ) <- assocs matrix , x == c ]
    vys r = [ v | (( x , y ), v ) <- assocs matrix , y == r ]
