module Matrix ( saddlePoints ) where
import Data.Array ( Array , ( ! ), bounds )
saddlePoints :: ( Ord e ) => Array ( Int , Int ) e -> [( Int , Int )]
saddlePoints matrix =
  [ ( r , c ) | r <- [ rf .. rt ], c <- [ cf .. ct ]
  , matrix ! ( r , c ) == maximum [ matrix ! ( r , c' ) | c' <- [ cf .. ct ]]
  , matrix ! ( r , c ) == minimum [ matrix ! ( r' , c ) | r' <- [ rf .. rt ]]
  ]
  where ( rf , cf ) = fst $ bounds matrix
    ( rt , ct ) = snd $ bounds matrix
