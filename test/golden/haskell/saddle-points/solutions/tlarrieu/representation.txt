{-# LANGUAGE TupleSections #-}
module Matrix
  ( saddlePoints
  ) where
import Data.Array
import Data.List ( foldl' , intersect , sort )
type Cell a = ( Point , a )
type Point = ( Int , Int )
data Order
  = Min
  | Max
  deriving ( Eq )
saddlePoints :: Ord a => Array Point a -> [ Point ]
saddlePoints mx = sort points
  where
    points = ( rows >>= optimums Max ) ` intersect ` ( cols >>= optimums Min )
    rows = map ( map mkCell . ( \ x -> map ( x , ) colIds )) rowIds
    cols = map ( map mkCell . ( \ y -> map (, y ) rowIds )) colIds
    mkCell p = ( p , mx ! p )
    ( rowIds , colIds ) = ([ i .. i' ], [ j .. j' ])
    (( i , j ), ( i' , j' )) = bounds mx
optimums :: Ord a => Order -> [ Cell a ] -> [ Point ]
optimums _ [] = []
optimums k (( p , e ) : cs ) = ids
  where
    ( _ , ids ) = foldl' select ( e , [ p ]) cs
    select ( actual , points ) ( point , next )
      | k == Min && next < actual = ( next , [ point ])
      | k == Max && next > actual = ( next , [ point ])
      | next == actual = ( actual , point : points )
      | otherwise = ( actual , points )
