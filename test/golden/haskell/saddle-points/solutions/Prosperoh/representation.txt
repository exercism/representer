module Matrix ( saddlePoints ) where
import Data.Array ( Array , indices , ( ! ))
type Coord = ( Int , Int )
type Matrix = Array ( Int , Int ) Int
isMinInColumn :: Matrix -> Coord -> Bool
isMinInColumn matrix coord = all ( ( >= ) ( matrix ! coord ) ) $ map ( ( ! ) matrix ) $ columnCoords
  where columnCoords = filter ( \ ( x , _ ) -> x == x0 ) $ indices matrix
    ( x0 , _ ) = coord
isMaxInRow :: Matrix -> Coord -> Bool
isMaxInRow matrix coord = all ( ( <= ) ( matrix ! coord ) ) $ map ( ( ! ) matrix ) $ rowCoords
  where rowCoords = filter ( \ ( _ , y ) -> y == y0 ) $ indices matrix
    ( _ , y0 ) = coord
saddlePoints :: Matrix -> [ Coord ]
saddlePoints matrix = filter ( \ c -> isMaxInRow matrix c && isMinInColumn matrix c ) $ indices matrix
