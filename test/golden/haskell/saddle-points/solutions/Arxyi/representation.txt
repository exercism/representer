module Matrix ( saddlePoints ) where
import Data.Array ( Array , bounds , indices , ( ! ))
import qualified Data.Array as A
saddlePoints matrix = filter ( isSaddlePoint matrix ) ( indices matrix )
isSaddlePoint matrix c @ ( x , y ) = matrix ! c == maxElemInSameRow && matrix ! c == minElemInSameColumn
  where
    (( sx , sy ),( ex , ey )) = bounds matrix
    maxElemInSameRow = maximum [ matrix ! ( x , k ) | k <- [ sy .. ey ]]
    minElemInSameColumn = minimum [ matrix ! ( k , y ) | k <- [ sx .. ex ]]
