module Matrix ( saddlePoints ) where
import qualified Data.Array as A
saddlePoints :: A . Array ( Int , Int ) Int -> [( Int , Int )]
saddlePoints matrix =
  [ i
  | r <- rowRange
  , c <- colRange
  , let i = ( r , c )
  , let v = matrix A .! i
  , v == rowMaxs A .! r
  , v == colMins A .! c
  ]
  where
    (( minRow , minCol ), ( maxRow , maxCol )) = A . bounds matrix
    rowBounds = ( minRow , maxRow )
    rowRange = [ minRow .. maxRow ]
    rowMax r = maximum [ matrix A .! ( r , c ) | c <- colRange ]
    rowMaxs = A . listArray rowBounds [ rowMax r | r <- rowRange ]
    colBounds = ( minCol , maxCol )
    colRange = [ minCol .. maxCol ]
    colMin c = minimum [ matrix A .! ( r , c ) | r <- rowRange ]
    colMins = A . listArray colBounds [ colMin c | c <- colRange ]
