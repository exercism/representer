module Matrix ( saddlePoints ) where
import Data.Array ( Array , bounds , ( ! ))
saddlePoints :: Array ( Int , Int ) Int -> [( Int , Int )]
saddlePoints matrix = [ ( x , y )
  | x <- [ zeroX .. width ]
  , y <- [ zeroY .. height ]
  , matrix ! ( x , y ) == maxRow x
  , matrix ! ( x , y ) == minCols y
  ]
  where
    (( zeroX , zeroY ), ( width , height )) = bounds matrix
    maxRow r = maximum [ matrix ! ( r , c ) | c <- [ zeroY .. height ] ]
    minCols c = minimum [ matrix ! ( r , c ) | r <- [ zeroX .. width ] ]
