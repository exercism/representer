module Matrix ( saddlePoints ) where
import Data.Array ( Array , bounds , ( ! ))
saddlePoints :: Array ( Int , Int ) Int -> [( Int , Int )]
saddlePoints matrix = [ ( i , j ) | i <- [ minX .. maxX ] , j <- [ minY .. maxY ]
  , matrix ! ( i , j ) == maxRow i , matrix ! ( i , j ) == minCol j ]
  where
    (( minX , minY ) , ( maxX , maxY )) = bounds matrix
    maxRow a = maximum [ matrix ! ( a , i ) | i <- [ minY .. maxY ]]
    minCol b = minimum [ matrix ! ( i , b ) | i <- [ minX .. maxX ]]
