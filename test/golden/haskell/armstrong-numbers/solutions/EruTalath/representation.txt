module ArmstrongNumbers where
armstrong :: Integral a => a -> Bool
armstrong n = n == ( sum $ fmap ( ^ length s ) s )
  where
    s = integralToList ( abs n ) []
    integralToList :: Integral a => a -> [ a ] -> [ a ]
    integralToList 0 acc = acc
    integralToList x acc = integralToList ( quot x 10 ) (( rem x 10 ) : acc )
