module Prime ( nth ) where
nth :: Int -> Maybe Integer
nth n
  | n < 1 = Nothing
  | n == 1 = Just 2
  | otherwise = Just $ filter isPrime [ 3 , 5 .. ] !! ( n - 2 )
  where
    isPrime :: Integer -> Bool
    isPrime i = all ( \ x -> rem i x > 0 ) [ 3 .. ( floor . sqrt ) ( fromIntegral i )]
