module ValentinesDay where
data Approval
  = Yes
  | No
  | Maybe
data Cuisine
  = Korean
  | Turkish
data Genre
  = Crime
  | Horror
  | Romance
  | Thriller
data Activity
  = BoardGame
  | Chill
  | Movie Genre
  | Restaurant Cuisine
  | Walk Int
rateActivity :: Activity -> Approval
rateActivity activity =
  case activity of
    BoardGame -> No
    Chill -> No
    Movie Romance -> Yes
    Movie _ -> No
    Restaurant Korean -> Yes
    Restaurant Turkish -> Maybe
    Walk w -> if w < 3 then Yes else if w > 5 then No else Maybe
