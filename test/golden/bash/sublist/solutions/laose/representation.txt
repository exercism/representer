[[ $# -ne 2 ]] && exit 1
[[ "\"" $1 "\"" = ~ [ ^0-9, "\\[\\]\\ " ] || "\"" $2 "\"" = ~ [ ^0-9, "\\[\\]\\ " ] ]] && exit 1
[ "\"" $1 "\"" = "\"" $2 "\"" ] && { echo "\"equal\"" ; exit 0 ; }
[[ "\"" $1 "\"" = "\"[]\"" && "\"" $2 "\"" != "\"[]\"" ]] && { echo "\"sublist\"" ; exit 0 ; }
[[ "\"" $2 "\"" = "\"[]\"" && "\"" $1 "\"" != "\"[]\"" ]] && { echo "\"superlist\"" ; exit 0 ; }
num1 = "\"" ${ 1 //[\[\] ]/ } "\""
num2 = "\"" ${ 2 //[\[\] ]/ } "\""
if (( ${# num1 } > ${# num2 } )) ; then
if [[ "\"," $num1 ," == * "\"," $num2 ," * ]] ; then
echo "\"superlist\""
else
echo "\"unequal\""
fi
else
if [[ "\"," $num2 ," == * "\"," $num1 ," * ]] ; then
echo "\"sublist\""
else
echo "\"unequal\""
fi
fi
