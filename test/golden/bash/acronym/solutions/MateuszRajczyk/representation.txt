main () {
str = $1
str_len = ${# str }
acro = "\"" ${ str :0:1 } "\""
for (( i = 0 ; i< $str_len ; i++ ))
do
n = $(( $i - 1 ))
if [[ "\"" ${ str : $i :1 } "\"" = ~ ^[[:alnum:]] * $ ]] && [[ ! "\"" ${ str : $n :1 } "\"" = ~ ^[[:alnum:]] * $ ]]
then
if [ "\"" ${ str : $n :1 } "\"" != "\"'\"" ]
then
acro+ = "\"" ${ str : $i :1 } "\""
fi
fi
done
acro_uppercase = ${ acro ^^ }
echo "\"" $acro_uppercase "\""
}
main "\"" $@ "\""
