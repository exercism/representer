(program
  (command set -ue)
  (variable_assignment PLACEHOLDER_1
    (string
      (expansion 1 :?
        (concatenation ERROR: " Positive number between 1 and 3999 to convert to roman is required."))))
  (variable_assignment PLACEHOLDER_2 "")
  (while_statement
    (arithmetic_command
      (binary_expression PLACEHOLDER_1 > 0))
    (do_group
      (case_statement
        (string
          (simple_expansion PLACEHOLDER_1))
        (case_item [1-3]
          (variable_assignment PLACEHOLDER_2 += "I")
          (list
            (command let remainder-=1) ||
            (command :)))
        (case_item 4
          (variable_assignment PLACEHOLDER_2 += "IV")
          (list
            (command let remainder-=4) ||
            (command :)))
        (case_item [5-8]
          (variable_assignment PLACEHOLDER_2 += "V")
          (list
            (command let remainder-=5) ||
            (command :)))
        (case_item 9
          (variable_assignment PLACEHOLDER_2 += "IX")
          (list
            (command let remainder-=9) ||
            (command :)))
        (case_item [1-3][0-9]
          (variable_assignment PLACEHOLDER_2 += "X")
          (list
            (command let remainder-=10) ||
            (command :)))
        (case_item
          (concatenation 4 [ 0-9 ])
          (variable_assignment PLACEHOLDER_2 += "XL")
          (list
            (command let remainder-=40) ||
            (command :)))
        (case_item [5-8][0-9]
          (variable_assignment PLACEHOLDER_2 += "L")
          (list
            (command let remainder-=50) ||
            (command :)))
        (case_item
          (concatenation 9 [ 0-9 ])
          (variable_assignment PLACEHOLDER_2 += "XC")
          (list
            (command let remainder-=90) ||
            (command :)))
        (case_item [1-3][0-9][0-9]
          (variable_assignment PLACEHOLDER_2 += "C")
          (list
            (command let remainder-=100) ||
            (command :)))
        (case_item
          (concatenation 4 [ 0-9 ] [ 0-9 ])
          (variable_assignment PLACEHOLDER_2 += "CD")
          (list
            (command let remainder-=400) ||
            (command :)))
        (case_item [5-8][0-9][0-9]
          (variable_assignment PLACEHOLDER_2 += "D")
          (list
            (command let remainder-=500) ||
            (command :)))
        (case_item
          (concatenation 9 [ 0-9 ] [ 0-9 ])
          (variable_assignment PLACEHOLDER_2 += "CM")
          (list
            (command let remainder-=900) ||
            (command :)))
        (case_item [1-3][0-9][0-9][0-9]
          (variable_assignment PLACEHOLDER_2 += "M")
          (list
            (command let remainder-=1000) ||
            (command :)))
        (case_item *
          (command echo
            (string "ERROR: Cannot represent number "
              (simple_expansion PLACEHOLDER_1) " in Roman."))
          (command exit 1)))))
  (command echo
    (string
      (simple_expansion PLACEHOLDER_2))))
