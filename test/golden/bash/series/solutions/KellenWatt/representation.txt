(program
  (variable_assignment PLACEHOLDER_1
    (string
      (simple_expansion 1)))
  (variable_assignment PLACEHOLDER_2
    (expansion # PLACEHOLDER_1))
  (variable_assignment PLACEHOLDER_3
    (string
      (simple_expansion 2)))
  (if_statement
    (test_command [[
      (binary_expression
        (simple_expansion PLACEHOLDER_2) -eq 0)) then
    (command
      (file_redirect >& 2) echo "series cannot be empty")
    (command exit 1)
    (elif_clause
      (test_command [[
        (binary_expression
          (string
            (simple_expansion PLACEHOLDER_3)) -gt
          (simple_expansion PLACEHOLDER_2))) then
      (command
        (file_redirect >& 2) echo "slice length cannot be greater than series length")
      (command exit 1))
    (elif_clause
      (test_command [[
        (binary_expression
          (string
            (simple_expansion PLACEHOLDER_3)) -le 0)) then
      (variable_assignment PLACEHOLDER_4 "negative")
      (if_statement
        (test_command [[
          (binary_expression
            (string
              (simple_expansion PLACEHOLDER_3)) -eq 0)) then
        (variable_assignment PLACEHOLDER_4 "zero"))
      (command
        (file_redirect >& 2) echo
        (string "slice length cannot be "
          (simple_expansion PLACEHOLDER_4)))
      (command exit 1)))
  (variable_assignment PLACEHOLDER_5
    (string
      (expansion PLACEHOLDER_1 : 0 :
        (simple_expansion PLACEHOLDER_3))))
  (c_style_for_statement
    (variable_assignment PLACEHOLDER_6 1) ;
    (binary_expression PLACEHOLDER_6 <
      (binary_expression PLACEHOLDER_2 -
        (binary_expression PLACEHOLDER_3 - 1))) ;
    (postfix_expression PLACEHOLDER_6 ++)
    (do_group
      (variable_assignment PLACEHOLDER_5
        (string
          (simple_expansion PLACEHOLDER_5)
          (expansion PLACEHOLDER_1 : PLACEHOLDER_6 :
            (simple_expansion PLACEHOLDER_3))))))
  (command echo
    (string
      (simple_expansion PLACEHOLDER_5))))
