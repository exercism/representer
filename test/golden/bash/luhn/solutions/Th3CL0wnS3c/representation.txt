      (arithmetic_command

          (binary_expression
            (string
              (simple_expansion PLACEHOLDER_2)) =~
            (simple_expansion PLACEHOLDER_1))))) then
    (command echo false)
    (else_clause
      (arithmetic_command
        (binary_expression PLACEHOLDER_4 =
          (binary_expression
            (expansion # PLACEHOLDER_2) % 2)))

        (variable_assignment PLACEHOLDER_5 0) ;
        (binary_expression PLACEHOLDER_5 <
          (expansion # PLACEHOLDER_2)) ;

          (list

              (binary_expression PLACEHOLDER_3 +=

                (binary_expression PLACEHOLDER_3 += PLACEHOLDER_6))))))
      (list
        (list
