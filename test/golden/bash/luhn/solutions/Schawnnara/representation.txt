(file "luhn.sh")
luhn_check () {
[[ $1 = ~ ^[[:digit:]] { 2, } $ ]] || return 1
declare -i sum = 0 digit is_double
for (( is_double = 0, i = ${# 1 } -1 ; i > -1 ; i--, is_double^ = 1, sum + = digit )) {
digit = "\"" ${ 1 :i:1 } "\""
(( is_double )) && {
(( digit << = 1 ))
(( digit > 9 )) && (( digit - = 9 ))
}
}
(( sum % 10 == 0 ))
}
luhn_check "\"" ${ *// / } "\"" && echo true || echo false
