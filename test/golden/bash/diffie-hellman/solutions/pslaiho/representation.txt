rand () {
echo "\"" $(( "\"" $1 "\"" + RANDOM % ( "\"" $2 "\"" - "\"" $1 "\"" + 1 ) )) "\""
}
calculate_key () {
echo "\"" $1 ^ $2 % $3 "\"" | bc
}
main () {
case "\"" $1 "\"" in
"\"privateKey\"" )
rand 2 $(( "\"" $2 "\"" - 1 )) ;;
"\"publicKey\"" )
calculate_key "\"" $3 "\"" "\"" $4 "\"" "\"" $2 "\"" ;;
"\"secret\"" )
calculate_key "\"" $3 "\"" "\"" $4 "\"" "\"" $2 "\"" ;;
esac
}
main "\"" $@ "\""
