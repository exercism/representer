(program
  (while_statement
    (command getopts w:b: PLACEHOLDER_1)
    (do_group
      (case_statement
        (string
          (simple_expansion PLACEHOLDER_1))
        (case_item w
          (command
            (variable_assignment IFS ,) read -r PLACEHOLDER_2 PLACEHOLDER_3
            (herestring_redirect
              (simple_expansion OPTARG))))
        (case_item b
          (command
            (variable_assignment IFS ,) read -r PLACEHOLDER_4 PLACEHOLDER_5
            (herestring_redirect
              (simple_expansion OPTARG))))
        (case_item *
          (command echo invalid option)
          (command exit 1)))))
  (list
    (arithmetic_command
      (binary_expression
        (binary_expression PLACEHOLDER_2 < 0) ||
        (binary_expression PLACEHOLDER_4 < 0))) &&
    (compound_statement
      (command echo row not positive)
      (command exit 1)))
  (list
    (arithmetic_command
      (binary_expression
        (binary_expression PLACEHOLDER_3 < 0) ||
        (binary_expression PLACEHOLDER_5 < 0))) &&
    (compound_statement
      (command echo column not positive)
      (command exit 1)))
  (list
    (arithmetic_command
      (binary_expression
        (binary_expression PLACEHOLDER_2 > 7) ||
        (binary_expression PLACEHOLDER_4 > 7))) &&
    (compound_statement
      (command echo row not on board)
      (command exit 1)))
  (list
    (arithmetic_command
      (binary_expression
        (binary_expression PLACEHOLDER_3 > 7) ||
        (binary_expression PLACEHOLDER_5 > 7))) &&
    (compound_statement
      (command echo column not on board)
      (command exit 1)))
  (list
    (arithmetic_command
      (binary_expression
        (binary_expression PLACEHOLDER_2 == PLACEHOLDER_4) &&
        (binary_expression PLACEHOLDER_3 == PLACEHOLDER_5))) &&
    (compound_statement
      (command echo same position)
      (command exit 1)))
  (variable_assignment PLACEHOLDER_6
    (arithmetic_expansion
      (binary_expression PLACEHOLDER_2 - PLACEHOLDER_4)))
  (variable_assignment PLACEHOLDER_7
    (arithmetic_expansion
      (binary_expression PLACEHOLDER_3 - PLACEHOLDER_5)))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression
            (binary_expression PLACEHOLDER_2 == PLACEHOLDER_4) ||
            (binary_expression PLACEHOLDER_3 == PLACEHOLDER_5)) ||
          (binary_expression
            (expansion PLACEHOLDER_6 # -) ==
            (expansion PLACEHOLDER_7 # -)))) &&
      (command echo true)) ||
    (command echo false)))
