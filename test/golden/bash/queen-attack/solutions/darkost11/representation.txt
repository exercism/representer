(program
  (variable_assignment IFS ",")
  (command read -r -a PLACEHOLDER_1
    (herestring_redirect
      (string
        (simple_expansion 2))))
  (command read -r -a PLACEHOLDER_2
    (herestring_redirect
      (string
        (simple_expansion 4))))
  (variable_assignment PLACEHOLDER_3
    (string
      (expansion
        (subscript PLACEHOLDER_1 0))))
  (variable_assignment PLACEHOLDER_4
    (string
      (expansion
        (subscript PLACEHOLDER_1 1))))
  (variable_assignment PLACEHOLDER_5
    (string
      (expansion
        (subscript PLACEHOLDER_2 0))))
  (variable_assignment PLACEHOLDER_6
    (string
      (expansion
        (subscript PLACEHOLDER_2 1))))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression PLACEHOLDER_3 < 0) ||
          (binary_expression PLACEHOLDER_5 < 0))) &&
      (command echo "row not positive")) &&
    (command exit 1))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression PLACEHOLDER_4 < 0) ||
          (binary_expression PLACEHOLDER_6 < 0))) &&
      (command echo "column not positive")) &&
    (command exit 1))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression PLACEHOLDER_3 > 7) ||
          (binary_expression PLACEHOLDER_5 > 7))) &&
      (command echo "row not on board")) &&
    (command exit 1))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression PLACEHOLDER_4 > 7) ||
          (binary_expression PLACEHOLDER_6 > 7))) &&
      (command echo "column not on board")) &&
    (command exit 1))
  (list
    (list
      (arithmetic_command
        (binary_expression
          (binary_expression PLACEHOLDER_3 == PLACEHOLDER_5) &&
          (binary_expression PLACEHOLDER_4 == PLACEHOLDER_6))) &&
      (command echo "same position")) &&
    (command exit 1))
  (if_statement
    (arithmetic_command
      (binary_expression
        (binary_expression
          (binary_expression
            (binary_expression PLACEHOLDER_3 == PLACEHOLDER_5) ||
            (binary_expression PLACEHOLDER_4 == PLACEHOLDER_6)) ||
          (binary_expression
            (binary_expression PLACEHOLDER_3 - PLACEHOLDER_5) ==
            (binary_expression PLACEHOLDER_4 - PLACEHOLDER_6))) ||
        (binary_expression
          (binary_expression PLACEHOLDER_3 - PLACEHOLDER_5) ==
          (binary_expression PLACEHOLDER_6 - PLACEHOLDER_4)))) then
    (command echo true)
    (else_clause
      (command echo false))))
