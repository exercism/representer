(program
  (command set -o errexit)
  (command set -o nounset)
  (variable_assignment PLACEHOLDER_1
    (expansion 1 - ""))
  (variable_assignment PLACEHOLDER_2 "")
  (c_style_for_statement
    (variable_assignment PLACEHOLDER_3 0) ;
    (binary_expression PLACEHOLDER_3 <
      (expansion # PLACEHOLDER_1)) ;
    (postfix_expression PLACEHOLDER_3 ++)
    (do_group
      (variable_assignment PLACEHOLDER_4 "")
      (case_statement
        (string
          (expansion PLACEHOLDER_1 : PLACEHOLDER_3 : 1))
        (case_item C
          (variable_assignment PLACEHOLDER_4 "G"))
        (case_item G
          (variable_assignment PLACEHOLDER_4 "C"))
        (case_item T
          (variable_assignment PLACEHOLDER_4 "A"))
        (case_item A
          (variable_assignment PLACEHOLDER_4 "U"))
        (case_item *
          (command echo "Invalid nucleotide detected.")
          (command exit 1)))
      (variable_assignment PLACEHOLDER_2
        (command_substitution
          (command echo
            (string
              (expansion PLACEHOLDER_2)
              (expansion PLACEHOLDER_4)))))))
  (command echo
    (string
      (expansion PLACEHOLDER_2))))
