if [ "\"$#\"" -ne 1 ] ; then
exit 1
fi
num = $1
out = "\"\""
if ! [[ "\"" $num "\"" = ~ ^[0-9]+ $ ]] ; then
echo "\"Argument must be an int\""
exit 1
fi
if [[ $(( "\"" $num "\"" % 3 )) == 0 ]] ; then
out = "\"" ${ out } Pling"
fi
if [[ $(( "\"" $num "\"" % 5 )) == 0 ]] ; then
out = "\"" ${ out } Plang"
fi
if [[ $(( "\"" $num "\"" % 7 )) == 0 ]] ; then
out = "\"" ${ out } Plong"
fi
if [ "\"" $out "\"" != "\"\"" ] ; then
echo "\"" $out "\""
else
echo "\"" $num "\""
fi
