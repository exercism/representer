(file "raindrops.sh")
set -o errexit
set -o nounset
usage () {
echo "\"Usage: ./raindrops <number>\""
exit 1
}
declare -A m =(
[ 3] = Pling
[ 5] = Plang
[ 7] = Plong
)
PlingPlangPlong () {
output = "\"\""
for k in 3 5 7 ;
do
if (( "\"" $1 "\"" % $k == 0 )) ; then
output+ = "\"" ${ m [ $k ] } "\""
fi
done
echo "\"" ${ output :- $1 } "\""
}
if (( $# != 1 )) ; then
usage
fi
if [[ "\"" $1 "\"" = ~ [ ^[:digit:]] ]] ; then
usage
fi
PlingPlangPlong "\"" $@ "\""
