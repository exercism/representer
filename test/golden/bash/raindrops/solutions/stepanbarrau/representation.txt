set -o errexit
set -o nounset
main () {
local sound = "\"\""
if [[ $(( $1 % 3 )) -eq 0 ]] ; then
sound = "\"" ${ sound } Pling"
fi
if [[ $(( $1 % 5 )) -eq 0 ]] ; then
sound = "\"" ${ sound } Plang"
fi
if [[ $(( $1 % 7 )) -eq 0 ]] ; then
sound = "\"" ${ sound } Plong"
fi
if [[ -z "\"" $sound "\"" ]] ; then
echo "\"" $1 "\""
else
echo "\"" $sound "\""
fi
}
main "\"" $@ "\""
