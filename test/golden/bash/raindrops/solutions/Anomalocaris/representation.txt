divisibleBy () {
(( $1 % $2 == 0 ))
}
number = $1
answer =
if divisibleBy "\"" $number "\"" 3 ; then
answer+ = "\"Pling\""
fi
if divisibleBy "\"" $number "\"" 5 ; then
answer+ = "\"Plang\""
fi
if divisibleBy "\"" $number "\"" 7 ; then
answer+ = "\"Plong\""
fi
if [[ "\"" $answer "\"" == "\"\"" ]] ; then
answer = "\"" $number "\""
fi
echo "\"" $answer "\""
