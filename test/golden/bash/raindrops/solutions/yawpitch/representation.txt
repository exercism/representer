(file "raindrops.sh")
set -o errexit
set -o nounset
PROGNAME = $( basename "\"" $0 "\")"
error_exit ()
{
printf "'Usage: %s <integer>\\n'" "\"" ${ PROGNAME } "\"" 1>&2
exit 1
}
pling_plang_plong ()
{
(( $1 % 3 )) || result+ = "\"Pling\""
(( $1 % 5 )) || result+ = "\"Plang\""
(( $1 % 7 )) || result+ = "\"Plong\""
printf '%s\n' "\"" ${ result :- $1 } "\""
}
main () {
if [ "\"$#\"" -eq 1 ] && [ ! -z "\"" $1 "\"" ] && [[ "\"" $1 "\"" = ~ ^[0-9] * $ ]] && [ "\"" $1 "\"" -eq "\"" $1 "\"" ]
then
pling_plang_plong "\"" $1 "\""
else
error_exit
fi
}
main "\"" $@ "\""
