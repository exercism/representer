set -o errexit
set -o nounset
main () {
input = "\"" $1 "\""
result = "\"\""
if (( "\"" $input "\"" % 3 == 0 )) ; then
result = "\"Pling\""
fi
if (( "\"" $input "\"" % 5 == 0 )) ; then
result+ = "\"Plang\""
fi
if (( "\"" $input "\"" % 7 == 0 )) ; then
result+ = "\"Plong\""
fi
if [[ "\"" $result "\"" == "\"\"" ]] ; then
result = "\"" $input "\""
fi
echo "\"" $result "\""
}
main "\"" $@ "\""
