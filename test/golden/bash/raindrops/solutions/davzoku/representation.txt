INPUT = $1
if (( $INPUT % 105 == 0 ))
then echo "\"PlingPlangPlong\""
elif (( $INPUT % 15 == 0 ))
then echo "\"PlingPlang\""
elif (( $INPUT % 21 == 0 ))
then echo "\"PlingPlong\""
elif (( $INPUT % 35 == 0 ))
then echo "\"PlangPlong\""
elif (( $INPUT % 3 == 0 ))
then echo "\"Pling\""
elif (( $INPUT % 5 == 0 ))
then echo "\"Plang\""
elif (( $INPUT % 7 == 0 ))
then echo "\"Plong\""
else
echo "\"" $INPUT "\""
fi
