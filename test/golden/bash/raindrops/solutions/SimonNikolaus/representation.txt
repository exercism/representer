set -o errexit
set -o pipefail
set -o nounset
function main () {
declare -i argNumber = "\"" ${ 1 } "\""
local rpSpeak = "\"\""
[[ $(( argNumber % 3 )) == 0 ]] && rpSpeak = "\"Pling\""
[[ $(( argNumber % 5 )) == 0 ]] && rpSpeak = "\"" ${ rpSpeak } Plang"
[[ $(( argNumber % 7 )) == 0 ]] && rpSpeak = "\"" ${ rpSpeak } Plong"
[[ -z ${ rpSpeak } ]] && rpSpeak = ${ argNumber }
echo "\"" ${ rpSpeak } "\""
}
main "\"" ${ @ } "\""
