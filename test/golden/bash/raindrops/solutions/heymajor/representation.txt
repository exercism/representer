(file "raindrops.sh")
IFS = "' '"
num = $1
factors = $( factor "\"" $num "\")"
factors = ${ factors ## $1 : }
output = "\"\""
for factor in $factors ; do
if [[ "\"" $factor "\"" == 3 ]] ; then
output = "\"" ${ output } Pling"
elif [[ "\"" $factor "\"" == 5 ]] ; then
output = "\"" ${ output } Plang"
elif [[ "\"" $factor "\"" == 7 ]] ; then
output = "\"" ${ output } Plong"
else
output = "\"" ${ output } "\""
fi
done
if [[ "\"" $output "\"" = * Pling * ]] ; then
while test "\"" $output "\"" != "\"" ${ output //PlingPling/Pling } "\"" ; do
output = "\"" ${ output //PlingPling/Pling } "\""
done
fi
if [[ "\"" $output "\"" = * Plang * ]] ; then
while test "\"" $output "\"" != "\"" ${ output //PlangPlang/Plang } "\"" ; do
output = "\"" ${ output //PlangPlang/Plang } "\""
done
fi
if [[ "\"" $output "\"" = * Plong * ]] ; then
while test "\"" $output "\"" != "\"" ${ output //PlongPlong/Plong } "\"" ; do
output = "\"" ${ output //PlongPlong/Plong } "\""
done
fi
if [[ "\"" $output "\"" == "\"\"" ]] ; then
echo "\"" $1 "\""
else
echo $output
fi
