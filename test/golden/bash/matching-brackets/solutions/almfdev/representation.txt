set -euo pipefail
set -f
main () {
local input = "\"" ${ 1 :-} "\""
local len = "\"" ${# input } "\""
local i char
local stack =()
local top = 0
for (( i = 0 ; i<len ; i++ )) ; do
char = "\"" ${ input :i:1 } "\""
case "\"" $char "\"" in
'(' )
stack[top++] = "\")\""
;;
'[' )
stack[top++] = "\"]\""
;;
'{' )
stack[top++] = "\"}\""
;;
')' | ']' | '}' )
if (( top == 0 )) ; then
echo "\"false\""
return 0
fi
(( top-- ))
if [[ "\"" ${ stack [top] } "\"" != "\"" $char "\"" ]] ; then
echo "\"false\""
return 0
fi
;;
* )
;;
esac
done
if (( top == 0 )) ; then
echo "\"true\""
else
echo "\"false\""
fi
}
main "\"" $@ "\""
