(program
  (variable_assignment PLACEHOLDER_1
    (array 10 5 1))
  (variable_assignment PLACEHOLDER_2
    (array 1 25 100))
  (if_statement
    (test_command [
      (binary_expression
        (string
          (simple_expansion #)) -ne "2")) then
    (command echo "Bad arg count")
    (command exit 1))
  (variable_assignment PLACEHOLDER_3 "^-?[0-9]+([.][0-9]+)?$")
  (if_statement
    (list
      (negated_command
        (test_command [[
          (binary_expression
            (string
              (simple_expansion 1)) =~
            (simple_expansion PLACEHOLDER_3)))) ||
      (negated_command
        (test_command [[
          (binary_expression
            (string
              (simple_expansion 2)) =~
            (simple_expansion PLACEHOLDER_3))))) then
    (command echo "Malformed argument")
    (command exit 1))
  (variable_assignment PLACEHOLDER_4
    (string "("
      (simple_expansion 1) ")"))
  (variable_assignment PLACEHOLDER_5
    (string "("
      (simple_expansion 2) ")"))
  (variable_assignment PLACEHOLDER_6
    (string
      (command_substitution
        (command bc
          (herestring_redirect
            (string "scale=4; "
              (simple_expansion PLACEHOLDER_4) " * "
              (simple_expansion PLACEHOLDER_4) " + "
              (simple_expansion PLACEHOLDER_5) " * "
              (simple_expansion PLACEHOLDER_5)))))))
  (for_statement PLACEHOLDER_7 in
    (brace_expression 0 2)
    (do_group
      (if_statement
        (test_command [[
          (binary_expression "1" ==
            (string
              (command_substitution
                (command bc
                  (herestring_redirect
                    (string
                      (expansion
                        (subscript PLACEHOLDER_2
                          (simple_expansion PLACEHOLDER_7))) " >= "
                      (simple_expansion PLACEHOLDER_6)))))))) then
        (command echo
          (string
            (expansion
              (subscript PLACEHOLDER_1
                (simple_expansion PLACEHOLDER_7)))))
        (command exit 0))))
  (command echo "0")
  (command exit 0))
