declare -i n = "\"" $1 "\""
if (( n<0 )) ; then
(( n * = -1 ))
suffix = "\"i\""
fi
xk = "\"" $n "\""
prev = $(( n+1 ))
while (( xk<prev )) ; do
prev = "\"" $xk "\""
(( xk =( xk+n/xk ) /2 ))
done
if (( xk * xk == n )) ; then
printf "\"%d%s\\n\"" "\"" $xk "\"" "\"" $suffix "\""
else
printf "\"%d is not a square\\n\"" "\"" $1 "\"" > &2
exit 1
fi
