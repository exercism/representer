error_exit () {
echo "\"Invalid number.  [1]NXX-NXX-XXXX N=2-9, X=0-9\"" > &2
exit 1
}
main () {
local input = "\"" $1 "\""
local cleaned = "\"" ${ input //[ -\.\(\)\+]/ } "\""
if [[ "\"" $cleaned "\"" = ~ [ ^0-9] ]] ; then
error_exit
fi
local len = ${# cleaned }
if (( len == 11 )) ; then
if [[ "\"" ${ cleaned :0:1 } "\"" != "\"1\"" ]] ; then
error_exit
fi
cleaned = "\"" ${ cleaned :1 } "\""
len = 10
fi
if (( len != 10 )) ; then
error_exit
fi
if [[ ! "\"" ${ cleaned :0:1 } "\"" = ~ [ 2-9] ]] || [[ ! "\"" ${ cleaned :3:1 } "\"" = ~ [ 2-9] ]] ; then
error_exit
fi
echo "\"" $cleaned "\""
}
main "\"" $@ "\""
