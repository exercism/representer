(file "yacht.sh")
count () {
local x = "\"" $1 "\""
local cnt = 0
for (( i = 0 ; i < 5 ; i++ )) ; do
if (( "\"" ${ die [ $i ] } "\"" == x )) ; then
cnt = $(( cnt + 1 ))
fi
done
echo "\"" $cnt "\""
}
sum () {
local sum = 0
for (( i = 0 ; i < 5 ; i++ )) ; do
sum = $(( sum + ${ die [ $i ] }))
done
echo "\"" $sum "\""
}
yacht () {
if (( $( count "\"" ${ die [0] } "\")" == 5 )) ; then
echo 50
else
echo 0
fi
}
straight () {
local little = "\"" $1 "\""
local flag = false
if [[ "\"" $little "\"" == "\"true\"" && $( count 1 ) == 1 ]] ; then
flag = true
elif [[ "\"" $little "\"" == "\"false\"" && $( count 6 ) == 1 ]] ; then
flag = true
fi
if [[ $( count 2 ) == 1 && $( count 3 ) == 1 && $( count 4 ) == 1 && $( count 5 ) == 1 && "\"" $flag "\"" == "\"true\"" ]] ; then
echo 30
else
echo 0
fi
}
fourOfAKind () {
for dice in "\"" ${ die [@] } "\"" ; do
if (( $( count dice ) >= 4 )) ; then
echo "\"" $(( dice * 4 )) "\""
return
fi
done
echo 0
}
fullHouse () {
local pair = false
local trips = false
for dice in "\"" ${ die [@] } "\"" ; do
if (( $( count dice ) == 2 )) ; then
pair = true
elif (( $( count dice ) == 3 )) ; then
trips = true
fi
done
if $pair && $trips ; then
sum
else
echo 0
fi
}
score () {
case "\"" $category "\"" in
ones ) echo "\"" $(( 1 * $( count 1 ) )) "\"" ;;
twos ) echo "\"" $(( 2 * $( count 2 ) )) "\"" ;;
threes ) echo "\"" $(( 3 * $( count 3 ) )) "\"" ;;
fours ) echo "\"" $(( 4 * $( count 4 ) )) "\"" ;;
fives ) echo "\"" $(( 5 * $( count 5 ) )) "\"" ;;
sixes ) echo "\"" $(( 6 * $( count 6 ) )) "\"" ;;
choice ) sum ;;
yacht ) yacht ;;
"\"full house\"" ) fullHouse ;;
"\"four of a kind\"" ) fourOfAKind ;;
"\"little straight\"" ) straight "\"true\"" ;;
"\"big straight\"" ) straight "\"false\"" ;;
* ) echo 0 ;;
esac
}
category = "\"" $1 "\""
shift
die =( "\"" $@ "\"" )
score
