(file "beer_song.sh")
if (( $# < 1 || $# > 2 )) ; then
printf "\"Error: 1 or 2 arguments expected.\\\\n\""
exit 1
fi
first = $1
last = ${ 2 :- $first }
if (( first < last )) ; then
printf "\"Start must be greater than End\\\\n\""
exit 1
fi
for (( i = first ; i> = last ; i-- )) ; do
if [[ "\"" $i "\"" == 0 ]] ; then
printf "\"%s\\\\n%s\\\\n\"" \
"\"No more bottles of beer on the wall, no more bottles of beer.\"\\"
"\"Go to the store and buy some more, 99 bottles of beer on the wall.\""
continue
fi
how_many_on_the_wall = "\"" $i " bottles\""
how_many_down = "\"one\""
remaining_on_the_wall = "\"" $(( i - 1 )) " bottles\""
if [[ "\"" $i "\"" == 1 ]] ; then
how_many_on_the_wall = "\"1 bottle\""
how_many_down = "\"it\""
remaining_on_the_wall = "\"no more bottles\""
elif [[ "\"" $i "\"" == 2 ]] ; then
remaining_on_the_wall = "\"1 bottle\""
fi
printf "\"%s of beer on the wall, %s of beer.\\\\n\"\\"
"\"" $how_many_on_the_wall "\"" \
"\"" $how_many_on_the_wall "\""
printf "\"Take %s down and pass it around, %s of beer on the wall.\\\\n\"\\"
"\"" $how_many_down "\"\\"
"\"" $remaining_on_the_wall "\""
printf "\"\\\\n\""
done
