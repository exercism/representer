main () {
local phrase = "\"" ${ 1 ,, } "\""
local letters = "\"" ${ phrase //[^a-z]/ } "\""
local seen = "\"\""
local len = ${# letters }
for (( i = 0 ; i<len ; i++ )) ; do
local char = "\"" ${ letters :i:1 } "\""
if [[ "\"" $seen "\"" == * "\"" $char "\"" * ]] ; then
echo "\"false\""
return 0
fi
seen+ = "\"" $char "\""
done
echo "\"true\""
}
if [[ "\"" ${ BASH_SOURCE [0] } "\"" == "\"" ${ 0 } "\"" ]] ; then
main "\"" $@ "\""
fi
