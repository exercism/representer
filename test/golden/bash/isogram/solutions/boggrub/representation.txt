main () {
declare -A chars
string = "\"" ${ 1 //[^[ :alpha:]] } "\""
string = "\"" ${ string ,, } "\""
for (( i = 0 ; i < ${# string } ; i++ ))
do
char = ${ string :i:1 }
if [ -n "\"" ${ chars [ ${ char } ] } "\"" ] ; then
echo "\"false\""
exit 0
else
chars[ "\"" ${ char } "\"" ]= 1
fi
done
echo "\"true\""
}
main "\"" $@ "\""
