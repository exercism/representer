(file "isogram.sh")
declare -A dict
declare input = "\"" ${ 1 ,, } "\""
input = "\"" ${ input // / } "\""
input = "\"" ${ input //-/ } "\""
[[ "\"" $input "\"" = ~ ${ input //?/(.) } ]]
for ch in "\"" ${ BASH_REMATCH [@] :1 } "\"" ; do
if (( ${ dict [ $ch ] : =0 } )) ; then
echo "\"false\""
exit
fi
(( dict[ $ch ] ++ ))
done
echo true
