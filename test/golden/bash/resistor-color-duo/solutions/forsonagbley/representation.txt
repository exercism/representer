color () {
case "\"" ${ 1 ,, } "\"" in
black ) echo 0 ;;
brown ) echo 1 ;;
red ) echo 2 ;;
orange ) echo 3 ;;
yellow ) echo 4 ;;
green ) echo 5 ;;
blue ) echo 6 ;;
violet ) echo 7 ;;
grey ) echo 8 ;;
white ) echo 9 ;;
* ) exit 1 ;;
esac
}
main () {
str = "\"\""
for item in "\"" ${ @ :1:2 } "\"" ; do
str = "\"" ${ str } $( color "\"" $item "\")\"" || { echo "\"invalid color\"" ; exit 1 ; }
done
echo $(( 10
}
main "\"" $@ "\""
