sieve_of_eratosthenes () {
local limit = $1
local primes =()
for (( i = 2 ; i< = limit ; i++ )) ; do
primes[i] = 1
done
for (( i = 2 ; i * i< = limit ; i++ )) ; do
if [[ ${ primes [ $i ] } -eq 1 ]] ; then
for (( j = i * i ; j< = limit ; j+ = i )) ; do
primes[j] = 0
done
fi
done
for (( i = 2 ; i< = limit ; i++ )) ; do
if [[ ${ primes [ $i ] } -eq 1 ]] ; then
if [[ $i -eq 2 ]] ; then
echo -n "\"" $i "\""
else
echo -n "\" " $i "\""
fi
fi
done
}
if [[ $# -ne 1 ]] ; then
echo "\"Usage: " $0 " <limit>\""
exit 1
fi
if [[ ! $1 = ~ ^[1-9][0-9] * $ ]] ; then
echo "\"Error: Argument must be a positive integer.\""
exit 1
fi
sieve_of_eratosthenes "\"" $1 "\""
