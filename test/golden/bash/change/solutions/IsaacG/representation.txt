(file "change.sh")
declare -a count =( 0 ) using =( 0 )
target = $1 ; shift ; coins =( $@ )
die () { echo "\"" $@ "\"" ; exit 1 ; }
compute () {
for (( i = 1 ; i < = target ; i++ )) ; do
min = 0
count[i] = using[i] =
for n in "\"" ${ coins [@] } "\"" ; do
(( n > i )) && continue
[[ ${ count [i-n] } ]] || continue
maybe = $(( ${ count [i-n] } + 1 ))
if (( ! min )) || (( maybe < min )) ; then
min = $maybe
count[i] = $maybe
using[i] = $n
fi
done
done
}
show () {
[[ -z ${ count [target] } ]] && die "\"can't make target with given coins\""
used =()
while (( target > 0 )) ; do
used+ =( "\"" ${ using [target] } "\"" )
(( target - = "\"" ${ using [target] } "\"" ))
done
sorted =()
for n in "\"" ${ coins [@] } "\"" ; do
for u in "\"" ${ used [@] } "\"" ; do
(( n == u )) && sorted+ =( $n )
done
done
echo "\"" ${ sorted [@] } "\""
}
main () {
(( target < 0 )) && die "\"target can't be negative\""
compute
show
}
main "\"" $@ "\""
