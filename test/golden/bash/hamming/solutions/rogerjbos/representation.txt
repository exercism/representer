(program
  (function_definition (function_name PLACEHOLDER_1)
    (compound_statement
      (if_statement
        (test_command [
          (binary_expression
            (simple_expansion #) -ne 2)) then
        (command echo "Usage: hamming.sh <string1> <string2>")
        (command exit 1))
      (if_statement
        (test_command [[
          (binary_expression
            (expansion # 1) -eq
            (expansion # 2))) then
        (variable_assignment PLACEHOLDER_2 0)
        (c_style_for_statement
          (variable_assignment PLACEHOLDER_3 0) ;
          (binary_expression PLACEHOLDER_3 <
            (expansion # 1)) ;
          (postfix_expression PLACEHOLDER_3 ++)
          (do_group
            (list
              (test_command [[
                (binary_expression
                  (string
                    (expansion 1 :
                      (simple_expansion PLACEHOLDER_3) : 1)) !=
                  (string
                    (expansion 2 :
                      (simple_expansion PLACEHOLDER_3) : 1)))) &&
              (arithmetic_command
                (binary_expression PLACEHOLDER_2 =
                  (binary_expression PLACEHOLDER_2 + 1))))))
        (command echo
          (simple_expansion PLACEHOLDER_2))
        (else_clause
          (command echo "left and right strands must be of equal length")
          (command exit 1)))))
  (command (command_name PLACEHOLDER_1)
    (string
      (simple_expansion @))))
