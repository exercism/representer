(program
  (function_definition (function_name PLACEHOLDER_1)
    (compound_statement
      (variable_assignment PLACEHOLDER_2 0)
      (variable_assignment PLACEHOLDER_3
        (expansion # 1))
      (variable_assignment PLACEHOLDER_4
        (expansion # 2))
      (if_statement
        (test_command [
          (binary_expression
            (simple_expansion PLACEHOLDER_3) -ne
            (simple_expansion PLACEHOLDER_4))) then
        (command echo "strands must be of equal length")
        (command exit 2))
      (c_style_for_statement
        (variable_assignment PLACEHOLDER_5 0) ;
        (binary_expression PLACEHOLDER_5 <
          (simple_expansion PLACEHOLDER_3)) ;
        (postfix_expression PLACEHOLDER_5 ++)
        (do_group
          (if_statement
            (test_command [
              (binary_expression
                (expansion 1 : PLACEHOLDER_5 : 1) !=
                (expansion 2 : PLACEHOLDER_5 : 1))) then
            (variable_assignment PLACEHOLDER_2
              (arithmetic_expansion
                (binary_expression PLACEHOLDER_2 + 1))))))
      (command echo
        (string
          (simple_expansion PLACEHOLDER_2)))))
  (if_statement
    (test_command [
      (binary_expression
        (simple_expansion #) != 2)) then
    (command echo "Usage: hamming.sh <string1> <string2>")
    (command exit 1)
    (else_clause
      (command (command_name PLACEHOLDER_1)
        (string
          (simple_expansion @))))))
