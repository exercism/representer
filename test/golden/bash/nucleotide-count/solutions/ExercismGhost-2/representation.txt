(program
  (variable_assignment PLACEHOLDER_1
    (simple_expansion 1))
  (variable_assignment PLACEHOLDER_2 0)
  (variable_assignment PLACEHOLDER_3 0)
  (variable_assignment PLACEHOLDER_4 0)
  (variable_assignment PLACEHOLDER_5 0)
  (list
    (test_command [[
      (binary_expression
        (simple_expansion PLACEHOLDER_1) =~ [^ACGT])) &&
    (compound_statement
      (command echo "Invalid nucleotide in strand")
      (command exit 1)))
  (c_style_for_statement
    (variable_assignment PLACEHOLDER_6 0) ;
    (binary_expression PLACEHOLDER_6 <=
      (expansion # PLACEHOLDER_1)) ;
    (postfix_expression PLACEHOLDER_6 ++)
    (do_group
      (case_statement
        (expansion PLACEHOLDER_1 : PLACEHOLDER_6 : 1)
        (case_item a A
          (arithmetic_command
            (unary_expression ++ PLACEHOLDER_2)))
        (case_item c C
          (arithmetic_command
            (unary_expression ++ PLACEHOLDER_3)))
        (case_item g G
          (arithmetic_command
            (unary_expression ++ PLACEHOLDER_4)))
        (case_item t T
          (arithmetic_command
            (unary_expression ++ PLACEHOLDER_5))))))
  (command printf "%s\\n%s\\n%s\\n%s\\n"
    (string "A: "
      (expansion PLACEHOLDER_2))
    (string "C: "
      (expansion PLACEHOLDER_3))
    (string "G: "
      (expansion PLACEHOLDER_4))
    (string "T: "
      (expansion PLACEHOLDER_5))))
