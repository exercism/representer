(source_file
  (unit_declaration unit module BinarySearch)
  (routine_declaration sub binary-search
    (signature "("
      (parameter_list
        (parameter :@array) ,
        (parameter :$value)) ")")
    (trait is export)
    (block {
      (assignment
        (variable_declaration
          (scope my)
          (type Int) (local_scalar PLACEHOLDER_1)) = 0)
      (assignment
        (variable_declaration
          (scope my)
          (type Int) (local_scalar PLACEHOLDER_2)) =
        (binary_expression
          (method_call @array . elems) - 1))
      (while_statement while
        (binary_expression (scalar PLACEHOLDER_1) <= (scalar PLACEHOLDER_2))
        (block {
          (assignment
            (variable_declaration
              (scope my)
              (type Int) (local_scalar PLACEHOLDER_3)) =
            (binary_expression
              (binary_expression (scalar PLACEHOLDER_1) + (scalar PLACEHOLDER_2)) div 2))
          (if_statement if
            (binary_expression
              (subscript @array [ (scalar PLACEHOLDER_3) ]) == $value)
            (block {
              (listop_call return (scalar PLACEHOLDER_3)) })
            (elsif_clause elsif
              (binary_expression
                (subscript @array [ (scalar PLACEHOLDER_3) ]) < $value)
              (block {
                (assignment (scalar PLACEHOLDER_1) =
                  (binary_expression (scalar PLACEHOLDER_3) + 1)) }))
            (else_clause else
              (block {
                (assignment (scalar PLACEHOLDER_2) =
                  (binary_expression (scalar PLACEHOLDER_3) - 1)) }))) }))
      (listop_call die "value not in array") })))
