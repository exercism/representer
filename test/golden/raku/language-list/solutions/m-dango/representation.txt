(source_file
  (unit_declaration unit module LanguageList)
  (routine_declaration sub add-language
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1)) ,
        (parameter (local_scalar PLACEHOLDER_2))) ")")
    (trait is export)
    (block {
      (listop_call return
        (list_expression
          (unary_expression | (array PLACEHOLDER_1)) , (scalar PLACEHOLDER_2))) }))
  (routine_declaration sub remove-language
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (listop_call return
        (method_call (array PLACEHOLDER_1) . head
          (argument_list "("
            (binary_expression
              (whatever *) - 1) ")"))) }))
  (routine_declaration sub first-language
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (listop_call return
        (method_call (array PLACEHOLDER_1) . head)) }))
  (routine_declaration sub last-language
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (listop_call return
        (method_call (array PLACEHOLDER_1) . tail)) }))
  (routine_declaration sub get-languages
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1)) ,
        (parameter (local_array PLACEHOLDER_3))) ")")
    (trait is export)
    (block {
      (listop_call return
        (subscript (array PLACEHOLDER_1) [
          (method_call (array PLACEHOLDER_3) . map
            (argument_list "("
              (binary_expression
                (whatever *) - 1) ")")) ])) }))
  (routine_declaration sub has-language
    (signature "("
      (parameter_list
        (parameter (local_array PLACEHOLDER_1)) ,
        (parameter (local_scalar PLACEHOLDER_2))) ")")
    (trait is export)
    (block {
      (binary_expression
        (method_call (array PLACEHOLDER_1) . any) eq (scalar PLACEHOLDER_2)) })))
