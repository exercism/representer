(source_file
  (unit_declaration unit module ProteinTranslation)
  (constant_declaration constant %CODON-MAP =
    (list_expression
      (assignment "AUG" => "Methionine") ,
      (assignment "UUU" => "Phenylalanine") ,
      (assignment "UUC" => "Phenylalanine") ,
      (assignment "UUA" => "Leucine") ,
      (assignment "UUG" => "Leucine") ,
      (assignment "UCU" => "Serine") ,
      (assignment "UCC" => "Serine") ,
      (assignment "UCA" => "Serine") ,
      (assignment "UCG" => "Serine") ,
      (assignment "UAU" => "Tyrosine") ,
      (assignment "UAC" => "Tyrosine") ,
      (assignment "UGU" => "Cysteine") ,
      (assignment "UGC" => "Cysteine") ,
      (assignment "UGG" => "Tryptophan") ,
      (assignment "UAA" => "STOP") ,
      (assignment "UAG" => "STOP") ,
      (assignment "UGA" => "STOP") ,))
  (routine_declaration sub protein-translation
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (variable_declaration
        (scope my) (local_array PLACEHOLDER_2))
      (assignment
        (variable_declaration
          (scope my) (local_scalar PLACEHOLDER_3)) = 0)
      (assignment
        (variable_declaration
          (scope my) (local_scalar PLACEHOLDER_4)) =
        (method_call (scalar PLACEHOLDER_1) . chars))
      (while_statement while
        (binary_expression (scalar PLACEHOLDER_3) < (scalar PLACEHOLDER_4))
        (block {
          (if_statement if
            (binary_expression
              (binary_expression (scalar PLACEHOLDER_3) + 3) > (scalar PLACEHOLDER_4))
            (block {
              (listop_call die "Incomplete RNA sequence") }))
          (assignment
            (variable_declaration
              (scope my) (local_scalar PLACEHOLDER_5)) =
            (method_call (scalar PLACEHOLDER_1) . substr
              (argument_list "("
                (list_expression (scalar PLACEHOLDER_3) , 3) ")")))
          (unless_statement unless
            (subscript %CODON-MAP { (scalar PLACEHOLDER_5) } :exists)
            (block {
              (listop_call die
                (interpolated_string "Invalid codon: "
                  (interpolation (scalar PLACEHOLDER_5)))) }))
          (assignment
            (variable_declaration
              (scope my) (local_scalar PLACEHOLDER_6)) =
            (subscript %CODON-MAP { (scalar PLACEHOLDER_5) }))
          (if_statement if
            (binary_expression (scalar PLACEHOLDER_6) eq "STOP")
            (block { last }))
          (method_call (array PLACEHOLDER_2) . push
            (argument_list "(" (scalar PLACEHOLDER_6) ")"))
          (assignment (scalar PLACEHOLDER_3) 3) }))
      (method_call (array PLACEHOLDER_2) . List) })))
