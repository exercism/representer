(source_file
  (unit_declaration unit module ProteinTranslation)
  (routine_declaration sub protein-translation
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (statement_prefix gather
        (for_statement for
          (method_call (scalar PLACEHOLDER_1) . comb
            (argument_list "(" 3 ")"))
          (block {
            (when_statement when "AUG"
              (block {
                (listop_call take "Methionine") }))
            (when_statement when "UGG"
              (block {
                (listop_call take "Tryptophan") }))
            (when_statement when
              (regex /UU<[UC]>/)
              (block {
                (listop_call take "Phenylalanine") }))
            (when_statement when
              (regex /UU<[AG]>/)
              (block {
                (listop_call take "Leucine") }))
            (when_statement when
              (regex /UA<[UC]>/)
              (block {
                (listop_call take "Tyrosine") }))
            (when_statement when
              (regex /UG<[UC]>/)
              (block {
                (listop_call take "Cysteine") }))
            (when_statement when
              (regex /UC<[UCAG]>/)
              (block {
                (listop_call take "Serine") }))
            (when_statement when
              (regex /U[A<[GA]>|GA]/)
              (block { last }))
            (default_statement default
              (block { die })) }))) })))
