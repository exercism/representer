(source_file
  (unit_declaration unit module NucleotideCount)
  (routine_declaration sub nucleotide-count
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (if_statement if
        (method_call
          (method_call (scalar PLACEHOLDER_1) . comb) . grep
          (argument_list "("
            (block {
              (binary_expression $_ !~~
                (regex /<[ACGT]>/)) }) ")"))
        (block {
          (listop_call die "Invalid nucleotide in strand") }))
      (assignment
        (variable_declaration
          (scope my) (local_scalar PLACEHOLDER_2)) =
        (method_call
          (method_call (scalar PLACEHOLDER_1) . comb) . Bag))
      (assignment
        (variable_declaration
          (scope my) (local_hash PLACEHOLDER_3)) =
        (list_expression
          (assignment A =>
            (binary_expression
              (subscript (scalar PLACEHOLDER_2) { "A" }) // 0)) ,
          (assignment C =>
            (binary_expression
              (subscript (scalar PLACEHOLDER_2) { "C" }) // 0)) ,
          (assignment G =>
            (binary_expression
              (subscript (scalar PLACEHOLDER_2) { "G" }) // 0)) ,
          (assignment T =>
            (binary_expression
              (subscript (scalar PLACEHOLDER_2) { "T" }) // 0)) ,))
      (listop_call return
        (method_call (hash PLACEHOLDER_3) . Bag)) })))
