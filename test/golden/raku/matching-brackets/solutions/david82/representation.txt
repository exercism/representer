(source_file
  (unit_declaration unit module MatchingBrackets)
  (constant_declaration constant %pairs =
    (list_expression
      (assignment "]" => "[") ,
      (assignment "}" => "{") ,
      (assignment ")" => "(") ,))
  (routine_declaration sub has-matching-brackets
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) -->
      (type Bool) ")")
    (trait is export)
    (block {
      (variable_declaration
        (scope my) (local_array PLACEHOLDER_2))
      (for_statement for
        (method_call (scalar PLACEHOLDER_1) . comb)
        (block {
          (when_statement when
            (regex "/<[ ( { [ ]>/")
            (block {
              (method_call (array PLACEHOLDER_2) . push
                (argument_list "(" $_ ")")) }))
          (when_statement when
            (regex "/<[ ) } \\] ]>/")
            (block {
              (unless_statement unless
                (binary_expression (array PLACEHOLDER_2) &&
                  (binary_expression
                    (method_call (array PLACEHOLDER_2) . pop) eq
                    (subscript %pairs { $_ })))
                (block {
                  (listop_call return False) })) })) }))
      (binary_expression
        (method_call (array PLACEHOLDER_2) . elems) == 0) })))
