(source_file
  (unit_declaration unit module Darts)
  (routine_declaration sub score-dart
    (signature "("
      (parameter_list
        (parameter :$x) ,
        (parameter :$y)) ")")
    (trait is export)
    (block {
      (assignment
        (variable_declaration
          (scope my) (local_scalar PLACEHOLDER_1)) =
        (binary_expression
          (binary_expression $x * $x) +
          (binary_expression $y * $y)))
      (assignment
        (variable_declaration
          (scope my) (local_scalar PLACEHOLDER_2)) = 10)
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) > 1)
        (block {
          (assignment (scalar PLACEHOLDER_2) = 5) }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) > 25)
        (block {
          (assignment (scalar PLACEHOLDER_2) = 1) }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) > 100)
        (block {
          (assignment (scalar PLACEHOLDER_2) = 0) }))
      (listop_call return (scalar PLACEHOLDER_2)) })))
