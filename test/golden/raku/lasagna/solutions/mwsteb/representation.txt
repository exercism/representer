(source_file
  (unit_declaration unit module Lasagna)
  (constant_declaration constant $EXPECTED-MINUTES-IN-OVEN = 40)
  (constant_declaration constant $MINUTES-PER-LASANA-LAYER = 2)
  (routine_declaration sub remaining-minutes-in-oven
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (listop_call return
        (binary_expression $EXPECTED-MINUTES-IN-OVEN - (scalar PLACEHOLDER_1))) }))
  (routine_declaration sub preparation-time-in-minutes
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_2))) ")")
    (trait is export)
    (block {
      (listop_call return
        (binary_expression (scalar PLACEHOLDER_2) * $MINUTES-PER-LASANA-LAYER)) }))
  (routine_declaration sub total-time-in-minutes
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_2)) ,
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (listop_call return
        (binary_expression
          (call_expression preparation-time-in-minutes
            (argument_list "(" (scalar PLACEHOLDER_2) ")")) + (scalar PLACEHOLDER_1))) }))
  (routine_declaration sub oven-alarm
    (signature "(" ")")
    (trait is export)
    (block {
      (listop_call return "Ding!") })))
