(source_file
  (unit_declaration unit class Clock)
  (variable_declaration
    (scope has) $.hour)
  (variable_declaration
    (scope has) $.minute)
  (routine_declaration method (definition PLACEHOLDER_1)
    (block {
      (if_statement if
        (binary_expression
          (binary_expression $!minute > 59) ||
          (binary_expression $!minute < 0))
        (block {
          (assignment
            (variable_declaration
              (scope my) (local_scalar PLACEHOLDER_2)) = $!minute)
          (assignment $!minute =
            (binary_expression (scalar PLACEHOLDER_2) % 60))
          (assignment $!hour
            (binary_expression (scalar PLACEHOLDER_2) div 60)) }))
      (if_statement if
        (binary_expression
          (binary_expression $!hour > 23) ||
          (binary_expression $!hour < 0))
        (block {
          (assignment $!hour =
            (binary_expression $!hour % 24)) })) }))
  (routine_declaration method time
    (block {
      (call_expression sprintf
        (argument_list "("
          (list_expression "%02d:%02d" ,
            (method_call self . hour) ,
            (method_call self . minute)) ")")) }))
  (routine_declaration method add
    (signature "("
      (parameter_list
        (parameter :$minutes)) -->
      (type Clock) ")")
    (block {
      (assignment $!minute $minutes)
      (method_call self . PLACEHOLDER_1
        (argument_list "(" ")"))
      (listop_call return self) }))
  (routine_declaration method subtract
    (signature "("
      (parameter_list
        (parameter :$minutes)) -->
      (type Clock) ")")
    (block {
      (method_call self . add
        (argument_list "("
          (colonpair :minutes
            (argument_list "("
              (binary_expression $minutes *
                (unary_expression - 1)) ")")) ")"))
      (listop_call return self) }))
  (routine_declaration submethod TWEAK
    (signature "(" ")")
    (block {
      (method_call self . PLACEHOLDER_1
        (argument_list "(" ")"))
      (listop_call return self) })))
