    (block {
      (list_expression
        (listop_call sprintf
          (list_expression "%02d:%02d" , $!hour)) , $!minute) }))

    (signature "("
      (parameter_list
        (parameter :$minutes)) -->
      (type Clock) ")")
    (block {
      (method_listop_call self . new :
        (list_expression
          (colonpair :hour
            (argument_list "(" $.hour ")")) ,
          (colonpair :minute
            (argument_list "("
              (binary_expression $.minute + $minutes) ")")))) }))

    (signature "("
      (parameter_list
        (parameter :$minutes)) -->
      (type Clock) ")")
    (block {
      (method_listop_call self . new :
        (list_expression
          (colonpair :hour
            (argument_list "(" $.hour ")")) ,
          (colonpair :minute
            (argument_list "("
              (binary_expression $.minute - $minutes) ")")))) }))

        (binary_expression
