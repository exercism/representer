  (routine_declaration sub handshake
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (variable_declaration
        (scope my) (local_array PLACEHOLDER_2))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) +& 1)
        (block {
          (method_listop_call (array PLACEHOLDER_2) . push : "wink") }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) +& 2)
        (block {
          (method_listop_call (array PLACEHOLDER_2) . push : "double blink") }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) +& 4)
        (block {
          (method_listop_call (array PLACEHOLDER_2) . push : "close your eyes") }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) +& 8)
        (block {
          (method_listop_call (array PLACEHOLDER_2) . push : "jump") }))
      (if_statement if
        (binary_expression (scalar PLACEHOLDER_1) +& 16)
        (block {
          (assignment (array PLACEHOLDER_2) =
            (method_call (array PLACEHOLDER_2) . reverse)) }))
      (listop_call return (array PLACEHOLDER_2)) })))
