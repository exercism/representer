(source_file
  (unit_declaration unit module SecretHandshake)
  (routine_declaration sub handshake
    (signature "("
      (parameter_list
        (parameter (local_scalar PLACEHOLDER_1))) ")")
    (trait is export)
    (block {
      (variable_declaration
        (scope my)
        (type Str) (local_array PLACEHOLDER_2))
      (listop_call say (scalar PLACEHOLDER_1))
      (unless_statement unless
        (binary_expression (scalar PLACEHOLDER_1) %% 2)
        (block {
          (method_call (array PLACEHOLDER_2) . push
            (argument_list "(" "wink" ")")) }))
      (unless_statement unless
        (binary_expression
          (binary_expression (scalar PLACEHOLDER_1) div 2) %% 2)
        (block {
          (method_call (array PLACEHOLDER_2) . push
            (argument_list "(" "double blink" ")")) }))
      (unless_statement unless
        (binary_expression
          (binary_expression (scalar PLACEHOLDER_1) div 4) %% 2)
        (block {
          (method_call (array PLACEHOLDER_2) . push
            (argument_list "(" "close your eyes" ")")) }))
      (unless_statement unless
        (binary_expression
          (binary_expression (scalar PLACEHOLDER_1) div 8) %% 2)
        (block {
          (method_call (array PLACEHOLDER_2) . push
            (argument_list "(" "jump" ")")) }))
      (unless_statement unless
        (binary_expression
          (binary_expression (scalar PLACEHOLDER_1) div 16) %% 2)
        (block {
          (assignment (array PLACEHOLDER_2) =
            (call_expression reverse
              (argument_list "(" (array PLACEHOLDER_2) ")"))) }))
      (listop_call return (array PLACEHOLDER_2)) })))
