(source_file
  (unit_declaration unit module SpaceAge)
  (constant_declaration constant $secs-per-year-on-earth = 31557600)
  (constant_declaration constant %secs-per-year-on-earth-on =
    (binary_expression
      (quote_words < Mercury Venus Earth Mars Jupiter Saturn Uranus Neptune >)
      (binary_expression
        (list_expression 0.2408467 , 0.61519726 , 1 , 1.8808158 , 11.862615 , 29.447498 , 84.016846 , 164.79132) $secs-per-year-on-earth)))
  (routine_declaration sub age-on
    (signature "("
      (parameter_list
        (parameter :$planet) ,
        (parameter :$seconds)) ")")
    (trait is export)
    (block {
      (statement_modifier
        (listop_call fail "not a planet") without
        (subscript %secs-per-year-on-earth-on { $planet }))
      (listop_call round
        (list_expression
          (binary_expression $seconds ÷
            (subscript %secs-per-year-on-earth-on { $planet })) , 0.01)) })))
