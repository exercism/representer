pub fn rotate(text: ByteArray, shift_key: u8) -> ByteArray {
// panic!("implement `rotate`")
let mut output: ByteArray = Default::default();
let mut index = 0;
let text_len = text.len();
loop {
if (index == text_len) {
break;
}
let mut chr = text.at(index).unwrap();
if (chr >= 97 && chr <= 122) {
chr = int_rotate(chr, 97, shift_key);
} else if (chr >= 65 && chr <= 90){
chr = int_rotate(chr, 65, shift_key);
}
output.append_word(chr.into(), 1);
index += 1;
};
return output;
}
fn int_rotate(chr: u8, rotor: u8, shift: u8) -> u8 {
let output = ((chr - rotor) + shift) % 26 + rotor;
return output;
}
