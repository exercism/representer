use core::byte_array::{ByteArray, ByteArrayTrait};
/// Uppercase acronym from a ByteArray phrase.
/// Word boundaries: space ' ', hyphen '-', underscore '_'.
/// Other punctuation is ignored. Only alphabetic ASCII are collected.
pub fn abbreviate(phrase: ByteArray) -> ByteArray {
let mut acronym: ByteArray = Default::default();
// Delimiters.
const SPACE: u8 = 32_u8; // ' '
const HYPHEN: u8 = 45_u8; // '-'
const UNDERSCORE: u8 = 95_u8; // '_'
// ASCII alpha & case.
const A_UP: u8 = 65_u8; const Z_UP: u8 = 90_u8;
const A_LO: u8 = 97_u8; const Z_LO: u8 = 122_u8;
const CASE_DIFF: u8 = 32_u8; // 'a' - 'A'
let n: usize = phrase.len();
if n == 0_usize { return acronym; }
let mut i: usize = 0_usize;
let mut at_word_start: bool = true;
loop {
if i == n { break; }
let b: u8 = phrase.at(i).unwrap();
// delimiters: ' ', '-', '_'
if b == SPACE || b == HYPHEN || b == UNDERSCORE {
at_word_start = true;
} else {
// alphabetic?
let is_alpha = (b >= A_UP && b <= Z_UP) || (b >= A_LO && b <= Z_LO);
if at_word_start && is_alpha {
// uppercase if needed
let up = if b >= A_LO && b <= Z_LO { b - CASE_DIFF } else { b };
acronym.append_byte(up);
}
at_word_start = false;
}
i += 1_usize;
};
acronym
}
