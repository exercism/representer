pub fn is_armstrong_number(num: u128) -> bool {
let mut data: ByteArray = format!("{}", num);
let mut count: u128 = 0;
let mut data_len = data.len().into();
let mut index = 0;
loop {
if index == data_len {
break;
}
let x: u128 = ((data.at(index).unwrap().into()) - 48) ;
let mut innercount: u128 = 1;
for _ in 0..data_len {
innercount *= x;
};
count += innercount;
index += 1;
};
count == num
}
