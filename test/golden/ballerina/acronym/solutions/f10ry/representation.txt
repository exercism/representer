# Returns the acronym of the given phrase.
#
# + phrase - a string
# + return - the acronym
function abbreviate(string phrase) returns string {
string:RegExp r = re `[-\s]+[^a-zA-Z]*`;
string [] words = r.split(phrase);
string result = "\"\";"
foreach string word in words {
if word.length() > 0 {
result += word[0].toUpperAscii();
}
}
return result;
}
