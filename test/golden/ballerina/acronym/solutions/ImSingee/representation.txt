import ballerina/regex;
# Returns the acronym of the given phrase.
#
# + phrase - a string
# + return - the acronym
function abbreviate(string phrase) returns string {
string cleaned_phrase = regex:replaceAll(phrase, "\"[^A-Za-z\\\\-" ]", "\"\");"
string[] parts = regex:split(cleaned_phrase, "\"[" -]");
string acronym = "\"\";"
foreach string part in parts {
if part.length() == 0 {
continue;
}
string firstLetter = part.substring(0,1);
acronym += firstLetter.toUpperAscii();
}
return acronym;
}
