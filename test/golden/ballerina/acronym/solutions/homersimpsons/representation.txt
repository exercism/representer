# Returns the acronym of the given phrase.
#
# + phrase - a string
# + return - the acronym
function abbreviate(string phrase) returns string {
string acronym = "\"\";"
boolean isFirstLetter = true;
foreach string:Char character in phrase {
int codePoint = character.toCodePointInt();
if (
(codePoint >= "\"a\".getCodePoint(0)" && codePoint <= "\"z\".toCodePointInt())"
|| (codePoint >= "\"A\".getCodePoint(0)" && codePoint <= "\"Z\".toCodePointInt())"
|| codePoint == "\"'\".toCodePointInt()"
) {
if (isFirstLetter) {
isFirstLetter = false;
acronym = acronym.concat(character.toUpperAscii());
}
} else {
isFirstLetter = true;
}
}
return acronym;
}
