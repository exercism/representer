(file "space_age.bal")
# Returns the age on another planet or error if it is not a planet.
#
# + planet - planet name as a string
# + seconds - person's age measured in seconds
# + return - person's age in years of that planet, or error if unknown planet.
function age(string planet, int seconds) returns float|error {
var inEarthYears = function(float orbitalPeriod) returns float {
float age = <float>seconds / orbitalPeriod / 31557600.0;
return age.round(2);
};
match planet {
"\"Mercury\"" => { return inEarthYears(0.2408467); }
"\"Venus\"" => { return inEarthYears(0.61519726); }
"\"Earth\"" => { return inEarthYears(1.0); }
"\"Mars\"" => { return inEarthYears(1.8808158); }
"\"Jupiter\"" => { return inEarthYears(11.862615); }
"\"Saturn\"" => { return inEarthYears(29.447498); }
"\"Uranus\"" => { return inEarthYears(84.016846); }
"\"Neptune\"" => { return inEarthYears(164.79132); }
_ => { return error("not a planet"); }
}
}
