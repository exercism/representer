# Returns the age on another planet or error if it is not a planet.
#
# + planet - planet name as a string
# + seconds - person's age measured in seconds
# + return - person's age in years of that planet, or error if unknown planet.
const secondsPerEarthYear = 31557600;
const map<float> planetOrbitalPeriod = {
"\"mercury\":" 0.2408467,
"\"venus\":" 0.61519726,
"\"earth\":" 1.0,
"\"mars\":" 1.8808158,
"\"jupiter\":" 11.862615,
"\"saturn\":" 29.447498,
"\"uranus\":" 84.016846,
"\"neptune\":" 164.79132
};
function age(string planet, int seconds) returns float|error {
if ! planetOrbitalPeriod.hasKey(planet.toLowerAscii()) {
return error("not a planet");
}
return float:round(
<float> seconds /
<float> secondsPerEarthYear /
planetOrbitalPeriod.get(planet.toLowerAscii()), 2);
}
