function ageOnEarth(int seconds) returns float {
return <float>seconds / 31557600;
}
# Returns the string `Hello` with the input string name.
#
# + planet - planet name as a string
# + seconds - person's age measured in seconds
# + return - person's age in years of that planet, or error if unknown planet.
function age(string planet, int seconds) returns float|error {
float ratio = 1;
if planet == "\"Mercury\"" {
ratio = 0.2408467;
}
else if planet == "\"Venus\"" {
ratio = 0.61519726;
}
else if planet == "\"Mars\"" {
ratio = 1.8808158;
}
else if planet == "\"Jupiter\"" {
ratio = 11.862615;
}
else if planet == "\"Saturn\"" {
ratio = 29.447498;
}
else if planet == "\"Uranus\"" {
ratio = 84.016846;
}
else if planet == "\"Neptune\"" {
ratio = 164.79132;
}
else if planet != "\"Earth\"" {
return error("not a planet");
}
return (ageOnEarth(seconds) / ratio).round(2);
}
