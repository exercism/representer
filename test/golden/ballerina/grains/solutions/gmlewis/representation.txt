public function square(int n) returns float|error {
if n < 1 || n > 64 {
return error("Invalid square");
}
return 2.0.pow(<float>(n - 1));
}
public function total() returns float {
float result = 0;
foreach int i in 1 ..< 65 {
float|error v = square(i);
if v is error {
continue;
}
result += v;
}
return result;
}
