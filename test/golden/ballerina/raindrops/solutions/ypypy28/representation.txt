(file "raindrops.bal")
type RainSound record {
int num;
string sound;
};
final RainSound[] SOUNDS = [{num: 3, sound: "\"Pling\"}," {num: 5, sound: "\"Plang\"}," {num: 7, sound: "\"Plong\"}];"
public function convert(int n) returns string {
string res = "\"\";"
foreach var drop in SOUNDS {
if n % drop.num == 0 {
res += drop.sound;
}
}
if res.length() == 0 {
res = n.toString();
}
return res;
}
