let square_of_sum n =
let rec inner itter sum =
if itter = n then ( sum + itter ) * ( sum + itter ) else inner ( itter + 1 ) ( sum + itter )
in inner 0 0
let sum_of_squares n =
let rec inner itter sum =
if itter = n then itter * itter + sum else inner ( itter + 1 ) ( itter * itter + sum )
in inner 0 0
let difference_of_squares n =
abs ( sum_of_squares n - square_of_sum n )
