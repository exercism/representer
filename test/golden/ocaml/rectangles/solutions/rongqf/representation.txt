(file "rectangles.ml")
open Base
let c_to_i s =
match s with
| ' + ' -> 1
| ' - ' -> 2
| ' | ' -> 3
|_-> 0
let at arr x y =
List . nth_exn ( List . nth_exn arr x ) y
let is_range arr x1 y1 x2 y2 =
let a = ref 0 in
for x = x1 to x2 do
let v1 = at arr x y1 in
let v2 = at arr x y2 in
if v1 = 0 || v1 = 2 || v2 = 0 || v2 = 2 then a := 1
done ;
for y = y1 to y2 do
let v1 = at arr x1 y in
let v2 = at arr x2 y in
if v1 = 0 || v1 = 3 || v2 = 0 || v2 = 3 then a := 1
done ;
! a = 0
let loop_count arr =
let row = List . length arr in
match row with
| x when x < 2 -> 0
|_-> let col = List . length @@ List . hd_exn arr in
match col with
| y when y < 2 -> 0
|_-> let n = ref 0 in
for x1 = 0 to row - 1 do
for y1 = 0 to col - 1 do
if at arr x1 y1 = 1 then
for x2 = x1 + 1 to row - 1 do
for y2 = y1 + 1 to col - 1 do
if at arr x2 y2 = 1 && at arr x1 y2 = 1 && at arr x2 y1 = 1 && is_range arr x1 y1 x2 y2 then n := ! n + 1
done
done
done
done ;
! n
let count_rectangles ss =
List . map ~ f : ( fun s -> ( List . map ~ f : c_to_i ) @@ String . to_list s ) ( Array . to_list ss )
|> loop_count
