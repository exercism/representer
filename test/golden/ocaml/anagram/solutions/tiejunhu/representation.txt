open Base
let lowercase = String . lowercase
let sort s = s |> String . to_list |> List . sort ~ compare : Char . compare
let anagrams target candidates =
let lowered_target = lowercase target in
let sorted_target = sort lowered_target in
let is_anagram s =
let lowered_s = lowercase s in
String .( lowered_s <> lowered_target ) && List . compare Char . compare ( sort lowered_s ) sorted_target = 0
in
List . filter ~ f : is_anagram candidates
