open Base
let letter_to_int c = Char . to_int c - 96
let factorize_string s = String . fold s ~ init : 1 ~ f : ( fun acc c -> acc * letter_to_int c )
let anagrams word candidates =
let normalized_word = String . lowercase word in
let factorized_word = factorize_string normalized_word in
candidates
|> List . filter ~ f : ( fun s -> not String .( lowercase s = normalized_word ))
|> List . filter ~ f : ( fun candidate -> factorize_string ( String . lowercase candidate ) = factorized_word )
