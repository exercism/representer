let explode s = List . init ( String . length s ) ( String . get s )
;;
let rec compareOrderedCharLists list1 list2 =
match ( list1 , list2 ) with
| ([] , [] ) -> true
| ([] , _ ) -> false
| ( _ , [] ) -> false
| ( x :: xs , y :: ys ) ->
if Char . equal x y
then compareOrderedCharLists xs ys
else false
;;
let isAnagram word1 word2 =
let word1L = String . lowercase_ascii word1 in
let word2L = String . lowercase_ascii word2 in
if ( String . equal word1L word2L ) then false
else if String . length word1 = String . length word2 then
let list1S = word1L |> explode |> ( List . sort Char . compare ) in
let list2S = word2L |> explode |> ( List . sort Char . compare ) in
compareOrderedCharLists list1S list2S
else false
;;
let rec myFilter filter candidateList =
match candidateList with
| [] -> []
| x :: xs ->
if filter x
then x :: ( myFilter filter xs )
else ( myFilter filter xs )
let anagrams word wordList =
myFilter ( isAnagram word ) wordList
;;
