type ' a cell =
{ id : int ;
is_input : bool ;
eq : ' a -> ' a -> bool ;
mutable start : ' a ;
mutable value : ' a ;
mutable prop : ( ' a -> ' a cell list ) list ;
mutable cbs : ( int * ( ' a -> unit )) list }
type callback_id = int
let value_of { value ; _ } = value
let ( <~ ) cell x =
if cell . eq x cell . value
then []
else begin
cell . value <- x ;
List .( map (( |> ) x ) cell . prop |> flatten |> cons cell )
end
let rec do_callbacks =
function
| [] -> ()
| cell :: cells ->
if not ( cell . eq cell . value cell . start )
then begin
cell . start <- cell . value ;
List . iter ( fun ( _ , cb ) -> cb cell . value ) cell . cbs ;
end ;
List . filter ( fun { id ; _ } -> id <> cell . id ) cells
|> do_callbacks
let set_value cell x =
if cell . is_input then cell <~ x |> do_callbacks
let next_cell_id_ref = ref 0
let create_cell is_input eq value =
let id = ! next_cell_id_ref in
next_cell_id_ref := id + 1 ;
{ id ; is_input ; eq ; start = value ; value ; prop = [] ; cbs = [] }
let create_input_cell ~ value ~ eq =
create_cell true eq value
let create_compute_cell_1 src ~ f ~ eq =
let cell = create_cell false eq ( f src . value ) in
src . prop <- ( fun x -> cell <~ f x ) :: src . prop ;
cell
let create_compute_cell_2 src1 src2 ~ f ~ eq =
let cell = create_cell false eq ( f src1 . value src2 . value ) in
src1 . prop <- ( fun x -> cell <~ f x src2 . value ) :: src1 . prop ;
src2 . prop <- ( fun x -> cell <~ f src1 . value x ) :: src2 . prop ;
cell
let add_callback cell ~ k =
let id = match cell . cbs with [] -> 0 | ( id , _ ) :: _ -> id + 1 in
cell . cbs <- ( id , k ) :: cell . cbs ;
id
let remove_callback cell id =
cell . cbs <- List . remove_assoc id cell . cbs
