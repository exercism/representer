type base = int
let rec check from digits = match digits with
| [] -> true
| first :: rest -> if first >= from || first < 0 then false
else check from rest ;;
let rec power n num = if n = 0 then 1
else num * ( power ( n - 1 ) num );;
let rec change from digits = match digits with
| [] -> 0
| first :: rest -> first * ( power ( List . length digits - 1 ) from ) + ( change from rest );;
let mod_cul n num = num - ( n * ( num / n ));;
let rec reduce target num = if num == 0 then []
else ( mod_cul target num ) :: ( reduce target ( num / target ));;
let rec reverse digits = match digits with
| [] -> []
| first :: rest -> List . rev digits ;;
let vanish digits = match digits with
| [] -> []
| first :: rest -> if first == 0 then rest
else digits ;;
let rec zero digits = match digits with
| [] -> true
| first :: rest -> if first == 0 then zero rest
else false ;;
let convert_bases ~ from : from ~ digits : digits ~ target : target =
if from < 2 then None
else if target < 2 then None
else if ( check from digits ) == false then None
else if digits == [] || zero digits then Some [ 0 ]
else Some ( reverse ( reduce target ( change from digits )));;
