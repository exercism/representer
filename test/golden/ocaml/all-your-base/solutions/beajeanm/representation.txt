open Base
type base = int
let convert_to_int from digits =
List . fold_left ~ init : 0 ~ f : ( fun acc digit -> digit + ( acc * from )) digits
let decompose target value =
let empty = Sequence . init ~ f : ( fun x -> x ) 1 in
let seq =
Sequence . unfold ~ init : value ~ f : ( fun value ->
if value = 0 then None else Some ( value % target , value / target ))
in
if value = 0 then empty else seq
let convert_bases ~ from ~ digits ~ target =
let all_digits_valid =
List . filter ~ f : ( fun n -> n < 0 || n >= from ) digits |> List . is_empty
in
let target_valid = target > 1 in
let from_valid = from > 1 in
if all_digits_valid && target_valid && from_valid then
convert_to_int from digits |> decompose target |> Sequence . to_list_rev
|> Option . return
else None
