type base = int
let rec to_int = fun digits radix acc -> match digits with
| [] -> acc
| digit :: rest -> to_int rest radix ( acc * radix + digit )
let rec to_list = fun num radix acc ->
if num >= radix then
let m = num mod radix in to_list (( num - m ) / radix ) radix ([ m ] @ acc )
else
[ num ] @ acc
let rec check_digit = fun digits radix -> match digits with
| [] -> true
| digit :: rest -> if digit < 0 || digit >= radix then false else check_digit rest radix
let convert_bases ~ from ~ digits ~ target =
if from < 2 || target < 2 then None else
if List . length digits < 1 then Some [ 0 ] else
if not ( check_digit digits from ) then None else
let num = to_int digits from 0 in
let list = to_list num target [] in
Some list
