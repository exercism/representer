type base = int
let rec pow base power =
match power with
| 0 -> 1
| 1 -> base
|_ -> if power mod 2 == 1 then base * ( pow base (( power - 1 ) / 2 )) * ( pow base (( power - 1 ) / 2 ))
else ( pow base ( power / 2 )) * ( pow base ( power / 2 ))
let rec sum l =
match l with
| [] -> 0
| x :: xs -> x + sum xs
let to10 base digits =
let digits_arr = Array . of_list digits in
let length = List . length digits in
let result =
let range = List . rev ( List . init ( List . length digits ) ( fun x -> x )) in
let powers = List . map ( fun x -> digits_arr . ( length - 1 - x ) * pow base x ) range in
sum powers ;
in
result
let from10 number target_base =
let length = Float . to_int (
Float . floor (
Float . log ( Int . to_float ( number )) /. Float . log ( Int . to_float ( target_base ))
)
) + 1 in
let digits = Array . init length ( fun x -> x ) in
digits . ( length - 1 ) <- number mod target_base ;
for i = 1 to length - 1 do
let remainder = number / ( pow target_base i ) mod target_base in
digits . ( length - 1 - i ) <- remainder ;
done ;
Array . to_list digits
let convert_bases ~ from ~ digits ~ target =
if from <= 1 then None
else if target <= 1 then None
else if not ( List . for_all ( fun x -> 0 <= x && x < from ) digits ) then None
else Some ( from10 ( to10 from digits ) target )
