open Base
open Option . Monad_infix
type base = int
let check_from digits ~ from =
if from > 1
then Some digits
else None
let check_base digits ~ target =
if target > 1
then Some digits
else None
let check_digits digits ~ from =
let rec aux l =
match l with
| [] -> Some []
| n ::_ tl when n >= from || n < 0 -> None
| n :: tl -> ( match aux tl with
| None -> None
| Some ds -> Some ( n :: ds ))
in
aux digits
let to_decimal digits ~ from =
let reducer ( num , cur_pow ) digit =
( num + ( cur_pow * digit ) , cur_pow * from )
in
List . rev digits
|> List . fold ~ init : ( 0 , 1 ) ~ f : reducer
|> fun ( n , _ ) -> Some n
let from_decimal num ~ target =
let rec aux rem =
if rem = 0
then []
else ( rem % target ) :: aux ( rem / target )
in
if num = 0
then Some [ 0 ]
else Some ( List . rev ( aux num ))
let convert_bases ~ from ~ digits ~ target =
check_digits ~ from digits
>>= check_from ~ from
>>= check_base ~ target
>>= to_decimal ~ from
>>= from_decimal ~ target
