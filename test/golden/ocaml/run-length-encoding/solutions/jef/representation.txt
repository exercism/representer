open Base
let add_to_encoding code char count =
match count with
| 0 -> code
| 1 -> code ^ Char . to_string char
| _ -> code ^ Int . to_string count ^ Char . to_string char
;;
let encode source =
let rec encode' result prev_char prev_count source_list =
match source_list with
| [] -> add_to_encoding result prev_char prev_count
| char :: tail when Char .( = ) char prev_char ->
encode' result prev_char ( prev_count + 1 ) tail
| char :: tail -> encode' ( add_to_encoding result prev_char prev_count ) char 1 tail
in
encode' "\"\"" '\000' 0 ( String . to_list source )
;;
let decode code =
let rec decode' result current_count code_list =
match code_list with
| [] -> result
| digit :: tail when Char . is_digit digit ->
decode' result (( current_count * 10 ) + Char . get_digit_exn digit ) tail
| char :: tail -> decode' ( result ^ String . make ( max 1 current_count ) char ) 0 tail
in
decode' "\"\"" 0 ( String . to_list code )
;;
