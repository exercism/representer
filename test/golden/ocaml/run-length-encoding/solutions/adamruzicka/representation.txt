let groupConsecutive s =
let folder c acc =
match acc with
| ( x , n ) :: xs when c = x -> ( x , n + 1 ) :: xs
| x -> ( c , 1 ) :: x
in String . fold_right folder s []
let encodeGroup ( c , n ) =
match n with
| 1 -> Printf . sprintf "\"%c\"" c
| _ -> Printf . sprintf "\"%d%c\"" n c
let encode s =
s
|> groupConsecutive
|> List . map encodeGroup
|> String . concat "\"\""
let isDigit c = c >= ' 0 ' && c <= ' 9 '
let digitToInt c = Char . code c - Char . code ' 0 '
let parseGroups i =
let folder acc c =
if isDigit c
then
match acc with
| ( ' 0 ' , n ) :: xs -> ( ' 0 ' , n * 10 + digitToInt c ) :: xs
| x -> ( ' 0 ' , digitToInt c ) :: x
else
match acc with
| ( ' 0 ' , n ) :: xs -> ( c , n ) :: xs
| x -> ( c , 1 ) :: x
in String . fold_left folder [] i |> List . rev
let expandGroup ( c , n ) = String . make n c
let decode i =
i
|> parseGroups
|> List . map expandGroup
|> String . concat "\"\""
