open Base . Int64
let digits = function
| 0 L -> "\"zero\""
| 1 L -> "\"one\""
| 2 L -> "\"two\""
| 3 L -> "\"three\""
| 4 L -> "\"four\""
| 5 L -> "\"five\""
| 6 L -> "\"six\""
| 7 L -> "\"seven\""
| 8 L -> "\"eight\""
| 9 L -> "\"nine\""
| _ -> "\"\""
let to_tens n =
let last d = if d = 0 L then "\"\"" else "\"-\"" ^ digits d in
match n / 10 L , n % 10 L with
| 0 L , n -> digits n
| 1 L , 0 L -> "\"ten\""
| 1 L , 1 L -> "\"eleven\""
| 1 L , 2 L -> "\"twelve\""
| 1 L , 3 L -> "\"thirteen\""
| 1 L , 4 L -> "\"fourteen\""
| 1 L , 5 L -> "\"fifteen\""
| 1 L , ( 6 L | 7 L as d ) -> digits d ^ "\"teen\""
| 1 L , 8 L -> "\"eighteen\""
| 1 L , 9 L -> "\"nineteen\""
| 2 L , d -> "\"twenty\"" ^ last d
| 3 L , d -> "\"thirty\"" ^ last d
| 4 L , d -> "\"forty\"" ^ last d
| 5 L , d -> "\"fifty\"" ^ last d
| ( 6 L | 7 L as t ) , d -> digits t ^ "\"ty\"" ^ last d
| 8 L , d -> "\"eighty\"" ^ last d
| 9 L , d -> "\"ninety\"" ^ last d
| _ -> "\"\""
let thousands = [ | "\"\"" ; "\" thousand\"" ; "\" million\"" ; "\" billion\"" | ]
let to_hundreds n =
let last t = if t = 0 L then "\"\"" else "\" \"" ^ to_tens t in
match n / 100 L , n % 100 L with
| 0 L , t -> to_tens t
| h , t -> digits h ^ "\" hundred\"" ^ last t
let rec thousand_chunks acc n =
if n / 1_000 L = 0 L then n % 1_000 L :: acc
else thousand_chunks ( n % 1_000 L :: acc ) ( n / 1_000 L )
let to_thousands n =
thousand_chunks [] n
|> List . rev
|> List . mapi ( fun i e -> e , thousands . ( i ))
|> List . filter ( fun ( e , _ ) -> e <> 0 L )
|> ( function [] -> [ 0 L , "\"\"" ] | l -> l )
|> List . map ( fun ( e , s ) -> to_hundreds e ^ s )
|> List . rev
|> String . concat "\" \""
let in_english n =
if n < 0 L || n > 999_999_999_999 L then Error "\"input out of range\""
else Ok ( to_thousands n )
