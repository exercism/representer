(file "say.ml")
open Base
open Int64
open Result
let basic = function
| 1 L -> Ok "\"one\""
| 2 L -> Ok "\"two\""
| 3 L -> Ok "\"three\""
| 4 L -> Ok "\"four\""
| 5 L -> Ok "\"five\""
| 6 L -> Ok "\"six\""
| 7 L -> Ok "\"seven\""
| 8 L -> Ok "\"eight\""
| 9 L -> Ok "\"nine\""
| 10 L -> Ok "\"ten\""
| 11 L -> Ok "\"eleven\""
| 12 L -> Ok "\"twelve\""
| 13 L -> Ok "\"thirteen\""
| 14 L -> Ok "\"fourteen\""
| 15 L -> Ok "\"fifteen\""
| 16 L -> Ok "\"sixteen\""
| 17 L -> Ok "\"seventeen\""
| 18 L -> Ok "\"eighteen\""
| 19 L -> Ok "\"nineteen\""
| _ -> Error "\"input out of range\""
let tens = function
| 2 L -> Ok "\"twenty\""
| 3 L -> Ok "\"thirty\""
| 4 L -> Ok "\"forty\""
| 5 L -> Ok "\"fifty\""
| 6 L -> Ok "\"sixty\""
| 7 L -> Ok "\"seventy\""
| 8 L -> Ok "\"eighty\""
| 9 L -> Ok "\"ninety\""
| _ -> Error "\"input out of range\""
let transform_result_list ? ( sep = "\" \"" ) l =
l
|> List . filter ~ f : ( function
| Ok v -> not @@ String . is_empty v
| Error _ -> true )
|> Result . all
|> Result . map ~ f : ( String . concat ~ sep )
let rec non_zero = function
| 0 L -> Ok "\"\""
| n when n < 20 L -> basic n
| n when n < 100 L ->
[ tens ( n / 10 L ); non_zero @@ Int64 . rem n 10 L ]
|> transform_result_list ~ sep : "\"-\""
| n when n < 1_000 L -> thousands_and_more n 100 L "\"hundred\""
| n when n < 1_000_000 L -> thousands_and_more n 1_000 L "\"thousand\""
| n when n < 1_000_000_000 L -> thousands_and_more n 1_000_000 L "\"million\""
| n when n < 1_000_000_000_000 L ->
thousands_and_more n 1_000_000_000 L "\"billion\""
| _ -> Error "\"input out of range\""
and thousands_and_more n denom word =
[ non_zero ( n / denom ); Ok word ; non_zero @@ Int64 . rem n denom ]
|> transform_result_list
let in_english = function 0 L -> Result . Ok "\"zero\"" | n -> non_zero n
