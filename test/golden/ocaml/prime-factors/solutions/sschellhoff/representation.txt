(file "prime_factors.ml")
let factors_of number =
let rec worker factor rest acc =
if ( Int64 . equal rest 1 L ) || ( Int64 . compare factor rest ) = 1 then
acc
else if Int64 . equal ( Int64 . rem rest factor ) 0 L then
worker factor ( Int64 . div rest factor ) ( factor :: acc )
else
worker ( Int64 . add factor 1 L ) rest acc
in
List . sort Int64 . compare ( worker 2 L number [] )
