(file "prime_factors.ml")
open Base
let factors_of num =
let rec aux n divisor acc = Int64 .(
match n with
| 1 L -> List . rev acc
| n when n % divisor = 0 L -> aux ( n / divisor ) divisor ( divisor :: acc )
| n ->
let inc = if divisor > 2 L then 2 L else 1 L in
aux n ( divisor + inc ) acc )
in
aux num 2 L []
