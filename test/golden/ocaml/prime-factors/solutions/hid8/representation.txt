open Base
open Int64
let two = ( of_int_exn 2 )
let factors_of ( number : int64 ) =
if number = one then [] else
let rec sub n m acc =
if n = one || m >= n then ( acc @ [ m ] ) else
let modd = n % m in
if equal modd zero
then sub ( n / m ) m ( acc @ [ m ] )
else sub n ( m + one ) acc
in sub number two []
