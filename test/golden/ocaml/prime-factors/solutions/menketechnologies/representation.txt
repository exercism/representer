open Base
let factors_of num =
let open Int64 . O in
let primes = function
| 2 L -> 3 L
| m -> m + 2 L
in
let rec aux n m acc =
if n = 1 L then acc
else if n % m = 0 L then aux ( n / m ) m ( m :: acc )
else aux n ( primes m ) acc
in
if num <= 0 L then
raise ( Invalid_argument "\"factors_of\"" )
else List . rev ( aux num 2 L [] )
