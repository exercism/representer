open Base
let empty = Map . empty ( module Char )
let valid_nucleotide = function
| ' A' | ' C' | ' G' | ' T' -> true
| _ -> false
let all_valid_nucleotides s =
String . for_all s ~ f : valid_nucleotide
let count_nucleotide s c =
if valid_nucleotide c
then if all_valid_nucleotides s
then Ok ( String . count s ~ f : ( Char . equal c ))
else Error ( ' X' )
else Error ( ' X' )
let count_nucleotides s =
if all_valid_nucleotides s
then
let records =
[ ' A' ; ' C' ; ' G' ; ' T' ]
|> List . map ~ f : ( fun c -> match count_nucleotide s c with
| Ok x -> ( c , x )
| _ -> failwith "\"does not happen\"" )
in
let foldf m ( key , data ) =
match data with
| 0 -> m
| _ -> Map . set m ~ key ~ data
in
Ok ( List . fold records ~ init : empty ~ f : foldf )
else Error ' X'
