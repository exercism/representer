(file "nucleotide_count.ml")
open Base
let empty = Map . empty ( module Char );;
let is_nucl c = String . contains "\"GATC\"" c ;;
let find_invalid_nucl = String . find ~ f : ( fun c -> not @@ is_nucl c );;
let get_count map dna = Map . find map dna |> Option . value ~ default : 0 ;;
let inc_count map nucl = Map . set map ~ key : nucl ~ data : ( get_count map nucl + 1 );;
let count_nucleotides = function
dna when find_invalid_nucl dna |> Option . is_none -> Ok ( String . fold ~ init : empty ~ f : inc_count dna )
| dna -> Error ( Caml . Option . get ( find_invalid_nucl dna ));;
let count_nucleotide s =
let open Result in
function
c when is_nucl c -> count_nucleotides s >>= fun x -> Ok ( get_count x c )
| c -> Error ( c );;
