open Base
let empty = Map . empty ( module Char )
let isValid c = match c with
| ' A' | ' C' | ' G' | ' T' -> true
| _ -> false
let map_increase map key =
Map . update key ~ f : ( Option . value_map ~ f : (( + ) 1 ) ~ default : 1 ) map
let count_nucleotides s = String . fold_result s
~ init : empty
~ f : ( fun map c -> if isValid c then Ok ( map_increase c map ) else Error c )
let count_nucleotide s c = if isValid c then match count_nucleotides s with
| Ok whole_map -> Ok ( Option . value_map ( Map . find whole_map c ) ~ f : ( fun x -> x ) ~ default : 0 )
| Error e -> Error e
else Error c
