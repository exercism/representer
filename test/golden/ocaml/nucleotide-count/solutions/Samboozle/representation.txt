(file "nucleotide_count.ml")
open Base
let empty = Map . empty ( module Char )
let check_nucleotide c =
match c with
| ' A' | ' C' | ' G' | ' T' -> Ok c
| _ -> Error c
let count_nucleotide str char =
match check_nucleotide char with
| Error c -> Error c
| _ -> String . fold_result str ~ init : 0 ~ f : ( fun acc c ->
check_nucleotide c
|> Result . map ~ f : ( fun c' -> acc + if phys_equal c' char then 1 else 0 )
)
let count_nucleotides =
String . fold_result ~ init : empty ~ f : ( fun m c ->
check_nucleotide c
|> Result . map ~ f : ( fun c' -> Map . update m c' ~ f : ( function
| Some prev -> prev + 1
| None -> 1
))
)
