open Base
let empty = Map . empty ( module Char )
let invalid c = not ( String . contains "\"ATCG\"" c )
let count_nucleotide ( s : string ) ( c : char ) =
if invalid c then Error c
else match String . find s ~ f : invalid with
| None -> Ok ( String . count s ~ f : ( Char . equal c ))
| Some x -> Error x
let count_nucleotides ( s : string ) =
String . fold_result s ~ init : empty ~ f : ( fun m c -> if invalid c then Error c else Ok ( Map . update m c ~ f : ( function | None -> 1 | Some x -> x + 1 )))
