open Base
let empty = Map . empty ( module Char )
let is_valid = function ' A' | ' C' | ' G' | ' T' -> true | _ -> false
let filter_str n =
let s = String . to_list n in
match List . find ~ f : ( fun x -> not ( is_valid x )) s with
| Some x -> Error x
| None -> Ok s
let occurrence s c =
List . fold ~ init : 0
~ f : ( fun init x -> if Char . equal x c then init + 1 else init )
s
let count_nucleotide s c =
if not ( is_valid c ) then Error c
else match filter_str s with Error _ as e -> e | Ok n -> Ok ( occurrence n c )
let count_nucleotides s =
let seen m c = match Map . find m c with Some x -> x | None -> 0 in
let countn m c = Map . set m ~ key : c ~ data : ( seen m c + 1 ) in
match filter_str s with
| Error _ as e -> e
| Ok n -> Ok ( List . fold ~ init : empty ~ f : ( fun init x -> countn init x ) n )
