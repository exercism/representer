IN: difference-of-squares
: square-of-sum ( n -- m )
dup 1 + * 2/ dup * ;
: sum-of-squares ( n -- m )
dup 1 + over 2 * 1 + * * 6 / ;
: difference-of-squares ( n -- m )
dup dup * dup 1 swap - -rot 3 * swap 2 * + * 12 / neg ;
