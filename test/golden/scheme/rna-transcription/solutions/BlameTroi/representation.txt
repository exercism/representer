(file "rna-transcription.scm")
( import ( rnrs ))
( define ( dna-character->rna-character dna )
( cond
(( equal? dna #\A ) ' #\U )
(( equal? dna #\T ) ' #\A )
(( equal? dna #\G ) ' #\C )
(( equal? dna #\C ) ' #\G )
( ' ( #t ) ' #\? )))
( define ( alternate-dna->rna dna )
( list->string ( map dna-character->rna-character ( string->list dna ))))
( define ( dna->rna dna )
( list->string ( map ( lambda ( x )
( cond
(( equal? x #\A ) ' #\U )
(( equal? x #\T ) ' #\A )
(( equal? x #\G ) ' #\C )
(( equal? x #\C ) ' #\G )
( ' ( #t ) ' #\? )))
( string->list dna ))))
