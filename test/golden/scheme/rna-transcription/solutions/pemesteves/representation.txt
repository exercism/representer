( import ( rnrs ))
( define ( rnaChar r )
( case r
(( #\G ) ' ( #\C ))
(( #\C ) ' ( #\G ))
(( #\T ) ' ( #\A ))
(( #\A ) ' ( #\U ))))
( define ( getRna rnaList )
( if ( = ( length rnaList ) 0 )
' ()
( append ( rnaChar ( car rnaList ))
( getRna ( cdr rnaList )))))
( define ( dna->rna dna )
( list->string ( getRna ( string->list dna ))))
