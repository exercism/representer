(file "rna-transcription.scm")
( import ( rnrs ))
( define ( string-map on-list-item the-string )
( list->string ( map on-list-item ( string->list the-string ))))
( define ( dna->rna dna )
( let* (( replace-table ' (( #\A . #\U )
( #\C . #\G )
( #\G . #\C )
( #\T . #\A )))
( replace ( lambda ( c ) ( cdr ( assv c replace-table )))))
( string-map replace dna )))
