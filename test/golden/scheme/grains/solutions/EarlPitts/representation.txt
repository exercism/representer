(file "grains.scm")
( import ( rnrs ))
( define ( reduce op id l )
( if ( null? l )
id
( op ( car l ) ( reduce op id ( cdr l )))))
( define ( square n )
( cond (( or ( > n 64 ) ( < n 0 )) ( error "\"Wrong board size\"" 1 ))
(( zero? ( 1- n )) 1 )
( else ( * 2 ( square ( 1- n ))))))
( define total
( reduce + 0 ( map square ( map 1 + ( iota 64 )))))
