( import ( rnrs ))
( define ( square n )
( if ( <= n 0 )
( assertion-violation
’ square "\"square 0\"" n ))
( if ( > n 64 )
( assertion-violation
’ square "\"square 64\"" n ))
( expt 2 ( sub1 n )))
( define ( all-but-last xs )
( cond [( null? xs ) xs ]
[ else ( take xs ( sub1 ( length xs )))]))
( define ( sum xs )
( apply + xs ))
( define ( range start end )
( cond [( = start end ) ' ()]
[ else ( cons start ( range ( add1 start )
end ))]))
( define ( total )
( sum ( map square ( range 1 65 ))))
