(file "scrabble-score.scm")
( import ( rnrs ))
( define 1 pt ( string->list "\"AEIOULNRSTaeioulnrst\"" ))
( define 2 pt ( string->list "\"DGdg\"" ))
( define 3 pt ( string->list "\"BCMPbcmp\"" ))
( define 4 pt ( string->list "\"FHVWYfhvwy\"" ))
( define 5 pt ( string->list "\"Kk\"" ))
( define 8 pt ( string->list "\"JXjx\"" ))
( define 10 pt ( string->list "\"QZqz\"" ))
( define ( char-score c )
( cond
(( member c 1 pt ) 1 )
(( member c 2 pt ) 2 )
(( member c 3 pt ) 3 )
(( member c 4 pt ) 4 )
(( member c 5 pt ) 5 )
(( member c 8 pt ) 8 )
(( member c 10 pt ) 10 )
( else ( error #f "\"NO\"" ))))
( define ( score word )
( apply + ( map char-score ( string->list word ))))
