(file "scrabble-score.scm")
( import ( rnrs ))
( define ( score word )
( define ( score total letter )
( cond
[( char=? #\D letter ) ( + total 2 )]
[( char=? #\G letter ) ( + total 2 )]
[( char=? #\B letter ) ( + total 3 )]
[( char=? #\C letter ) ( + total 3 )]
[( char=? #\M letter ) ( + total 3 )]
[( char=? #\P letter ) ( + total 3 )]
[( char=? #\F letter ) ( + total 4 )]
[( char=? #\H letter ) ( + total 4 )]
[( char=? #\V letter ) ( + total 4 )]
[( char=? #\W letter ) ( + total 4 )]
[( char=? #\Y letter ) ( + total 4 )]
[( char=? #\K letter ) ( + total 5 )]
[( char=? #\J letter ) ( + total 8 )]
[( char=? #\X letter ) ( + total 8 )]
[( char=? #\Q letter ) ( + total 10 )]
[( char=? #\Z letter ) ( + total 10 )]
[ else ( + total 1 )]
)
)
( fold-left score 0 ( string->list ( string-upcase word )))
)
