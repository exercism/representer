(file "scrabble-score.scm")
( import ( rnrs ))
( define ( letter-value letter )
( case ( char-upcase letter )
(( #\A #\E #\I #\O #\U
#\L #\N #\R #\S #\T ) 1 )
(( #\D #\G ) 2 )
(( #\B #\C #\M #\P ) 3 )
(( #\F #\H #\V #\W #\Y ) 4 )
(( #\K ) 5 )
(( #\J #\X ) 8 )
(( #\Q #\Z ) 10 )))
( define ( score word )
( string-fold ( lambda ( letter total )
( + total ( letter-value letter )))
0
word ))
