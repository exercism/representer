(program
  (list define
    (list
      (definition PLACEHOLDER_1)
      (definition PLACEHOLDER_2))
    (list let
      (definition PLACEHOLDER_3)
      (list
        (list
          (definition PLACEHOLDER_4)
          (list string-length PLACEHOLDER_2))
        (list
          (definition PLACEHOLDER_5) 0)
        (list
          (definition PLACEHOLDER_6) 0)
        (list
          (definition PLACEHOLDER_7) 0)
        (list
          (definition PLACEHOLDER_8) 0))
      (list if
        (list zero? PLACEHOLDER_4)
        (list list
          (list cons "#\\A" PLACEHOLDER_5)
          (list cons "#\\C" PLACEHOLDER_6)
          (list cons "#\\G" PLACEHOLDER_7)
          (list cons "#\\T" PLACEHOLDER_8))
        (list let
          (list
            (list
              (definition PLACEHOLDER_4)
              (list - PLACEHOLDER_4 1)))
          (list case
            (list string-ref PLACEHOLDER_2 PLACEHOLDER_4)
            (list
              (list "#\\A")
              (list PLACEHOLDER_3 PLACEHOLDER_4
                (list + PLACEHOLDER_5 1) PLACEHOLDER_6 PLACEHOLDER_7 PLACEHOLDER_8))
            (list
              (list "#\\C")
              (list PLACEHOLDER_3 PLACEHOLDER_4 PLACEHOLDER_5
                (list + PLACEHOLDER_6 1) PLACEHOLDER_7 PLACEHOLDER_8))
            (list
              (list "#\\G")
              (list PLACEHOLDER_3 PLACEHOLDER_4 PLACEHOLDER_5 PLACEHOLDER_6
                (list + PLACEHOLDER_7 1) PLACEHOLDER_8))
            (list
              (list "#\\T")
              (list PLACEHOLDER_3 PLACEHOLDER_4 PLACEHOLDER_5 PLACEHOLDER_6 PLACEHOLDER_7
                (list + PLACEHOLDER_8 1)))
            (list else
              (list error
                (list quote PLACEHOLDER_1) "Invalid nucleotide in strand"))))))))
