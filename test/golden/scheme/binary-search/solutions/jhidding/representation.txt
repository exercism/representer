# | Binary search implementation . Assumes ` a ` and ` b ` are within range
| and array [ a ] and array [ b ] straddle the target .
| #
( define ( binary-search* array a b target )
( if ( < ( - b a ) 2 )
'not-found
( let* (( middle ( div ( + a b ) 2 ))
( value ( vector-ref array middle )))
( cond
(( = value target ) middle )
(( > value target ) ( binary-search* array a middle target ))
( else ( binary-search* array middle b target ))))))
( define ( binary-search array target )
( let (( a 0 )
( b ( - ( vector-length array ) 1 )))
( cond
(( zero? ( vector-length array )) 'not-found )
(( = target ( vector-ref array a )) a )
(( = target ( vector-ref array b )) b )
(( < target ( vector-ref array a )) 'not-found )
(( > target ( vector-ref array b )) 'not-found )
( else ( binary-search* array a b target )))))
