( import ( rnrs ))
( define ( plants diagram student )
( let* (( lines ( string-split diagram #\newline ))
( l1 ( list-ref lines 0 ))
( l2 ( list-ref lines 1 ))
( idx ( index-of *students* student )))
( if ( number? idx )
( let (( c1 ( string-ref l1 ( * 2 idx )))
( c2 ( string-ref l1 ( 1+ ( * 2 idx ))))
( c3 ( string-ref l2 ( * 2 idx )))
( c4 ( string-ref l2 ( 1+ ( * 2 idx )))))
( list ( plant-name c1 )
( plant-name c2 )
( plant-name c3 )
( plant-name c4 )))
' ())))
( define *students*
( list "\"Alice\"" "\"Bob\"" "\"Charlie\"" "\"David\""
"\"Eve\"" "\"Fred\"" "\"Ginny\"" "\"Harriet\""
"\"Ileana\"" "\"Joseph\"" "\"Kincaid\"" "\"Larry\"" ))
( define ( plant-name c )
( case c
(( #\G ) "\"grass\"" )
(( #\C ) "\"clover\"" )
(( #\R ) "\"radishes\"" )
(( #\V ) "\"violets\"" )))
( define ( index-of lst x )
( let loop (( lst lst ) ( idx 0 ))
( cond
(( null? lst ) # nil )
(( equal? x ( car lst )) idx )
( else ( loop ( cdr lst ) ( 1+ idx ))))))
