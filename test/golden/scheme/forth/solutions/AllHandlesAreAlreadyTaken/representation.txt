(file "forth.scm")
( use-modules ( srfi srfi-11 ))
( use-modules ( ice-9 receive ))
( define ( tokenize tokens )
"\"Split program to tokens.\""
( let loop ([ tokens tokens ]
[ in-definition #f ]
[ parts ' ()])
( cond
[( null? tokens )
( reverse parts )]
[( equal? ( car tokens ) "\":\"" )
( loop ( cdr tokens ) #t ( cons ' () parts ))]
[( equal? ( car tokens ) "\";\"" )
( loop ( cdr tokens ) #f ( cons ( reverse ( car parts )) ( cdr parts )))]
[ in-definition
( loop ( cdr tokens ) #t ( cons ( cons ( car tokens ) ( car parts )) ( cdr parts )))]
[ else
( loop ( cdr tokens ) #f ( cons ( car tokens ) parts ))])))
( define ( resolve-recursive tokens context )
"\"Resolve token again given context. If TOKENS is a list, resolve each token in the list.\n  \""
( cond
[( null? tokens ) ' ()]
[( pair? tokens )
( cons ( resolve-recursive ( car tokens ) context )
( resolve-recursive ( cdr tokens ) context ))]
[( string->number tokens ) tokens ]
[( procedure? tokens ) tokens ]
[ else
( resolve-recursive ( assoc-ref context tokens context ) context )]))
( define ( compose-with-context fns context )
( apply compose
( map ( lambda ( token )
( let ([ evaluated ( evaluate token context )])
( if ( number? evaluated )
( make-stack-appender evaluated )
evaluated )))
( reverse fns ))))
( define ( evaluate token context )
"\"Convert parsed token in given context.\n\n   When token is a list - it's definition and context will be modified.\n   When token is number-like, it's converted to number.\n   When token is a symbol-like, it's evaluated in the given context.\n   If everything fails, error raised.\""
( cond
[( pair? token )
( when ( string->number ( car token ))
( error 'evaluate "\"cannot redefine numbers\"" ))
( let ([ composed ( compose-with-context ( cdr token ) context )])
( values #f
( assoc-set! context ( car token ) composed )))]
[( string->number token )
( values ( string->number token ) context )]
[( assoc token context )
( values ( assoc-ref context token ) context )]
[ else
( error 'evaluate ( format #f "\"Unknown token: ~a\"" token ))]))
( define ( make-stack-appender value )
( lambda ( stack )
( cons value stack )))
( define ( f-plus stack )
( cons ( + ( cadr stack ) ( car stack )) ( cddr stack )))
( define ( f-minus stack )
( cons ( - ( cadr stack ) ( car stack )) ( cddr stack )))
( define ( f-multiply stack )
( cons ( * ( cadr stack ) ( car stack )) ( cddr stack )))
( define ( f-divide stack )
( cons ( floor/ ( cadr stack ) ( car stack )) ( cddr stack )))
( define ( f-dup stack )
( cons ( car stack ) stack ))
( define ( f-drop stack )
( cdr stack ))
( define ( f-swap stack )
( cons ( cadr stack ) ( cons ( car stack ) ( cddr stack ))))
( define ( f-over stack )
( cons ( cadr stack ) stack ))
( define ( make-default-context )
` ([ "\"+\"" . , f-plus ]
[ "\"-\"" . , f-minus ]
[ "\"*\"" . , f-multiply ]
[ "\"/\"" . , f-divide ]
[ "\"dup\"" . , f-dup ]
[ "\"drop\"" . , f-drop ]
[ "\"swap\"" . , f-swap ]
[ "\"over\"" . , f-over ]))
( define ( forth program )
( let* ([ source ( string-split ( string-downcase ( string-join program "\" \"" )) #\Space )]
[ preprocessed-program ( tokenize source )])
( let loop ([ program preprocessed-program ]
[ context ( make-default-context )]
[ stack ' ()])
( if ( null? program )
stack
( receive ( token context )
( evaluate ( car program ) context )
( cond
[( number? token )
( loop ( cdr program ) context ( cons token stack ))]
[( procedure? token )
( loop ( cdr program )
context
( token stack ))]
[ else
( loop ( cdr program ) context stack )]))))))
