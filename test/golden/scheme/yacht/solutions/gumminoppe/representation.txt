( import ( rnrs ))
( define ( histogram dice )
( let (( hist ( make-vector 7 0 )))
( for-each ( lambda ( n ) ( vector-set! hist n ( 1+ ( vector-ref hist n )))) dice )
( vector->list hist )))
( define ( score category dice )
( let* (( dice ( vector->list dice ))
( hist ( histogram dice )))
( cond (( eq? category 'ones ) ( list-ref hist 1 ))
(( eq? category 'twos ) ( * 2 ( list-ref hist 2 )))
(( eq? category 'threes ) ( * 3 ( list-ref hist 3 )))
(( eq? category 'fours ) ( * 4 ( list-ref hist 4 )))
(( eq? category 'fives ) ( * 5 ( list-ref hist 5 )))
(( eq? category 'sixes ) ( * 6 ( list-ref hist 6 )))
(( eq? category 'full-house )
( if ( and ( member 2 hist ) ( member 3 hist )) ( apply + dice ) 0 ))
(( eq? category 'four-of-a-kind )
( let (( tail ( memp ( lambda ( n ) ( <= 4 n )) hist )))
( if tail ( * 4 ( - 7 ( length tail ))) 0 )))
(( eq? category 'little-straight )
( if ( equal? ( cdr hist ) ' ( 1 1 1 1 1 0 )) 30 0 ))
(( eq? category 'big-straight )
( if ( equal? ( cdr hist ) ' ( 0 1 1 1 1 1 )) 30 0 ))
(( eq? category 'choice ) ( apply + dice ))
(( eq? category 'yacht ) ( if ( member 5 hist ) 50 0 ))
( else 0 ))))
