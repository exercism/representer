CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
data ( inp ) = input .
data ( divisor ) = 2 .
while inp > 1 .
while inp mod divisor <> 0 .
divisor = divisor + 1 .
endwhile .
append divisor to result .
inp = inp div divisor .
endwhile .
ENDMETHOD .
ENDCLASS .
