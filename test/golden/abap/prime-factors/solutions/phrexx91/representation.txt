(file "zcl_prime_factors.clas.abap")
CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
DATA ( num ) = input .
DATA ( factor ) = 2 .
WHILE factor <= num .
IF num MOD factor = 0 .
num = num / factor .
INSERT factor INTO TABLE result .
factor = 2 .
CONTINUE .
ENDIF .
factor += 1 .
ENDWHILE .
ENDMETHOD .
ENDCLASS .
