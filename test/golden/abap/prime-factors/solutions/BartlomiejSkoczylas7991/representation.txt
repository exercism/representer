CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
DATA : over TYPE abap_bool VALUE abap_false ,
lv_input TYPE int8 VALUE input ,
lv_result TYPE integertab ,
incr TYPE int8 VALUE 2 .
WHILE lv_input > 1 .
IF lv_input MOD incr = 0 .
APPEND incr TO lv_result .
lv_input = lv_input / incr .
ELSE .
incr = incr + 1 .
ENDIF .
IF lv_input = 1 .
over = abap_true .
ENDIF .
ENDWHILE .
result = lv_result .
ENDMETHOD .
ENDCLASS .
