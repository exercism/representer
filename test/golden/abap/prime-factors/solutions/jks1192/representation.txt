CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
data : i type i .
while input gt 1 .
i = 2 .
while i le input .
if input mod i eq 0 .
append i to result .
input = input / i .
exit .
endif .
i = i + 1 .
endwhile .
endwhile .
ENDMETHOD .
ENDCLASS .
