CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
DATA ( remainder ) = input .
DATA ( factor ) = 2 .
WHILE remainder > 1 .
IF remainder MOD factor = 0 .
remainder = remainder / factor .
APPEND factor TO result .
ELSE .
factor = factor + 1 .
ENDIF .
endwhile .
ENDMETHOD .
ENDCLASS .
