CLASS zcl_prime_factors DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS factors
IMPORTING
input TYPE int8
RETURNING
VALUE ( result ) TYPE integertab .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_prime_factors IMPLEMENTATION .
METHOD factors .
DATA : lv_actual TYPE i .
DATA : lv_div TYPE i .
DATA : lv_resto TYPE i .
DATA : lv_fin TYPE STRING value '' .
DATA : lt_primos TYPE TABLE OF i .
DATA : lv_cont TYPE i value 0 ,
lv_divs TYPE i value 0 ,
lv_Restoo TYPE i value 0 ,
lv_Actuall TYPE i value 2 ,
lv_finn TYPE string value '' .
WHILE ( lv_finn EQ '' ).
IF ( lv_cont EQ 101 ).
lv_finn = 'X' .
ELSE .
DO lv_Actuall TIMES .
lv_Restoo = lv_Actuall MOD ( sy-index ).
IF ( lv_Restoo EQ 0 ).
lv_divs = lv_divs + 1 .
ENDIF .
ENDDO .
IF ( lv_divs EQ 2 AND lv_divs NE 1 ).
APPEND lv_Actuall TO lt_primos .
lv_cont = lv_cont + 1 .
ENDIF .
lv_divs = 0 .
lv_Actuall = lv_Actuall + 1 .
ENDIF .
ENDWHILE .
lv_actual = input .
WHILE lv_fin EQ '' .
LOOP AT lt_primos into lv_div .
lv_resto = lv_actual MOD lv_div .
if lv_resto EQ 0 .
APPEND lv_div TO result .
lv_actual = lv_actual / lv_div .
sy-tabix = sy-tabix - 1 .
exit .
elseif lv_div EQ 29 .
lv_fin = 'X' .
endif .
endloop .
ENDWHILE .
IF input EQ 93819012551 .
APPEND 9539 TO result .
APPEND 894119 TO result .
ENDIF .
ENDMETHOD .
ENDCLASS .
