(file "zcl_crypto_square.clas.abap")
CLASS zcl_crypto_square DEFINITION PUBLIC .
PUBLIC SECTION .
METHODS encode
IMPORTING
plain_text TYPE string
RETURNING
VALUE ( crypto_text ) TYPE string .
PRIVATE SECTION .
TYPES :
BEGIN OF rank_type ,
rows TYPE i ,
cols TYPE i ,
END OF rank_type .
TYPES :
integertab TYPE STANDARD TABLE OF i
WITH EMPTY KEY .
METHODS sanitize
IMPORTING
text TYPE string
RETURNING
VALUE ( result ) TYPE string .
METHODS extract_candidates
IMPORTING
n TYPE i
RETURNING
VALUE ( candidates ) TYPE integertab .
METHODS optimal_rank
IMPORTING
length TYPE i
RETURNING
VALUE ( rank ) TYPE rank_type .
METHODS matricize
IMPORTING
text TYPE string
rows TYPE i
cols TYPE i
RETURNING
VALUE ( result ) TYPE string .
ENDCLASS .
CLASS zcl_crypto_square IMPLEMENTATION .
METHOD encode .
DATA ( lv_text ) = sanitize ( plain_text ).
DATA ( lv_text_len ) = STRLEN ( lv_text ).
IF lv_text_len LT 1 .
crypto_text = '' .
ELSEIF lv_text_len EQ 1 .
crypto_text = lv_text .
ELSE .
DATA ( lv_optimal_rank ) = optimal_rank ( lv_text_len ).
crypto_text = matricize ( text = lv_text
rows = lv_optimal_rank-rows
cols = lv_optimal_rank-cols ).
ENDIF .
ENDMETHOD .
METHOD sanitize .
IF STRLEN ( text ) LT 1 .
result = '' . RETURN .
ENDIF .
result = text .
REPLACE ALL OCCURRENCES OF REGEX |[,<>!&@#$%^&*()_+=':;/.?]|
IN result WITH || .
REPLACE ALL OCCURRENCES OF '-' IN result WITH '' .
CONDENSE result NO-GAPS .
IF STRLEN ( result ) LT 1 .
RETURN .
ENDIF .
TRANSLATE result TO LOWER CASE .
ENDMETHOD .
METHOD extract_candidates .
DATA :
lt_factors TYPE integertab ,
lv_i TYPE i VALUE 2 .
WHILE lv_i LT n .
IF n MOD lv_i EQ 0 .
APPEND lv_i TO lt_factors .
ENDIF .
lv_i = lv_i + 1 .
ENDWHILE .
DATA :
lt_rows_cols TYPE integertab ,
lv_idx TYPE i VALUE 1 ,
lv_divisor TYPE i ,
lv_dividend TYPE i .
WHILE lv_idx LE LINES ( lt_factors ).
lv_divisor = lt_factors [ lv_idx ].
lv_dividend = n DIV lv_divisor .
IF ABS ( lv_divisor - lv_dividend ) LT 2 .
APPEND lv_divisor TO lt_rows_cols .
APPEND lv_dividend TO lt_rows_cols .
ENDIF .
lv_idx = lv_idx + 1 .
ENDWHILE .
SORT lt_rows_cols .
DELETE ADJACENT DUPLICATES FROM lt_rows_cols .
candidates = lt_rows_cols .
ENDMETHOD .
METHOD optimal_rank .
DATA ( lv_lowerbound ) = length .
DATA ( lv_default_RC ) = CEIL ( SQRT ( length ) ).
DATA ( lv_upperbound ) = IPOW ( base = lv_default_RC exp = 2 ).
WHILE lv_lowerbound LE lv_upperbound .
DATA ( lt_candidates ) = extract_candidates ( lv_lowerbound ).
DATA ( lv_candidates_len ) = LINES ( lt_candidates ).
IF lv_candidates_len EQ 1 .
rank-rows = lt_candidates [ 1 ]. rank-cols = lt_candidates [ 1 ].
RETURN .
ELSEIF lv_candidates_len EQ 2 .
IF lt_candidates [ 1 ] LE lt_candidates [ 2 ].
rank-rows = lt_candidates [ 1 ]. rank-cols = lt_candidates [ 2 ].
ELSE .
rank-rows = lt_candidates [ 2 ]. rank-cols = lt_candidates [ 1 ].
ENDIF .
RETURN .
ENDIF .
lv_lowerbound = lv_lowerbound + 1 .
ENDWHILE .
rank-rows = lv_default_RC . rank-cols = lv_default_RC .
ENDMETHOD .
METHOD matricize .
CONSTANTS :
lv_fillchar TYPE string VALUE 'X' .
DATA :
lv_text_len TYPE i ,
lv_tbl_len TYPE i ,
lv_i TYPE i ,
lv_row TYPE i ,
lv_col TYPE i ,
lv_j TYPE i .
lv_text_len = STRLEN ( text ).
lv_tbl_len = rows * cols .
DATA :
ls_tbl TYPE string .
lv_i = 0 .
WHILE lv_i LT lv_tbl_len .
ls_tbl = ls_tbl && lv_fillchar .
lv_i = lv_i + 1 .
ENDWHILE .
CONDENSE ls_tbl NO-GAPS .
lv_i = 0 .
WHILE lv_i LT lv_text_len .
IF lv_i MOD cols EQ 0 .
lv_col = 0 .
lv_row = lv_i DIV cols .
ENDIF .
lv_j = lv_col * rows + lv_row .
ls_tbl + lv_j ( 1 ) = text + lv_i ( 1 ).
lv_col = lv_col + 1 .
lv_i = lv_i + 1 .
ENDWHILE .
lv_j = 0 .
DO cols TIMES .
result = result && ls_tbl + lv_j ( rows ) && "` `" .
lv_j = lv_j + rows .
ENDDO .
CONDENSE result .
REPLACE ALL OCCURRENCES OF lv_fillchar IN result
WITH SPACE .
ENDMETHOD .
ENDCLASS .
