CLASS zcl_crypto_square DEFINITION .
PUBLIC SECTION .
METHODS encode IMPORTING plain_text TYPE string
RETURNING VALUE ( crypto_text ) TYPE string .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_crypto_square IMPLEMENTATION .
METHOD encode .
DATA : string TYPE string ,
columns TYPE i VALUE 1 ,
rows TYPE i VALUE 1 ,
x TYPE i ,
y TYPE i VALUE 0 ,
itab TYPE TABLE OF string ,
wa TYPE string ,
result TYPE string ,
table_entry TYPE string ,
letter TYPE string ,
counter1 TYPE i VALUE 1 ,
counter2 TYPE i VALUE 0 .
string = plain_text .
REPLACE ALL OCCURRENCES OF REGEX '[^\w]+' IN string WITH "| |" .
CONDENSE string NO-GAPS .
TRANSLATE string TO LOWER CASE .
DATA ( len ) = STRLEN ( string ).
IF len <> 0 .
WHILE rows * columns < len .
IF columns = rows .
columns = columns + 1 .
ELSE .
rows = rows + 1 .
ENDIF .
ENDWHILE .
WHILE len <> rows * columns .
string = string && "|  |" .
len = len + 1 .
ENDWHILE .
DO rows TIMES .
APPEND string + y ( columns ) TO itab .
y = y + columns .
ENDDO .
DO columns TIMES .
DO rows TIMES .
table_entry = itab [ counter1 ].
letter = table_entry + counter2 ( 1 ).
result = result && letter .
counter1 = counter1 + 1 .
ENDDO .
counter1 = 1 .
counter2 = counter2 + 1 .
result = result && "| |" .
ENDDO .
IF rows = 1 .
CONDENSE result NO-GAPS .
ELSE .
len = STRLEN ( result ).
len = len - 1 .
result = result + 0 ( len ).
ENDIF .
crypto_text = result .
WRITE : / crypto_text .
ELSE .
crypto_text = '' .
ENDIF .
ENDMETHOD .
ENDCLASS .
