CLASS zcl_phone_number DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS clean
IMPORTING
! number TYPE string
RETURNING
VALUE ( result ) TYPE string
RAISING
cx_parameter_invalid .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_phone_number IMPLEMENTATION .
METHOD clean .
DATA : v_syindex TYPE sy-index .
DATA ( v_temp ) = number .
CONDENSE v_temp NO-GAPS .
TRY .
DO STRLEN ( v_temp ) TIMES .
v_syindex = sy-index - 1 .
CHECK v_temp + v_syindex ( 1 ) CN '0123456789' .
REPLACE ALL OCCURRENCES OF v_temp + v_syindex ( 1 ) IN v_temp WITH '' .
ENDDO .
CATCH cx_root .
ENDTRY .
IF v_temp + 0 ( 1 ) EQ '1' .
REPLACE FIRST OCCURRENCE OF v_temp + 0 ( 1 ) IN v_temp WITH '' .
ENDIF .
IF strlen ( v_temp ) NE 10 OR
v_temp + 0 ( 1 ) CO '01' OR
v_temp + 3 ( 1 ) CO '01' .
RAISE EXCEPTION TYPE cx_parameter_invalid .
ELSE .
result = v_temp .
ENDIF .
ENDMETHOD .
ENDCLASS .
