(file "zcl_anagram.clas.abap")
CLASS zcl_anagram DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS anagram
IMPORTING
input TYPE string
candidates TYPE string_table
RETURNING
VALUE ( result ) TYPE string_table .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_anagram IMPLEMENTATION .
METHOD anagram .
DATA : lv_flag TYPE c ,
lt_source TYPE string_table ,
lt_actual TYPE string_table .
LOOP AT candidates INTO DATA ( ls_candidate ).
CHECK to_lower ( ls_candidate ) <> to_lower ( input ).
IF lv_flag = '' .
DO strlen ( input ) TIMES .
DATA ( lv_index ) = sy-index - 1 .
APPEND to_lower ( input + lv_index ( 1 ) ) TO lt_source .
ENDDO .
SORT lt_source .
ENDIF .
CLEAR lv_index .
DO strlen ( ls_candidate ) TIMES .
lv_index = sy-index - 1 .
APPEND to_lower ( ls_candidate + lv_index ( 1 ) ) TO lt_actual .
ENDDO .
SORT lt_actual .
IF lt_source = lt_actual .
APPEND ls_candidate TO result .
ENDIF .
CLEAR lt_actual .
lv_flag = abap_true .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
