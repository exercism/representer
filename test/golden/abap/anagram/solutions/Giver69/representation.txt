CLASS zcl_anagram DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS anagram
IMPORTING
input TYPE string
candidates TYPE string_table
RETURNING
VALUE ( result ) TYPE string_table .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_anagram IMPLEMENTATION .
METHOD anagram .
DATA : lt_candidates_aux TYPE string_table .
DATA ( lv_long_input ) = strlen ( input ).
DATA ( input_aux ) = input .
TRANSLATE input_aux TO UPPER CASE .
LOOP AT candidates INTO DATA ( wa_candidates ).
DATA ( wa_candidates_aux ) = wa_candidates .
TRANSLATE wa_candidates_aux TO UPPER CASE .
IF lv_long_input = strlen ( wa_candidates ) AND wa_candidates_aux <> input_aux .
DATA ( lv_pos_letra ) = 0 .
DO lv_long_input TIMES .
DATA ( lv_abuscar_input ) = substring ( val = wa_candidates_aux off = lv_pos_letra len = 1 ).
FIND ALL OCCURRENCES OF lv_abuscar_input IN input_aux MATCH COUNT DATA ( lv_cnt_input ).
FIND ALL OCCURRENCES OF lv_abuscar_input IN wa_candidates_aux MATCH COUNT DATA ( lv_cnt_candidate ).
IF lv_cnt_input NE lv_cnt_candidate .
EXIT .
ENDIF .
IF SY-INDEX EQ lv_long_input .
INSERT wa_candidates INTO result .
ENDIF .
lv_pos_letra = lv_pos_letra + 1 .
ENDDO .
ENDIF .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
