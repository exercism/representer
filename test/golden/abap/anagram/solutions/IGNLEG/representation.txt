(file "zcl_anagram.clas.abap")
CLASS zcl_anagram DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS anagram
IMPORTING
input TYPE string
candidates TYPE string_table
RETURNING
VALUE ( result ) TYPE string_table .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_anagram IMPLEMENTATION .
METHOD anagram .
LOOP AT candidates INTO DATA ( candidate ).
DATA ( lv_cur_pos ) = 0 .
DATA ( lv_flag ) = 1 .
IF strlen ( input ) = strlen ( candidate ) AND to_lower ( candidate ) <> to_lower ( input ).
DO strlen ( candidate ) TIMES .
FIND ALL OCCURRENCES OF to_lower ( candidate + lv_cur_pos ( 1 ) ) in to_lower ( candidate ) MATCH COUNT DATA ( lcnt ).
FIND ALL OCCURRENCES OF to_lower ( candidate + lv_cur_pos ( 1 ) ) in to_lower ( input ) MATCH COUNT DATA ( rcnt ).
IF lcnt <> rcnt .
lv_flag = 0 .
EXIT .
ENDIF .
add 1 to lv_cur_pos .
ENDDO .
if lv_flag = 1 .
APPEND candidate TO result .
endif .
ENDIF .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
