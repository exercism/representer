CLASS zcl_anagram DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS anagram
IMPORTING
input TYPE string
candidates TYPE string_table
RETURNING
VALUE ( result ) TYPE string_table .
PROTECTED SECTION .
PRIVATE SECTION .
types :
begin of ty_attribute ,
character type string ,
number_of_occorance type i ,
end of ty_attribute .
types ty_attribute_tab type sorted table of ty_attribute with unique key character .
methods extract_string_component importing input type string
returning value ( attr_tab ) type ty_attribute_tab .
ENDCLASS .
CLASS zcl_anagram IMPLEMENTATION .
METHOD anagram .
data ( input_attr ) = me -> extract_string_component ( input ).
LOOP AT candidates ASSIGNING FIELD-SYMBOL ( < candidate > ).
check to_lower ( input ) <> to_lower ( < candidate > ).
DATA ( candidate_attr ) = me -> extract_string_component ( < candidate > ).
IF me -> extract_string_component ( < candidate > ) = input_attr .
APPEND < candidate > TO result .
ENDIF .
ENDLOOP .
ENDMETHOD .
METHOD extract_string_component .
TYPES :
BEGIN OF ty_char ,
character TYPE c LENGTH 1 ,
END OF ty_char .
TYPES ty_char_tab TYPE SORTED TABLE OF ty_char WITH NON-UNIQUE KEY character .
DATA lt_chars TYPE ty_char_tab .
lt_chars = VALUE ty_char_tab (
FOR j = 0 THEN j + 1 WHILE j < strlen ( input )
( character = to_lower ( input + j ( 1 ) ) )
).
LOOP AT lt_chars ASSIGNING FIELD-SYMBOL ( < char > )
GROUP BY ( character = < char >- character
size = GROUP SIZE ) ASCENDING
WITHOUT MEMBERS
ASSIGNING FIELD-SYMBOL ( < key > ).
APPEND VALUE # ( character = < key >- character number_of_occorance = < key >- size ) TO attr_tab .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
