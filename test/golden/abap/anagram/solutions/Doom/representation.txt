CLASS zcl_anagram IMPLEMENTATION .
METHOD anagram .
LOOP AT candidates INTO DATA ( candidate ).
CHECK strlen ( candidate ) = strlen ( input )
AND to_lower ( candidate ) <> to_lower ( input ).
DATA ( tmp_input ) = to_lower ( input ).
DO strlen ( input ) TIMES .
DATA ( str_index ) = sy-index - 1 .
IF find ( val = tmp_input sub = to_lower ( candidate + str_index ( 1 ) ) ) <> - 1 .
REPLACE to_lower ( candidate + str_index ( 1 ) ) IN tmp_input WITH '' .
ENDIF .
ENDDO .
IF ( tmp_input = '' ).
APPEND candidate TO result .
ENDIF .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
