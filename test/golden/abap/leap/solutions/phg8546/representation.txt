CLASS zcl_leap DEFINITION PUBLIC .
PUBLIC SECTION .
METHODS leap
IMPORTING
year TYPE i
RETURNING
VALUE ( result ) TYPE abap_bool .
ENDCLASS .
CLASS zcl_leap IMPLEMENTATION .
METHOD leap .
result = abap_false .
IF year MOD 4 = 0 .
IF year MOD 100 <> 0 .
result = abap_true .
EXIT .
ENDIF .
ENDIF .
IF year MOD 100 = 0 .
IF year MOD 400 = 0 .
result = abap_true .
ENDIF .
ENDIF .
ENDMETHOD .
ENDCLASS .
