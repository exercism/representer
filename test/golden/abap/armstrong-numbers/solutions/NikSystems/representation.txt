CLASS zcl_armstrong_numbers IMPLEMENTATION .
METHOD is_armstrong_number .
DATA numb TYPE i .
DATA exponent TYPE i .
DATA narc TYPE i .
numb = num .
exponent = strlen ( CONV string ( num ) ) - 1 .
WHILE numb IS NOT INITIAL .
narc = narc + ( numb MOD 10 ) ** exponent .
numb = round ( val = ( numb / 10 ) dec = 0 mode = 5 ).
ENDWHILE .
result = COND # ( WHEN narc = num THEN abap_true ELSE abap_false ).
ENDMETHOD .
ENDCLASS .
