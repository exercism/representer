CLASS zcl_raindrops DEFINITION PUBLIC .
PUBLIC SECTION .
METHODS raindrops
IMPORTING
input TYPE i
RETURNING
VALUE ( result ) TYPE string .
ENDCLASS .
CLASS zcl_raindrops IMPLEMENTATION .
METHOD raindrops .
CLEAR result .
IF input MOD 3 = 0 .
CONCATENATE result 'Pling' INTO result .
ENDIF .
IF input MOD 5 = 0 .
CONCATENATE result 'Plang' INTO result .
ENDIF .
IF input MOD 7 = 0 .
CONCATENATE result 'Plong' INTO result .
ENDIF .
IF result IS INITIAL .
result = input .
ENDIF .
ENDMETHOD .
ENDCLASS .
