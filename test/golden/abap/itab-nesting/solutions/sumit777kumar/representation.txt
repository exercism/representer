CLASS zcl_itab_nesting IMPLEMENTATION .
METHOD perform_nesting .
DATA w_album_nested type album_song_nested_type .
DATA w_nested_data TYPE LINE OF nested_data .
LOOP AT artists INTO DATA ( w_artist ).
w_nested_data-artist_id = w_artist-artist_id .
w_nested_data-artist_name = w_artist-artist_name .
w_nested_data-albums = VALUE # ( FOR w_album IN albums
WHERE ( artist_id = w_artist-artist_id )
( album_id = w_album-album_id
album_name = w_album-album_name
songs = VALUE # ( FOR w_song IN songs
WHERE ( artist_id = w_artist-artist_id AND
album_id = w_album-album_id )
( song_id = w_song-song_id
song_name = w_song-song_name
) ) ) ).
APPEND w_nested_data TO nested_data .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
