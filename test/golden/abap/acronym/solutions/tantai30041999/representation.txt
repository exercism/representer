(file "zcl_acronym.clas.abap")
CLASS zcl_acronym DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS parse IMPORTING phrase TYPE string
RETURNING VALUE ( acronym ) TYPE string .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_acronym IMPLEMENTATION .
METHOD parse .
DATA : lv_str TYPE string .
lv_str = phrase .
REPLACE ALL OCCURRENCES OF '-' IN lv_str WITH "' _'" .
REPLACE ALL OCCURRENCES OF '_' IN lv_str WITH "' '" .
SPLIT lv_str AT "' '" INTO TABLE DATA ( lt_str ).
LOOP AT lt_str ASSIGNING FIELD-SYMBOL ( < fs_str > ).
acronym = acronym && to_upper ( < fs_str >+ 0 ( 1 ) ).
ENDLOOP .
CONDENSE acronym .
ENDMETHOD .
ENDCLASS .
