(file "zcl_acronym.clas.abap")
CLASS zcl_acronym DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS parse IMPORTING phrase TYPE string
RETURNING VALUE ( acronym ) TYPE string .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_acronym IMPLEMENTATION .
METHOD parse .
DATA ( lv_phrase ) = phrase .
REPLACE ALL OCCURENCES OF REGEX ',|-|_' IN lv_phrase WITH "' '" .
SPLIT lv_phrase AT "' '" INTO TABLE DATA ( lt_splitted_phrase ).
DELETE lt_splitted_phrase WHERE table_line = "' '" .
LOOP AT lt_splitted_phrase REFERENCE INTO DATA ( lr_phrase_word ).
acronym = "|{ acronym }{ to_upper( substring( val = lr_phrase_word->* len = 1 ) ) }|" .
ENDLOOP .
ENDMETHOD .
ENDCLASS .
