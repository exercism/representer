CLASS zcl_binary_search DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
TYPES integertab TYPE STANDARD TABLE OF i WITH EMPTY KEY .
METHODS binary_search
IMPORTING
val TYPE i
table TYPE integertab
RETURNING
VALUE ( result ) TYPE i
RAISING
cx_sy_itab_line_not_found .
PROTECTED SECTION .
PRIVATE SECTION .
ENDCLASS .
CLASS zcl_binary_search IMPLEMENTATION .
METHOD binary_search .
DATA ( lt_sorted_table ) = table .
SORT lt_sorted_table .
DATA ( low ) = 1 .
DATA ( high ) = lines ( lt_sorted_table ).
WHILE low <= high .
DATA ( middle ) = ( low + high ) DIV 2 .
IF lt_sorted_table [ middle ] = val .
result = middle .
RETURN .
ELSEIF lt_sorted_table [ middle ] < val .
low = middle + 1 .
ELSE .
high = middle - 1 .
ENDIF .
ENDWHILE .
IF low > high OR high = 0 .
RAISE EXCEPTION TYPE cx_sy_itab_line_not_found .
ENDIF .
ENDMETHOD .
ENDCLASS .
