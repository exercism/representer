(file "zcl_two_fer.clas.abap")
CLASS zcl_two_fer DEFINITION PUBLIC .
PUBLIC SECTION .
METHODS two_fer
IMPORTING
input TYPE string OPTIONAL
RETURNING
VALUE ( result ) TYPE string .
ENDCLASS .
CLASS zcl_two_fer IMPLEMENTATION .
METHOD two_fer .
result = cond # (
when input is not initial
then "|One for { input }, one for me.|"
when input is initial
then "|One for you, one for me.|" ).
ENDMETHOD .
ENDCLASS .
