CLASS zcl_rle DEFINITION
PUBLIC
FINAL
CREATE PUBLIC .
PUBLIC SECTION .
METHODS encode IMPORTING input TYPE string
RETURNING VALUE ( result ) TYPE string .
METHODS decode IMPORTING input TYPE string
RETURNING VALUE ( result ) TYPE string .
ENDCLASS .
CLASS zcl_rle IMPLEMENTATION .
METHOD encode .
DATA ( offset ) = 1 .
DATA ( count ) = 1 .
DATA ( chr ) = input ( 1 ).
WHILE offset < strlen ( input ).
IF chr = input + offset ( 1 ).
count = count + 1 .
ELSE .
IF count > 1 .
result = "|{ result }{ count }{ chr }|" .
ELSE .
result = "|{ result }{ chr }|" .
ENDIF .
count = 1 .
ENDIF .
chr = input + offset ( 1 ).
offset = offset + 1 .
ENDWHILE .
IF count > 1 .
result = "|{ result }{ count }{ chr }|" .
ELSE .
result = "|{ result }{ chr }|" .
ENDIF .
ENDMETHOD .
METHOD decode .
DO .
IF input IS INITIAL .
EXIT .
ENDIF .
FIND REGEX "'^([0-9]+)([A-Za-z ])'" IN input SUBMATCHES DATA ( len ) DATA ( chr ).
IF sy-subrc = 0 .
DO len TIMES .
result = result && chr .
ENDDO .
SHIFT input LEFT BY ( strlen ( len ) + 1 ) PLACES .
ELSE .
result = result && input ( 1 ).
SHIFT input LEFT .
ENDIF .
ENDDO .
ENDMETHOD .
ENDCLASS .
