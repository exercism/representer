(file "resistor-color-duo.awk")
BEGIN {
color [ "\"black\"" ] = "\"0\""
color [ "\"brown\"" ] = "\"1\""
color [ "\"red\"" ] = "\"2\""
color [ "\"orange\"" ] = "\"3\""
color [ "\"yellow\"" ] = "\"4\""
color [ "\"green\"" ] = "\"5\""
color [ "\"blue\"" ] = "\"6\""
color [ "\"violet\"" ] = "\"7\""
color [ "\"grey\"" ] = "\"8\""
color [ "\"white\"" ] = "\"9\""
}
{
if ( $1 in color ) {
if ( color [ $1 ] == "\"0\"" ) {
print color [ $2 ]
} else {
print color [ $1 ] color [ $2 ]
}
} else {
print "\"invalid color\""
exit 1
}
}
