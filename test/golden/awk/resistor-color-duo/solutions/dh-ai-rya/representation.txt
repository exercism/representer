(file "resistor-color-duo.awk")
BEGIN {
res = "\"\""
t = 0
}
{
for ( i = 1 ; i <= 2 ; i ++ ){
if ( $i == "\"black\"" ){
if ( res != "\"\"" ){
res = res "\"0\""
}
}
else if ( $i == "\"brown\"" ){
res = res "\"1\""
}
else if ( $i == "\"red\"" ){
res = res "\"2\""
}
else if ( $i == "\"orange\"" ){
res = res "\"3\""
}
else if ( $i == "\"yellow\"" ){
res = res "\"4\""
}
else if ( $i == "\"green\"" ){
res = res "\"5\""
}
else if ( $i == "\"blue\"" ){
res = res "\"6\""
}
else if ( $i == "\"violet\"" ){
res = res "\"7\""
}
else if ( $i == "\"grey\"" ){
res = res "\"8\""
}
else if ( $i == "\"white\"" ){
res = res "\"9\""
}
else {
t ++
}
}
}
END {
if ( t > 0 ){
print ( "\"invalid color\"" )
exit ( 1 )
}
else {
print ( res )
}
}
