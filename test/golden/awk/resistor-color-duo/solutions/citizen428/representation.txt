BEGIN {
bands [ "\"black\"" ] = 0
bands [ "\"brown\"" ] = 1
bands [ "\"red\"" ] = 2
bands [ "\"orange\"" ] = 3
bands [ "\"yellow\"" ] = 4
bands [ "\"green\"" ] = 5
bands [ "\"blue\"" ] = 6
bands [ "\"violet\"" ] = 7
bands [ "\"grey\"" ] = 8
bands [ "\"white\"" ] = 9
}
{
print resistance ( $1 ) * 10 + resistance ( $2 )
}
function resistance ( color ) {
if ( color in bands )
return bands [ color ]
print ( "\"invalid color: \"" , color )
exit ( 1 )
}
