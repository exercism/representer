BEGIN {
colors [ "\"black\"" ] = 0 ;
colors [ "\"brown\"" ] = 1 ;
colors [ "\"red\"" ] = 2 ;
colors [ "\"orange\"" ] = 3 ;
colors [ "\"yellow\"" ] = 4 ;
colors [ "\"green\"" ] = 5 ;
colors [ "\"blue\"" ] = 6 ;
colors [ "\"violet\"" ] = 7 ;
colors [ "\"grey\"" ] = 8 ;
colors [ "\"white\"" ] = 9 ;
}
{
if ( ! ( $1 in colors ) || ( ! $2 in colors )) {
print "\"invalid color\"" ;
exit 1 ;
}
color1 = colors [ $1 ];
color2 = colors [ $2 ];
print ( color1 * 10 + color2 );
}
