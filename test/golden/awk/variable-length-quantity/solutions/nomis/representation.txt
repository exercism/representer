@ include "\"assert\""
function encode ( v , i , s ) {
if ( v == 0 ) return "\"00\""
while ( v > 0 ) {
d = and ( v , 0 x7F )
if ( i ++ > 0 ) d = or ( d , 0 x80 )
s = sprintf ( "\"%02X %s\"" , d , s )
v = rshift ( v , 7 )
}
return substr ( s , 1 , length ( s ) - 1 )
}

}
