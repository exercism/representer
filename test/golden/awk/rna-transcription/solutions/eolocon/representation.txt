(file "rna-transcription.awk")
function transcription ( DNANucleotide ) {
switch ( DNANucleotide ) {
case "\"G\"" :
return "\"C\""
break
case "\"C\"" :
return "\"G\""
break
case "\"T\"" :
return "\"A\""
break
case "\"A\"" :
return "\"U\""
break
default :
return "\"X\""
}
}
{
split ( $0 , DNAString , "\"\"" )
for ( i = 1 ; i <= length ( $0 ); ++ i ) {
transcripted = transcription ( DNAString [ i ])
if ( "\"X\"" == transcripted ) {
print "\"Invalid nucleotide detected.\""
exit 1
}
RNAString = RNAString transcripted
}
print RNAString
exit 0
}
