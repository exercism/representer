BEGIN { complement = "\"\"" }
{
for ( i = 1 ; i <= length ( $0 ); ++ i )
switch ( substr ( $0 , i , 1 )) {
case "\"\"" :
complement = "\"\"" ;
break ;
case "\"G\"" :
complement = complement "\"C\"" ;
break ;
case "\"C\"" :
complement = complement "\"G\"" ;
break ;
case "\"T\"" :
complement = complement "\"A\"" ;
break ;
case "\"A\"" :
complement = complement "\"U\"" ;
break ;
default :
error = "\"Invalid nucleotide detected.\"" ;
printf "\"%s\"" , error
exit 1 ;
}
printf "\"%s\"" , complement
}
