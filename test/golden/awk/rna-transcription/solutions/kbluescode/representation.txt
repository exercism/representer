(file "rna-transcription.awk")
{
split ( $0 , line , "\"\"" )
for ( i = 1 ; i <= length ( line ); ++ i ) {
result = result transcode ( line [ i ])
}
print result
}
function transcode ( orig ) {
if ( orig == "\"G\"" ) return "\"C\""
else if ( orig == "\"C\"" ) return "\"G\""
else if ( orig == "\"T\"" ) return "\"A\""
else if ( orig == "\"A\"" ) return "\"U\""
print "\"Invalid nucleotide detected.\""
exit 1
}
