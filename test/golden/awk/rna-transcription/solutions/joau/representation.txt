(file "rna-transcription.awk")
BEGIN {
FS = "\"\""
OFS = "\"\""
} END {
for ( i = 1 ; i <= NF ; i ++ ) {
if ( $i == "\"G\"" ) {
$i = "\"C\""
} else if ( $i == "\"C\"" ) {
$i = "\"G\""
} else if ( $i == "\"T\"" ) {
$i = "\"A\""
} else if ( $i == "\"A\"" ) {
$i = "\"U\""
} else {
print "\"Invalid nucleotide detected.\""
exit 1
}
}
print $0
exit 0
}
