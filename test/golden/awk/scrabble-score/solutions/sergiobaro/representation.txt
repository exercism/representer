(file "scrabble-score.awk")
function populate_score_table ( table ) {
score_table [ 1 ] = "\"A,E,I,O,U,L,N,R,S,T\""
score_table [ 2 ] = "\"D,G\""
score_table [ 3 ] = "\"B,C,M,P\""
score_table [ 4 ] = "\"F,H,V,W,Y\""
score_table [ 5 ] = "\"K\""
score_table [ 8 ] = "\"J,X\""
score_table [ 10 ] = "\"Q,Z\""
for ( score in score_table ) {
split ( score_table [ score ], letters , "\",\"" )
for ( letter_index in letters ) {
letter = letters [ letter_index ]
table [ letter ] = score
}
}
}
function calculate_score ( word ) {
populate_score_table ( table )
total_score = 0
split ( word , letters , "\"\"" )
for ( letter_index in letters ) {
letter = letters [ letter_index ]
letter_score = table [ letter ]
total_score += letter_score
}
return total_score
}
END {
word = toupper ( $0 )
score = calculate_score ( word )
printf "\"%s,%d\"" , word , score
}
