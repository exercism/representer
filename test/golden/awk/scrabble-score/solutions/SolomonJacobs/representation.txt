(file "scrabble-score.awk")
BEGIN {
score [ "\"A\"" ] = 1
score [ "\"E\"" ] = 1
score [ "\"I\"" ] = 1
score [ "\"O\"" ] = 1
score [ "\"U\"" ] = 1
score [ "\"L\"" ] = 1
score [ "\"N\"" ] = 1
score [ "\"R\"" ] = 1
score [ "\"S\"" ] = 1
score [ "\"T\"" ] = 1
score [ "\"D\"" ] = 2
score [ "\"G\"" ] = 2
score [ "\"B\"" ] = 3
score [ "\"C\"" ] = 3
score [ "\"M\"" ] = 3
score [ "\"P\"" ] = 3
score [ "\"F\"" ] = 4
score [ "\"H\"" ] = 4
score [ "\"V\"" ] = 4
score [ "\"W\"" ] = 4
score [ "\"Y\"" ] = 4
score [ "\"K\"" ] = 5
score [ "\"J\"" ] = 8
score [ "\"X\"" ] = 8
score [ "\"Q\"" ] = 10
score [ "\"Z\"" ] = 10
score [ "\" \"" ] = 0
}
{
split ( toupper ( $0 ), chars , "\"\"" )
x = 0
for ( i = 1 ; i <= length ( chars ); ++ i ) {
c = chars [ i ]
if ( c in score ) { x = x + score [ c ] }
}
printf ( "\"%s,%s\"" , toupper ( $0 ), x )
}
