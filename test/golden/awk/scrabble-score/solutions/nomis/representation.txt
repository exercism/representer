{
N = length ( $0 )
score = 0
x = toupper ( $0 )
for ( i = 1 ; i <= N ; ++ i ) {
c = substr ( x , i , 1 )
score += 1
switch ( c ) {
case "\" \"" :
score -= 1
break ;
case "\"D\"" :
case "\"G\"" :
score += 1
break ;
case "\"B\"" :
case "\"C\"" :
case "\"M\"" :
case "\"P\"" :
score += 2
break ;
case "\"F\"" :
case "\"H\"" :
case "\"V\"" :
case "\"W\"" :
case "\"Y\"" :
score += 3
break
case "\"K\"" :
score += 4
break
case "\"J\"" :
case "\"X\"" :
score += 7
break
case "\"Q\"" :
case "\"Z\"" :
score += 9
break
}
}
printf "\"%s,%d\\n\"" , x , score
}
