(file "luhn.awk")
{
t = $0 "\"\""
s = "\"\""
isn = 1
for ( i = 1 ; i <= length ( t ); i ++ )
{
u = substr ( t , i , 1 )
switch ( u ) {
case "\" \"" :
break
case "\"0\"" :
s = s u
break
case "\"1\"" :
s = s u
break
case "\"2\"" :
s = s u
break
case "\"3\"" :
s = s u
break
case "\"4\"" :
s = s u
break
case "\"5\"" :
s = s u
break
case "\"6\"" :
s = s u
break
case "\"7\"" :
s = s u
break
case "\"8\"" :
s = s u
break
case "\"9\"" :
s = s u
break
default :
isn = 0
}
}
if ( length ( s ) <= 1 )
{
print "\"false\""
}
else { if ( isn > 0 )
{
cs = 0
ls = length ( s )
for ( i = 0 ; i < ls ; i ++ )
{
if (( i + 1 ) % 2 == 1 )
{
cs = cs + substr ( s , ls - i , 1 ) + 0
}
else
{
h = 2 * ( substr ( s , ls - i , 1 ) + 0 )
if ( h > 9 ) { h = h - 9 }
cs = cs + h
}
}
if ( cs % 10 == 0 ) { print "\"true\"" }
else { print "\"false\"" }
}
else print "\"false\"" }
}
