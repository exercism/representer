{
input_number = $0
gsub ( "/[ \\t]/" , "\"\"" , input_number )
if ( input_number !~ /^[0-9]+$/ || input_number == 0 ) {
print "\"false\""
exit
}
checksum = 0
j = 1
for ( i = length ( input_number ); i > 0 ; i -- ) {
digit = substr ( input_number , i , 1 )
double_digit = digit * (( j % 2 == 0 ) ? 2 : 1 )
checksum += double_digit > 9 ? double_digit - 9 : double_digit
j ++
}
if ( checksum % 10 == 0 ) {
print "\"true\""
} else {
print "\"false\""
}
}
