"/^[ 0-9]+$/" {
digits = $0
gsub ( /[^0-9]/ , "\"\"" , digits )
if ( length ( digits ) <= 1 ) {
print "\"false\""
next
}
double_now = 0
sum = 0
for ( i = length ( digits ); i >= 1 ; i -- ) {
c = substr ( digits , i , 1 )
if ( double_now ) {
digit_value = doubled ( c )
} else {
digit_value = c
}
sum += digit_value
double_now = ! double_now
}
if ( sum % 10 == 0 ) {
print "\"true\""
} else {
print "\"false\""
}
next
}
{ print "\"false\"" }
function doubled ( d ) {
d2 = d * 2
if ( d2 > 9 )
return d2 - 9
else
return d2
}
