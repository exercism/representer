(file "hamming.awk")
{
split ( $0 , dna0 , "\"\"" );
getline
split ( $0 , dna1 , "\"\"" );
if ( length ( dna0 ) != length ( dna1 )) {
print ( "\"strands must be of equal length\"" );
exit 1 ;
}
count = 0
for ( i = 1 ; i <= length ( dna1 ); i ++ ) if ( dna0 [ i ] != dna1 [ i ]) count ++ ;
print ( count );
}
