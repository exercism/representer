(file "roman-numerals.awk")
{
number = int ( $0 )
result = "\"\""
while ( number > 0 ) {
switch ( number ) {
case / [ 1 - 9 ][ 0 - 9 ][ 0 - 9 ][ 0 - 9 ] / :
result = result "\"M\""
number = number - 1000
break
case / [ 9 ][ 0 - 9 ][ 0 - 9 ] / :
result = result "\"CM\""
number = number - 900
break
case / [ 5 - 9 ][ 0 - 8 ][ 0 - 9 ] / :
result = result "\"D\""
number = number - 500
break
case / [ 4 ][ 0 - 9 ][ 0 - 9 ] / :
result = result "\"CD\""
number = number - 400
break
case / [ 1 - 4 ][ 0 - 9 ][ 0 - 9 ] / :
result = result "\"C\""
number = number - 100
break
case / [ 9 ][ 0 - 9 ] / :
result = result "\"XC\""
number = number - 90
break
case / [ 5 - 8 ][ 0 - 9 ] / :
result = result "\"L\""
number = number - 50
break
case / [ 4 ][ 0 - 9 ] / :
result = result "\"XL\""
number = number - 40
break
case / [ 1 - 3 ][ 0 - 9 ] / :
result = result "\"X\""
number = number - 10
break
case / [ 9 ] / :
result = result "\"IX\""
number = number - 9
break
case / [ 8 ] / :
result = result "\"VIII\""
number = number - 8
break
case / [ 7 ] / :
result = result "\"VII\""
number = number - 7
break
case / [ 6 ] / :
result = result "\"VI\""
number = number - 6
break
case / [ 5 ] / :
result = result "\"V\""
number = number - 5
break
case / [ 4 ] / :
result = result "\"IV\""
number = number - 4
break
case / [ 1 - 3 ] / :
result = result "\"I\""
number --
break
}
}
print result
}
