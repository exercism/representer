{
  format: ( name , number ) ->
    d0 = number % 10
    d00 = number % 100
    ord = "\"th\""
    if 1 <= d0 and d0 <= 3 and ( d00 < 10 or 20 < d00 )
      ord = ({ "\"st\"" , "\"nd\"" , "\"rd\"" })[ d0 ]
    "\"\#{" name "}, you are the \#{" number }#{ ord "} customer we serve today. Thank you!\""
}
