{
  format: ( name , number ) ->
    nMod10 = number % 10
    nMod100 = number % 100
    suffix = if nMod100 >= 11 and nMod100 <= 13 then 'th'
    else if nMod10 == 1 then 'st'
    else if nMod10 == 2 then 'nd'
    else if nMod10 == 3 then 'rd'
    else 'th'
    "\"\#{" name "}, you are the \#{" number }#{ suffix "} customer we serve today. Thank you!\""
}
