rotate_char = ( char , key ) ->
  byte = char \ byte !
  base = if byte >= 'A' \ byte ! and byte <= 'Z' \ byte ! then 'A'
  else if byte >= 'a' \ byte ! and byte <= 'z' \ byte ! then 'a'
  return char unless base
  base = base \ byte !
  string.char ( byte - base + key ) % 26 + base
rotate: ( text , shift_key ) ->
  table.concat [ rotate_char char , shift_key for char in text \ gmatch utf8.charpattern ]
