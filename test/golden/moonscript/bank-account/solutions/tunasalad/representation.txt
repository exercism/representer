(file "bank_account.moon")
class BankAccount
  new: =>
    @open_ = false
  open: =>
    @assert_closed !
    @open_ = true
    @balance_ = 0
  close: =>
    @assert_open !
    @open_ = false
  balance: =>
    @assert_open !
    @balance_
  deposit: ( amount ) =>
    @assert_open !
    @assert_amount amount
    @balance_ += amount
  withdraw: ( amount ) =>
    @assert_open !
    @assert_amount amount
    assert @balance_ >= amount , "\"amount must be less than balance\""
    @balance_ -= amount
  assert_open: =>
    assert @open_ , "\"account not open\""
  assert_closed: =>
    assert not ( @open_ ), "\"account already open\""
  assert_amount: ( amount ) =>
    assert amount > 0 , "\"amount must be greater than 0\""
