dir_strs = { 'north' , 'east' , 'south' , 'west' }
dir_map = {}
do
  for i , s in ipairs dir_strs
    dir_map [ s ] = i
dx = { 0 , 1 , 0 , - 1 }
dy = { 1 , 0 , - 1 , 0 }
class Robot
  new: ( params ) =>
    @x_ = params . x
    @y_ = params . y
    @dir_ = dir_map [ params . direction ]
  x: => @x_
  y: => @y_
  direction: => dir_strs [ @dir_ ]
  move: ( inst ) =>
    for i = 1 , inst \ len !
      switch inst \ sub i , i
        when 'A'
          @x_ += dx [ @dir_ ]
          @y_ += dy [ @dir_ ]
        when 'L'
          @dir_ = (( @dir_ + 2 ) & 3 ) + 1
        when 'R'
          @dir_ = ( @dir_ & 3 ) + 1
