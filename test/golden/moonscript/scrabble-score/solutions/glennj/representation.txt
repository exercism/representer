import fold from require 'moon'
sum = ( t ) ->
  fold { 0 , table.unpack t }, ( s , n ) -> s + n
product = ( t ) ->
  fold { 1 , table.unpack t }, ( prod , n ) -> prod * n
gcd = ( a , b ) ->
  b == 0 and a or gcd b , a % b
{ :sum , :product , :gcd }
import sum from require 'lib/math'
tiles =
  A : 1 , E : 1 , I : 1 , O : 1 , U : 1 ,
  L : 1 , N : 1 , R : 1 , S : 1 , T : 1 ,
  D : 2 , G : 2 ,
  B : 3 , C : 3 , M : 3 , P : 3 ,
  F : 4 , H : 4 , V : 4 , W : 4 , Y : 4 ,
  K : 5 ,
  J : 8 , X : 8 ,
  Q : 10 , Z : 10 ,
{
  score: ( word ) ->
    sum [ tiles [ c ] or 0 for c in word \ upper !\ gmatch '.' ]
}
