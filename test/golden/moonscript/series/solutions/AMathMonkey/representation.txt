slices: ( series , length ) ->
  assert length != 0 , "'slice length cannot be zero'"
  assert length > 0 , "'slice length cannot be negative'"
  assert # series > 0 , "'series cannot be empty'"
  assert length <= # series , "'slice length cannot be greater than series length'"
  coroutine.wrap ->
    for offset = 1 , # series - length + 1
      coroutine.yield series \ sub ( offset , offset + length - 1 )
