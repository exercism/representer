(file "nth_prime.moon")
{
  prime: ( n ) ->
    assert n > 0 , "'there is no zeroth prime'"
    return 2 if n == 1
    with res = 1
      while n > 1
        res += 2
        i = 3
        ok = true
        while i * i <= res
          if res % i == 0
            ok = false
            break
          i += 2
        n -= 1 if ok
}
