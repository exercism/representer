(file "spiral_matrix.moon")
{
  spiral_matrix: ( size ) ->
    result = [[ size * size for _ = 1 , size ] for _ = 1 , size ]
    value = 1
    row = 1
    column = 1
    side = size - 1
    while side >= 1
      for _ = 1 , side
        result [ row ][ column ] = value
        value += 1
        column += 1
      for _ = 1 , side
        result [ row ][ column ] = value
        value += 1
        row += 1
      for _ = 1 , side
        result [ row ][ column ] = value
        value += 1
        column -= 1
      for _ = 1 , side
        result [ row ][ column ] = value
        value += 1
        row -= 1
      row += 1
      column += 1
      side -= 2
    result
}
