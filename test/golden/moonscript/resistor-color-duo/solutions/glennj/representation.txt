colors = -> {
  'black' , 'brown' , 'red' , 'orange' , 'yellow' ,
  'green' , 'blue' , 'violet' , 'grey' , 'white'
}
color_code = ( color ) ->
  for i , c in ipairs colors !
    if c == color
      return i - 1
  return - 1
{ :colors , :color_code }
import color_code from require './resistor_color'
value = ( band1 , band2 ) -> 10 * color_code ( band1 ) + color_code ( band2 )
{ :value }
