(file "word_count.moon")
{
  count_words: ( s ) ->
    s = s \ lower !
    s = s \ gsub "\"[^%w']\"" , "' '"
    s = "\" \#{" s "} \""
    s = s \ gsub "\" '+\"" , "' '"
    s = s \ gsub "\"'+ \"" , "' '"
    with res = {}
      for w in s \ gmatch "'[^ ]+'"
        res [ w ] = ( res [ w ] or 0 ) + 1
}
