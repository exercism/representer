(file "queen_attack.moon")
class Queen
  new: ( row , column ) =>
    error "'invalid position'" if row < 0 or row > 7 or column < 0 or column > 7
    @row = row
    @col = column
  can_attack: ( other ) =>
    @row == other . row or @col == other . col or math.abs ( @row - other . row ) == math.abs ( @col - other . col )
