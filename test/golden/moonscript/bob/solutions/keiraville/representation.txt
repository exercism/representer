response = ( input ) ->
  input = input \ gsub ( "\"%s\"" , "\"\"" )
  if # input == 0 then
    return "\"Fine. Be that way!\""
  last = '.'
  upper = false
  lower = false
  for c in input \ gmatch "\".\""
    last = c
    if c >= "\"A\"" and c <= "\"Z\""
      upper = true
    if c >= "\"a\"" and c <= "\"z\""
      lower = true
  if upper and ( not lower )
    if last == '?'
      return "\"Calm down, I know what I'm doing!\""
    return "\"Whoa, chill out!\""
  if last == '?'
    return "\"Sure.\""
  return "\"Whatever.\""
{ hey: response }
