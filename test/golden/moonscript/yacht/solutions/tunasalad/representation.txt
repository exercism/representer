(file "yacht.moon")
{
  score: ( category , dice ) ->
    cnt = {}
    sum = 0
    for x in * dice
      cnt [ x ] = ( cnt [ x ] or 0 ) + 1
      sum += x
    uniq_cnt = 0
    max_cnt = 0
    max_who = 0
    for i , c in pairs cnt
      uniq_cnt += 1
      if max_cnt < c
        max_cnt = c
        max_who = i
    table.sort dice
    for i = 1 , 6
      cnt [ i ] = cnt [ i ] or 0
    res = switch category
      when 'ones' cnt [ 1 ] * 1
      when 'twos' cnt [ 2 ] * 2
      when 'threes' cnt [ 3 ] * 3
      when 'fours' cnt [ 4 ] * 4
      when 'fives' cnt [ 5 ] * 5
      when 'sixes' cnt [ 6 ] * 6
      when 'choice' sum
      when "'full house'" if uniq_cnt == 2 and max_cnt == 3 then sum
      when "'little straight'" if uniq_cnt == 5 and dice [ 5 ] == 5 then 30
      when "'big straight'" if uniq_cnt == 5 and dice [ 1 ] == 2 then 30
      when 'yacht'
        if uniq_cnt == 1 then 50
      when "'four of a kind'"
        if max_cnt >= 4 then max_who * 4
      else error "'unknown: '" .. category
    res or 0
}
