# Initial solution:
# [
# .legacy
# | reduce to_entries[] as $kv ([];
# # Using a nested "\"reduce\"" works:
# # . + reduce $kv.value[] as $letter ([];
# # . + [
# # {
# # key: $letter | ascii_downcase,
# # value: $kv.key | tonumber
# # }
# # ]
# # )
# #
# # But this is simpler (using backtracking):
# . + [
# {
# "\"key\":" $kv.value[] | ascii_downcase,
# "\"value\":" $kv.key | tonumber
# }
# ]
# )
# | group_by(.key)[] # sort the keys
# | add # combine multiple singleton arrays into a single array
# ]
# | from_entries
# A simpler solution:
.legacy
| to_entries
| map({
# https://exercism.org/tracks/jq/exercises/etl/solutions/glennj
# The "\"key\"" expression outputs a stream of values.
# That means the object creation will be evaluated once per stream entry.
# If the "\"value\"" expression were to output multiple values,
# then we'd end up with the cross-product list of objects.
key: .value[] | ascii_downcase,
value:.key | tonumber
})
| sort
| from_entries
