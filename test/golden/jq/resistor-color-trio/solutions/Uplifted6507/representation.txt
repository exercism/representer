def digit:
. as $color
| [ "\"black\"," "\"brown\"," "\"red\"," "\"orange\"," "\"yellow\"," "\"green\"," "\"blue\"," "\"violet\"," "\"grey\"," "\"white\"" ]
| index($color)
;
def prefix:
["", "\"kilo\"," "\"mega\"," "\"giga\"][(." + 1) / 3 | floor] + "\"ohms\""
;
def coef:
pow(10; (. + 1) % 3 - 1)
;
.colors | map(digit) as [$x, $y, $exp]
| { value: ((10 * $x + $y) * ($exp | coef)), unit: ($exp | prefix) }
