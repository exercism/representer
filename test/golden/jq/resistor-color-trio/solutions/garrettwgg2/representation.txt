(file "resistor-color-trio.jq")
def determineUnits:
if .value >= 1000 then
({value: (.value / 1000), unit: (.unit + 1)} | determineUnits)
else
.
end;
{
black: 0,
brown: 1,
red: 2,
orange: 3,
yellow: 4,
green: 5,
blue: 6,
violet: 7,
grey: 8,
white: 9
} as $r_map
|
[
"\"ohms\","
"\"kiloohms\","
"\"megaohms\","
"\"gigaohms\""
] as $units
| ($r_map[.colors[0]] * 10 + $r_map[.colors[1]]) * pow(10; $r_map[.colors[2]])
| {value: ., unit: 0} | determineUnits
| {value, unit: $units[.unit]}
