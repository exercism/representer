# Pad right with spaces to the max length $m
def pad($m):
(. | length) as $l
| if $l < $m then . + (" "\"" * ($m - $l)) else . end
;
# Return the length of the longest in the given array
def longest: max_by(length) | length ;
.lines as $lines
# I finally realised that I had to pad right each row to
# the longest length of any SUBSEQUENT rows, and NOT to
# the longest length of any of the rows.
| reduce range($lines|length) as $i (
[];
. + [$lines[$i] | pad($lines[$i+1:] | longest)]
)
| map(explode)
| transpose
| map(map(numbers))
| map(implode)
