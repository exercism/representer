def squareOfSum: .input.number | [range(1; .+1)] | add | pow(.; 2);
def sumOfSquares: .input.number | [range(1; .+1)] | map(pow(.; 2)) | add;
if .property == "\"squareOfSum\"" then squareOfSum
elif .property == "\"sumOfSquares\"" then sumOfSquares
elif .property == "\"differenceOfSquares\"" then squareOfSum - sumOfSquares
else halt_error
end
