(program
  (pipeline
    (binding_expression
      (binary_expression
        (subscript_expression . "actual_minutes_in_oven") // 0) (variable PLACEHOLDER_1))
    (pipeline
      (binding_expression
        (binary_expression
          (subscript_expression . "number_of_layers") // 1) (variable PLACEHOLDER_2))
      (object
        (pair "expected_minutes_in_oven" 40)
        (pair "remaining_minutes_in_oven"
          (binary_expression 40 - (variable PLACEHOLDER_1)))
        (pair "preparation_time"
          (binary_expression (variable PLACEHOLDER_2) * 2))
        (pair "total_time"
          (binary_expression
            (subscript_expression . "actual_minutes_in_oven") +
            (binary_expression (variable PLACEHOLDER_2) * 2)))))))
