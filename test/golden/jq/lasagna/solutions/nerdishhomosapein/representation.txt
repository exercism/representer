(program
  (pipeline
    (assignment_expression
      (subscript_expression . "actual_minutes_in_oven") //= 0)
    (pipeline
      (assignment_expression
        (subscript_expression . "number_of_layers") //= 1)
      (pipeline
        (binding_expression 40 (variable PLACEHOLDER_1))
        (pipeline
          (binding_expression 2 (variable PLACEHOLDER_2))
          (pipeline
            (binding_expression
              (binary_expression (variable PLACEHOLDER_2) *
                (subscript_expression . "number_of_layers")) (variable PLACEHOLDER_3))
            (object
              (pair "expected_minutes_in_oven" (variable PLACEHOLDER_1))
              (pair "remaining_minutes_in_oven"
                (binary_expression (variable PLACEHOLDER_1) -
                  (subscript_expression . "actual_minutes_in_oven")))
              (pair "preparation_time" (variable PLACEHOLDER_3))
              (pair "total_time"
                (binary_expression (variable PLACEHOLDER_3) +
                  (subscript_expression . "actual_minutes_in_oven"))))))))))
