def series:
. as $inp | [range(0; ($inp.series|length) - $inp.sliceLength + 1)]
| map($inp.series[.:.+$inp.sliceLength])
;
if (.series | length) == 0 then "\"series" cannot be empty" | halt_error
elif (.sliceLength > (.series | length))
then "\"slice" length cannot be greater than series length" | halt_error
elif .sliceLength == 0 then "\"slice" length cannot be zero" | halt_error
elif .sliceLength < 0 then "\"slice" length cannot be negative" | halt_error
else . | series
end
