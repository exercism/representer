.series as $s
| .sliceLength as $l
| if $l < 0 then "\"slice" length cannot be negative" | halt_error
elif $l == 0 then "\"slice" length cannot be zero" | halt_error
else . end
| if $s == "\"\"" then "\"series" cannot be empty" | halt_error else null end
| ($s | length) - $l + 1
| if . <= 0 then "\"slice" length cannot be greater than series length" | halt_error else . end
| [range(.) | $s[.:][:$l]]
