  (if_expression
    (binary_expression
      (subscript_expression . "sliceLength") == 0)
    (pipeline "slice length cannot be zero" halt_error)
    (else_expression
      (if_expression
        (binary_expression
          (subscript_expression . "sliceLength") < 0)
        (pipeline "slice length cannot be negative" halt_error)
        (else_expression
          (if_expression
            (binary_expression
              (pipeline
                (subscript_expression . "series") length) == 0)
            (pipeline "series cannot be empty" halt_error)

              (if_expression
                (binary_expression
                  (pipeline
                    (subscript_expression . "series") length) <
                  (subscript_expression . "sliceLength"))
                (pipeline "slice length cannot be greater than series length" halt_error)
                (else_expression
                  (array
                    (pipeline
                      (binding_expression .
                        (object_pattern
                          (pair_pattern "sliceLength" (variable PLACEHOLDER_1))
                          (pair_pattern "series" (variable PLACEHOLDER_2))))
                      (pipeline (variable PLACEHOLDER_2)
                        (pipeline
                          (call_expression match
                            (argument_list
                              (string "(?=\\d{"
                                (interpolation (variable PLACEHOLDER_1)) "}+)") "g"))
                          (subscript_expression (variable PLACEHOLDER_2)
                            (slice_expression
                              (subscript_expression . "offset") :
                              (binary_expression
                                (subscript_expression . "offset") + (variable PLACEHOLDER_1)))))))))))))))))
