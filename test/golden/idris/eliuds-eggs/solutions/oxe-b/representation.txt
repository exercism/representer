module EliudsEggs
import Data.Bits
export
eggCount : Int -> Int
eggCount number = ( z * 0 x0101010101010101 ) ` shiftR ` 56
  where
    x = ( number .&. 0 x5555555555555555 ) + (( number ` shiftR ` 1 ) .&. 0 x5555555555555555 )
    y = ( x .&. 0 x3333333333333333 ) + (( x ` shiftR ` 2 ) .&. 0 x3333333333333333 )
    z = (( y + ( y ` shiftR ` 4 )) .&. 0 x0F0F0F0F0F0F0F0F )
