module Dominoes
import Data.Vect
import Data.List
export
canChain : List ( Vect 2 ( Fin 7 )) -> Bool
canChain [] = True
canChain ( z :: zs ) = go ( head z ) zs
  where go : Fin 7 -> List ( Vect 2 ( Fin 7 )) -> Lazy Bool
go l [] = l == last z
  go l xs = or [ go ( if a == l then b else a ) $ delete x xs | x @ [ a , b ] <- xs , elem l x ]
