module Hamming
import Data.Vect
public export
data Nucleotide = A | C | G | T
public export
implementation Eq Nucleotide where
  A == A = True
  C == C = True
  G == G = True
  T == T = True
  _ == _ = False
export
hamming_distance : Eq a => Vect n a -> Vect n a -> Nat
hamming_distance s1 s2 = go 0 s1 s2
  where
    go : Nat -> Vect m a -> Vect m a -> Nat
go acc [] [] = acc
  go acc ( x :: xs ) ( y :: ys ) = if x /= y
    then go ( acc + 1 ) xs ys
    else go acc xs ys
