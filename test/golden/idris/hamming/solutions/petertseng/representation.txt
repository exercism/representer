module Hamming
public export
data Nucleotide = A | C | G | T
public export
implementation Eq Nucleotide where
  A == A = True
  C == C = True
  G == G = True
  T == T = True
  A == _ = False
  C == _ = False
  G == _ = False
  T == _ = False
export
hamming_distance : Vect n Nucleotide -> Vect n Nucleotide -> Nat
hamming_distance xs ys = fst ( filter id ( zipWith ( /= ) xs ys ))
