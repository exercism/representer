module Hamming
import Data.Vect
public export
data Nucleotide = A | C | G | T
Eq Nucleotide where
  ( == ) A A = True
  ( == ) C C = True
  ( == ) G G = True
  ( == ) T T = True
  ( == ) _ _ = False
  ( /= ) x y = not ( x == y )
export
hamming_distance : Vect n Nucleotide -> Vect n Nucleotide -> Nat
hamming_distance s1 s2 = fst $ filter ( \ ( n1 , n2 ) => n1 /= n2 ) ( zip s1 s2 )
