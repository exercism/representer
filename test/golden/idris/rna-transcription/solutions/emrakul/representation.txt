(file "src/RnaTranscription.idr")
module RnaTranscription
import Data.Vect
%default total
namespace DnaNucleotide
  public export
  data DnaNucleotide = A | G | C | T
namespace RnaNucleotide
  public export
  data RnaNucleotide = A | G | C | U
export
implementation Eq RnaNucleotide where
  ( == ) A A = True
  ( == ) G G = True
  ( == ) C C = True
  ( == ) U U = True
  ( == ) _ _ = False
export
implementation Show RnaNucleotide where
  show A = "\"A\""
  show G = "\"G\""
  show C = "\"C\""
  show U = "\"U\""
translate : DnaNucleotide -> RnaNucleotide
translate A = U
translate G = C
translate C = G
translate T = A
export
toRna : Vect n DnaNucleotide -> Vect n RnaNucleotide
toRna l = map translate l
