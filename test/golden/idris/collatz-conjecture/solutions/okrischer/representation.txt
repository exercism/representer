module CollatzConjecture
import Data.List
export
steps : Int -> Maybe Int
steps number = if number <= 0 then Nothing else Just ( countSteps number )
  where
  next : Int -> Maybe Int
  next 1 = Nothing
  next n = if n ` mod ` 2 == 0 then Just ( n ` div ` 2 ) else Just ( n * 3 + 1 )
  countSteps : Int -> Int
  countSteps n = cast $ drop 1 ( iterate next n ) |> length
