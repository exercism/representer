module ArmstrongNumbers
export
digs : Integer -> List Integer
digs 0 = []
digs x = digs ( x ` div ` 10 ) ++ [ x ` mod ` 10 ]
export
isArmstrongNumber : Integer -> Bool
isArmstrongNumber i = let lstNat = digs i in
  let ac = aCandidate lstNat 0 ( length lstNat )
    in
      ( cast ac ) == i
      where
      aCandidate : List Integer -> Nat -> Nat -> Nat
      aCandidate [] ys len = ys
      aCandidate ( x :: xs ) ys len = aCandidate xs (( power ( cast x ) len ) + ys ) len
