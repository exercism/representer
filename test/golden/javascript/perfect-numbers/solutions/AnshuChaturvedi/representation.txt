  (program

        (variable_declarator classify =
          (arrow_function PLACEHOLDER_1
            (statement_block

                (lexical_declaration let
                  (variable_declarator PLACEHOLDER_3 = 1))

                (binary_expression PLACEHOLDER_2 === PLACEHOLDER_1)
