(program
  (export_statement
    (lexical_declaration const
      (variable_declarator EXPECTED_MINUTES_IN_OVEN = 40)))
  (export_statement
    (lexical_declaration const
      (variable_declarator actualMinutesInOven = 25)))
  (export_statement
    (lexical_declaration const
      (variable_declarator PREPARATION_MINUTES_PER_LAYER = 2)))
  (export_statement
    (lexical_declaration const
      (variable_declarator numberOfLayer = 4)))
  (export_statement
    (function_declaration remainingMinutesInOven
      (formal_parameters actualMinutesInOven)
      (statement_block
        (return_statement
          (binary_expression EXPECTED_MINUTES_IN_OVEN - actualMinutesInOven)))))
  (export_statement
    (function_declaration preparationTimeInMinutes
      (formal_parameters numberOfLayer)
      (statement_block
        (return_statement
          (binary_expression numberOfLayer * PREPARATION_MINUTES_PER_LAYER)))))
  (export_statement
    (function_declaration totalTimeInMinutes
      (formal_parameters numberOfLayer actualMinutesInOven)
      (statement_block
        (return_statement
          (binary_expression
            (call_expression preparationTimeInMinutes
              (arguments numberOfLayer)) + actualMinutesInOven))))))
