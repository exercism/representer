class Anagram ( val source : String ) {
private val sourceLetterCounts = source
. groupingBy { it . lowercaseChar () }
. eachCount ()
fun match ( anagrams : Collection < String >): Set < String > = anagrams
. filterNot { it . lowercase () == source . lowercase () }
. filter { word ->
word
. groupingBy { it . lowercaseChar () }
. eachCount () == sourceLetterCounts
}. toSet ()
}
