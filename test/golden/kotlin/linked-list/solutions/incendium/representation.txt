class Deque < T > {
private var firstNode : Node < T >? = null
private var lastNode : Node < T >? = null
fun push ( value : T ) {
val originalLast = lastNode
lastNode = Node ( value ). apply {
prev = originalLast
originalLast ?. next = this
}
if ( firstNode == null ) {
firstNode = lastNode
}
}
fun pop (): T ? = lastNode ?. let { node ->
lastNode = node . prev
lastNode ?. next = null
node . value
}
fun unshift ( value : T ) {
val originalFirst = firstNode
firstNode = Node ( value ). apply {
next = originalFirst
originalFirst ?. prev = this
}
if ( lastNode == null ) {
lastNode = firstNode
}
}
fun shift (): T ? = firstNode ?. let { node ->
firstNode = node . next
firstNode ?. prev = null
node . value
}
}
class Node < T >( val value : T ) {
var prev : Node < T >? = null
var next : Node < T >? = null
}
