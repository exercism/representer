class CustomSet ( vararg input : Int ) {
val nums : MutableSet < Int > = input . toMutableSet ()
fun isEmpty (): Boolean =
nums . isEmpty ()
fun isSubset ( other : CustomSet ): Boolean =
nums . all { other . contains ( it ) }
fun isDisjoint ( other : CustomSet ): Boolean =
nums . all { ! other . contains ( it ) }
fun contains ( other : Int ): Boolean =
nums . contains ( other )
fun intersection ( other : CustomSet ): CustomSet =
CustomSet (* nums . filter { other . contains ( it ) }. toIntArray ())
fun add ( other : Int ) {
nums . add ( other )
}
override fun equals ( other : Any ?): Boolean =
if ( other is CustomSet )
nums . all { other . contains ( it ) } &&
other . nums . all { nums . contains ( it ) }
else
false
operator fun plus ( other : CustomSet ): CustomSet =
CustomSet (*( nums + other . nums ). toIntArray ())
operator fun minus ( other : CustomSet ): CustomSet =
CustomSet (*( nums - other . nums ). toIntArray ())
}
