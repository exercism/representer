object PrimeFactorCalculator {
fun primeFactors ( int : Int ): List < Int > {
return primeFactors ( int . toLong ()). map { it . toInt ()}
}
tailrec fun primeFactors ( long : Long , acc : List < Long > = emptyList ()): List < Long > {
val nextPrime = ( 2 .. long ). firstOrNull { long % it == 0L } ?: return acc
return primeFactors ( long / nextPrime , acc + nextPrime )
}
}
