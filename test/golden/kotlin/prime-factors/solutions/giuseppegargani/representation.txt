object PrimeFactorCalculator {
fun primeFactors ( int : Int ): List < Int > {
var finale = mutableListOf < Int >()
var fattore = 2
fun dividi ( int : Int ): List < Int > {
when {
int == 1 -> return finale
fattore >= int -> {
finale . add ( int )
return finale
}
int % fattore == 0 -> {
finale . add ( fattore )
return dividi ( int = int / fattore )
}
else -> {
fattore ++
return dividi ( int = int )
}
}
}
dividi ( int )
return finale
}
fun primeFactors ( long : Long ): List < Long > {
var finale = mutableListOf < Long >()
var fattore = 2L
tailrec fun dividi ( lungo : Long ): List < Long > {
when {
lungo == 1 . toLong () -> return finale
fattore >= lungo -> {
finale . add ( lungo )
return finale
}
lungo % fattore == 0L -> {
finale . add ( fattore )
return dividi ( lungo / fattore )
}
else -> {
fattore ++
return dividi ( lungo )
}
}
}
dividi ( long )
return finale
}
}
