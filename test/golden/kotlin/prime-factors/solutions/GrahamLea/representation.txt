object PrimeFactorCalculator {
fun primeFactors ( number : Int ) = primeFactors ( number . toLong ()). map { it . toInt () }
fun primeFactors ( number : Long ): List < Long > =
generateSequence < Triple < Long , Long ?, Long ?>>( Triple ( number , null , 2L )) {
( remainingDividend , factorJustFound , nextUntestedNumber ) ->
when {
remainingDividend == 1L -> null
nextUntestedNumber == remainingDividend -> Triple ( 1 , remainingDividend , null )
else -> {
( factorJustFound ?: nextUntestedNumber !! ). let { divisor ->
if ( remainingDividend % divisor == 0L )
Triple ( remainingDividend / divisor , divisor , null )
else
Triple ( remainingDividend , null , divisor + 1L )
}
}
}
}. mapNotNull { it . second }. toList ()
}
