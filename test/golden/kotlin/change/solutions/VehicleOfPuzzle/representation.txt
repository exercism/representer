private fun < TInput , TOutput > memoize ( fn : ( TInput ) -> TOutput ): ( TInput ) -> TOutput {
val cache = mutableMapOf < TInput , TOutput >()
return {
if ( cache . containsKey ( it )) {
cache . get ( it ) as TOutput
} else {
val output = fn ( it )
cache . put ( it , output )
output
}
}
}
class ChangeCalculator ( coinDenominations : List < Int >) {
private val coinDenominations = coinDenominations . toSortedSet ()
private val getNumberOfStepsAndFirstCoin : ( Int ) -> Pair < Int ?, Int ?> =
memoize ({ amount ->
when ( amount ) {
0 -> Pair ( 0 , 0 )
else ->
run {
var bestOption : Pair < Int , Int >? = null
for ( coin in this . coinDenominations ) {
if ( coin > amount ) {
break
}
val ( steps ) = this . getNumberOfStepsAndFirstCoin ( amount - coin )
if ( steps == null ) {
continue
}
if ( bestOption == null || bestOption . first > 1 + steps ) {
bestOption = Pair ( 1 + steps , coin )
}
}
bestOption ?: Pair ( null , null )
}
}
})
fun computeMostEfficientChange ( grandTotal : Int ): List < Int > = buildList {
require ( grandTotal >= 0 ) { "\"Negative totals are not allowed.\"" }
var amount = grandTotal
while ( amount > 0 ) {
val coin = this @ChangeCalculator . getNumberOfStepsAndFirstCoin ( amount ). second
requireNotNull ( coin ) { "\"The total $grandTotal cannot be represented in the given currency.\"" }
amount -= coin
this . add ( coin )
}
}
}
