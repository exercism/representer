class ChangeCalculator ( coins : List < Int >) {
private val sortedCoins : List < Int > = coins . sortedDescending ()
fun computeMostEfficientChange ( grandTotal : Int ): List < Int > {
if ( grandTotal == 0 ) return listOf ()
if ( grandTotal < 0 ) {
throw IllegalArgumentException ( "\"Negative totals are not allowed.\"" )
}
if ( sortedCoins . all { it > grandTotal }) {
throw IllegalArgumentException ( "\"The total $grandTotal cannot be represented in the given currency.\"" )
}
val result : List < Int > = calc ( grandTotal )
if ( result . isEmpty ()) {
throw IllegalArgumentException ( "\"The total $grandTotal cannot be represented in the given currency.\"" )
}
return result
}
private fun calc ( grandTotal : Int ): List < Int > {
val usedCoins : MutableList < Int > = ( 1 .. grandTotal + 1 ). map { 0 }. toMutableList ()
val minCountCoins : MutableList < Int > = ( 1 .. grandTotal + 1 ). map { 0 }. toMutableList ()
minCountCoins [ 0 ] = Int . MAX_VALUE
for ( cent in 1 .. grandTotal ) {
var curCount = cent
var newCoin = 0
for ( coin in sortedCoins . filter { it <= cent }) {
if ( minCountCoins [ cent - coin ] + 1 <= curCount ) {
curCount = minCountCoins [ cent - coin ] + 1
newCoin = coin
}
}
minCountCoins [ cent ] = curCount
usedCoins [ cent ] = newCoin
}
return transformUsedCoins ( usedCoins , grandTotal )
}
private fun transformUsedCoins ( usedCoins : MutableList < Int >, grandTotal : Int ): List < Int > {
var coin = grandTotal
val result : MutableList < Int > = mutableListOf ()
while ( coin > 0 ) {
if ( usedCoins [ coin ] == 0 ) {
throw IllegalArgumentException ( "\"The total $grandTotal cannot be represented in the given currency.\"" )
}
result . add ( 0 , usedCoins [ coin ])
coin -= usedCoins [ coin ]
}
return result
}
}
