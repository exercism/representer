class DiamondPrinter {
fun printToList ( letter : Char ): List < String > {
val topAndCenter = ( 'A' .. letter ). map { currentLetter -> line ( currentLetter , letter ) }
val bottom = topAndCenter . dropLast ( 1 ). reversed ()
return topAndCenter + bottom
}
private fun line ( currentLetter : Char , letter : Char ): String {
val centerAndRight = ( 'A' .. letter ). map { if ( it == currentLetter ) currentLetter else "' '" }. joinToString ( "\"\"" )
val left = centerAndRight . drop ( 1 ). reversed ()
return left + centerAndRight
}
}
