class DiamondPrinter {
fun printToList ( letter : Char ) : List < String > {
require ( letter . isLetter () && letter . isUpperCase () ) { "\"Need an upper case letter for this to work\"" }
val centre = letter - 'A' + 1
val width = ( 2 * centre ) - 1
return ( 'A' .. letter ). mapIndexed { index , c ->
val row = CharArray ( width ) { "' '" }
row [ centre - index - 1 ] = c
row [ centre + index - 1 ] = c
row . joinToString ( "\"\"" )
}. let { it . plus ( it . dropLast ( 1 ). reversed () ) }
}
}
