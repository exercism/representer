class DiamondPrinter {
fun printToList ( maxChar : Char ): List < String > =
mirror ( topHalf ( maxChar ))
private fun topHalf ( maxChar : Char ) =
( 'A' .. maxChar ). map { line ( maxChar , it ) }
private fun mirror ( topHalf : List < String >) =
topHalf + reflection ( topHalf )
private fun reflection ( topHalf : List < String >) =
topHalf . reversed (). drop ( 1 )
private fun line ( maxChar : Char , rowChar : Char ) =
mirror ( leftHalf ( maxChar , rowChar ))
private fun leftHalf ( maxChar : Char , rowChar : Char ): String {
val before = maxChar - rowChar
val after = rowChar - 'A'
return spaces ( before ) + rowChar + spaces ( after )
}
private fun spaces ( num : Int ) = "\" \"" . repeat ( num )
private fun mirror ( leftHalf : String ) =
leftHalf + reflection ( leftHalf )
private fun reflection ( leftHalf : String ) =
leftHalf . reversed (). drop ( 1 )
}
