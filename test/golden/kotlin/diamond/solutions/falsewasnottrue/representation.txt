class DiamondPrinter {
fun printToList ( end : Char ): List < String > {
require ( end in 'A' .. 'Z' )
val width = 2 * ( end - 'A' ) - 1
val halfDiamond = ( 'B' .. end ). fold ( mutableListOf ( "\" \"" . repeat ( end - 'A' ) + "\"A\"" + "\" \"" . repeat ( end - 'A' ))) {
acc , c ->
val offset = end - c
val row =
"\" \"" . repeat ( offset ) +
c +
"\" \"" . repeat ( width - 2 * offset ) +
c +
"\" \"" . repeat ( offset )
acc . add ( row )
acc
}
return halfDiamond + halfDiamond . dropLast ( 1 ). reversed ()
}
}
