class DiamondPrinter {
val alphabet = "\"abcdefghijklmnopqrstuvwxyz\""
fun getLetterPosition ( letter : Char ) : Int {
return alphabet . indexOf ( letter . lowercaseChar ())
}
fun getLetterAtPosition ( pos : Int ) : String {
return alphabet [ pos ]. uppercase ()
}
fun printToList ( letter : Char ) : List < String > {
val pos = getLetterPosition ( letter )
val size = pos + pos + 1
val line = Array ( size ) { "\".\"" }
val doubleArray = MutableList ( size ){ line . joinToString ( "\"\"" ) }
var incDecr = 0
while ( pos-incDecr >= 0 ) {
val lineIndexLow = pos-incDecr
val lineIndexHigh = pos + incDecr
val currentLetter = getLetterAtPosition ( lineIndexLow )
line . forEachIndexed { index , _ ->
if ( index == incDecr || index == ( line . size-1-incDecr )) {
line [ index ] = currentLetter
} else {
line [ index ] = "\" \""
}
}
doubleArray [ lineIndexLow ] = line . joinToString ( "\"\"" )
doubleArray [ lineIndexHigh ] = line . joinToString ( "\"\"" )
incDecr ++
}
println ( doubleArray )
return doubleArray
}
}
