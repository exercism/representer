private fun Char . intValue () = Character . getNumericValue ( this )
class DiamondPrinter {
fun printToList ( input : Char ): List < String > {
val chr = input . toUpperCase ()
val size = chr . intValue () - 'A' . intValue ()
val top = ( 'A' .. chr ). mapIndexed { index , c ->
pad ( size - index ) + line ( index , c ) + pad ( size - index )
}. toList ()
return if ( top . size == 1 ) top else top + top . reversed (). drop ( 1 )
}
private fun pad ( size : Int ) = "\" \"" . repeat ( size )
private fun line ( index : Int , c : Char ) =
if ( index == 0 ) c . toString () else c + pad (( index - 1 ) * 2 + 1 ) + c
}
