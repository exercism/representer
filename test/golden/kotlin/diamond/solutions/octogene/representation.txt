class DiamondPrinter {
fun printToList ( upToLetter : Char ): List < String > {
val range = ( 'A' .. upToLetter ). toList ()
val result = mutableListOf < String >()
var externalSpaces = range . size - 1
var internalSpaces = 0
range . forEachIndexed { index , c ->
result += "\" \"" . repeat ( externalSpaces )
. plus ( "\"$c\"" )
. plus ( "\" \"" . repeat ( internalSpaces ))
. plus ( "\"$c\"" . takeIf { index > 0 } ?: "\"\"" )
. plus ( "\" \"" . repeat ( externalSpaces ))
externalSpaces--
if ( internalSpaces == 0 ) {
internalSpaces = 1
} else {
internalSpaces += 2
}
}
return result . plus ( result . subList ( 0 , result . size - 1 ). reversed ())
}
}
