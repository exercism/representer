class DiamondPrinter {
fun printToList ( c : Char ) : List < String > =
( 'A' .. c ). mapIndexed { i , v ->
"\" \"" . repeat ( getLeadingSpacesForRow ( i , c ))
. plus ( v )
. plus ( "\" \"" . repeat ( i ))
. toList ()
. map { it . toString () }
. let ( this :: mirror )
. let { it . joinToString ( "\"\"" ) }
}. let ( this :: mirror )
private fun getLeadingSpacesForRow ( row : Int , c : Char ) = ( c - 'A' ). toInt () - row
private fun mirror ( input : List < String >) : List < String > =
input . plus ( input . take ( input . size - 1 ). reversed ())
}
