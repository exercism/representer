class DiamondPrinter {
fun printToList ( c : Char ): List < String > {
if ( c == 'A' ) {
return listOf ( c . toString ())
}
else {
val diamondWidth = ( c . toInt () - ( 'A' ). toInt ())* 2 + 1
return diamondBuilder ( c , diamondWidth )
}
}
private fun lineList ( c : Char ): List < Char > {
var done = false
var forward = true
var index = 'A'
val lb = ArrayList < Char >()
lb . add ( index )
while (! done ) {
if ( forward ) {
index = index . inc ()
}
else {
index = index . dec ()
}
if ( index == 'A' ) {
done = true
}
else {
lb . add ( index )
}
if ( index == c ) {
forward = false
}
}
lb . add ( index )
return lb . toList ()
}
private fun diamondBuilder ( c : Char , width : Int ) : List < String > {
val diamondList = ArrayList < String >()
val lineList = lineList ( c )
lineList . forEach { letter ->
diamondList . add ( buildLine ( letter , c , width ))
}
return diamondList . toList ()
}
private fun buildLine ( lc : Char , c : Char , width : Int ) : String {
val lb = StringBuilder ()
if ( lc == 'A' ) {
for ( d in 1 .. width / 2 ) { lb . append ( "' '" ) }
lb . append ( 'A' )
for ( d in 1 .. width / 2 ) { lb . append ( "' '" ) }
}
else {
val distance = c . toInt () - lc . toInt ()
for ( d in 1 .. distance ) { lb . append ( "' '" ) }
lb . append ( lc )
for ( d in 1 .. ( width - distance * 2 - 2 )) { lb . append ( "' '" ) }
lb . append ( lc )
for ( d in 1 .. distance ) { lb . append ( "' '" ) }
}
return lb . toString ()
}
}
