class DiamondPrinter {
val regex = Regex ( "\"^ | $\"" )
fun printToList ( char : Char ) =
generateSequence ( Pair ( 0 , true )) {
if ( it . first < char - 'A' && it . second ) {
Pair ( it . first + 1 , true )
} else {
Pair ( it . first - 1 , false )
}
}
. map () { it . first }
. takeWhile () { it >= 0 }
. map () {
val c = ( 'A' + it ). toChar (). toString ()
val l = it * 2 + 1
"\" \"" . repeat ( char - 'A' - it ) +
"\" \"" . repeat ( l ). replace ( regex , c ) +
"\" \"" . repeat ( char - 'A' - it )
}
. toList ()
}
