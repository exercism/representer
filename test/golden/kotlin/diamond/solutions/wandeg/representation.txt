class DiamondPrinter {
fun printToList ( letter : Char ): MutableList < String > {
var letters = mutableListOf < String >()
val numSpaces = (( letter . toByte (). toInt () - ( 'A' ). toByte (). toInt ())* 2 )+ 1
var lettersAscending = ascendStrings ( letter , numSpaces )
val lettersDescending = lettersAscending . reversed (). slice ( 1 .. lettersAscending . size-1 )
letters . addAll ( lettersAscending )
letters . addAll ( lettersDescending )
return letters
}
fun ascendStrings ( finalLetter : Char , spaces : Int ): MutableList < String >{
var prespace = 0
var postspace = 0
var midspace = 0
var letter = 'A'
var row = "\"\""
var spacesLeft = 0
var finalData = mutableListOf < String >()
while ( letter <= finalLetter ){
if ( letter == 'A' ){
postspace = spaces / 2
prespace = spaces / 2
row = createPadString ( prespace ) + letter + createPadString ( postspace )
finalData . add ( row )
}
else {
midspace = ((( letter ). toByte (). toInt () - ( 'A' ). toByte (). toInt ())* 2 )- 1
spacesLeft = spaces - ( midspace + 2 )
postspace = spacesLeft / 2
prespace = spacesLeft / 2
row = createPadString ( prespace ) + letter + createPadString ( midspace ) + letter + createPadString ( postspace )
finalData . add ( row )
}
++ letter
}
return finalData
}
fun createPadString ( length : Int ) : String {
var start = "\"\""
var i = 0
while ( i < length ){
start += "\" \""
++ i
}
return start
}
}
