import kotlin.math.max
class DiamondPrinter {
fun printToList ( letter : Char ): List < String > {
val top = buildTop ( letter )
val bottom = top . reversed (). drop ( 1 )
return top + bottom
}
private fun buildTop ( letter : Char ): List < String > {
var numLeadingAndTrailinSpaces = letter - 'A'
var interiorSpaces = - 1
return ( 'A' .. letter )
. toList ()
. map {
val leadingAndTrailingSpaces = "\" \"" . repeat ( numLeadingAndTrailinSpaces )
val line = if ( interiorSpaces <= 0 )
leadingAndTrailingSpaces + it + leadingAndTrailingSpaces
else
leadingAndTrailingSpaces + it + "\" \"" . repeat ( max ( interiorSpaces , 0 )) + it + leadingAndTrailingSpaces
numLeadingAndTrailinSpaces -= 1
interiorSpaces += 2
return @map line
}
}
}
