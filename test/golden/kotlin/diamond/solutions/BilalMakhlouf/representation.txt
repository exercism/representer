class DiamondPrinter {
fun printToList ( c : Char ): List < String > = ( 'A' .. c ). plus (( 'A' until c ). reversed ()). map {
when {
getAlphabetPosition ( c ) - getAlphabetPosition ( it ) == getAlphabetPosition ( c ) -> "\" \"" . repeat ( getAlphabetPosition ( c )) + it + "\" \"" . repeat ( getAlphabetPosition ( c ))
else -> "\" \"" . repeat ( getAlphabetPosition ( c ) - getAlphabetPosition ( it )) + it + "\" \"" . repeat ( 2 * getAlphabetPosition ( it ) - 1 ) + it + "\" \"" . repeat ( getAlphabetPosition ( c ) - getAlphabetPosition ( it ))
}
}
private fun getAlphabetPosition ( c : Char ) = c . toInt () - 65
}
