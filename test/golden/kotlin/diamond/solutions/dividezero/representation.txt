import java.lang.IllegalArgumentException
class DiamondPrinter {
fun printToList ( input : Char ): List < String > {
if ( input . toUpperCase () ! in 'A' .. 'Z' )
throw IllegalArgumentException ( "\"Input must be an alphabet\"" )
val characters = ( 'A' .. input ). toList ()
val result : ArrayList < String > = ArrayList ( characters . mapIndexed { idx , c -> getRowForChar ( c . toString (), idx , characters . size ) })
return result . apply {
addAll ( result . subList ( 0 , result . size - 1 ). reversed ())
}
}
private fun getRowForChar ( input : String , idx : Int , max : Int ): String =
if ( idx == 0 ) padAround ( input , max )
else padAround ( input . padEnd ( idx * 2 , "' '" ) + input , max )
private fun padAround ( input : String , max : Int ): String {
val totalLength = (( max * 2 - 1 ) - input . length )
return input . padStart ( input . length + ( totalLength / 2 )). padEnd ( input . length + totalLength )
}
}
