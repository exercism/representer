class DiamondPrinter {
fun printToList ( char : Char ): List < String >? {
val diamondTop = ( 'A' .. char ). mapIndexed { index , charToAdd ->
val before = char - charToAdd
val after = charToAdd - 'A'
val row = spaces ( before ) + charToAdd + spaces ( after )
row + row . reversed (). substring ( 1 )
}. toList ()
val diamondBottom = diamondTop . asReversed (). drop ( 1 )
return diamondTop . plus ( diamondBottom )
}
private fun spaces ( times : Int ) = "\" \"" . repeat ( times )
}
