class DiamondPrinter {
fun printToList ( c : Char ): List < String >? {
val lineLength = (( 'A' .. c ). toList (). size * 2 ) - 1
return ( 'A' .. c ). mapIndexed { index , char ->
expand ( getCharacterSpacing ( index , char ), lineLength )
} + ( 'A' until c ). mapIndexed { index , char ->
expand ( getCharacterSpacing ( index , char ), lineLength )
}. reversed ()
}
private fun getCharacterSpacing ( index : Int , char : Char ): String {
if ( index == 0 ) {
return "\"$char\""
} else {
return "\"$char${(1 until (index * 2)).joinToString(\"\") { \"" "\" }}$char\""
}
}
private fun expand ( str : String , size : Int ): String {
val spaces = "\" \"" . repeat (( size - str . length ) / 2 )
return "\"$spaces$str$spaces\""
}
}
