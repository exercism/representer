class DiamondPrinter {
private val alphabetRange = 'A' .. 'Z'
private val space = "\" \""
fun printToList ( letter : Char ): List < String > {
require ( letter in alphabetRange )
val width = alphabetRange . indexOf ( letter ) * 2 - 1
val letterRange : List < String > = (( 'A' until letter ) + ( letter downTo 'A' ))
. map {
val spaceIn = alphabetRange . indexOf ( it ) * 2 - 1
val spaceOut = ( width - spaceIn ) / 2
buildString {
repeat ( spaceOut ) { append ( space ) }
append ( it )
repeat ( spaceIn ) { append ( space ) }
if ( it != 'A' ) append ( it )
repeat ( spaceOut ) { append ( space ) }
}
}
return letterRange
}
}
