class DiamondPrinter {
fun printToList ( letter : Char ): List < String > {
val list = mutableListOf < String >()
val letterPosition = ( 'A' .. 'Z' ). indexOf ( letter )
for (( index , char ) in ( 'A' .. letter ). withIndex ()) {
val left = "\" \"" . repeat ( letterPosition - index ) + char + "\" \"" . repeat ( index )
list . add ( left + left . reversed (). substring ( 1 ))
}
list . addAll ( list . reversed (). slice ( 1 until list . size ))
return list
}
}
