import kotlin.math.max
import kotlin.math.min
class DiamondPrinter {
fun printToList ( c : Char ): List < String > {
val chars = ( 'A' .. c ). toList ()
val range = listOf ( 0 .. chars . size-1 , ( chars . size - 2 ) downTo 0 ). flatten ()
return ( range ). map {
val s = "\"\"" . padStart (( chars . size * 2 ) - 1 , "' '" ). toCharArray ()
val mid : Int = ( range . size / 2 ) + ( range . size % 2 ) - 1
val offset = it
s [ mid + offset ] = chars [ it ]
s [ mid - offset ] = chars [ it ]
return @map s . joinToString ( "\"\"" )
}
}
}
