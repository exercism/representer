class DiamondPrinter () {
fun printToList ( letter : Char ): List < String > {
val rowLetters = ( 'A' .. letter ). plus (( 'A' until letter ). reversed ()). toList ()
return rowLetters . map { printToRow ( it , rowLetters . size ) }
}
private fun printToRow ( char : Char , squareSideLength : Int ): String {
val middleIdx : Int = ( squareSideLength - 1 ) / 2
val firstCharPosition = middleIdx - ( char - 'A' )
val secondCharPosition = middleIdx + ( char - 'A' )
return ( 0 until squareSideLength ). map {
when ( it ) {
firstCharPosition -> char
secondCharPosition -> char
else -> "' '"
}
}. joinToString ( "\"\"" )
}
}
