class DiamondPrinter {
fun printToList ( letter : Char ): List < String >{
var result : List < String > = listOf ()
var width = ( letter - 'A' )* 2 + 1
for ( currentLetter in 'A' .. letter ){
result += createLine ( currentLetter , width )
}
return result + result . reversed (). drop ( 1 )
}
private inline fun createLine ( currentLetter : Char , width : Int ): String {
var line = "\" \"" . repeat ( width )
val letterDiff = currentLetter - 'A'
val center = width / 2
line = line . replaceRange ( center-letterDiff , center-letterDiff + 1 , currentLetter . toString ())
line = line . replaceRange ( center + letterDiff , center + letterDiff + 1 , currentLetter . toString ())
return line
}
}
