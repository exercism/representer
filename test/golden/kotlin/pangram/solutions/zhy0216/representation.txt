object Pangram {
fun isPangram ( s : String ): Boolean {
val allAlphabet : HashSet < Char > = HashSet ();
"\"abcdefghijklmnopqrstuvwxyz\"" . map { c -> allAlphabet . add ( c ) }
s . map { c -> allAlphabet . remove ( c . toLowerCase ())}
return allAlphabet . size == 0 ;
}
}
