import java.math.BigInteger
object Board {
fun getGrainCountForSquare ( number : Int ): BigInteger {
validateInput ( number )
if ( number == 1 ) return number . toBigInteger ()
return 2.0 . toBigDecimal (). pow ( number-1 ). toBigInteger ()
}
private fun validateInput ( number : Int ) {
if ( number <= 0 ) throw IllegalArgumentException ()
if ( number > 64 ) throw IllegalArgumentException ()
}
fun getTotalGrainCount (): BigInteger {
return IntRange ( 1 , 64 ). fold ( 0 . toBigInteger ()) { sum , number ->
sum + getGrainCountForSquare ( number )
}
}
}
