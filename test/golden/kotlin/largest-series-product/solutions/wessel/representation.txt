class Series {
var input : String
val inputList : List < Long >
var span : Int = 0
constructor ( input : String , span : Int = 0 ) {
this . input = input
this . span = span
if (! input . matches ( "\"-?\\\\d*\"" . toRegex ())) {
throw IllegalArgumentException ()
}
this . inputList = input . toList (). map { it . toString (). toLong () }
}
fun getLargestProduct ( span : Int ): Long {
return when {
span < 0 || span > input . length -> throw IllegalArgumentException ()
span == 0 -> 1
else -> inputList
. windowed ( span , 1 , false )
. map { it . fold ( 1 , { acc : Long , i : Long -> acc * i }) }
. max () ?: 1
}
}
}
