object Transpose {
fun transpose ( input : List < String >): List < String > {
if ( input . isEmpty ())
return input
val charMatrix = input . map { it . toCharArray () }
val transposedMatrixRowSize = charMatrix . maxOf { it . size }
val transposedMatrixColumnSize = input . size
val result = mutableListOf < String >()
for ( i in 0 until transposedMatrixRowSize ) {
val sb = StringBuilder ()
for ( j in 0 until transposedMatrixColumnSize ) {
if ( j < charMatrix . size && i < charMatrix [ j ]. size ) {
if ( j > 0 && sb . length < j )
sb . append ( "\" \"" . repeat ( j - sb . length ))
sb . append ( charMatrix [ j ][ i ])
}
}
result . add ( sb . toString ())
}
return result
}
}
