object Transpose {
fun transpose ( input : List < String >): List < String > =
( 0 until ( input . maxOfOrNull { it . length } ?: 0 ))
. map { row : Int ->
input . indices . joinToString ( "\"\"" ) { col : Int ->
if ( input [ col ]. length > row ) input [ col ][ row ]. toString () else "\"\u2672\""
}. trimEnd ( '♲' ). replace ( '♲' , "' '" )
}
}
