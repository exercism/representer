object Transpose {
fun transpose ( input : List < String >): List < String > {
var res = mutableListOf < String >()
input . forEachIndexed { i , it ->
it . forEachIndexed { index , c ->
if ( res . size <= index ) res . add ( index , "\"\"" . padEnd ( i , "' '" ) + c )
else res [ index ] = res [ index ]. padEnd ( i , "' '" ) + c
}
}
return res
}
}
