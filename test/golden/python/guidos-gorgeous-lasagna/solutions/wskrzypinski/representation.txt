(module
  (assignment EXPECTED_BAKE_TIME = 40)
  (assignment PLACEHOLDER_1 = 2)
  (function_definition bake_time_remaining
    (parameters PLACEHOLDER_2) :
    (block
      (return_statement
        (binary_operator EXPECTED_BAKE_TIME - PLACEHOLDER_2))))
  (function_definition preparation_time_in_minutes
    (parameters PLACEHOLDER_3) :
    (block
      (return_statement
        (binary_operator PLACEHOLDER_3 * PLACEHOLDER_1))))
  (function_definition elapsed_time_in_minutes
    (parameters PLACEHOLDER_3 PLACEHOLDER_2) :
    (block
      (return_statement
        (binary_operator
          (call preparation_time_in_minutes
            (argument_list PLACEHOLDER_3)) + PLACEHOLDER_2)))))
