(module
  (assignment EXPECTED_BAKE_TIME = 40)
  (function_definition bake_time_remaining
    (parameters PLACEHOLDER_1) :
    (block
      (return_statement
        (binary_operator EXPECTED_BAKE_TIME - PLACEHOLDER_1))))
  (function_definition preparation_time_in_minutes
    (parameters PLACEHOLDER_2) :
    (block
      (return_statement
        (binary_operator PLACEHOLDER_2 * 2))))
  (function_definition elapsed_time_in_minutes
    (parameters PLACEHOLDER_2 PLACEHOLDER_3) :
    (block
      (return_statement
        (binary_operator
          (call preparation_time_in_minutes
            (argument_list PLACEHOLDER_2)) + PLACEHOLDER_3)))))
