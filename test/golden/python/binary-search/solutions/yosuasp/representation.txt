(module
  (function_definition find
    (parameters PLACEHOLDER_1 PLACEHOLDER_2) :
    (block
      (if_statement
        (not_operator not PLACEHOLDER_1) :
        (block
          (raise_statement
            (call ValueError
              (argument_list "value not in array")))))
      (assignment PLACEHOLDER_3 =
        (binary_operator
          (call len
            (argument_list PLACEHOLDER_1)) // 2))
      (assignment PLACEHOLDER_4 =
        (subscript PLACEHOLDER_1 PLACEHOLDER_3))
      (if_statement
        (comparison_operator PLACEHOLDER_2 > PLACEHOLDER_4) :
        (block
          (return_statement
            (binary_operator
              (binary_operator PLACEHOLDER_3 + 1) +
              (call find
                (argument_list
                  (subscript PLACEHOLDER_1
                    (slice
                      (binary_operator PLACEHOLDER_3 + 1) :)) PLACEHOLDER_2))))))
      (if_statement
        (comparison_operator PLACEHOLDER_2 < PLACEHOLDER_4) :
        (block
          (return_statement
            (call find
              (argument_list
                (subscript PLACEHOLDER_1
                  (slice : PLACEHOLDER_3)) PLACEHOLDER_2)))))
      (return_statement PLACEHOLDER_3))))
