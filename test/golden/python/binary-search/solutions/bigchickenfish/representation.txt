(module
  (function_definition find
    (parameters PLACEHOLDER_1 PLACEHOLDER_2) :
    (block
      (assignment PLACEHOLDER_3 = 0)
      (assignment PLACEHOLDER_4 =
        (binary_operator
          (call len
            (argument_list PLACEHOLDER_1)) - 1))
      (while_statement
        (comparison_operator PLACEHOLDER_3 <= PLACEHOLDER_4) :
        (block
          (assignment PLACEHOLDER_5 =
            (binary_operator
              (binary_operator PLACEHOLDER_3 + PLACEHOLDER_4) // 2))
          (assignment PLACEHOLDER_6 =
            (subscript PLACEHOLDER_1 PLACEHOLDER_5))
          (if_statement
            (comparison_operator PLACEHOLDER_6 == PLACEHOLDER_2) :
            (block
              (return_statement PLACEHOLDER_5))
            (else_clause :
              (block
                (if_statement
                  (comparison_operator PLACEHOLDER_6 < PLACEHOLDER_2) :
                  (block
                    (assignment PLACEHOLDER_3 =
                      (binary_operator PLACEHOLDER_5 + 1)))
                  (else_clause :
                    (block
                      (assignment PLACEHOLDER_4 =
                        (binary_operator PLACEHOLDER_5 - 1))))))))))
      (raise_statement
        (call ValueError
          (argument_list "value not in array"))))))
