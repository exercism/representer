(module
  (function_definition square_of_sum
    (parameters PLACEHOLDER_1) :
    (block
      (assignment sum = 0)
      (assignment PLACEHOLDER_2 = 0)
      (while_statement
        (comparison_operator PLACEHOLDER_2 <= PLACEHOLDER_1) :
        (block
          (assignment sum =
            (binary_operator sum + PLACEHOLDER_2))
          (augmented_assignment PLACEHOLDER_2 += 1)))
      (return_statement
        (binary_operator sum * sum))))
  (function_definition sum_of_squares
    (parameters PLACEHOLDER_1) :
    (block
      (assignment sum = 0)
      (assignment PLACEHOLDER_2 = 0)
      (while_statement
        (comparison_operator PLACEHOLDER_2 <= PLACEHOLDER_1) :
        (block
          (assignment sum =
            (binary_operator sum +
              (binary_operator PLACEHOLDER_2 * PLACEHOLDER_2)))
          (augmented_assignment PLACEHOLDER_2 += 1)))
      (return_statement sum)))
  (function_definition difference_of_squares
    (parameters PLACEHOLDER_1) :
    (block
      (return_statement
        (binary_operator
          (call square_of_sum
            (argument_list PLACEHOLDER_1)) -
          (call sum_of_squares
            (argument_list PLACEHOLDER_1)))))))
