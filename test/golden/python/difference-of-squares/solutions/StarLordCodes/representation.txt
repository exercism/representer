(module
  (import_from_statement from
    (dotted_name math)
    (dotted_name pow))
  (function_definition square_of_sum
    (parameters PLACEHOLDER_1) :
    (block
      (assignment sum =
        (binary_operator
          (binary_operator PLACEHOLDER_1 *
            (binary_operator PLACEHOLDER_1 + 1)) / 2))
      (assignment PLACEHOLDER_2 =
        (call pow
          (argument_list sum 2)))
      (return_statement PLACEHOLDER_2)))
  (function_definition sum_of_squares
    (parameters PLACEHOLDER_1) :
    (block
      (assignment PLACEHOLDER_3 =
        (binary_operator
          (binary_operator
            (binary_operator PLACEHOLDER_1 *
              (binary_operator PLACEHOLDER_1 + 1)) *
            (binary_operator
              (binary_operator 2 * PLACEHOLDER_1) + 1)) / 6))
      (return_statement PLACEHOLDER_3)))
  (function_definition difference_of_squares
    (parameters PLACEHOLDER_1) :
    (block
      (return_statement
        (binary_operator
          (call square_of_sum
            (argument_list PLACEHOLDER_1)) -
          (call sum_of_squares
            (argument_list PLACEHOLDER_1)))))))
