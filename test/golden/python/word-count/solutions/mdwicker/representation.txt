(module
  (function_definition count_words
    (parameters PLACEHOLDER_1) :
    (block
      (assignment PLACEHOLDER_2 =
        (call
          (attribute re . findall)
          (argument_list "[a-zA-Z0-9]+(?:\\'?[a-zA-Z0-9]+)?" PLACEHOLDER_1)))
      (assignment word_count =
        (dictionary))
      (for_statement PLACEHOLDER_3 in PLACEHOLDER_2 :
        (block
          (assignment PLACEHOLDER_4 =
            (call
              (attribute PLACEHOLDER_3 . lower)
              (argument_list)))
          (assignment
            (subscript word_count PLACEHOLDER_4) =
            (binary_operator
              (call
                (attribute word_count . get)
                (argument_list PLACEHOLDER_4 0)) + 1))))
      (return_statement word_count))))
