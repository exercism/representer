(module
  (function_definition capitalize_title
    (parameters PLACEHOLDER_1) :
    (block
      (assignment PLACEHOLDER_1 =
        (call
          (attribute PLACEHOLDER_1 . split)
          (argument_list)))
      (return_statement
        (call
          (attribute " " . join)
          (argument_list
            (list_comprehension
              (call
                (attribute PLACEHOLDER_2 . capitalize)
                (argument_list))
              (for_in_clause PLACEHOLDER_2 in PLACEHOLDER_1)))))))
  (function_definition check_sentence_ending
    (parameters PLACEHOLDER_3) :
    (block
      (return_statement
        (call
          (attribute PLACEHOLDER_3 . endswith)
          (argument_list ".")))))
  (function_definition clean_up_spacing
    (parameters PLACEHOLDER_3) :
    (block
      (assignment PLACEHOLDER_3 =
        (call
          (attribute PLACEHOLDER_3 . strip)
          (argument_list)))
      (return_statement PLACEHOLDER_3)))
  (function_definition replace_word_choice
    (parameters PLACEHOLDER_3 PLACEHOLDER_4 PLACEHOLDER_5) :
    (block
      (return_statement
        (call
          (attribute PLACEHOLDER_3 . replace)
          (argument_list PLACEHOLDER_4 PLACEHOLDER_5))))))
