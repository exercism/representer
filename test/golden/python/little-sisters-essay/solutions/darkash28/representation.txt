(module
  (function_definition capitalize_title
    (parameters PLACEHOLDER_1) :
    (block
      (return_statement
        (call
          (attribute PLACEHOLDER_1 . title)
          (argument_list)))))
  (function_definition check_sentence_ending
    (parameters PLACEHOLDER_2) :
    (block
      (if_statement
        (call
          (attribute PLACEHOLDER_2 . endswith)
          (argument_list
            (tuple "." "!" "?"))) :
        (block
          (return_statement True)))
      (return_statement False)))
  (function_definition clean_up_spacing
    (parameters PLACEHOLDER_2) :
    (block
      (assignment PLACEHOLDER_3 =
        (list_comprehension PLACEHOLDER_4
          (for_in_clause PLACEHOLDER_4 in
            (call
              (attribute PLACEHOLDER_2 . split)
              (argument_list " ")))
          (if_clause
            (comparison_operator PLACEHOLDER_4 != ""))))
      (return_statement
        (call
          (attribute " " . join)
          (argument_list PLACEHOLDER_3)))))
  (function_definition replace_word_choice
    (parameters PLACEHOLDER_2 PLACEHOLDER_5 PLACEHOLDER_6) :
    (block
      (return_statement
        (call
          (attribute PLACEHOLDER_2 . replace)
          (argument_list PLACEHOLDER_5 PLACEHOLDER_6))))))
