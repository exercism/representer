(module
  (function_definition is_criticality_balanced
    (parameters PLACEHOLDER_1 neutrons_emitted) :
    (block
      (return_statement
        (boolean_operator
          (boolean_operator
            (comparison_operator PLACEHOLDER_1 < 800) and
            (comparison_operator neutrons_emitted > 500)) and
          (comparison_operator
            (binary_operator PLACEHOLDER_1 * neutrons_emitted) < 500000)))))
  (function_definition reactor_efficiency
    (parameters voltage current theoretical_max_power) :
    (block
      (assignment PLACEHOLDER_2 =
        (binary_operator voltage * current))
      (assignment PLACEHOLDER_3 =
        (binary_operator
          (binary_operator PLACEHOLDER_2 / theoretical_max_power) * 100))
      (if_statement
        (comparison_operator PLACEHOLDER_3 >= 80) :
        (block
          (return_statement "green"))
        (else_clause :
          (block
            (if_statement
              (boolean_operator
                (comparison_operator PLACEHOLDER_3 < 80) and
                (comparison_operator PLACEHOLDER_3 >= 60)) :
              (block
                (return_statement "orange"))
              (elif_clause
                (boolean_operator
                  (comparison_operator PLACEHOLDER_3 < 60) and
                  (comparison_operator PLACEHOLDER_3 >= 30)) :
                (block
                  (return_statement "red")))
              (elif_clause
                (comparison_operator PLACEHOLDER_3 < 30) :
                (block
                  (return_statement "black")))))))))
  (function_definition fail_safe
    (parameters PLACEHOLDER_1 PLACEHOLDER_4 threshold) :
    (block
      (assignment PLACEHOLDER_5 =
        (binary_operator PLACEHOLDER_1 * PLACEHOLDER_4))
      (if_statement
        (comparison_operator PLACEHOLDER_5 <
          (binary_operator 0.9 * threshold)) :
        (block
          (return_statement "LOW"))
        (else_clause :
          (block
            (if_statement
              (comparison_operator PLACEHOLDER_5 <
                (binary_operator 1.1 * threshold)) :
              (block
                (return_statement "NORMAL"))
              (elif_clause
                (comparison_operator PLACEHOLDER_5 >
                  (binary_operator 0.1 * threshold)) :
                (block
                  (return_statement "DANGER")))
              (else_clause :
                (block
                  (return_statement "DANGER"))))))))))
