(module
  (function_definition is_criticality_balanced
    (parameters PLACEHOLDER_1 neutrons_emitted) :
    (block
      (assignment PLACEHOLDER_2 =
        (comparison_operator PLACEHOLDER_1 < 800))
      (assignment PLACEHOLDER_3 =
        (comparison_operator neutrons_emitted > 500))
      (assignment PLACEHOLDER_4 =
        (comparison_operator
          (binary_operator neutrons_emitted * PLACEHOLDER_1) < 500000))
      (return_statement
        (boolean_operator
          (boolean_operator PLACEHOLDER_2 and PLACEHOLDER_3) and PLACEHOLDER_4))))
  (function_definition reactor_efficiency
    (parameters voltage current theoretical_max_power) :
    (block
      (assignment PLACEHOLDER_5 =
        (binary_operator voltage * current))
      (assignment PLACEHOLDER_6 =
        (binary_operator
          (binary_operator PLACEHOLDER_5 / theoretical_max_power) * 100.0))
      (assignment PLACEHOLDER_7 = "black")
      (if_statement
        (comparison_operator PLACEHOLDER_6 >= 80.0) :
        (block
          (assignment PLACEHOLDER_7 = "green"))
        (else_clause :
          (block
            (if_statement
              (comparison_operator PLACEHOLDER_6 >= 60.0) :
              (block
                (assignment PLACEHOLDER_7 = "orange"))
              (elif_clause
                (comparison_operator PLACEHOLDER_6 >= 30.0) :
                (block
                  (assignment PLACEHOLDER_7 = "red")))))))
      (return_statement PLACEHOLDER_7)))
  (function_definition fail_safe
    (parameters PLACEHOLDER_1 PLACEHOLDER_8 threshold) :
    (block
      (assignment PLACEHOLDER_9 =
        (binary_operator PLACEHOLDER_1 * PLACEHOLDER_8))
      (assignment PLACEHOLDER_10 =
        (binary_operator 0.9 * threshold))
      (assignment PLACEHOLDER_11 =
        (binary_operator 1.1 * threshold))
      (assignment PLACEHOLDER_12 =
        (comparison_operator PLACEHOLDER_10 < PLACEHOLDER_9 < PLACEHOLDER_11))
      (assignment PLACEHOLDER_13 =
        (comparison_operator PLACEHOLDER_10 > PLACEHOLDER_9))
      (assignment PLACEHOLDER_14 = "DANGER")
      (if_statement PLACEHOLDER_12 :
        (block
          (assignment PLACEHOLDER_14 = "NORMAL"))
        (else_clause :
          (block
            (if_statement PLACEHOLDER_13 :
              (block
                (assignment PLACEHOLDER_14 = "LOW"))))))
      (return_statement PLACEHOLDER_14))))
