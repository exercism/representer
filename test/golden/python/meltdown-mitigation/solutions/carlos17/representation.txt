(file "conditionals.py"
  (module
    (function_definition is_criticality_balanced
      (parameters PLACEHOLDER_1 neutrons_emitted) :
      (block
        (return_statement
          (boolean_operator
            (boolean_operator
              (comparison_operator PLACEHOLDER_1 < 800) and
              (comparison_operator neutrons_emitted > 500)) and
            (comparison_operator
              (binary_operator PLACEHOLDER_1 * neutrons_emitted) < 500000)))))
    (function_definition reactor_efficiency
      (parameters voltage current theoretical_max_power) :
      (block
        (assignment PLACEHOLDER_2 =
          (binary_operator
            (binary_operator
              (binary_operator voltage * current) / theoretical_max_power) * 100))
        (if_statement
          (comparison_operator PLACEHOLDER_2 >= 80) :
          (block
            (return_statement "green")))
        (if_statement
          (comparison_operator PLACEHOLDER_2 >= 60) :
          (block
            (return_statement "orange")))
        (if_statement
          (comparison_operator PLACEHOLDER_2 >= 30) :
          (block
            (return_statement "red")))
        (return_statement "black")))
    (function_definition fail_safe
      (parameters PLACEHOLDER_1 PLACEHOLDER_3 threshold) :
      (block
        (assignment PLACEHOLDER_4 =
          (binary_operator PLACEHOLDER_1 * PLACEHOLDER_3))
        (if_statement
          (comparison_operator PLACEHOLDER_4 <
            (binary_operator threshold * 0.9)) :
          (block
            (return_statement "LOW")))
        (if_statement
          (boolean_operator
            (comparison_operator PLACEHOLDER_4 >=
              (binary_operator threshold -
                (binary_operator threshold * 0.1))) and
            (comparison_operator PLACEHOLDER_4 <=
              (binary_operator threshold +
                (binary_operator threshold * 0.1)))) :
          (block
            (return_statement "NORMAL")))
        (return_statement "DANGER")))))
