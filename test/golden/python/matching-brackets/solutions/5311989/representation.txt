(file "matching_brackets.py"
  (module
    (function_definition is_paired
      (parameters PLACEHOLDER_1) :
      (block
        (assignment PLACEHOLDER_2 =
          (dictionary
            (pair "{" : "}")
            (pair "[" : "]")
            (pair "(" : ")")))
        (assignment PLACEHOLDER_3 =
          (list))
        (for_statement PLACEHOLDER_4 in PLACEHOLDER_1 :
          (block
            (if_statement
              (comparison_operator PLACEHOLDER_4 in
                (call
                  (attribute PLACEHOLDER_2 . keys)
                  (argument_list))) :
              (block
                (call
                  (attribute PLACEHOLDER_3 . append)
                  (argument_list PLACEHOLDER_4))))
            (if_statement
              (comparison_operator PLACEHOLDER_4 in
                (call
                  (attribute PLACEHOLDER_2 . values)
                  (argument_list))) :
              (block
                (if_statement
                  (boolean_operator
                    (not_operator not PLACEHOLDER_3) or
                    (comparison_operator PLACEHOLDER_4 !=
                      (subscript PLACEHOLDER_2
                        (subscript PLACEHOLDER_3
                          (unary_operator - 1))))) :
                  (block
                    (return_statement False)))
                (call
                  (attribute PLACEHOLDER_3 . pop)
                  (argument_list))))))
        (return_statement
          (not_operator not PLACEHOLDER_3))))))
