            (comparison_operator
              (binary_operator PLACEHOLDER_3 - PLACEHOLDER_7) >= 0) :
            (block "first: make a list of all possible combinations of items when leaving x items behind"

              (for_statement PLACEHOLDER_8 in PLACEHOLDER_6 :
                (block
                  (assignment PLACEHOLDER_9 = 0)
                  (assignment PLACEHOLDER_10 = 0)
                  (for_statement PLACEHOLDER_11 in PLACEHOLDER_8 :
                    (block
                      (assignment PLACEHOLDER_9 =
                        (binary_operator PLACEHOLDER_9 +
                          (subscript PLACEHOLDER_11 "weight")))
                      (assignment PLACEHOLDER_10 =
                        (binary_operator PLACEHOLDER_10 +
                          (subscript PLACEHOLDER_11 "value")))))
                  (if_statement
                    (comparison_operator PLACEHOLDER_9 <= PLACEHOLDER_1) :
                    (block
                      (if_statement
                        (comparison_operator PLACEHOLDER_10 > PLACEHOLDER_5) :
                        (block
                          (assignment PLACEHOLDER_5 = PLACEHOLDER_10)))))))))))
      (return_statement PLACEHOLDER_5))))
