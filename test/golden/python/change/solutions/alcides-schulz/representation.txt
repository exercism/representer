(module
  (function_definition find_fewest_coins
    (parameters PLACEHOLDER_1 PLACEHOLDER_2) :
    (block
      (if_statement
        (comparison_operator PLACEHOLDER_2 < 0) :
        (block
          (raise_statement
            (call ValueError
              (argument_list "target can't be negative")))))
      (assignment PLACEHOLDER_3 =
        (binary_operator PLACEHOLDER_2 + 1))
      (assignment PLACEHOLDER_4 =
        (binary_operator
          (list PLACEHOLDER_3) * PLACEHOLDER_3))
      (assignment PLACEHOLDER_5 =
        (binary_operator
          (list
            (list)) * PLACEHOLDER_3))
      (assignment
        (subscript PLACEHOLDER_4 0) = 0)
      (for_statement PLACEHOLDER_6 in
        (call range
          (argument_list 1 PLACEHOLDER_3)) :
        (block
          (for_statement PLACEHOLDER_7 in PLACEHOLDER_1 :
            (block
              (if_statement
                (boolean_operator
                  (comparison_operator PLACEHOLDER_6 >= PLACEHOLDER_7) and
                  (comparison_operator
                    (binary_operator
                      (subscript PLACEHOLDER_4
                        (binary_operator PLACEHOLDER_6 - PLACEHOLDER_7)) + 1) <
                    (subscript PLACEHOLDER_4 PLACEHOLDER_6))) :
                (block
                  (assignment
                    (subscript PLACEHOLDER_4 PLACEHOLDER_6) =
                    (binary_operator
                      (subscript PLACEHOLDER_4
                        (binary_operator PLACEHOLDER_6 - PLACEHOLDER_7)) + 1))
                  (assignment
                    (subscript PLACEHOLDER_5 PLACEHOLDER_6) =
                    (binary_operator
                      (list PLACEHOLDER_7) +
                      (subscript PLACEHOLDER_5
                        (binary_operator PLACEHOLDER_6 - PLACEHOLDER_7))))))))))
      (if_statement
        (comparison_operator
          (subscript PLACEHOLDER_4 PLACEHOLDER_2) == PLACEHOLDER_3) :
        (block
          (raise_statement
            (call ValueError
              (argument_list "can't make target with given coins")))))
      (return_statement
        (subscript PLACEHOLDER_5 PLACEHOLDER_2)))))
