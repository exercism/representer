(module
  (function_definition add_prefix_un
    (parameters word) :
    (block
      (return_statement
        (binary_operator "un" + word))))
  (function_definition make_word_groups
    (parameters PLACEHOLDER_1) :
    (block
      (assignment PLACEHOLDER_2 =
        (call
          (attribute PLACEHOLDER_1 . pop)
          (argument_list 0)))
      (assignment PLACEHOLDER_3 =
        (binary_operator PLACEHOLDER_2 + " :: "))
      (for_statement word in PLACEHOLDER_1 :
        (block
          (augmented_assignment PLACEHOLDER_3 +=
            (binary_operator
              (binary_operator PLACEHOLDER_2 + word) + " :: "))))
      (return_statement
        (subscript PLACEHOLDER_3
          (slice :
            (unary_operator - 4))))))
  (function_definition remove_suffix_ness
    (parameters word) :
    (block
      (assignment word =
        (subscript word
          (slice :
            (unary_operator - 4))))
      (if_statement
        (comparison_operator
          (subscript word
            (unary_operator - 1)) == "i") :
        (block
          (assignment word =
            (binary_operator
              (subscript word
                (slice :
                  (unary_operator - 1))) + "y"))))
      (return_statement word)))
  (function_definition adjective_to_verb
    (parameters sentence index) :
    (block
      (assignment sentence =
        (call
          (attribute
            (call
              (attribute sentence . replace)
              (argument_list "." "")) . split)
          (argument_list " ")))
      (return_statement
        (binary_operator
          (call str
            (argument_list
              (subscript sentence index))) + "en")))))
