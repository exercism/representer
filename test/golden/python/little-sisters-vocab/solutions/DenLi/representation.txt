(module
  (function_definition add_prefix_un
    (parameters word) :
    (block
      (return_statement
        (binary_operator "un" + word))))
  (function_definition make_word_groups
    (parameters PLACEHOLDER_1) :
    (block
      (assignment PLACEHOLDER_2 =
        (list))
      (call
        (attribute PLACEHOLDER_2 . append)
        (argument_list
          (subscript PLACEHOLDER_1 0)))
      (for_statement index in
        (call range
          (argument_list 1
            (call len
              (argument_list PLACEHOLDER_1)))) :
        (block
          (call
            (attribute PLACEHOLDER_2 . append)
            (argument_list
              (binary_operator
                (subscript PLACEHOLDER_1 0) +
                (subscript PLACEHOLDER_1 index))))))
      (return_statement
        (call
          (attribute " :: " . join)
          (argument_list PLACEHOLDER_2)))))
  (function_definition remove_suffix_ness
    (parameters word) :
    (block
      (assignment word =
        (subscript word
          (slice :
            (unary_operator - 4))))
      (if_statement
        (comparison_operator
          (subscript word
            (unary_operator - 1)) == "i") :
        (block
          (return_statement
            (binary_operator
              (subscript word
                (slice :
                  (unary_operator - 1))) + "y"))))
      (return_statement word)))
  (function_definition adjective_to_verb
    (parameters sentence index) :
    (block
      (assignment word =
        (subscript
          (call
            (attribute sentence . split)
            (argument_list)) index))
      (if_statement
        (comparison_operator
          (subscript word
            (unary_operator - 1)) in
          (list "." "," "!" "?")) :
        (block
          (assignment word =
            (subscript word
              (slice :
                (unary_operator - 1))))))
      (return_statement
        (binary_operator word + "en")))))
