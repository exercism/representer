(module
  (function_definition add_prefix_un
    (parameters word) :
    (block
      (return_statement
        (fstring "un"
          (interpolation word)))))
  (function_definition make_word_groups
    (parameters PLACEHOLDER_1) :
    (block
      (return_statement
        (call
          (attribute
            (fstring " :: "
              (interpolation
                (subscript PLACEHOLDER_1 0))) . join)
          (argument_list PLACEHOLDER_1)))))
  (function_definition remove_suffix_ness
    (parameters word) :
    (block
      (if_statement
        (call
          (attribute
            (call
              (attribute word . rstrip)
              (argument_list "ness")) . endswith)
          (argument_list "i")) :
        (block
          (return_statement
            (call
              (attribute
                (call
                  (attribute word . rstrip)
                  (argument_list "ness")) . replace)
              (argument_list "i" "y")))))
      (return_statement
        (call
          (attribute word . rstrip)
          (argument_list "ness")))))
  (function_definition adjective_to_verb
    (parameters sentence index) :
    (block
      (assignment PLACEHOLDER_2 =
        (call
          (attribute sentence . split)
          (argument_list)))
      (return_statement
        (fstring
          (interpolation
            (call
              (attribute
                (subscript PLACEHOLDER_2 index) . rstrip)
              (argument_list "."))) "en")))))
