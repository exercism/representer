    (parameters word) :

      (return_statement
        (binary_operator "un" + word))))

    (parameters PLACEHOLDER_1) :
    (block
      (for_statement word in
        (call range
          (argument_list 1
            (call len
              (argument_list PLACEHOLDER_1)))) :
        (block
          (assignment
            (subscript PLACEHOLDER_1 word) =
            (binary_operator
              (subscript PLACEHOLDER_1 0) +
              (subscript PLACEHOLDER_1 word)))))
      (return_statement
        (call
          (attribute " :: " . join)
          (argument_list PLACEHOLDER_1)))))
  (function_definition remove_suffix_ness
    (parameters word) :
    (block
      (if_statement
        (comparison_operator
          (subscript word
            (unary_operator - 5)) == "i") :
        (block
          (return_statement
            (binary_operator
              (subscript word
                (slice 0 :
                  (unary_operator - 5))) + "y"))))
      (return_statement
        (subscript word
          (slice 0 :
            (unary_operator - 4))))))

    (parameters sentence index) :
    (block

            (block

                (block
                  (return_statement

                      (subscript
                        (subscript
                          (call
                            (attribute sentence . split)

                (binary_operator
                  (subscript
                    (subscript
                      (call

                      (unary_operator - 1))) + "en"))))))
      (return_statement

          (subscript
            (call
              (attribute sentence . split)
