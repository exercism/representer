(file "strings.py"
  (module
    (import_from_statement from
      (dotted_name operator)
      (dotted_name concat))
    (import_statement
      (dotted_name re))
    (function_definition add_prefix_un
      (parameters word) :
      (block
        (return_statement
          (call concat
            (argument_list "un" word)))))
    (function_definition make_word_groups
      (parameters PLACEHOLDER_1) :
      (block
        (call
          (attribute PLACEHOLDER_1 . insert)
          (argument_list 1 ""))
        (assignment
          (tuple_pattern PLACEHOLDER_2
            (list_splat_pattern * PLACEHOLDER_3)) = PLACEHOLDER_1)
        (return_statement
          (call
            (attribute " :: " . join)
            (generator_expression
              (call concat
                (argument_list PLACEHOLDER_2 word))
              (for_in_clause word in PLACEHOLDER_3))))))
    (function_definition remove_suffix_ness
      (parameters word) :
      (block
        (return_statement
          (conditional_expression
            (call concat
              (argument_list
                (subscript word
                  (slice :
                    (unary_operator - 5))) "y"))
            (call
              (attribute word . endswith)
              (argument_list "iness"))
            (subscript word
              (slice :
                (unary_operator - 4)))))))
    (function_definition adjective_to_verb
      (parameters sentence index) :
      (block
        (return_statement
          (call concat
            (argument_list
              (subscript
                (call
                  (attribute re . findall)
                  (argument_list "\\w+" sentence)) index) "en")))))))
