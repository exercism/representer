  (function_definition steps
    (parameters PLACEHOLDER_1) :
    (block
      (if_statement
        (comparison_operator PLACEHOLDER_1 < 1) :
        (block
          (raise_statement
            (call ValueError
              (argument_list "Only positive integers are allowed")))))
      (assignment sum = 1)
      (if_statement
        (comparison_operator PLACEHOLDER_1 == 1) :
        (block
          (return_statement 0)))
      (if_statement
        (comparison_operator
          (binary_operator PLACEHOLDER_1 % 2) == 0) :
        (block
          (augmented_assignment sum +=
            (call steps
              (argument_list
                (binary_operator PLACEHOLDER_1 / 2))))
          (return_statement sum))
        (else_clause :
          (block
            (augmented_assignment sum +=
              (call steps

                  (binary_operator
                    (binary_operator PLACEHOLDER_1 * 3) + 1))))
            (return_statement sum)))))))
