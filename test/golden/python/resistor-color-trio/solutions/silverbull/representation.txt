(file "resistor_color_trio.py"
  (module
    (function_definition label
      (parameters PLACEHOLDER_1) :
      (block
        (assignment PLACEHOLDER_2 =
          (list "black" "brown" "red" "orange" "yellow" "green" "blue" "violet" "grey" "white"))
        (assignment PLACEHOLDER_3 =
          (binary_operator
            (binary_operator
              (binary_operator 10 *
                (call
                  (attribute PLACEHOLDER_2 . index)
                  (argument_list
                    (subscript PLACEHOLDER_1 0)))) +
              (call
                (attribute PLACEHOLDER_2 . index)
                (argument_list
                  (subscript PLACEHOLDER_1 1)))) *
            (binary_operator 10 **
              (call
                (attribute PLACEHOLDER_2 . index)
                (argument_list
                  (subscript PLACEHOLDER_1 2))))))
        (if_statement
          (comparison_operator PLACEHOLDER_3 > 1000000000) :
          (block
            (assignment PLACEHOLDER_4 = "giga")
            (augmented_assignment PLACEHOLDER_3 //= 1000000000))
          (else_clause :
            (block
              (if_statement
                (comparison_operator PLACEHOLDER_3 > 1000000) :
                (block
                  (assignment PLACEHOLDER_4 = "mega")
                  (augmented_assignment PLACEHOLDER_3 //= 1000000))
                (elif_clause
                  (comparison_operator PLACEHOLDER_3 > 1000) :
                  (block
                    (assignment PLACEHOLDER_4 = "kilo")
                    (augmented_assignment PLACEHOLDER_3 //= 1000)))
                (else_clause :
                  (block
                    (assignment PLACEHOLDER_4 = "")))))))
        (return_statement
          (fstring
            (interpolation PLACEHOLDER_3) " "
            (interpolation PLACEHOLDER_4) "ohms"))))))
