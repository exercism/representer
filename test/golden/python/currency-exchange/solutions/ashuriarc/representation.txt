(module
  (function_definition exchange_money
    (parameters budget exchange_rate) :
    (block
      (return_statement
        (binary_operator budget / exchange_rate))))
  (function_definition get_change
    (parameters budget exchanging_value) :
    (block
      (return_statement
        (binary_operator budget - exchanging_value))))
  (function_definition get_value_of_bills
    (parameters denomination number_of_bills) :
    (block
      (return_statement
        (binary_operator denomination * number_of_bills))))
  (function_definition get_number_of_bills
    (parameters amount denomination) :
    (block
      (return_statement
        (call int
          (argument_list
            (binary_operator amount // denomination))))))
  (function_definition get_leftover_of_bills
    (parameters amount denomination) :
    (block
      (return_statement
        (binary_operator amount % denomination))))
  (function_definition exchangeable_value
    (parameters budget exchange_rate spread denomination) :
    (block
      (assignment PLACEHOLDER_1 =
        (binary_operator exchange_rate *
          (binary_operator 1 +
            (binary_operator spread / 100))))
      (assignment PLACEHOLDER_2 =
        (binary_operator budget / PLACEHOLDER_1))
      (assignment PLACEHOLDER_3 =
        (binary_operator PLACEHOLDER_2 // denomination))
      (assignment PLACEHOLDER_4 =
        (binary_operator PLACEHOLDER_3 * denomination))
      (return_statement PLACEHOLDER_4))))
