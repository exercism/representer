(module
  (import_from_statement from
    (dotted_name statistics)
    (dotted_name mean))
  (function_definition get_rounds
    (parameters number) :
    (block
      (return_statement
        (list number
          (binary_operator number + 1)
          (binary_operator number + 2)))))
  (function_definition concatenate_rounds
    (parameters rounds_1 rounds_2) :
    (block
      (call
        (attribute rounds_1 . extend)
        (argument_list rounds_2))
      (return_statement rounds_1)))
  (function_definition list_contains_round
    (parameters rounds number) :
    (block
      (return_statement
        (comparison_operator number in rounds))))
  (function_definition card_average
    (parameters hand) :
    (block
      (return_statement
        (call mean
          (argument_list hand)))))
  (function_definition approx_average_is_average
    (parameters hand) :
    (block
      (return_statement
        (boolean_operator
          (comparison_operator
            (named_expression PLACEHOLDER_1 :=
              (call card_average
                (argument_list hand))) ==
            (binary_operator
              (binary_operator
                (subscript hand 0) +
                (subscript hand
                  (unary_operator - 1))) / 2)) or
          (comparison_operator PLACEHOLDER_1 ==
            (subscript hand
              (binary_operator
                (call len
                  (argument_list hand)) // 2)))))))
  (function_definition average_even_is_average_odd
    (parameters hand) :
    (block
      (return_statement
        (comparison_operator
          (call card_average
            (argument_list
              (subscript hand
                (slice : : 2)))) ==
          (call card_average
            (argument_list
              (subscript hand
                (slice 1 : : 2))))))))
  (function_definition maybe_double_last
    (parameters hand) :
    (block
      (if_statement
        (comparison_operator
          (named_expression PLACEHOLDER_2 :=
            (subscript hand
              (unary_operator - 1))) == 11) :
        (block
          (assignment
            (subscript hand
              (unary_operator - 1)) =
            (binary_operator PLACEHOLDER_2 * 2))))
      (return_statement hand))))
