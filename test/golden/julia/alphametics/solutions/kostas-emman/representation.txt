include ( "\"permutations.jl\"" )
const coefficients = [ 10 ^ i for i in 9 :- 1 : 0 ]
function arrayofchars2int ( char_arr :: Vector { Char }, mapping :: Dict { Char , Int })
digits = [ mapping [ c ] for c in char_arr ]
len = length ( digits )
result = sum ( digits .* coefficients [ end - len + 1 : end ])
end
function solve ( str :: AbstractString )
words = split ( str , ! isletter , keepempty = false )
char_arrays = words .|> collect
result_char_arr = pop! ( char_arrays )
initials = Set ( word [ begin ] for word in words )
chars = unique ( string ( words ... ))
perms = permutations ( 0 : 9 , length ( chars ))
for perm in perms
mappings = Dict { Char , Int }()
valid = true
for ( c , i ) in zip ( chars , perm )
in ( c , initials ) && i == 0 && ( valid = false ; break )
mappings [ c ] = i
end
valid || continue
last_digits_sum = sum ([ mappings [ c ] for c in last . ( char_arrays )])
iszero (( last_digits_sum - mappings [ last ( result_char_arr )]) % 10 ) || continue
acc = sum ( arrayofchars2int . ( char_arrays , ( mappings ,)))
arrayofchars2int ( result_char_arr , mappings ) == acc && return mappings
end
return nothing
end
