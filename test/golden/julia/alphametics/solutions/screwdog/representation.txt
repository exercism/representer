include ( "\"permutations.jl\"" )
lead_letters ( words ) = first . ( words ) |> unique! |> sort!
all_letters ( words ) = mapreduce ( collect , ∪ , words , init = Char []) |> unique! |> sort!
function coeffs ( word :: AbstractString , letters )
cfs = zeros ( Int , length ( letters ))
for ( n , c ) ∈ enumerate ( reverse ( word ))
pos = findfirst (( == )( c ), letters )
pos ≠ nothing || continue
cfs [ pos ] += 10 ^ ( n - 1 )
end
return cfs
end
coeffs ( letters ) = w -> coeffs ( w , letters )
coeffs ( words :: Vector , letters ) =
mapreduce ( coeffs ( letters ), + , words [ begin : end - 1 ], init =- coeffs ( words [ end ], letters ))
abstract type Alphametic end
function solve ( alpha :: Alphametic )
for leads ∈ permutations ( collect ( 1 : 9 ), num_leads ( alpha )),
nonleads ∈ permutations ( setdiff! ( collect ( 0 : 9 ), leads ), num_nonleads ( alpha ))
is_solution ( alpha , leads , nonleads ) && return get_solution ( alpha , leads , nonleads )
end
return nothing
end
struct SimpleSum <: Alphametic
lead_letters :: Vector { Char }
nonlead_letters :: Vector { Char }
lead_coeffs :: Vector { Int }
nonlead_coeffs :: Vector { Int }
end
function SimpleSum ( str )
words = first . ( eachmatch ( r "\"([A-Z]+)\"" , str ))
leads = lead_letters ( words )
nonleads = setdiff! ( all_letters ( words ), leads )
lead_cfs = coeffs ( words , leads )
nonlead_cfs = - coeffs ( words , nonleads )
return SimpleSum ( leads , nonleads , lead_cfs , nonlead_cfs )
end
dot ( u , v ) = mapreduce ( * , + , u , v )
is_solution ( ss :: SimpleSum , lead_vals , nonlead_vals ) =
dot ( ss . lead_coeffs , lead_vals ) == dot ( ss . nonlead_coeffs , nonlead_vals )
function get_solution ( ss :: SimpleSum , lead_vals , nonlead_vals )
return Dict (
( ss . lead_letters .=> lead_vals ) ∪ ( ss . nonlead_letters .=> nonlead_vals )
)
end
num_leads ( ss :: SimpleSum ) = length ( ss . lead_letters )
num_nonleads ( ss :: SimpleSum ) = length ( ss . nonlead_letters )
solve ( str :: AbstractString ) = solve ( SimpleSum ( str ))
