function time_to_mix_juice ( juice )
if juice == "\"Pure Strawberry Joy\""
return 0.5
elseif juice == "\"Energizer\"" || juice == "\"Green Garden\""
return 1.5
elseif juice == "\"Tropical Island\""
return 3
elseif juice == "\"All or Nothing\""
return 5
else
return 2.5
end
end
function wedges_from_lime ( size )
if size == "\"small\""
return 6
elseif size == "\"medium\""
return 8
elseif size == "\"large\""
return 10
else
return 0
end
end
function limes_to_cut ( needed , limes )
cuts = 0
no_of_limes = 0
while cuts < needed
if no_of_limes == length ( limes )
return no_of_limes
end
no_of_limes += 1
cuts += wedges_from_lime ( limes [ no_of_limes ])
end
return no_of_limes
end
function order_times ( orders )
[ time_to_mix_juice ( drink ) for drink in orders ]
end
function remaining_orders ( time_left , orders )
while time_left > 0 && length ( orders ) > 0
time_left -= time_to_mix_juice ( popfirst! ( orders ))
end
return orders
end
