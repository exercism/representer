(file "clock.jl")
using Dates
struct Clock
hour :: Int
minute :: Int
function Clock ( h , m )
m_ = mod ( m , 60 )
h = ( h + ( m - m_ ) ÷ 60 ) % 24
new ( mod ( h , 24 ), m_ )
end
end
import Base : + , -
+ ( x :: Clock , y :: Minute ) = Clock ( x . hour , x . minute + y . value )
- ( x :: Clock , y :: Minute ) = Clock ( x . hour , x . minute - y . value )
function Base.show ( io :: IO , time :: Clock )
print ( io , "\"\\\"$(lpad(time.hour, 2, '0')):$(lpad(time.minute, 2, '0'))\\\"\"" )
end
