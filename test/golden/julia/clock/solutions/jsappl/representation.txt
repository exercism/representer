(file "clock.jl")
using Dates
import Base : == , + , - , show
mutable struct Clock
hours :: Int
minutes :: Int
Clock ( hours , minutes ) =
let
remainder = mod ( hours * 60 + minutes , 24 * 60 )
hh , mm = divrem ( remainder , 60 )
new ( hh , mm )
end
end
== ( first_clock :: Clock , second_clock :: Clock ) = ( first_clock . hours == second_clock . hours ) && ( first_clock . minutes == second_clock . minutes )
+ ( clock :: Clock , minutes :: Minute ) = Clock ( clock . hours , clock . minutes + Dates . value ( minutes ))
- ( clock :: Clock , minutes :: Minute ) = clock + - minutes
show ( io :: IO , clock :: Clock ) = print ( io , "\"\\\"$(lpad(clock.hours, 2, '0')):$(lpad(clock.minutes, 2, '0'))\\\"\"" )
