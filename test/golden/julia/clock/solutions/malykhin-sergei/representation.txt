(file "clock.jl")
using Printf , Dates
struct Clock
hours :: Int64
minutes :: Int64
function Clock ( h , m )
h += m ≥ 0 ? m ÷ 60 : m ÷ 60 - ( m % 60 == 0 ? 0 : 1 )
h = h ≥ 0 ? h % 24 : ( h % 24 + 24 ) % 24
m = m ≥ 0 ? m % 60 : ( m % 60 + 60 ) % 60
new ( h , m )
end
end
Base .: ( + )( x :: Clock , y :: Minute ) = Clock ( x . hours , x . minutes + y . value )
Base .: ( - )( x :: Clock , y :: Minute ) = Clock ( x . hours , x . minutes - y . value )
Base .: ( == )( x :: Clock , y :: Clock ) = x . hours == y . hours && x . minutes == y . minutes
Base . show ( io :: IO , x :: Clock ) = @printf ( io , "\"\\\"%02i:%02i\\\"\"" , x . hours , x . minutes )
