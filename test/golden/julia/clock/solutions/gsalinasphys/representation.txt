import Dates
function Clock ( hours , minutes )
extra_hours = minutes÷60
hours += extra_hours
hours %= 24
minutes %= 60
if minutes < 0
minutes += 60
hours -= 1
end
if hours < 0
hours += 24
end
Dates . Time ( hours , minutes )
end
function show ( DT :: Time )
Dates . format ( DT , "\"\\\"HH:MM\\\"\"" )
end
function sprint ( func :: Function , DT :: Time )
show ( DT )
end
