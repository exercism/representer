(file "clock.jl")
import Dates
import Base .+
import Base .-
struct Clock
hour :: Int8
minute :: Int8
Clock ( hours_ , minutes_ ) = (
minutes = mod ( minutes_ , 60 );
hours = hours_ + div ( minutes_ , 60 );
if ( minutes_ < 0 && minutes != 0 )
hours -= 1 ;
end ;
hours = mod ( hours , 24 );
return new ( hours , minutes )
)
end
( + )( x :: Clock , d :: Dates . Minute ) = begin
minutes_ = x . minute + d . value
Clock ( x . hour , minutes_ )
end
( - )( x :: Clock , d :: Dates . Minute ) = begin
minutes_ = x . minute - d . value
Clock ( x . hour , minutes_ )
end
show ( io :: IO , clock :: Clock ) = print ( io , "\"\\\"$(lpad(clock.hour, 2, \"" 0 "\")):$(lpad(clock.minute, 2, \"" 0 "\"))\\\"\"" )
