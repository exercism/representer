(file "clock.jl")
using Dates : Minute
using Printf : @printf
import Base : + , - , show
struct Clock
minutes :: Int
Clock ( hours :: Int , minutes :: Int ) = new ( mod ( hours * 60 + minutes , 60 * 24 ))
Clock ( minutes :: Int ) = new ( mod ( minutes , 60 * 24 ))
end
+ ( c :: Clock , mins :: Minute ) = Clock ( c . minutes + mins . value )
- ( c :: Clock , mins :: Minute ) = Clock ( c . minutes - mins . value )
show ( io :: IO , c :: Clock ) = @printf ( io , "\"\\\"%02d:%02d\\\"\"" , div ( c . minutes , 60 ), c . minutes % 60 )
