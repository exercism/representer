(file "clock.jl")
import Dates
using Printf
import Base .!=
import Base .==
import Base .+
import Base .-
"\"\"\"Represents the time of day.\n\nfields: hour, minute\n\"\"\""
mutable struct Clock
hour :: Int64
minute :: Int64
function Clock ( hour = 0 , minute = 0 )
minutes = hour * 60 + minute
while minutes < 0
minutes += 60
end
while hour < 0
hour += 24
end
minutes = hour * 60 + minute
while minutes < 0
minutes += 24 * 60
end
hours , minute = divrem ( minutes , 60 )
days , hour = divrem ( hours , 24 )
new ( hour , minute )
end
end
function Base.show ( io :: IO , clock :: Clock )
@printf ( io , "\"\\\"%02d:%02d\\\"\"" , clock . hour , clock . minute )
end
function clocktoint ( clock :: Clock )
minutes = clock . hour * 60 + clock . minute
while minutes < 0
minutes += 60
end
while clock . hour < 0
clock . hour += 24
end
minutes = clock . hour * 60 + clock . minute
while minutes < 0
minutes += 24 * 60
end
return minutes
end
function inttoclock ( minutes :: Int ) :: Clock
hours , minute = divrem ( minutes , 60 )
days , hour = divrem ( hours , 24 )
return Clock ( hour , minute )
end
function + ( clock :: Clock , minute :: Dates . Minute ) :: Clock
inttoclock ( clocktoint ( clock ) + clocktoint ( Clock ( 0 , minute . value )))
end
function + ( minute :: Dates . Minute , clock :: Clock ) :: Clock
inttoclock ( clocktoint ( clock ) + clocktoint ( Clock ( 0 , minute . value )))
end
function -( clock :: Clock , minute :: Dates . Minute ) :: Clock
inttoclock ( clocktoint ( clock ) - clocktoint ( Clock ( 0 , minute . value )))
end
function -( minute :: Dates . Minute , clock :: Clock ) :: Clock
inttoclock ( clocktoint ( clock ) - clocktoint ( Clock ( 0 , minute . value )))
end
function == ( c1 :: Clock , c2 :: Clock )
hour_check = inttoclock ( clocktoint ( c1 )) . hour == inttoclock ( clocktoint ( c2 )) . hour
minute_check = inttoclock ( clocktoint ( c1 )) . minute == inttoclock ( clocktoint ( c2 )) . minute
return hour_check && minute_check
end
function != ( c1 :: Clock , c2 :: Clock )
hour_check = inttoclock ( clocktoint ( c1 )) . hour != inttoclock ( clocktoint ( c2 )) . hour
minute_check = inttoclock ( clocktoint ( c1 )) . minute != inttoclock ( clocktoint ( c2 )) . minute
return hour_check || minute_check
end
