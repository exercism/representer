struct Clock
h :: Int
m :: Int
function Clock ( hours , minutes )
totalmins = mod ( 60 hours + minutes , 24 * 60 )
return new ( divrem ( totalmins , 60 ) ... )
end
end
Base .:+ ( a :: Clock , b :: Clock ) = Clock ( a . h + b . h , a . m + b . m )
Base .:+ ( c :: Clock , m :: Dates . Minute ) = Clock ( c . h , c . m + m . value )
Base .:- ( c :: Clock , m :: Dates . Minute ) = Clock ( c . h , c . m - m . value )
function Base.show ( io :: IO , c :: Clock )
hours = lpad ( c . h , 2 , '0' )
mins = lpad ( c . m , 2 , '0' )
print ( io , "\"\\\"$hours:$mins\\\"\"" )
end
