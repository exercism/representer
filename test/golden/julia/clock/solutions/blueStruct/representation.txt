using Printf
struct Clock
hour :: Int
min :: Int
function Clock ( x , y )
if y > 59
inc , y = divrem ( y , 60 )
x += inc
end
while y < 0
x , y = x - 1 , y + 60
end
while x > 23 || x < 0
x += 24
x %= 24
end
return new ( x , y )
end
end
Base .:+ ( c :: Clock , m :: Dates . Minute ) = Clock ( c . hour , c . min + m . value )
Base .:- ( c :: Clock , m :: Dates . Minute ) = Clock ( c . hour , c . min - m . value )
Base .: sprint ( f :: Function , c :: Clock ) = @sprintf "\"\\\"%02d:%02d\\\"\"" c . hour c . min
