(file "clock.jl")
import Base : + , - , show
using Dates
using Printf
const HOURS_PER_DAY = 24
const MINUTES_PER_HOUR = 60
const MINUTES_PER_DAY = HOURS_PER_DAY * MINUTES_PER_HOUR
struct Clock
hours
minutes
function Clock ( hours :: Integer , minutes :: Integer )
if minutes < 0
minutes += - Integer ( floor ( minutes / MINUTES_PER_DAY )) * MINUTES_PER_DAY
end
hours = mod ( hours + minutes ÷ MINUTES_PER_HOUR , HOURS_PER_DAY )
minutes = mod ( minutes , MINUTES_PER_HOUR )
new ( hours , minutes )
end
end
function (+) ( clock :: Clock , minutes :: Dates . Minute )
Clock ( clock . hours , clock . minutes + minutes . value )
end
function (-) ( clock :: Clock , minutes :: Dates . Minute )
clock + ( - minutes )
end
function show ( x :: Clock )
show ( stdout , x )
end
function show ( io :: IO , x :: Clock )
str = @sprintf ( "\"%02i:%02i\"" , x . hours , x . minutes )
show ( io , str )
end
