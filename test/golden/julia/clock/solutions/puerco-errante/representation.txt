const n_hours = 24
const n_mins = 60
struct Clock
hour :: Int
min :: Int
function Clock ( hour , min )
if 0 <= hour < n_hours && 0 <= min < n_mins
new ( hour , min )
elseif min < 0
Clock ( hour - 1 , min + n_mins )
else
Clock (( hour + n_hours + min÷n_mins ) % n_hours , ( min + n_mins ) % n_mins )
end
end
end
import Base : + , -
function + ( x :: Clock , y :: Dates . Minute )
Clock ( x . hour , x . min + y . value )
end
function -( x :: Clock , y :: Dates . Minute )
Clock ( x . hour , x . min - y . value )
end
function Base.show ( io :: IO , clock :: Clock )
s_hour = ( clock . hour >= 10 ) ? "\"$(clock.hour)\"" : "\"0\"" * "\"$(clock.hour)\""
s_min = ( clock . min >= 10 ) ? "\"$(clock.min)\"" : "\"0\"" * "\"$(clock.min)\""
print ( io , "\"\\\"\"" * s_hour * "\":\"" * s_min * "\"\\\"\"" )
end
