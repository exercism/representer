(file "clock.jl")
using Printf
struct Clock
hour :: Int
min :: Int
function Clock ( h , m )
hr , m = divrem ( m , 60 )
h += m < 0 ? hr - 1 : hr
h = h < 0 ? ( 24 + h % 24 ) % 24 : h % 24
m = m < 0 ? 60 + m : m
new ( h , m )
end
end
Base .:+ ( c :: Clock , m :: Dates . Minute ) = Clock ( c . hour , c . min + m . value )
Base .:- ( c :: Clock , m :: Dates . Minute ) = Clock ( c . hour , c . min - m . value )
Base . show ( io :: IO , c :: Clock ) = @printf ( io , "\"\\\"%02d:%02d\\\"\"" , c . hour , c . min )
