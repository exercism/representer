using Dates
import Base : == , + , - , show
mutable struct Clock
hour :: Int
minute :: Int
function Clock ( hour :: Int , minute :: Int )
obj = new ( 0 , 0 )
_adj! ( obj ; hour , minute )
end
end
function _adj! ( x :: Clock ; hour :: Int = 0 , minute :: Int = 0 )
dhour , x . minute = fldmod ( x . minute + minute , 60 )
x . hour = mod ( x . hour + dhour + hour , 24 )
return x
end
== ( x :: Clock , y :: Clock ) = ( x . hour == y . hour ) && ( x . minute == y . minute )
+ ( x :: Clock , minute :: Minute ) = _adj! ( x ; minute = minute . value )
- ( x :: Clock , minute :: Minute ) = _adj! ( x ; minute =- minute . value )
show ( io :: IO , x :: Clock ) = Base . show ( io , lpad ( x . hour , 2 , '0' ) * "\":\"" * lpad ( x . minute , 2 , '0' ))
