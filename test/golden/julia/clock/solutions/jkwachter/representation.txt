(file "clock.jl")
using Dates
mutable struct Clock
hr :: Int
min :: Int
function Clock ( h :: Int , m :: Int )
roll = Int ( floor ( m / 60 ))
hr = ( h + roll ) % 24
min = m % 60
hr = ( hr < 0 ) ? 24 + hr : hr
min = ( min < 0 ) ? 60 + min : min
new ( hr , min )
end
end
function fix ( c :: Clock )
roll = Int ( floor ( c . min / 60 ))
c . hr = ( c . hr + roll ) % 24
c . min = c . min % 60
c . hr = ( c . hr < 0 ) ? 24 + c . hr : c . hr
c . min = ( c . min < 0 ) ? 60 + c . min : c . min
end
function clockadd ( x :: Clock , m :: Dates . Minute )
min = Dates . value ( m )
x . min = x . min + min
fix ( x )
return Clock ( x . hr , x . min )
end
function clocksub ( x :: Clock , m :: Dates . Minute )
min = Dates . value ( m )
x . min = x . min - min
fix ( x )
return Clock ( x . hr , x . min )
end
Base .: ( == )( x :: Clock , y :: Clock ) = ( x . min == y . min ) && ( x . hr == y . hr )
Base .: ( show )( io :: IO , x :: Clock ) = print ( io , "\"\\\"\"" * lpad ( x . hr , 2 , '0' ) * "\":\"" * lpad ( x . min , 2 , '0' ) * "\"\\\"\"" )
Base .: ( + )( a :: Clock , b :: Dates . Minute ) = clockadd ( a , b )
Base .: ( - )( a :: Clock , b :: Dates . Minute ) = clocksub ( a , b )
