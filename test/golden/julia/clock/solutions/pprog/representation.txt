import Base .+ , Base .- , Base . show
using Dates , Printf
const HOUR_MINS = 60
const DAY_HOURS = 24
struct Clock
hour :: Int
minute :: Int
function Clock ( hour , minute )
( x , minute ) = fldmod ( minute , HOUR_MINS )
new ( mod ( hour + x , DAY_HOURS ), minute )
end
end
function + ( c :: Clock , m :: Dates . Minute )
Clock ( c . hour , c . minute + Dates . value ( m ))
end
function -( c :: Clock , m :: Dates . Minute )
Clock ( c . hour , c . minute - Dates . value ( m ))
end
function show ( io :: IO , c :: Clock )
@printf io "\"\\\"%02d:%02d\\\"\"" c . hour c . minute
end
