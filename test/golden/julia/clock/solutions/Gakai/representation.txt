(file "clock.jl")
using Dates
struct Clock
hours :: Integer
minutes :: Integer
Clock ( h , m ) = let minutes = h * 60 + m
while minutes < 0
minutes += 60 * 24
end
new ( mod ( div ( minutes , 60 ), 24 ), mod ( minutes , 60 ))
end
end
Base .:+ ( self :: Clock , x :: Dates . Minute ) = begin
Clock ( self . hours , self . minutes + Dates . value ( x ))
end
Base .:- ( self :: Clock , x :: Dates . Minute ) = begin
Clock ( self . hours , self . minutes - Dates . value ( x ))
end
Base .: ( == )( a :: Clock , b :: Clock ) = begin
a . hours == b . hours && a . minutes == b . minutes
end
Base . show ( io :: IO , c :: Clock ) = print (
io ,
'"' ,
lpad ( c . hours , 2 , '0' ),
"\":\"" ,
lpad ( c . minutes , 2 , '0' ),
'"'
)
