function matrix ( strmatrix )
strmatrix = [ parse . ( Int , split ( line , "\" \"" )) for line in split ( strmatrix , "\"\\n\"" ) if ! isempty ( line )]
rows = copy ( strmatrix )
columns = []
if ! isempty ( rows )
for i in 1 : length ( rows [ 1 ])
column = [ row [ i ] for row in rows ]
push! ( columns , column )
end
end
return rows , columns
end
