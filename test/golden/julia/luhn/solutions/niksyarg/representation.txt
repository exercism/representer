function luhn ( cardnum )
cleaned = replace ( cardnum , "\" \"" => "\"\"" )
if length ( cleaned ) <= 1
return false
end
if ! all ( isdigit , cleaned )
return false
end
total_sum = 0
double = false
for i in length ( cleaned ) :- 1 : 1
digit = parse ( Int , cleaned [ i ])
if double
digit *= 2
if digit > 9
digit -= 9
end
end
total_sum += digit
double = ! double
end
return total_sum % 10 == 0
end
