function luhn ( s :: AbstractString )
digits = filter ( ! isspace , s )
if length ( digits ) < 2 || any ( ! isdigit , digits )
return false
end
nums = reverse ([ parse ( Int , c ) for c in digits ])
total = 0
for ( i , n ) in enumerate ( nums )
if isodd ( i )
total += n
else
d = n * 2
total += d > 9 ? d - 9 : d
end
end
return total % 10 == 0
end
