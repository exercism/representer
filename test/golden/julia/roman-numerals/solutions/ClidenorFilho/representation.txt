const ROMAN_MAP = [
1000 => "\"M\"" ,
900 => "\"CM\"" ,
500 => "\"D\"" ,
400 => "\"CD\"" ,
100 => "\"C\"" ,
90 => "\"XC\"" ,
50 => "\"L\"" ,
40 => "\"XL\"" ,
10 => "\"X\"" ,
9 => "\"IX\"" ,
5 => "\"V\"" ,
4 => "\"IV\"" ,
1 => "\"I\""
]
"\"\"\"\n    to_roman(number)\n\nConvert an integer to its Roman numeral representation.\n\nThe function handles numbers from 1 to 3999. It will throw an `ErrorException`\nfor non-positive integers.\n\"\"\""
function to_roman ( number )
if number <= 0
throw ( ErrorException ( "\"Input must be a positive integer.\"" ))
end
buf = IOBuffer ()
for ( value , symbol ) in ROMAN_MAP
while number >= value
print ( buf , symbol )
number -= value
end
end
return String ( take! ( buf ))
end
