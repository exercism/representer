function modifier ( ability )
floor ( Int , ( ability - 10 ) / 2 )
end
function ability ()
sum ( sort ( rand ( 1 : 6 , 4 )[ 2 : end ]))
end
mutable struct DNDCharacter
strength :: Int
dexterity :: Int
constitution :: Int
intelligence :: Int
wisdom :: Int
charisma :: Int
hitpoints :: Int
function DNDCharacter ()
draws = rand ( 1 : 6 , ( 4 , 6 ))
foo ( x ) = sum ( sort ( x )[ 2 : end ])
sums = [ foo ( draws [ : , i ]) for i in 1 : 6 ]
push! ( sums , 10 + modifier ( sums [ 3 ]))
new ( sums ... )
end
end
