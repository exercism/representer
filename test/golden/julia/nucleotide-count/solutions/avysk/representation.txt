"\"\"\"\n    count_nucleotides(strand)\n\nThe count of each nucleotide within `strand` as a dictionary.\n\nInvalid strands raise a `DomainError`.\n\n\"\"\""
function count_nucleotides ( strand )
bad_re = r "\"[^ACGT]\""
if occursin ( bad_re , strand )
throw ( DomainError ( "\"Invalid strand\"" ))
end
res = Dict ( 'A' => 0 , 'C' => 0 , 'G' => 0 , 'T' => 0 )
for nuc ∈ strand
res [ nuc ] += 1
end
res
end
