function acronym ( phrase :: AbstractString )
cleaned = replace ( phrase , r "\"[-_,]\"" => "\" \"" )
words = split ( cleaned )
letters = []
for word in words
if ! isempty ( word ) && occursin ( r "\"[A-Za-z]\"" , word )
for char in word
if isletter ( char )
push! ( letters , char )
break
end
end
end
end
return uppercase ( join ( letters ))
end
