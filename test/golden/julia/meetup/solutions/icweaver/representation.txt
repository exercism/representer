function meetup ( year , month , week , day )
x = if week == "\"first\""
1 : 7
elseif week == "\"second\""
8 : 14
elseif week == "\"third\""
15 : 21
elseif week == "\"teenth\""
13 : 19
elseif week == "\"fourth\""
22 : 28
elseif week == "\"last\""
22 : daysinmonth ( year , month )
else
throw ( "\"todo\"" )
end
days = map ( dayname , ( Date ( year , month , i ) for i in x ))
x_day = x [ findlast ( == ( day ), days )]
return Date ( year , month , x_day )
end
