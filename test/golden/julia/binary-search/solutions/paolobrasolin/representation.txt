function binarysearch ( haystack , needle ; rev = false , by = identity , lt = rev ? ( > ) : ( < ))
lo = 1
hi = length ( haystack )
haystack = by . ( haystack )
needle = by ( needle )
while lo <= hi
i = div ( lo + hi , 2 )
if needle == haystack [ i ]
return i : i
elseif lt ( needle , haystack [ i ])
hi = i - 1
elseif lt ( haystack [ i ], needle )
lo = i + 1
end
end
lo : hi
end
