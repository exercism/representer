function binarysearch ( sorted_list , item , lt = < )
if sorted_list == []
return 1 : 0
end
function binarysearch_helper ( low , high )
if lt ( sorted_list [ high ], item )
return high + 1 : high
end
if lt ( item , sorted_list [ low ])
return low : low - 1
end
middle = max ( convert ( Int , floor (( low + high ) / 2 )), 1 )
if sorted_list [ middle ] == item
return middle : middle
elseif lt ( sorted_list [ middle ], item )
return binarysearch_helper ( middle + 1 , high )
else
return binarysearch_helper ( low , middle - 1 )
end
end
return binarysearch_helper ( 1 , length ( sorted_list ))
end
