function binarysearch ( arr , num ; rev = false , by = round , lt = < )
if rev == true
arr = reverse ( arr )
end
if by != nothing
arr = map ( Int ∘ by , arr )
num = map ( Int ∘ by , num )
end
l = 1
r = length ( arr )
while l <= r
i = l + (( r - l ) ÷ 2 )
if ! lt ( num , arr [ i ]) && ! lt ( arr [ i ], num )
l = r = i
break
elseif lt ( num , arr [ i ])
r = i - 1
elseif lt ( arr [ i ], num )
l = i + 1
end
end
if rev == true
l = length ( arr ) - l + 1
r = length ( arr ) - r + 1
l , r = r , l
end
return ( l : r )
end
