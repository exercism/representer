function binarysearch ( a :: Vector , x )
indx = cld ( length ( a ), 2 )
offs = 0
match = false
while ( indx > 0 )
val = a [ offs + indx ]
if val == x
match = true
break
end
if val < x
offs += indx
end
indx = fld ( indx , 2 )
end
if match == true
offs + indx : offs + indx
else
offs + indx : offs + indx - 1
end
end
