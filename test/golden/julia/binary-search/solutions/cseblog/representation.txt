function binarysearch ( arr , key )
count = 0 ;
while length ( arr ) > 0
if length ( arr ) == 1
if ( arr [ 1 ] == key )
return count + 1 : count + 1
else
return 1 : 0
end
end
i = floor ( Int64 , length ( arr ) / 2 )
if ( arr [ i ] == key )
return count + i : count + i
end
if ( arr [ i ] > key )
arr = arr [ 1 : i - 1 ]
else
count += i ;
arr = arr [ i + 1 : floor ( Int64 , length ( arr ))]
end
end
return 1 : 0
end
