function binarysearch ( list , element )
isempty ( list ) && return 1 : 0
mid = floor ( Int , length ( list ) + 1 / 2 )
mid_element = @show list [ mid ]
if mid_element == element
mid : mid
elseif mid_element > element
binarysearch ( list [ begin : mid - 1 ], element )
else
binarysearch ( list [ mid + 1 : end ], element )
end
end
