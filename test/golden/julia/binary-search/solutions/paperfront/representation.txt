function binarysearch ( arr , target ; by = x -> x , lt = < , rev = false )
if length ( arr ) == 0
return 1 : 0
end
arr = map ( by , arr )
target = by ( target )
left = 1
right = lastindex ( arr )
while left < right - 1
current = ( left + right ) ÷ 2
if arr [ current ] == target
return current : current
elseif lt ( arr [ current ], target )
if ! rev
left = current
else
right = current
end
else
if ! rev
right = current
else
left = current
end
end
end
if arr [ left ] == target
return left : left
elseif arr [ right ] == target
return right : right
end
if left == right
return right : left - 1
elseif ! rev
if lt ( lt ( arr [ left ], target ), arr [ right ])
return right : left
elseif lt ( target , arr [ left ])
return left : left - 1
else
return right + 1 : right
end
else
if lt ( lt ( arr [ right ], target ), arr [ left ])
return right : left
elseif lt ( arr [ left ], target )
return left : left - 1
else
return right + 1 : right
end
end
end
