function find_multiple_matches ( V , x , ind , by )
n = length ( V )
n == 1 && return ind
@show ind V [ ind [ 1 ]]
value = by ( V [ ind [ 1 ]])
lower = ind [ 1 ]
upper = ind [ end ]
while lower > 1
lower -= 1
if by ( V [ lower ]) != value
lower += 1
break
end
end
while upper < n
upper += 1
if by ( V [ upper ]) != value
upper -= 1
break
end
end
return lower : upper
end
function binarysearch2 ( V :: Array , x ;
rev = false ,
by = identity ,
lt = ifelse ( rev , > , < ))
n = length ( V )
n == 0 && return 1 : 0
b_x = by ( x )
n == 1 && (( b_x == by ( V [ 1 ]) && return 1 : 1 ) || return 1 : 0 )
ind = ( n + 1 ) ÷ 2
mid_value = V [ ind ]
b_m = by ( mid_value )
b_m == b_x && return ind : ind
if lt ( b_x , b_m )
return binarysearch2 ( V [ 1 : ind - 1 ], x , rev = rev , by = by , lt = lt )
else
return ind .+ binarysearch2 ( V [ ind + 1 : end ], x , rev = rev , by = by , lt = lt )
end
end
