function binarysearch ( sortedlist , num )
len = length ( sortedlist )
if len == 0
return 1 : 0
end
low = 1
high = len
if high == 1
key = 1
else
key = len÷2
end
found = false
while ! found
candidate = sortedlist [ key ]
if candidate == num
found = true
break
elseif key == high
break
else
if candidate > num
high = key
key = low + ( key - low ) ÷ 2
elseif candidate < num
low = key
if high - key == 1
key = high
else
key += ( high - key ) ÷ 2
end
end
end
end
if found
return key : key
elseif key == 1
return 1 : 0
elseif key == len
return len + 1 : len
else
return key : key - 1
end
end
