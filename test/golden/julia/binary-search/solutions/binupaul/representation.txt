function binarysearch ( arr , key ; rev = false , lt = < , by = identity )
high = length ( arr )
low = 1
while low <= high
current = div ( high + low , 2 )
if by ( arr [ current ]) == by ( key )
return current : current
elseif lt ( by ( arr [ current ]), by ( key ))
if rev
high = current - 1
else
low = current + 1
end
else
if rev
low = current + 1
else
high = current - 1
end
end
end
return low : high
end
