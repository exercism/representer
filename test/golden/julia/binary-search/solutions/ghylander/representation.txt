function binarysearchr ( dictionary , item ; lt )

function lookahead ( dictionary , item , midpoint )
hb = 1
while (( midpoint + ( hb + 1 ) < length ( dictionary )) && ( dictionary [ midpoint + ( hb + 1 )] == item ))
hb += 1
end
return midpoint + hb
end
function lookbehind ( dictionary , item , midpoint )
lb = 1
while (( midpoint - ( lb + 1 ) > 0 ) && ( dictionary [ midpoint - ( lb + 1 )] == item ))
lb += 1
end
return midpoint - lb
end
