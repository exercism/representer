function binarysearch ( a , x ; rev = false , by = identity , lt =< )
i = 1
j = length ( a )
if rev
a = a [ end :- 1 : 1 ]
end
a = map ( by , a )
x = by ( x )
println ( "\"Start at $i:$j\"" )
while i < j
if lt ( x , a [ i ])
j = i - 1
break
elseif lt ( a [ j ], x )
i = j + 1
break
end
m = ( i + j ) ÷ 2
if lt ( a [ m ], x )
i = m + 1
elseif lt ( x , a [ m ])
j = m - 1
else
i = m
j = m
end
println ( "\"$i:$j\"" )
end
if rev
n = length ( a )
return ( n + 1 - j ) : ( n + 1 - i )
else
return i : j
end
end
