function binarysearch ( arr :: AbstractVector , target ; rev :: Bool = false , by :: Function = identity , lt :: Function = ( < ))
bytarget = by ( target )
if rev
lo = bisect_left ( arr , bytarget , 1 , length ( arr ) + 1 , by , ( a , b ) -> lt ( b , a ))
hi = bisect_right ( arr , bytarget , 1 , length ( arr ) + 1 , by , ( a , b ) -> lt ( b , a ))
return lo : hi - 1
else
lo = bisect_left ( arr , bytarget , 1 , length ( arr ) + 1 , by , lt )
hi = bisect_right ( arr , bytarget , 1 , length ( arr ) + 1 , by , lt )
return lo : hi - 1
end
end
function bisect_left ( arr :: AbstractVector , target , lo :: Integer , hi :: Integer , by :: Function , lt :: Function ) :: Int
while lo < hi
mid = ( lo + hi ) ÷ 2
if lt ( by ( arr [ mid ]), target )
lo = mid + 1
else
hi = mid
end
end
return lo
end
function bisect_right ( arr :: AbstractVector , target , lo :: Integer , hi :: Integer , by :: Function , lt :: Function ) :: Int
while lo < hi
mid = ( lo + hi ) ÷ 2
if lt ( target , by ( arr [ mid ]))
hi = mid
else
lo = mid + 1
end
end
return lo
end
