function binarysearch ( list , key ; rev = false , by = identity , lt = ifelse ( rev , > , < ))
mid = ( length ( list ) + 1 ) ÷ 2
mid == 0 && return 1 : 0
elm = by ( list [ mid ])
elm == by ( key ) && return mid : mid
if lt ( elm , by ( key ))
mid .+ binarysearch ( @view ( list [ mid + 1 : end ]), key , rev = rev , by = by , lt = lt )
else
binarysearch ( @view ( list [ 1 : mid - 1 ]), key , rev = rev , by = by , lt = lt )
end
end
