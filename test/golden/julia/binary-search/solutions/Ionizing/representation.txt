function binarysearch (
arr :: AbstractArray ,
needle ;
i :: Integer = 1 ,
j :: Integer = length ( arr ),
rev :: Bool = false ,
lt = < ,
by = identity
) :: UnitRange { Integer }
if rev
lt = >
end
while i <= j
mid = i + Int ( ceil (( j - i ) / 2 ) )
if by ( arr [ mid ]) == by ( needle )
return mid : mid ;
elseif lt ( by ( arr [ mid ]), by ( needle ))
i = mid + 1
else
j = mid - 1
end
end
return i : j
end
