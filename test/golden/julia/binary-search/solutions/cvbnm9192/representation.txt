@views function binarysearch ( list , value ; rev = false , by = x -> x , lt = < )
l = rev ? reverse ( by . ( list )) : by . ( list )
v = by . ( value )
r = binarysearch ( l [ 1 : end ], v , lt )
L = length ( l ) + 1
rev ? ( L - r . stop : L - r . start ) : r
end
@views function binarysearch ( list , value , lt )
if list == [] || list == [ value ]
return parentindices ( list )[ 1 ]
elseif length ( list ) == 1
p = parentindices ( list )[ 1 ]
if ! lt ( value , list [ 1 ])
p = 1 .+ p
end
return p . start : p . start - 1
end
mid = length ( list ) ÷ 2
left = list [ 1 : mid ]
right = list [ mid + 1 : end ]
sublist = ( left != [] && lt ( left [ end ], value )) ? right : left
return binarysearch ( sublist , value , lt )
end
