function binarysearch ( A , x ; rev = false , by = identity , lt = < )
A = by . ( A );
x = by ( x );
if ( rev )
lt = > ;
end
low = 1 ;
high = length ( A );
while ( low <= high )
mid = div ( low + high , 2 );
if ( A [ mid ] == x )
return mid : mid ;
elseif ( lt ( A [ mid ], x ))
low = mid + 1 ;
else
high = mid - 1 ;
end
end
return low : high ;
end
