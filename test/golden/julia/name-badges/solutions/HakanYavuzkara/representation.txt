function print_name_badge ( id , name , department )
ismissing ( id ) ? "\"$(name) - $(uppercase(department))\"" :
isnothing ( department ) ? "\"[$(id)] - $(name) - OWNER\"" :
"\"[$(id)] - $(name) - $(uppercase(department))\""
end
function salaries_no_id ( ids , salaries )
sum ( ismissing ( ids [ i ]) && salaries [ i ] for i in 1 : length ( ids ))
end
