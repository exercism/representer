#ifndef DARTS_H
#define DARTS_H
#include <stdint.h>
typedef struct {
float x ;
float y ;
} coordinate_t ;
uint8_t score ( coordinate_t landing_position );
#endif
#include "\"darts.h\""
#include <math.h>
uint8_t score ( coordinate_t position ) {
float distance = hypot ( position . x , position . y );
if ( distance <= 1 . 0 F ) {
return 10 ;
}
if ( distance <= 5 . 0 F ) {
return 5 ;
}
if ( distance <= 10 . 0 F ) {
return 1 ;
}
return 0 ;
}
