#ifndef RESISTOR_COLOR_DUO_H
#define RESISTOR_COLOR_DUO_H
typedef enum {
BLACK , BROWN , RED , ORANGE , YELLOW , GREEN , BLUE , VIOLET , GREY , WHITE
} resistor_band_t ;
int color_code ( resistor_band_t * resisten );
#endif
int color_code ( resistor_band_t * resisten ){
if ( resisten == NULL ){
return - 1 ;
}
return ( resisten [ 0 ] * 10 ) + resisten [ 1 ];
}
