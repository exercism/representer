#ifndef REVERSE_STRING_H
#define REVERSE_STRING_H
char * reverse ( const char * value );
#endif
char * reverse ( const char * value ) {
size_t len = strlen ( value );
char * reversed = calloc ( len + 1 , sizeof ( char ));
for ( size_t ir = 0 , iv = len - 1 ; ir < len ; ir ++ , iv -- ) {
reversed [ ir ] = value [ iv ];
}
return reversed ;
}
