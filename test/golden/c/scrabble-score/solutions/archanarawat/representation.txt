unsigned int score ( const char * word ) {
unsigned int result = 0 ;
if ( word == NULL ) return 0 ;
for ( size_t i = 0 ; i < strlen ( word ); i ++ )
result += SCRABBLE_SCORE [ toupper ( word [ i ]) - 'A' ];
return result ;
}
