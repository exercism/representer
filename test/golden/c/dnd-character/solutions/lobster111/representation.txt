#ifndef DND_CHARACTER_H
#define DND_CHARACTER_H
typedef struct {
int strength ;
int dexterity ;
int constitution ;
int intelligence ;
int wisdom ;
int charisma ;
int hitpoints ;
} dnd_character_t ;
void bubble_sort ( int arr [], int length );
int ability ( void );
int modifier ( int score );
dnd_character_t make_dnd_character ( void );
#endif
void bubble_sort ( int arr [], int length ){
for ( int i = 1 ; i < length ; i ++ ){
for ( int j = i ; j < length ; j ++ ){
if ( arr [ j ] > arr [ j - 1 ]){
int tmp = arr [ j ];
arr [ j ] = arr [ j - 1 ];
arr [ j - 1 ] = tmp ;
}
}
}
}
dnd_character_t make_dnd_character ( void ){
dnd_character_t c = { ability (), ability (), ability (), ability (), ability (), ability (), 10 };
c . hitpoints += modifier ( c . constitution );
return c ;
}
int modifier ( int score ){
return ( int ) floor (( score - 10 ) / 2 . 0 F );
}
int ability ( void ){
const int roll_count = 4 ;
int rolls [ roll_count ];
for ( int i = 0 ; i < roll_count ; i ++ ){
rolls [ i ] = rand () % 6 + 1 ;
}
bubble_sort ( rolls , roll_count );
int sum = 0 ;
for ( int i = 0 ; i < roll_count - 1 ; i ++ ){
sum += rolls [ i ];
}
return sum ;
}
