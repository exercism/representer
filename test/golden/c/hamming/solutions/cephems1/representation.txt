#ifndef HAMMING_H
#define HAMMING_H
int compute ( const char * lhs , const char * rhs );
#endif
int compute ( const char * lhs , const char * rhs ){
int length = strlen ( lhs );
if ( length != ( int ) strlen ( rhs ))
return - 1 ;
size_t result = 0 ;
while ( -- length >= 0 ){
if ( lhs [ length ] != rhs [ length ])
result ++ ;
}
return result ;
}
