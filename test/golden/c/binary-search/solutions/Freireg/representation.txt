#ifndef BINARY_SEARCH_H
#define BINARY_SEARCH_H
const int * binary_search ( int value , const int * arr , size_t length );
#endif
const int * binary_search ( int value , const int * arr , size_t length ){
size_t right = length , left = 0 ;
while ( left < right ){
size_t mid = left + ( right - left ) / 2 ;
if ( arr [ mid ] == value ){
return & arr [ mid ];
} else if ( arr [ mid ] < value ){
left = mid + 1 ;
} else {
right = mid ;
}
}
return NULL ;
}
