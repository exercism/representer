bool is_pangram ( const char * sentence ){
if ( sentence == NULL ) return false ;
for ( size_t i = 0 , char_set_count = 0 ; sentence [ i ]; ++ i ){
if ( ! isalpha ( sentence [ i ])) continue ;
int ch = tolower ( sentence [ i ]);
for ( size_t j = 0 ; j < i ; ++ j ){
if ( ! isalpha ( sentence [ j ])) continue ;
if ( ch == tolower ( sentence [ j ])){
-- char_set_count ;
break ;
}
}
if ( 26 == ++ char_set_count ) return true ;
}
return false ;
}
