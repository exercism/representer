#ifndef SPACE_AGE_H
#define SPACE_AGE_H
#define MERCURY_ORBIT_IN_EARTH_YEARS 0.2408467
#define VENUS_ORBIT_IN_EARTH_YEARS 0.61519726
#define MARS_ORBIT_IN_EARTH_YEARS 1.8808158
#define JUPITER_ORBIT_IN_EARTH_YEARS 11.862615
#define SATURN_ORBIT_IN_EARTH_YEARS 29.447498
#define URANUS_ORBIT_IN_EARTH_YEARS 84.016846
#define NEPTUNE_ORBIT_IN_EARTH_YEARS 164.79132
#define EARTH_YEAR_IN_SECONDS 31557600
typedef enum planet {
MERCURY ,
VENUS ,
EARTH ,
MARS ,
JUPITER ,
SATURN ,
URANUS ,
NEPTUNE ,
} planet_t ;
float seconds_to_earth_years ( float seconds );
float age ( planet_t planet , int64_t seconds );
#endif
float seconds_to_earth_years ( float seconds )
{
return seconds / EARTH_YEAR_IN_SECONDS ;
}
float age ( planet_t planet , int64_t seconds )
{
switch ( planet )
{
case MERCURY :
{
return seconds_to_earth_years ( seconds ) / MERCURY_ORBIT_IN_EARTH_YEARS ;
}
case VENUS :
{
return seconds_to_earth_years ( seconds ) / VENUS_ORBIT_IN_EARTH_YEARS ;
}
case EARTH :
{
return seconds_to_earth_years ( seconds );
}
case MARS :
{
return seconds_to_earth_years ( seconds ) / MARS_ORBIT_IN_EARTH_YEARS ;
}
case JUPITER :
{
return seconds_to_earth_years ( seconds ) / JUPITER_ORBIT_IN_EARTH_YEARS ;
}
case SATURN :
{
return seconds_to_earth_years ( seconds ) / SATURN_ORBIT_IN_EARTH_YEARS ;
}
case URANUS :
{
return seconds_to_earth_years ( seconds ) / URANUS_ORBIT_IN_EARTH_YEARS ;
}
case NEPTUNE :
{
return seconds_to_earth_years ( seconds ) / NEPTUNE_ORBIT_IN_EARTH_YEARS ;
}
default: return - 1 ;
}
}
