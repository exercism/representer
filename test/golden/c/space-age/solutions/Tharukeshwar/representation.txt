#include "\"space_age.h\""
#include <math.h>
#define EARTH_SECONDS 31557600
float orbital_periods [] = { 0 . 2408467 f , 0 . 61519726 f , 1 . 0 f , 1 . 8808158 f , 11 . 862615 f , 29 . 447498 f , 84 . 016 846 f , 164 . 79132 f };
float age ( planet_t planet , int64_t seconds ) {
if (( int ) planet < 0 ) {
return - 1 . 0 f ;
}
float planet_earth_seconds = orbital_periods [( int ) planet ] * EARTH_SECONDS ;
float age = seconds / planet_earth_seconds ;
return age ;
}
