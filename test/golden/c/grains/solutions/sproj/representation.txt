#include "\"grains.h\""
uint64_t square ( uint8_t index ) {
return ( index > 0 && index < 65 ) ? 1ul << ( index - 1 ) : 0 ;
uint64_t count = 1 ;
while ( -- index ) {
count *= 2 ;
}
return count ;
}
uint64_t total ( void ) {
uint8_t squareIndex = 64 ;
uint64_t sum = square ( squareIndex );
return sum * 2ull - 1 ;
}
