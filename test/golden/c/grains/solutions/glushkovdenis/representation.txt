#include "\"grains.h\""
#include <math.h>
uint64_t square ( uint8_t index ) {
if ( index > 64 || index == 0 )
return 0 ;
return ( uint64_t ) powl ( 2 , index - 1 );
}
uint64_t total ( void ) {
uint64_t res = 0 ;
for ( int i = 0 ; i < 64 ; i ++ ) {
res += powl ( 2 , i );
}
return res ;
}
