#include "\"grains.h\""
#include <stdint.h>
uint64_t square ( uint8_t index ) {
if ( index > 64 || index == 0 ) {
return 0 ;
}
return 1ULL << ( index - 1 );
}
uint64_t total ( void ) {
uint64_t total = 0 ;
for ( uint8_t i = 0 ; i <= 64 ; i ++ ) {
total += square ( i );
}
return total ;
}
