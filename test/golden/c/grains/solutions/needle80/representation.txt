#include "\"grains.h\""
#include <stdio.h>
uint64_t square ( uint8_t index )
{
uint64_t sq = 0 ;
if ( index == 0 || index > 64 )
return 0 ;
sq = sq | ( 1ull << ( index - 1 ));
return sq ;
}
uint64_t total ( void )
{
uint64_t sq = 0 ;
for ( int i = 1 ; i <= 64 ; i ++ ) {
sq += square ( i );
}
return sq ;
}
