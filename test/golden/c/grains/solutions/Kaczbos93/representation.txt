uint64_t square ( uint8_t index )
{
uint64_t grains ;
if ( ! index )
return 0 ;
if ( index == 1 )
grains = 1 ;
if ( index > 1 )
{
grains = 1 ;
for ( int i = 0 ; i < index - 1 ; i ++ )
grains *= 2 ;
}
return grains ;
}

{
return UINT64_MAX ;
}
