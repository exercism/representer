#ifndef GRAINS_H
#define GRAINS_H
#include <stdint.h>
uint64_t square ( uint8_t index );
uint64_t total ( void );
#endif
#include "\"grains.h\""
uint64_t square ( uint8_t index ){
if ( index == 1 ){
return 1 ;
} else {
return 2 * square ( index - 1 );
}
}
uint64_t total ( void ){
uint64_t total_val = 0 ;
for ( int i = 1 ; i <= 64 ; i ++ )
total_val += square ( i );
return total_val ;
}
