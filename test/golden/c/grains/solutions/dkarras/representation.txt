uint64_t square ( uint8_t square )
{
if ( square > 0 && square <= 64 ) {
uint8_t bit = square - 1 ;
return ( 1ULL << bit );
}
return 0 ;
}
uint64_t total ( void )
{
uint64_t total = 0 ;
for ( int i = 1 ; i < 65 ; i ++ ) {
total += square ( i );
}
return total ;
}
