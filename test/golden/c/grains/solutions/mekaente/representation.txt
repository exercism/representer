#ifndef GRAINS_H
#define GRAINS_H
#include <stdint.h>
uint64_t square ( uint8_t index );
uint64_t total ( void );
#endif
#include "\"grains.h\""
uint64_t square ( uint8_t index ) {
if (( index > 64 ) || ( index < 1 )) {
return 0 ;
}
return ( 1ULL << ( index - 1 ));
}
uint64_t total ( void ) {
uint64_t seeds = 0 ;
for ( int i = 1 ; i < 65 ; i ++ ) {
seeds += square ( i );
}
return seeds ;
}
