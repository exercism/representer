uint64_t
square ( uint8_t index )
{
uint64_t prod = 1 ;
if ( index == 0 || index > 64 ) {
return 0 ;
}
index -- ;
return ( prod << index );
}
uint64_t
total ( void )
{
uint64_t sum = 0 ;
for ( uint8_t i = 1 ; i <= SQUARES ; i ++ ) {
sum += square ( i );
}
return sum ;
}
(file "grains.h")
#ifndef GRAINS_H
#define GRAINS_H
#include <stdint.h>
#define SQUARES 64
uint64_t square ( uint8_t index );
uint64_t total ( void );
#endif
