#ifndef GRAINS_H
#define GRAINS_H
#include <stdint.h>
uint64_t square ( uint8_t index );
uint64_t total ( void );
#endif
#include "\"grains.h\""
uint64_t
square ( uint8_t index )
{
uint64_t result = 1 ;
for ( uint8_t j = 1 ; j < index ; ++ j )
{
result *= 2 ;
}
return result = ( index == 0 ? 0 : result );
}
uint64_t
total ( void )
{
uint64_t total_grains = 0 ;
uint8_t number_of_cells = 64 ;
for ( uint8_t i = 1 ; i <= number_of_cells ; ++ i )
{
total_grains += square ( i );
}
return total_grains ;
}
