uint64_t square ( uint8_t index )
{
if ( ! index )
return ( 0 );
int i = 1 ;
uint64_t sum_square = 1 ;
while ( i < index )
{
sum_square += sum_square ;
i ++ ;
}
return ( sum_square );
}
uint64_t total ( void )
{
int i = 0 ;
int square = 1 ;
int sum = 1 ;
while ( ++ i < 64 )
{
square *= 2 ;
sum += square ;
}
return ( sum );
}
