uint64_t pow64 ( uint64_t num , uint64_t p );
uint64_t pow64 ( uint64_t num , uint64_t p ) {
if ( p == 0 )
return 1 ;
return num * pow64 ( num , p - 1 );
}
uint64_t square ( uint8_t index ) {
if ( index == 0 )
return 0 ;
return pow64 ( 2 , index - 1 );
}
uint64_t total ( void ) {
uint64_t tot = 0 ;
for ( int i = 1 ; i <= 64 ; i ++ ) {
tot += square ( i );
}
return tot ;
}
