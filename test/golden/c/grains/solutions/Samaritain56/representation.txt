uint64_t total ( void );
uint64_t square ( uint8_t index ){
uint64_t n = 1 , i = index , sum = 1 ;
while ( i > 0 ){
sum = n ;
n += n ;
-- i ;
}
if ( index == 0 ){
return 0 ;
}
return sum ;
}
uint64_t total ( void ){
int total = 0 ;
for ( int i = 1 ; i <= 64 ; i ++ ){
total += square ( i );
}
return total ;
}
