uint64_t square ( uint8_t index ) {
if ( index == 0 || index > 64 ) {
return 0 ;
}
unsigned long i = 0 , num = 0 ;
for ( i = 1 ; i <= index ; i ++ ) {
num = pow ( 2 , ( i - 1 ));
}
return ( uint64_t ) num ;
}
uint64_t total ( void )
{
unsigned long i = 0 , total = 0 ;
for ( i = 0 ; i <= 64 ; i ++ ) {
total += square (( uint8_t ) i );
}
return ( uint64_t ) total ;
}
