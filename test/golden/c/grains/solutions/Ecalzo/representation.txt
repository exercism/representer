uint64_t square ( uint8_t index )
{
if ( index == 0 )
{
return 0 ;
}
if ( index == 1 )
{
return 1 ;
}
return ( 2 * square ( index - 1 ));
}
uint64_t total_iter ( uint8_t index )
{
if ( index == 0 )
{
return square ( 0 );
}
return ( square ( index ) + total_iter ( index - 1 ));
}
uint64_t total ( void )
{
return total_iter ( 64 );
}
