#include "\"grains.h\""
uint64_t square ( uint8_t index )
{
return pow ( 2 , index - 1 );
}
uint64_t total ( void )
{
int sum = 0 ;
int i = 0 ;
while ( i <= 64 )
{
sum += square ( i );
i ++ ;
}
return sum ;
}
