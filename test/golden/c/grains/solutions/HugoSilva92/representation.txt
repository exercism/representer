uint64_t square ( uint8_t index )
{
uint64_t i ;
uint64_t grains ;
grains = 1 ;
i = 1 ;
if ( index < 1 || index > 64 )
return 0 ;
while ( i < index )
{
grains *= 2 ;
i ++ ;
}
return ( grains );
}
uint64_t total ( void )
{
uint64_t i ;
uint64_t grains ;
uint64_t total ;
i = 1 ;
grains = 1 ;
total = grains ;
while ( i <= 64 )
{
grains *= 2 ;
total += grains ;
i ++ ;
}
return ( total );
}
