#include <math.h>
#include "\"grains.h\""
uint64_t square ( const uint8_t index )
{
if ( 1 > index || 64 < index ) {
return 0 ;
}
return pow ( 2 , index - 1 );
}
uint64_t total ( void ) {
uint64_t accumulator = 0 ;
for ( int idx = 1 ; idx < 65 ; idx ++ ) {
accumulator += square ( idx );
}
return accumulator ;
}
