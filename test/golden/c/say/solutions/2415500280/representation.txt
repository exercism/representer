#define num_elements(_arr) (sizeof(_arr) / sizeof(_arr[0]))

static const char * group_name [] = {
"\"\"" ,
"\"thousand\"" ,
"\"million\"" ,
"\"billion\""
};
static void append ( char ** current , const char * s )
{
* current += sprintf ( * current , "\"%s\"" , s );
}
static void say_group ( int64_t n , char ** current )
{
if ( n >= 100 ) {
append ( current , ones_and_teens [ n / 100 ]);
append ( current , "\" hundred\"" );
n -= ( n / 100 ) * 100 ;
if ( n > 0 ) append ( current , "\" \"" );
}
if ( n >= 20 ) {
append ( current , tens [ n / 10 ]);
n -= ( n / 10 ) * 10 ;
if ( n > 0 ) append ( current , "\"-\"" );
}
append ( current , ones_and_teens [ n ]);
}
int say ( int64_t n , char ** ans )
{
* ans = calloc ( 1000 , sizeof ( char ));
char * current = * ans ;
if ( n < 0 ) return - 1 ;
if ( n > 999999999999ll ) return - 1 ;
for ( size_t i = num_elements ( group_name ); i > 0 ; i -- ) {
int64_t group_value = n / pow ( 1000 , i - 1 );
if ( group_value ) {
say_group ( group_value , & current );
append ( & current , "\" \"" );
append ( & current , group_name [ i - 1 ]);
append ( & current , "\" \"" );
n -= group_value * pow ( 1000 , i - 1 );
}
}
while (( current > * ans ) && ( current [ - 1 ] == "' '" )) {
*-- current = 0 ;
}
if ( !* ans [ 0 ]) {
append ( & current , "\"zero\"" );
}
return 0 ;
}
