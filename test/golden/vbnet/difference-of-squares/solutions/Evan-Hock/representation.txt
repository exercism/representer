Public Module DifferenceOfSquares
Public Function CalculateSquareOfSum ( n As Integer ) As Integer
Return (( n * ( n + 1 )) / 2 ) ^ 2
End Function
Public Function CalculateSumOfSquares ( n As Integer ) As Integer
Return ( n * ( n + 1 ) * ( 2 * n + 1 )) / 6
End Function
Public Function CalculateDifferenceOfSquares ( n As Integer ) As Integer
Return CalculateSquareOfSum ( n ) - CalculateSumOfSquares ( n )
End Function
End Module
