Public Module BottleSong
Public Function Recite ( ByVal startBottles As Integer , ByVal takeDown As Integer ) As String
Dim numbers = { "\"One\"" , "\"Two\"" , "\"Three\"" , "\"Four\"" , "\"Five\"" , "\"Six\"" , "\"Seven\"" , "\"Eight\"" , "\"Nine\"" , "\"Ten\"" }
Dim lyrics = "\"\""
For i As Integer = startBottles - 1 To startBottles - takeDown Step - 1
Dim pluralSuffix1 = If ( i = 0 , "\"\"" , "\"s\"" )
Dim pluralSuffix2 = If ( i = 1 , "\"\"" , "\"s\"" )
Dim nextNumber = "\"no\""
If i > 0 Then nextNumber = numbers ( i - 1 ). ToLower
lyrics +=
$ "\"{numbers(i)} green bottle{pluralSuffix1} hanging on the wall,\"" & vbLf &
$ "\"{numbers(i)} green bottle{pluralSuffix1} hanging on the wall,\"" & vbLf &
"\"And if one green bottle should accidentally fall,\"" & vbLf &
$ "\"There'll be {nextNumber} green bottle{pluralSuffix2} hanging on the wall.\"" &
If ( i = startBottles - takeDown , "\"\"" , vbLf & vbLf )
Next
Return lyrics
End Function
End Module
