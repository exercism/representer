Public Module ArmstrongNumbers
Public Function IsArmstrongNumber ( number As Integer ) As Boolean
If number < 10 Then Return True
Dim digitCount = Math . Floor ( Math . Log10 ( number ) + 1 )
Dim num = number
Dim pow_total = 0
While num > 0
Dim pow_temp = num Mod 10
Dim pow_temp_total = 1
For i As Integer = 1 To digitCount
pow_temp_total *= pow_temp
Next i
pow_total += pow_temp_total
num \= 10
End While
Return number = pow_total
End Function
End Module
