Public Class Anagram
Private _strWord As String
Public Sub New ( baseWord As String )
Me . _strWord = baseWord . ToLower
End Sub
Public Function Match ( potentialMatches As String ()) As String ()
Dim arrWord As Char () = _strWord . ToArray ()
Dim arrMatch As Char ()
Dim arrAnagrams As IList ( Of String ) = New List ( Of String )
For Each strMatch As String In potentialMatches
If StrComp ( strMatch . ToLower , _strWord ) = 0 Then
Continue For
End If
arrMatch = strMatch . ToLower . ToArray ()
Array . Sort ( arrMatch )
Array . Sort ( arrWord )
If StrComp ( arrMatch , arrWord ) = 0 AndAlso Not arrAnagrams . Contains ( arrMatch ) Then
arrAnagrams . Add ( strMatch )
End If
Next strMatch
arrAnagrams = arrAnagrams . ToArray ()
Array . Sort ( arrAnagrams )
Return arrAnagrams
End Function
End Class
