Public Class Anagram
Public Property start As String
Public Sub New ( input As String )
start = input . ToLower ()
End Sub
Private Function CompareCharAsString ( ByVal c1 As Char , ByVal c2 As Char ) As Integer
Dim str1 As String = c1
Dim str2 As String = c2
Return str1 . CompareTo ( str2 )
End Function
Public Function ToComparableString ( word As String ) As String
Dim charArray As Char () = word . ToCharArray ()
Array . Sort ( charArray , AddressOf CompareCharAsString )
Return New String ( charArray )
End Function
Public Function Match ( words As String ()) As String ()
Dim matchMe = ToComparableString ( start )
Dim results As List ( Of String ) = New List ( Of String )()
For Each word As String In words
Dim compareStr = ToComparableString ( word . ToLower ())
If matchMe . Equals ( compareStr ) AndAlso Not start . Equals ( word . ToLower ()) Then
results . Add ( word )
End If
Next
Dim resultArray = results . ToArray ()
Array . Sort ( resultArray )
Return resultArray
End Function
End Class
