Public Class Allergies
Public AllergyList As List ( Of String )
Public Sub New ( allergyScore As Integer )
Dim list = New List ( Of String )
Console . WriteLine ( allergyScore )
If allergyScore > 255 Then
allergyScore = allergyScore Mod 256
Console . WriteLine ( allergyScore )
End If
If allergyScore >= 128 Then
allergyScore -= 128
list . Add ( "\"cats\"" )
End If
If allergyScore >= 64 Then
allergyScore -= 64
list . Add ( "\"pollen\"" )
End If
If allergyScore >= 32 Then
allergyScore -= 32
list . Add ( "\"chocolate\"" )
End If
If allergyScore >= 16 Then
allergyScore -= 16
list . Add ( "\"tomatoes\"" )
End If
If allergyScore >= 8 Then
allergyScore -= 8
list . Add ( "\"strawberries\"" )
End If
If allergyScore >= 4 Then
allergyScore -= 4
list . Add ( "\"shellfish\"" )
End If
If allergyScore >= 2 Then
allergyScore -= 2
list . Add ( "\"peanuts\"" )
End If
If allergyScore >= 1 Then
allergyScore -= 1
list . Add ( "\"eggs\"" )
End If
list . Reverse ()
AllergyList = list
End Sub
Public Function AllergicTo ( input As String )
Return Me . AllergyList . Contains ( input )
End Function
Public Function List ()
Return Me . AllergyList
End Function
End Class
