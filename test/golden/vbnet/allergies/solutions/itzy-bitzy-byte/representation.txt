(file "Allergies.vb")
Public Class Allergies
Private _score As Integer
Private _allergiesDictionary AS Dictionary ( Of String , Integer )
Public Sub New ( score As Integer )
_score = score
_allergiesDictionary = New Dictionary ( Of String , Integer )
_allergiesDictionary . Add ( "\"eggs\"" , 1 )
_allergiesDictionary . Add ( "\"peanuts\"" , 2 )
_allergiesDictionary . Add ( "\"shellfish\"" , 4 )
_allergiesDictionary . Add ( "\"strawberries\"" , 8 )
_allergiesDictionary . Add ( "\"tomatoes\"" , 16 )
_allergiesDictionary . Add ( "\"chocolate\"" , 32 )
_allergiesDictionary . Add ( "\"pollen\"" , 64 )
_allergiesDictionary . Add ( "\"cats\"" , 128 )
End Sub
Public Function AllergicTo ( allergy As String ) As Boolean
Return _allergiesDictionary . Item ( allergy ) And _score
End Function
Public Function List () As IList ( Of String )
Dim allergiesList As New List ( Of String )
For Each item As KeyValuePair ( Of String , Integer ) In _allergiesDictionary
If AllergicTo ( item . Key ) = True Then
allergiesList . Add ( item . Key )
End if
Next
List = allergiesList
End Function
End Class
