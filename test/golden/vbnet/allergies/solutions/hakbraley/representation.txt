(file "Allergies.vb")
Public Class Allergies
Protected allergiesList as String () = {
"\"eggs\"" ,
"\"peanuts\"" ,
"\"shellfish\"" ,
"\"strawberries\"" ,
"\"tomatoes\"" ,
"\"chocolate\"" ,
"\"pollen\"" ,
"\"cats\""
}
Private allergyScore as Integer
Private Function StrToScore ( allergy as String ) As Integer
Dim index As Integer = Array . IndexOf ( allergiesList , allergy . ToLower ())
If index = - 1
Return 0
Else
Return 1 << index
End If
End Function
Public Sub New ( score As Integer )
allergyScore = score
End Sub
Public Function AllergicTo ( allergy As String ) As Boolean
Return Convert . ToBoolean ( allergyScore And StrToScore ( allergy ))
End Function
Public Function List () As List ( Of String )
Dim allergies As New List ( Of String )
Dim index As Integer = 0
For Each allergen As String In allergiesList
If allergyScore And ( 1 << index )
allergies . Add ( allergiesList ( index ))
End If
index += 1
Next
Return allergies
End Function
End Class
