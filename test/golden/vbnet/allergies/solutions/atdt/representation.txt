Enum Allergen As Byte
eggs = 1
peanuts = 2
shellfish = 4
strawberries = 8
tomatoes = 16
chocolate = 32
pollen = 64
cats = 128
End Enum
Public Class Allergies
Protected Property Score As Integer
Public Sub New ( score As Integer )
Me . Score = score
End Sub
Public Function AllergicTo ( allergy As String ) As Boolean
Return Me . Score And [ Enum ]. Parse ( GetType ( Allergen ), allergy )
End Function
Public Function List () As IList ( Of String )
Return ( From v In System . Enum . GetValues ( GetType ( Allergen ))
Where Me . Score and v
Select allergen = v . ToString ). ToList ()
End Function
End Class
