﻿ Imports System.Runtime.CompilerServices
< Extension >
Public Module Int32Extensions
< Extension >
Public Function GetAliquotSum ( positiveNumber As Int32 ) As UInt32
If ( positiveNumber < 1 ) Then Throw New ArgumentOutOfRangeException ( NameOf ( positiveNumber ), positiveNumber , $ "\"A positive number expected!\"" )
If ( positiveNumber = 1 ) Then Return 0
Dim result As UInt32 = 1
Dim sqrt As Int32 = IntSqrt ( positiveNumber )
For i As Int32 = 2 To sqrt
Dim remainder As Int32
Dim quotient As Int32 = Math . DivRem ( positiveNumber , i , remainder )
If ( remainder = 0 ) Then
result += i
If ( i <> quotient ) Then result += quotient
End If
Next
Return result
End Function
Private Function IntSqrt ( positiveNumber As Int32 ) As Int32
If ( positiveNumber < 1 ) Then Throw New ArgumentOutOfRangeException ( NameOf ( positiveNumber ), positiveNumber , $ "\"A positive number expected!\"" )
Dim exactRoot As Double = Math . Sqrt ( positiveNumber )
Return CInt ( Math . Truncate ( exactRoot ))
End Function
End Module

Public Module PerfectNumbers
Public Function Classify ( ByVal number As Int32 ) As Classification
If ( number < 1 ) Then Throw New ArgumentOutOfRangeException ( NameOf ( number ), number , $ "\"The {NameOf(number)} must be a positive {NameOf(Int32)}!\"" )
Dim sum As UInt32 = number . GetAliquotSum ()
Return Classify ( number , sum )
End Function
Private Function Classify ( ByVal number As Int32 , aliquotSum As UInt32 ) As Classification
If ( aliquotSum > number ) Then Return Classification . Abundant
If ( aliquotSum < number ) Then Return Classification . Deficient
Return Classification . Perfect
End Function
End Module
