Public Class Binary
Dim isBinary as Boolean = True
Dim binArr as Char ()
Public Sub New ( value As String )
binArr = New Char () {}
binArr = value . ToCharArray
For i as Integer = 0 to binArr . Length - 1
If binArr ( i ) <> "\"0\"" And binArr ( i ) <> "\"1\"" Then isBinary = False
Next
End Sub
Public Function ToDecimal () As Integer
If isBinary = False Then Return 0
Dim result as integer = 0
For i As Integer = 0 To binArr . Length - 1
result += ( Convert . ToInt32 ( binArr ( i )) - 48 ) * 2 ^ ( binArr . Length - 1 - i )
Next
Return result
End Function
End Class
