Public Class Binary
Private _strBinary As Char ()
Public Sub New ( value As String )
_strBinary = StrReverse ( value ). ToCharArray ()
End Sub
Public Function ToDecimal () As Integer
Dim i As Integer
Dim decDecimal As Decimal
For Each digit As Char In _strBinary
If digit = "\"0\"" c Then
decDecimal += 0
ElseIf digit = "\"1\"" c Then
decDecimal += 1 << i
Else
Return 0
End If
i += 1
Next
Return decDecimal
End Function
End Class
