(file "Binary.vb")
Public Class Binary
Private stringValue as String
Public Sub New ( value As String )
stringValue = value
End Sub
Public Function ToDecimal () As Integer
Dim decimalValue As Integer = 0
Dim index As Integer = stringValue . Length - 1
For Each ch In stringValue
If ch <> "\"0\"" And ch <> "\"1\""
Return 0
Else If ch = "\"1\""
decimalValue += 1 << index
End If
index -= 1
Next
Return decimalValue
End Function
End Class
