Public Class SpiralMatrix
Public Shared Function GetMatrix ( ByVal size As Integer ) As Integer (,)
Dim matrix ( size - 1 , size - 1 ) As Integer
Dim i As Integer = 1
Dim minX As Integer = 0
Dim minY As Integer = 0
Dim maxX As Integer = size - 1
Dim maxY As Integer = size - 1
While minX <= maxX
For y As Integer = minY To maxY
matrix ( minX , y ) = i
i += 1
Next
minX += 1
For x As Integer = minX To maxX
matrix ( x , maxY ) = i
i += 1
Next
maxY -= 1
For y As Integer = maxY To minY Step - 1
matrix ( maxX , y ) = i
i += 1
Next
maxX -= 1
For x As Integer = maxX To minX Step - 1
matrix ( x , minY ) = i
i += 1
Next
minY += 1
End While
Return matrix
End Function
End Class
