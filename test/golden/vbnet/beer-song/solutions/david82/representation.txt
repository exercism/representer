Public Module BeerSong
Public Function Recite ( ByVal startBottles As Integer , ByVal takeDown As Integer ) As String
Dim verses = Enumerable . Range ( startBottles - ( takeDown - 1 ), takeDown ). Reverse (). Select ( Function ( n ) Verse ( n ))
Return String . Join ( Environment . NewLine & Environment . NewLine , verses )
End Function
Private Function Verse ( bottles As Integer ) As String
Select Case bottles
Case 2
Return $ "\"{bottles} bottles of beer on the wall, {bottles} bottles of beer.\"" & Environment . NewLine &
$ "\"Take one down and pass it around, {bottles - 1} bottle of beer on the wall.\""
Case 1
Return $ "\"{bottles} bottle of beer on the wall, {bottles} bottle of beer.\"" & Environment . NewLine &
$ "\"Take it down and pass it around, no more bottles of beer on the wall.\""
Case 0
Return "\"No more bottles of beer on the wall, no more bottles of beer.\"" & Environment . NewLine &
"\"Go to the store and buy some more, 99 bottles of beer on the wall.\""
Case Else
Return $ "\"{bottles} bottles of beer on the wall, {bottles} bottles of beer.\"" & Environment . NewLine &
$ "\"Take one down and pass it around, {bottles - 1} bottles of beer on the wall.\""
End Select
End Function
End Module
