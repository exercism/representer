Imports System.Text
Module Crypto
Public Function Ciphertext ( ByVal plaintext As String ) As String
If ( String . IsNullOrEmpty ( plaintext )) Then Return plaintext
Dim rawText As String = RemoveUnsupportedChars ( plaintext )
If ( rawText . Length = 0 ) Then Return String . Empty
Dim length As Int32 = rawText . Length
Dim cols As Int32
Dim rows As Int32
DetermineRowsAndColumns ( length , columns : = cols , rows : = rows )
If ( length > rawText . Length ) Then
rawText &= New String ( "\" \"" c , length - rawText . Length )
End If
Dim result As New StringBuilder ( length + cols )
For offset As Int32 = 0 To cols - 1
For i As Int32 = offset To length - 1 Step cols
result . Append ( rawText ( i ))
Next
result . Append ( "\" \"" c )
Next
result . Length -= 1
Return result . ToString ()
End Function
Private Function RemoveUnsupportedChars ( plaintext As String ) As String
Const alphabet As String = "\"abcdefghijklmnopqrstuvwxyz0123456789\""
Dim chars As IEnumerable ( Of Char ) = From c In plaintext Where Char . IsLetterOrDigit ( c ) Select c = Char . ToLowerInvariant ( c ) Where alphabet . IndexOf ( c ) > - 1
Return New String ( chars . ToArray ())
End Function
Private Sub DetermineRowsAndColumns ( ByRef length As Int32 , ByRef columns As Int32 , ByRef rows As Int32 )
Dim root As Double = Math . Sqrt ( length )
rows = CInt ( Math . Truncate ( root ))
columns = rows
If (( root Mod 1 ) = 0 ) Then Return
columns += 1
If ( rows * columns < length ) Then
rows += 1
End If
length = rows * columns
End Sub
End Module
