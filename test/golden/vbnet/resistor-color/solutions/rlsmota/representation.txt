Imports System
Public Module ResistorColor
Dim arrCol () As String = { "\"black\"" , "\"brown\"" , "\"red\"" , "\"orange\"" , "\"yellow\"" , "\"green\"" , "\"blue\"" , "\"violet\"" , "\"grey\"" , "\"white\"" }
Public Function ColorCode ( ByVal color As String ) As Integer
For i = 0 To arrCol . Length - 1
If arrCol ( i ) = color Then
Return i
End If
Next
End Function
Public Function Colors () As String ()
Return arrCol
End Function
End Module
