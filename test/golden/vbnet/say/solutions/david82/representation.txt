Imports System
Imports System.Collections.Generic
Public Module Say
Private digitNames As New Dictionary ( Of Long , String )()
Private upToTwentyNames As New Dictionary ( Of Long , String )()
Private tensNames As New Dictionary ( Of Long , String )()
Sub New ()
digitNames . Add ( 0 , "\"zero\"" )
digitNames . Add ( 1 , "\"one\"" )
digitNames . Add ( 2 , "\"two\"" )
digitNames . Add ( 3 , "\"three\"" )
digitNames . Add ( 4 , "\"four\"" )
digitNames . Add ( 5 , "\"five\"" )
digitNames . Add ( 6 , "\"six\"" )
digitNames . Add ( 7 , "\"seven\"" )
digitNames . Add ( 8 , "\"eigth\"" )
digitNames . Add ( 9 , "\"nine\"" )
upToTwentyNames . Add ( 0 , "\"ten\"" )
upToTwentyNames . Add ( 1 , "\"eleven\"" )
upToTwentyNames . Add ( 2 , "\"twelve\"" )
upToTwentyNames . Add ( 3 , "\"thirteen\"" )
upToTwentyNames . Add ( 4 , "\"fourteen\"" )
upToTwentyNames . Add ( 5 , "\"fifteen\"" )
upToTwentyNames . Add ( 6 , "\"sixteen\"" )
upToTwentyNames . Add ( 7 , "\"seventeen\"" )
upToTwentyNames . Add ( 8 , "\"eigteen\"" )
upToTwentyNames . Add ( 9 , "\"nineteen\"" )
tensNames . Add ( 1 , "\"ten\"" )
tensNames . Add ( 2 , "\"twenty\"" )
tensNames . Add ( 3 , "\"thirty\"" )
tensNames . Add ( 4 , "\"forty\"" )
tensNames . Add ( 5 , "\"fifty\"" )
tensNames . Add ( 6 , "\"sixty\"" )
tensNames . Add ( 7 , "\"seventy\"" )
tensNames . Add ( 8 , "\"eighty\"" )
tensNames . Add ( 9 , "\"ninety\"" )
tensNames . Add ( 10 , "\"hundred\"" )
End Sub
Const tens As Long = 10
Const hundreds As Long = 10 * 10
Const thousands As Long = hundreds * tens
Const millions As Long = thousands * hundreds * tens
Const billions As Long = millions * hundreds * tens
Public Function InEnglish ( ByVal number As Long ) As String
If number < 0 OrElse number > 999999999999L Then
Throw New ArgumentOutOfRangeException ()
End If
If number < tens Then
Return GetDigit ( number )
End If
If number < hundreds Then
Return GetTens ( number )
End If
If number < thousands Then
Return GetRecurse ( number , hundreds , "\" hundred\"" )
End If
If number < millions Then
Return GetRecurse ( number , thousands , "\" thousand\"" )
End If
If number < billions Then
Return GetRecurse ( number , millions , "\" million\"" )
End If
If number <= 999999999999L Then
Return GetRecurse ( number , billions , "\" billion\"" )
End If
Throw New ArgumentOutOfRangeException ()
End Function
Public Function GetRecurse ( ByVal ths As Long , ByVal divider As Long , ByVal marker As String ) As String
Dim hund As Long = ths \ divider
Dim remainder As Long = ths Mod divider
Dim dec = InEnglish ( hund ) & marker
If remainder > 0 Then
dec &= "\" \"" & InEnglish ( remainder )
End If
Return dec
End Function
Public Function GetTens ( ByVal ten As Long ) As String
Dim hund As Long = ten \ tens
Dim remainder As Long = ten Mod tens
If hund = 1 Then
Return upToTwentyNames ( remainder )
End If
Dim dec = tensNames ( hund )
If remainder > 0 Then
dec &= "\"-\"" & GetDigit ( remainder )
End If
Return dec
End Function
Public Function GetDigit ( ByVal ten As Long ) As String
Return digitNames ( ten )
End Function
End Module
