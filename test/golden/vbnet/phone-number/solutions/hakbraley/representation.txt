Public Class PhoneNumber
Public Shared Function Clean ( ByVal phoneNumber As String ) As String
If Regex . IsMatch ( phoneNumber , "\"[^0-9+. \\()\\-]\"" )
Throw New ArgumentException ( "\"Invalid characters found in phone number.\"" )
End If
phoneNumber = Regex . Replace ( phoneNumber , "\"[^\\d]\"" , "\"\"" )
If Len ( phoneNumber ) > 11 Or Len ( phoneNumber ) < 10
Throw New ArgumentException ( "\"Invalid phone number length.\"" )
End If
If Len ( phoneNumber ) = 11
If phoneNumber ( 0 ) <> "\"1\""
Throw New ArgumentException ( "\"11 digit phone number does not start with a 1.\"" )
End If
phoneNumber = phoneNumber . substring ( 1 )
End If
If phoneNumber ( 0 ) < "\"2\""
Throw New ArgumentException ( String . Format ( "\"Area Code cannot start with a {0}\"" , phoneNumber ( 0 )))
End If
If phoneNumber ( 3 ) < "\"2\""
Throw New ArgumentException ( String . Format ( "\"Exchange Code cannot start with a {0}\"" , phoneNumber ( 3 )))
End If
Return phoneNumber
End Function
End Class
